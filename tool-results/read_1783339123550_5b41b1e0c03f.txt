   138→   138→export function ProductionModule() {

   140→   140→  const [search, setSearch] = React.useState('')
   141→   141→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   142→   142→  const [loading, setLoading] = React.useState(true)
   143→   143→  const [formOpen, setFormOpen] = React.useState(false)
   144→   144→  const [editingProduction, setEditingProduction] = React.useState<Production | null>(null)
   145→   145→  const [formData, setFormData] = React.useState<ProductionFormData>(emptyForm)
   146→   146→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   147→   147→  const [deleteTarget, setDeleteTarget] = React.useState<Production | null>(null)
   148→   148→  const [deleting, setDeleting] = React.useState(false)
   149→   149→  const [deleteAllOpen, setDeleteAllOpen] = React.useState(false)
   150→   150→  const [deletingAll, setDeletingAll] = React.useState(false)
   151→   151→  const [deleteAllConfirm, setDeleteAllConfirm] = React.useState('')
   152→   152→
   153→   153→  // ── Multi-select state ──────────────────────────────────────────────────
   154→   154→  // Tracks which production rows the user has ticked in the table. The
   155→   155→  // "Delete Selected" button appears in the header whenever this set is
   156→   156→  // non-empty. We use a Set for O(1) toggle/lookup, then convert to an
   157→   157→  // array when sending to the API.
   158→   158→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   159→   159→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   160→   160→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   161→   161→
   162→   162→  const toggleSelect = (id: string) => {
   163→   163→    setSelectedIds((prev) => {
   164→   164→      const next = new Set(prev)
   165→   165→      if (next.has(id)) next.delete(id)
   166→   166→      else next.add(id)
   167→   167→      return next
   168→   168→    })
   169→   169→  }
   170→   170→
   171→   171→  const toggleSelectAll = () => {
   172→   172→    setSelectedIds((prev) => {
   173→   173→      // If everything is already selected, clear. Otherwise select all
   174→   174→      // currently-filtered rows (so the user can select-all within a search
   175→   175→      // result without picking rows they can't see).
   176→   176→      if (prev.size === filteredProductions.length && filteredProductions.length > 0) {
   177→   177→        return new Set()
   178→   178→      }
   179→   179→      return new Set(filteredProductions.map((p) => p.id))
   180→   180→    })
   181→   181→  }
   182→   182→
   183→   183→  const clearSelection = () => setSelectedIds(new Set())
   184→   184→
   185→   185→  const handleBulkDelete = async () => {
   186→   186→    const ids = Array.from(selectedIds)
   187→   187→    if (ids.length === 0) return
   188→   188→    setBulkDeleting(true)
   189→   189→    try {
   190→   190→      const res = await api.bulkDeleteProductions(ids)
   191→   191→      toast({
   192→   192→        title: 'Success',
   193→   193→        description: `${res.deletedCount} of ${ids.length} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   194→   194→      })
   195→   195→      setBulkDeleteOpen(false)
   196→   196→      clearSelection()
   197→   197→      fetchProductions()
   198→   198→    } catch (err) {
   199→   199→      toast({
   200→   200→        title: 'Error',
   201→   201→        description: err instanceof Error ? err.message : 'Failed to delete selected production entries',
   202→   202→        variant: 'destructive',
   203→   203→      })
   204→   204→    } finally {
   205→   205→      setBulkDeleting(false)
   206→   206→    }
   207→   207→  }
   208→   208→
   209→   209→  const fetchProductions = React.useCallback(async () => {
   210→   210→    setLoading(true)
   211→   211→    try {
   212→   212→      const res = await api.getProduction()
   213→   213→      const data = (res.productions as Production[]).sort(
   214→   214→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   215→   215→      )
   216→   216→      setProductions(data)
   217→   217→    } catch (err) {
   218→   218→      toast({
   219→   219→        title: 'Error',
   220→   220→        description: err instanceof Error ? err.message : 'Failed to fetch production data',
   221→   221→        variant: 'destructive',
   222→   222→      })
   223→   223→    } finally {
   224→   224→      setLoading(false)
   225→   225→    }
   226→   226→  }, [])
   227→   227→
   228→   228→  // Debounced search
   229→   229→  React.useEffect(() => {
   230→   230→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   231→   231→    return () => clearTimeout(t)
   232→   232→  }, [search])
   233→   233→
   234→   234→  // Client-side filter
   235→   235→  const filteredProductions = React.useMemo(() => {
   236→   236→    if (!debouncedSearch.trim()) return productions
   237→   237→    const q = debouncedSearch.toLowerCase()
   238→   238→    return productions.filter((item: any) =>
   239→   239→      ['date', 'remarks'].some((f) =>
   240→   240→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   241→   241→      )
   242→   242→    )
   243→   243→  }, [productions, debouncedSearch])
   244→   244→
   245→   245→  React.useEffect(() => {
   246→   246→    fetchProductions()
   247→   247→  }, [fetchProductions])
   248→   248→
   249→   249→  // Excel import
   250→   250→  const [importOpen, setImportOpen] = React.useState(false)
   251→   251→
   252→   252→  const openAddDialog = () => {
   253→   253→    setEditingProduction(null)
   254→   254→    setFormData(emptyForm)
   255→   255→    setFormOpen(true)
   256→   256→  }
   257→   257→
   258→   258→  const openEditDialog = (prod: Production) => {
   259→   259→    setEditingProduction(prod)
   260→   260→    setFormData({
   261→   261→      date: prod.date ? prod.date.split('T')[0] : '',
   262→   262→      cement: String(prod.cement || ''),
   263→   263→      zigZagGrey80: String(prod.zigZagGrey80 || ''),
   264→   264→      zigZagRed80: String(prod.zigZagRed80 || ''),
   265→   265→      zigZagYellow80: String(prod.zigZagYellow80 || ''),
   266→   266→      zigZagGrey60: String(prod.zigZagGrey60 || ''),
   267→   267→      zigZagRed60: String(prod.zigZagRed60 || ''),
   268→   268→      zigZagYellow60: String(prod.zigZagYellow60 || ''),
   269→   269→      curveStone: String(prod.curveStone || ''),
   270→   270→      chequreTile: String(prod.chequreTile || ''),
   271→   271→      dumbleGrey80: String(prod.dumbleGrey80 || ''),
   272→   272→      dumbleRed80: String(prod.dumbleRed80 || ''),
   273→   273→      dumbleYellow80: String(prod.dumbleYellow80 || ''),
   274→   274→      transportationCharge: String(prod.transportationCharge || ''),
   275→   275→      remarks: prod.remarks || '',
   276→   276→    })
   277→   277→    setFormOpen(true)
   278→   278→  }
   279→   279→
   280→   280→  const handleFormChange = (field: keyof ProductionFormData, value: string) => {
   281→   281→    setFormData((prev) => ({ ...prev, [field]: value }))
   282→   282→  }
   283→   283→
   284→   284→  const handleSubmit = async () => {
   285→   285→    if (!formData.date) {
   286→   286→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   287→   287→      return
   288→   288→    }
   289→   289→
   290→   290→    setFormSubmitting(true)
   291→   291→    try {
   292→   292→      const payload = {
   293→   293→        date: formData.date,
   294→   294→        cement: Number(formData.cement) || 0,
   295→   295→        zigZagGrey80: Number(formData.zigZagGrey80) || 0,
   296→   296→        zigZagRed80: Number(formData.zigZagRed80) || 0,
   297→   297→        zigZagYellow80: Number(formData.zigZagYellow80) || 0,
   298→   298→        zigZagGrey60: Number(formData.zigZagGrey60) || 0,
   299→   299→        zigZagRed60: Number(formData.zigZagRed60) || 0,
   300→   300→        zigZagYellow60: Number(formData.zigZagYellow60) || 0,
   301→   301→        curveStone: Number(formData.curveStone) || 0,
   302→   302→        chequreTile: Number(formData.chequreTile) || 0,
   303→   303→        dumbleGrey80: Number(formData.dumbleGrey80) || 0,
   304→   304→        dumbleRed80: Number(formData.dumbleRed80) || 0,
   305→   305→        dumbleYellow80: Number(formData.dumbleYellow80) || 0,
   306→   306→        transportationCharge: Number(formData.transportationCharge) || 0,
   307→   307→        remarks: formData.remarks.trim(),
   308→   308→      }
   309→   309→
   310→   310→      if (editingProduction) {
   311→   311→        await api.updateProduction(editingProduction.id, payload)
   312→   312→        toast({ title: 'Success', description: 'Production entry updated successfully' })
   313→   313→      } else {
   314→   314→        await api.createProduction(payload)
   315→   315→        toast({ title: 'Success', description: 'Production entry created successfully' })
   316→   316→      }
   317→   317→
   318→   318→      setFormOpen(false)
   319→   319→      setFormData(emptyForm)
   320→   320→      setEditingProduction(null)
   321→   321→      fetchProductions()
   322→   322→    } catch (err) {
   323→   323→      toast({
   324→   324→        title: 'Error',
   325→   325→        description: err instanceof Error ? err.message : 'Failed to save production entry',
   326→   326→        variant: 'destructive',
   327→   327→      })
   328→   328→    } finally {
   329→   329→      setFormSubmitting(false)
   330→   330→    }
   331→   331→  }
   332→   332→
   333→   333→  const handleDelete = async () => {
   334→   334→    if (!deleteTarget) return
   335→   335→    setDeleting(true)
   336→   336→    try {
   337→   337→      await api.deleteProduction(deleteTarget.id)
   338→   338→      toast({ title: 'Success', description: 'Production entry deleted successfully' })
   339→   339→      setDeleteTarget(null)
   340→   340→      fetchProductions()
   341→   341→    } catch (err) {
   342→   342→      toast({
   343→   343→        title: 'Error',
   344→   344→        description: err instanceof Error ? err.message : 'Failed to delete production entry',
   345→   345→        variant: 'destructive',
   346→   346→      })
   347→   347→    } finally {
   348→   348→      setDeleting(false)
   349→   349→    }
   350→   350→  }
   351→   351→
   352→   352→  const handleDeleteAll = async () => {
   353→   353→    // No more 'DELETE ALL' typing requirement — the dialog now uses a
   354→   354→    // simple Yes / No choice. We keep the deleteAllConfirm state around
   355→   355→    // only so existing code that resets it (e.g. on dialog close) doesn't
   356→   356→    // break, but it's no longer gating the delete.
   357→   357→    setDeletingAll(true)
   358→   358→    try {
   359→   359→      const res = await api.deleteAllProductions()
   360→   360→      toast({
   361→   361→        title: 'Success',
   362→   362→        description: `${res.deletedCount} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   363→   363→      })
   364→   364→      setDeleteAllOpen(false)
   365→   365→      setDeleteAllConfirm('')
   366→   366→      fetchProductions()
   367→   367→    } catch (err) {
   368→   368→      toast({
   369→   369→        title: 'Error',
   370→   370→        description: err instanceof Error ? err.message : 'Failed to delete all production entries',
   371→   371→        variant: 'destructive',
   372→   372→      })
   373→   373→    } finally {
   374→   374→      setDeletingAll(false)
   375→   375→    }
   376→   376→  }
   377→   377→
   378→   378→  const renderSkeletons = () =>
   379→   379→    Array.from({ length: 3 }).map((_, i) => (
   380→   380→      <TableRow key={i}>
   381→   381→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   382→   382→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   383→   383→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   384→   384→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   385→   385→        {PRODUCT_FIELDS.map((_, j) => (
   386→   386→          <TableCell key={j}><Skeleton className="h-4 w-16" /></TableCell>
   387→   387→        ))}
   388→   388→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   389→   389→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   390→   390→      </TableRow>
   391→   391→    ))
   392→   392→
   393→   393→  return (
   394→   394→    <div className="space-y-6 relative">
   395→   395→      {/* Full-screen loading overlay shown during any delete operation.
   396→   396→          The spinner + message stay visible until the API call completes and
   397→   397→          the table refreshes, so the user always knows "something is happening". */}
   398→   398→      {(deleting || deletingAll || bulkDeleting) && (
   399→   399→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   400→   400→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   401→   401→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   402→   402→            <div className="text-center">
   403→   403→              <p className="text-lg font-semibold">
   404→   404→                {bulkDeleting
   405→   405→                  ? `Deleting ${selectedIds.size} entries...`
   406→   406→                  : deletingAll
   407→   407→                  ? 'Deleting all entries...'
   408→   408→                  : 'Deleting entry...'}
   409→   409→              </p>
   410→   410→              <p className="text-sm text-muted-foreground mt-1">
   411→   411→                Please wait while records are removed and stock is re-synced.
   412→   412→              </p>
   413→   413→            </div>
   414→   414→          </div>
   415→   415→        </div>
   416→   416→      )}
   417→   417→      {/* Header */}
   418→   418→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   419→   419→        <div className="flex items-center gap-3">
   420→   420→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   421→   421→            <Factory className="size-5" />
   422→   422→          </div>
   423→   423→          <div>
   424→   424→            <h2 className="text-2xl font-bold tracking-tight">Production Management</h2>
   425→   425→            <p className="text-sm text-muted-foreground">
   426→   426→              Track daily paver block production entries
   427→   427→            </p>
   428→   428→          </div>
   429→   429→        </div>
   430→   430→        <div className="flex flex-wrap gap-2 w-full sm:w-auto">
   431→   431→          <Button
   432→   432→            variant="outline"
   433→   433→            onClick={() => setImportOpen(true)}
   434→   434→            className="w-full sm:w-auto"
   435→   435→          >
   436→   436→            <Upload className="size-4 mr-2" />
   437→   437→            Import Excel
   438→   438→          </Button>
   439→   439→          {/* Bulk-delete button — only visible when at least one row is selected.
   440→   440→              Clicking it opens the confirmation dialog below. */}
   441→   441→          {selectedIds.size > 0 && (
   442→   442→            <Button
   443→   443→              variant="outline"
   444→   444→              onClick={() => setBulkDeleteOpen(true)}
   445→   445→              disabled={bulkDeleting || loading}
   446→   446→              className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   447→   447→            >
   448→   448→              <Trash2 className="size-4 mr-2" />
   449→   449→              Delete Selected
   450→   450→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">
   451→   451→                {selectedIds.size}
   452→   452→              </Badge>
   453→   453→            </Button>
   454→   454→          )}
   455→   455→          {/* Clear-selection button — small, ghost-styled, only visible when rows are selected */}
   456→   456→          {selectedIds.size > 0 && (
   457→   457→            <Button
   458→   458→              variant="ghost"
