   366→// Format a 10-digit Indian mobile number as "+91 XXXXX XXXXX" for display.
   367→// Falls back to the original string if it doesn't look like a 10-digit number.
   368→function formatPhone(phone: string): string {
   369→  const digits = phone.replace(/\D/g, '')
   370→  // Handle "9572831213" or "919572831213" or "+919572831213"
   371→  if (digits.length === 10) {
   372→    return `+91 ${digits.slice(0, 5)} ${digits.slice(5)}`
   373→  }
   374→  if (digits.length === 12 && digits.startsWith('91')) {
   375→    return `+91 ${digits.slice(2, 7)} ${digits.slice(7)}`
   376→  }
   377→  return phone
   378→}
