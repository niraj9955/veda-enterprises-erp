   131→// ── Helpers ─────────────────────────────────────────────────────────────────
   132→
   133→const formatDate = (dateStr: string): string => {
   134→  if (!dateStr) return '—'
   135→  const date = new Date(dateStr + 'T00:00:00')
   136→  return date.toLocaleDateString('en-IN', {
   137→    day: '2-digit',
   138→    month: 'short',
   139→    year: 'numeric',
   140→  })
   141→}

   143→// ── Component ───────────────────────────────────────────────────────────────
   144→
   145→export function DispatchModule() {
   146→  // ── State ─────────────────────────────────────────────────────────────
   147→  const [dispatches, setDispatches] = React.useState<Dispatch[]>([])
   148→  const [search, setSearch] = React.useState('')
   149→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   150→  const [loading, setLoading] = React.useState(true)
   151→
   152→  // Create dialog
   153→  const [createOpen, setCreateOpen] = React.useState(false)
   154→  const [formData, setFormData] = React.useState<DispatchFormData>(emptyForm)
   155→  const [submitting, setSubmitting] = React.useState(false)
   156→
   157→  // Dropdown data
   158→  const [customers, setCustomers] = React.useState<Customer[]>([])
   159→  const [orders, setOrders] = React.useState<Order[]>([])
   160→  const [dropdownsLoading, setDropdownsLoading] = React.useState(false)
   161→
   162→  // Challan dialog
   163→  const [challanDispatch, setChallanDispatch] = React.useState<Dispatch | null>(null)
   164→
   165→  // Excel import
   166→  const [importOpen, setImportOpen] = React.useState(false)
   167→
   168→  // Delete confirmation
   169→  const [deleteTarget, setDeleteTarget] = React.useState<Dispatch | null>(null)
   170→  const [deleting, setDeleting] = React.useState(false)
   171→
   172→  // ── Fetch dispatches ──────────────────────────────────────────────────
   173→  const fetchDispatches = React.useCallback(async () => {
   174→    setLoading(true)
   175→    try {
   176→      const res = await api.getDispatches()
   177→      setDispatches(res.dispatches as Dispatch[])
   178→    } catch (err) {
   179→      toast({
   180→        title: 'Error',
   181→        description: err instanceof Error ? err.message : 'Failed to fetch dispatches',
   182→        variant: 'destructive',
   183→      })
   184→    } finally {
   185→      setLoading(false)
   186→    }
   187→  }, [])
   188→
   189→  // Debounced search
   190→  React.useEffect(() => {
   191→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   192→    return () => clearTimeout(t)
   193→  }, [search])
   194→
   195→  // Client-side filter
   196→  const filteredDispatches = React.useMemo(() => {
   197→    if (!debouncedSearch.trim()) return dispatches
   198→    const q = debouncedSearch.toLowerCase()
   199→    return dispatches.filter((item: any) =>
   200→      ['dispatchNumber', 'truckNumber', 'driverName', 'brickType', 'date'].some((f) =>
   201→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   202→      )
   203→    )
   204→  }, [dispatches, debouncedSearch])
   205→
   206→  React.useEffect(() => {
   207→    fetchDispatches()
   208→  }, [fetchDispatches])
   209→
   210→  // ── Fetch dropdown data when create dialog opens ──────────────────────
   211→  React.useEffect(() => {
   212→    if (!createOpen) return
   213→
   214→    const fetchDropdownData = async () => {
   215→      setDropdownsLoading(true)
   216→      try {
   217→        const [custRes, ordRes] = await Promise.all([
   218→          api.getCustomers(),
   219→          api.getOrders(),
   220→        ])
   221→        setCustomers(custRes.customers as Customer[])
   222→        setOrders(ordRes.orders as Order[])
   223→      } catch (err) {
   224→        toast({
   225→          title: 'Error',
   226→          description: 'Failed to load dropdown data',
   227→          variant: 'destructive',
   228→        })
   229→      } finally {
   230→        setDropdownsLoading(false)
   231→      }
   232→    }
   233→
   234→    fetchDropdownData()
   235→  }, [createOpen])
   236→
   237→  // ── Filter orders by selected customer ────────────────────────────────
   238→  const pendingOrdersForCustomer = React.useMemo(() => {
   239→    if (!formData.customerId) return []
   240→    return orders.filter(
   241→      (o) =>
   242→        o.customerId === formData.customerId &&
   243→        (o.status === 'Pending' || o.status === 'Processing')
   244→    )
   245→  }, [orders, formData.customerId])
   246→
   247→  // ── Auto-fill brick type when order is selected ───────────────────────
   248→  const handleOrderChange = (orderId: string) => {
   249→    setFormData((prev) => {
   250→      if (!orderId) return { ...prev, orderId: '' }
   251→      const selectedOrder = orders.find((o) => o.id === orderId)
   252→      return {
   253→        ...prev,
   254→        orderId,
   255→        brickType: selectedOrder?.brickType || prev.brickType,
   256→        quantity: selectedOrder ? String(selectedOrder.quantity) : prev.quantity,
   257→      }
   258→    })
   259→  }
   260→
   261→  // ── Create dispatch ───────────────────────────────────────────────────
   262→  const handleCreate = async () => {
   263→    // Unified empty-form check — show ONE popup instead of cascading errors
   264→    if (isFormEmpty([formData.customerId, formData.orderId, formData.brickType, formData.quantity, formData.truckNumber, formData.driverName, formData.date])) {
   265→      toast(showPleaseFillDataToast())
   266→      return
   267→    }
   268→    if (!formData.customerId) {
   269→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   270→      return
   271→    }
   272→    if (!formData.brickType) {
   273→      toast({ title: 'Validation Error', description: 'Please select a brick type', variant: 'destructive' })
   274→      return
   275→    }
   276→    if (!formData.quantity || Number(formData.quantity) <= 0) {
   277→      toast({ title: 'Validation Error', description: 'Please enter a valid quantity', variant: 'destructive' })
   278→      return
   279→    }
   280→    if (!formData.truckNumber.trim()) {
   281→      toast({ title: 'Validation Error', description: 'Please enter the truck number', variant: 'destructive' })
   282→      return
   283→    }
   284→    if (!formData.driverName.trim()) {
   285→      toast({ title: 'Validation Error', description: 'Please enter the driver name', variant: 'destructive' })
   286→      return
   287→    }
   288→    if (!formData.date) {
   289→      toast({ title: 'Validation Error', description: 'Please select a date', variant: 'destructive' })
   290→      return
   291→    }
   292→
   293→    setSubmitting(true)
   294→    try {
   295→      const payload = {
   296→        customerId: formData.customerId,
   297→        orderId: formData.orderId || null,
   298→        brickType: formData.brickType,
   299→        quantity: Number(formData.quantity),
   300→        truckNumber: formData.truckNumber.trim(),
   301→        driverName: formData.driverName.trim(),
   302→        date: formData.date,
   303→      }
   304→
   305→      await api.createDispatch(payload)
   306→      toast({ title: 'Success', description: 'Dispatch created successfully' })
   307→      setCreateOpen(false)
   308→      setFormData(emptyForm)
   309→      fetchDispatches()
   310→    } catch (err) {
   311→      toast({
   312→        title: 'Error',
   313→        description: err instanceof Error ? err.message : 'Failed to create dispatch',
   314→        variant: 'destructive',
   315→      })
   316→    } finally {
   317→      setSubmitting(false)
   318→    }
   319→  }
   320→
   321→  // ── Delete dispatch ───────────────────────────────────────────────────
   322→  const handleDelete = async () => {
   323→    if (!deleteTarget) return
   324→    setDeleting(true)
   325→    try {
   326→      await api.deleteDispatch(deleteTarget.id)
   327→      toast({ title: 'Success', description: 'Dispatch deleted successfully' })
   328→      setDeleteTarget(null)
   329→      fetchDispatches()
   330→    } catch (err) {
   331→      toast({
   332→        title: 'Error',
   333→        description: err instanceof Error ? err.message : 'Failed to delete dispatch',
   334→        variant: 'destructive',
   335→      })
   336→    } finally {
   337→      setDeleting(false)
   338→    }
   339→  }
   340→
   341→  // ── Open create dialog ────────────────────────────────────────────────
   342→  const openCreateDialog = () => {
   343→    setFormData(emptyForm)
   344→    setCreateOpen(true)
   345→  }
   346→
   347→  // ── Print challan ─────────────────────────────────────────────────────
   348→  const handlePrint = () => {
   349→    window.print()
   350→  }
   351→
   352→  // ── Render: Loading skeletons ─────────────────────────────────────────
   353→  const renderSkeletons = () =>
   354→    Array.from({ length: 5 }).map((_, i) => (
   355→      <TableRow key={i}>
   356→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   357→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   358→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   359→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   360→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   361→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   362→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   363→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   364→      </TableRow>
   365→    ))
   366→
   367→  // ── Render: Create dispatch dialog ────────────────────────────────────
   368→  const renderCreateDialog = () => (
   369→    <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   370→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   371→        <DialogHeader>
   372→          <DialogTitle className="flex items-center gap-2">
   373→            <Truck className="size-5 text-emerald-600" />
   374→            Create Dispatch
   375→          </DialogTitle>
   376→          <DialogDescription>
   377→            Fill in the details to create a new dispatch entry.
   378→          </DialogDescription>
   379→        </DialogHeader>
   380→
   381→        <div className="grid gap-4 py-2">
   382→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   383→          <CustomerSearchInput
   384→            value={formData.customerId}
   385→            onSelect={(c) =>
   386→              setFormData((prev) => ({ ...prev, customerId: c.id, orderId: '', brickType: '', quantity: '' }))
   387→            }
   388→            onClear={() =>
   389→              setFormData((prev) => ({ ...prev, customerId: '', orderId: '', brickType: '', quantity: '' }))
   390→            }
   391→            required
   392→            label="Customer"
   393→            placeholder="Type customer name or mobile to search..."
   394→          />
   395→
   396→          {/* Order (optional) */}
   397→          <div className="grid gap-2">
   398→            <Label>
   399→              Order <span className="text-muted-foreground text-xs font-normal">(optional)</span>
   400→            </Label>
   401→            <Select
   402→              value={formData.orderId}
   403→              onValueChange={handleOrderChange}
   404→              disabled={!formData.customerId || pendingOrdersForCustomer.length === 0}
   405→            >
   406→              <SelectTrigger className="w-full">
   407→                <SelectValue
   408→                  placeholder={
   409→                    !formData.customerId
   410→                      ? 'Select a customer first'
   411→                      : pendingOrdersForCustomer.length === 0
   412→                        ? 'No pending orders'
   413→                        : 'Select order'
   414→                  }
   415→                />
   416→              </SelectTrigger>
   417→              <SelectContent>
   418→                {pendingOrdersForCustomer.map((o) => (
   419→                  <SelectItem key={o.id} value={o.id}>
   420→                    {o.orderNumber} — {o.brickType} ({o.quantity.toLocaleString()})
   421→                  </SelectItem>
   422→                ))}
   423→              </SelectContent>
   424→            </Select>
   425→          </div>
   426→
   427→          {/* Brick Type */}
   428→          <div className="grid gap-2">
   429→            <Label>
   430→              Brick Type <span className="text-destructive">*</span>
   431→            </Label>
   432→            <Select
   433→              value={formData.brickType}
   434→              onValueChange={(val) => setFormData((prev) => ({ ...prev, brickType: val }))}
   435→            >
   436→              <SelectTrigger className="w-full">
   437→                <SelectValue placeholder="Select brick type" />
   438→              </SelectTrigger>
   439→              <SelectContent>
   440→                {BRICK_TYPES.map((type) => (
   441→                  <SelectItem key={type} value={type}>
   442→                    {type}
   443→                  </SelectItem>
   444→                ))}
   445→              </SelectContent>
   446→            </Select>
   447→          </div>
   448→
   449→          {/* Quantity */}
   450→          <div className="grid gap-2">
   451→            <Label htmlFor="dispatch-quantity">
   452→              Quantity <span className="text-destructive">*</span>
   453→            </Label>
   454→            <Input
   455→              id="dispatch-quantity"
   456→              type="number"
   457→              min="1"
   458→              placeholder="Enter quantity"
   459→              value={formData.quantity}
   460→              onChange={(e) => setFormData((prev) => ({ ...prev, quantity: e.target.value }))}
   461→            />
   462→          </div>
   463→
   464→          {/* Truck Number */}
   465→          <div className="grid gap-2">
   466→            <Label htmlFor="dispatch-truck">
   467→              Truck Number <span className="text-destructive">*</span>
   468→            </Label>
   469→            <Input
   470→              id="dispatch-truck"
   471→              placeholder="e.g. RJ-14-AB-1234"
   472→              value={formData.truckNumber}
   473→              onChange={(e) => setFormData((prev) => ({ ...prev, truckNumber: e.target.value }))}
   474→            />
   475→          </div>
   476→
   477→          {/* Driver Name */}
   478→          <div className="grid gap-2">
   479→            <Label htmlFor="dispatch-driver">
   480→              Driver Name <span className="text-destructive">*</span>
   481→            </Label>
   482→            <Input
   483→              id="dispatch-driver"
   484→              placeholder="Enter driver name"
   485→              value={formData.driverName}
   486→              onChange={(e) => setFormData((prev) => ({ ...prev, driverName: e.target.value }))}
   487→            />
   488→          </div>
   489→
   490→          {/* Date */}
   491→          <div className="grid gap-2">
   492→            <Label htmlFor="dispatch-date">
   493→              Date <span className="text-destructive">*</span>
   494→            </Label>
   495→            <Input
   496→              id="dispatch-date"
   497→              type="date"
   498→              value={formData.date}
   499→              onChange={(e) => setFormData((prev) => ({ ...prev, date: e.target.value }))}
   500→            />
   501→          </div>
   502→        </div>
   503→
   504→        <DialogFooter>
   505→          <Button
   506→            variant="outline"
   507→            onClick={() => setCreateOpen(false)}
   508→            disabled={submitting}
   509→          >
   510→            Cancel
   511→          </Button>
   512→          <Button
   513→            onClick={handleCreate}
   514→            disabled={submitting}
   515→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   516→          >
   517→            {submitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   518→            Create Dispatch
   519→          </Button>
   520→        </DialogFooter>
   521→      </DialogContent>
   522→    </Dialog>
   523→  )
   524→
   525→  // ── Render: Challan / Print View dialog ───────────────────────────────
   526→  const renderChallanDialog = () => {
   527→    if (!challanDispatch) return null
   528→    const d = challanDispatch
   529→
   530→    return (
   531→      <Dialog open={!!challanDispatch} onOpenChange={(open) => !open && setChallanDispatch(null)}>
   532→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto print:max-w-none print:max-h-none print:p-0 print:shadow-none print:border-none">
   533→          {/* Screen-only close button header */}
   534→          <DialogHeader className="print:hidden">
   535→            <DialogTitle className="flex items-center gap-2">
   536→              <Eye className="size-5 text-emerald-600" />
   537→              Dispatch Challan
   538→            </DialogTitle>
   539→            <DialogDescription>
   540→              Preview and print the dispatch slip
   541→            </DialogDescription>
   542→          </DialogHeader>
   543→
   544→          {/* ── Challan Content (Print-friendly) ─────────────────────── */}
   545→          <div id="challan-content" className="print:p-6">
   546→            {/* Company Header */}
   547→            <div className="text-center mb-6 print:mb-8">
   548→              <div className="flex items-center justify-center gap-3 mb-2">
   549→                <Building2 className="size-8 text-emerald-600 print:text-black" />
   550→                <h1 className="text-2xl font-bold tracking-wide text-emerald-800 print:text-black">
   551→                  Veda Enterprises
   552→                </h1>
   553→              </div>
   554→              <p className="text-sm text-muted-foreground print:text-gray-600">
   555→                Brick Manufacturing &amp; Supply
   556→              </p>
   557→              <Separator className="mt-4 print:mt-6" />
   558→            </div>
   559→
   560→            {/* Title */}
   561→            <div className="text-center mb-6">
   562→              <h2 className="text-lg font-semibold underline underline-offset-4 print:text-black">
   563→                DISPATCH SLIP
   564→              </h2>
   565→            </div>
   566→
   567→            {/* Dispatch & Customer Info Grid */}
   568→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-6 text-sm print:text-black">
   569→              {/* Left column: Dispatch info */}
   570→              <div className="space-y-2">
   571→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   572→                  <span className="text-muted-foreground print:text-gray-500">Dispatch No:</span>
   573→                  <span className="font-semibold">{d.dispatchNumber}</span>
   574→                </div>
   575→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   576→                  <span className="text-muted-foreground print:text-gray-500">Date:</span>
   577→                  <span className="font-semibold">{formatDate(d.date)}</span>
   578→                </div>
   579→                {d.order && (
   580→                  <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   581→                    <span className="text-muted-foreground print:text-gray-500">Order Ref:</span>
   582→                    <span className="font-semibold">{d.order.orderNumber}</span>
   583→                  </div>
   584→                )}
   585→              </div>
   586→
   587→              {/* Right column: Customer info */}
   588→              <div className="space-y-2">
   589→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   590→                  <span className="text-muted-foreground print:text-gray-500">Customer:</span>
   591→                  <span className="font-semibold">{d.customer?.name || '—'}</span>
   592→                </div>
   593→                {d.customer?.mobile && (
   594→                  <div className="flex items-center justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   595→                    <span className="text-muted-foreground print:text-gray-500 flex items-center gap-1">
   596→                      <Phone className="size-3" /> Phone:
   597→                    </span>
   598→                    <span>{d.customer.mobile}</span>
   599→                  </div>
   600→                )}
   601→                {d.customer?.address && (
   602→                  <div className="flex items-start justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   603→                    <span className="text-muted-foreground print:text-gray-500 flex items-center gap-1 shrink-0">
   604→                      <MapPin className="size-3" /> Address:
   605→                    </span>
   606→                    <span className="text-right max-w-[60%]">{d.customer.address}</span>
   607→                  </div>
   608→                )}
   609→              </div>
   610→            </div>
   611→
   612→            <Separator className="mb-6 print:border-gray-400" />
   613→
   614→            {/* Transport Info */}
   615→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-6 text-sm print:text-black">
   616→              <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   617→                <span className="text-muted-foreground print:text-gray-500">Truck Number:</span>
   618→                <span className="font-semibold">
   619→                  <Badge
   620→                    variant="outline"
   621→                    className="font-mono text-xs print:border-black print:text-black print:bg-transparent"
   622→                  >
   623→                    {d.truckNumber}
   624→                  </Badge>
   625→                </span>
   626→              </div>
   627→              <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   628→                <span className="text-muted-foreground print:text-gray-500">Driver Name:</span>
   629→                <span className="font-semibold">{d.driverName}</span>
   630→              </div>
   631→            </div>
   632→
   633→            <Separator className="mb-6 print:border-gray-400" />
   634→
   635→            {/* Material Details */}
   636→            <div className="mb-8 print:text-black">
   637→              <h3 className="text-sm font-semibold mb-3 text-muted-foreground print:text-gray-500 uppercase tracking-wider">
   638→                Material Details
   639→              </h3>
   640→              <Table>
   641→                <TableHeader>
   642→                  <TableRow className="print:border-gray-400">
   643→                    <TableHead className="print:text-black print:border-gray-400">Brick Type</TableHead>
   644→                    <TableHead className="text-right print:text-black print:border-gray-400">Quantity</TableHead>
   645→                  </TableRow>
   646→                </TableHeader>
   647→                <TableBody>
   648→                  <TableRow className="print:border-gray-400">
   649→                    <TableCell className="font-medium print:text-black print:border-gray-400">
   650→                      {d.brickType}
   651→                    </TableCell>
   652→                    <TableCell className="text-right font-mono font-semibold print:text-black print:border-gray-400">
   653→                      {d.quantity.toLocaleString()}
   654→                    </TableCell>
   655→                  </TableRow>
   656→                </TableBody>
   657→              </Table>
   658→            </div>
   659→
   660→            {/* Signature area */}
   661→            <div className="grid grid-cols-2 gap-8 mt-12 print:text-black">
   662→              <div className="text-center">
   663→                <div className="border-t border-gray-400 pt-2 text-sm text-muted-foreground print:text-gray-500">
   664→                  Driver Signature
   665→                </div>
   666→              </div>
   667→              <div className="text-center">
   668→                <div className="border-t border-gray-400 pt-2 text-sm text-muted-foreground print:text-gray-500">
   669→                  Authorized Signature
   670→                </div>
   671→              </div>
   672→            </div>
   673→
   674→            {/* Footer */}
   675→            <div className="mt-8 text-center text-xs text-muted-foreground print:text-gray-400 print:mt-12">
   676→              <p>This is a computer-generated dispatch slip from Veda Enterprises ERP.</p>
   677→            </div>
   678→          </div>
   679→
   680→          {/* Screen-only footer with print button */}
   681→          <DialogFooter className="print:hidden">
   682→            <Button
   683→              variant="outline"
   684→              onClick={() => setChallanDispatch(null)}
   685→            >
   686→              Close
   687→            </Button>
   688→            <Button
   689→              onClick={handlePrint}
   690→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   691→            >
   692→              <Printer className="size-4" />
   693→              Print Challan
   694→            </Button>
   695→          </DialogFooter>
   696→        </DialogContent>
   697→      </Dialog>
   698→    )
   699→  }
   700→
   701→  // ── Render: Delete confirmation ───────────────────────────────────────
   702→  const renderDeleteDialog = () => (
   703→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   704→      <AlertDialogContent>
   705→        <AlertDialogHeader>
   706→          <AlertDialogTitle>Delete Dispatch</AlertDialogTitle>
   707→          <AlertDialogDescription>
   708→            Are you sure you want to delete dispatch{' '}
   709→            <strong>{deleteTarget?.dispatchNumber}</strong>? This action cannot
   710→            be undone and the stock will not be restored.
   711→          </AlertDialogDescription>
   712→        </AlertDialogHeader>
   713→        <AlertDialogFooter>
   714→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   715→          <AlertDialogAction
   716→            onClick={handleDelete}
   717→            disabled={deleting}
   718→            className="bg-destructive text-white hover:bg-destructive/90"
   719→          >
   720→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   721→            Delete
   722→          </AlertDialogAction>
   723→        </AlertDialogFooter>
   724→      </AlertDialogContent>
   725→    </AlertDialog>
   726→  )
   727→
   728→  // ── Render: Main ──────────────────────────────────────────────────────
   729→  return (
   730→    <div className="space-y-6 print:hidden">
   731→      {/* Header */}
   732→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   733→        <div className="flex items-center gap-3">
   734→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   735→            <Truck className="size-5" />
   736→          </div>
   737→          <div>
   738→            <h2 className="text-2xl font-bold tracking-tight">Dispatch Management</h2>
   739→            <p className="text-sm text-muted-foreground">
   740→              Manage dispatch entries and generate challans
   741→            </p>
   742→          </div>
   743→        </div>
   744→        <div className="flex gap-2 w-full sm:w-auto">
   745→          <Button
   746→            variant="outline"
   747→            onClick={() => setImportOpen(true)}
   748→            className="w-full sm:w-auto"
   749→          >
   750→            <Upload className="size-4 mr-2" />
   751→            Import Excel
   752→          </Button>
   753→          <Button
   754→            onClick={openCreateDialog}
   755→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   756→          >
   757→            <Plus className="size-4" />
   758→            Create Dispatch
   759→          </Button>
   760→        </div>
   761→      </div>
   762→
   763→      {/* Search */}
   764→      <Card>
   765→        <CardContent className="pt-6">
   766→          <div className="relative">
   767→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   768→            <Input
   769→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   770→              value={search}
   771→              onChange={(e) => setSearch(e.target.value)}
   772→              className="pl-9"
   773→            />
   774→          </div>
   775→        </CardContent>
   776→      </Card>
   777→
   778→      {/* Table */}
   779→      <Card>
   780→        <CardHeader>
   781→          <CardTitle className="flex items-center justify-between">
   782→            <span>Dispatches</span>
   783→            <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   784→              {filteredDispatches.length} of {dispatches.length} record{dispatches.length !== 1 ? 's' : ''}
   785→            </Badge>
   786→          </CardTitle>
   787→        </CardHeader>
   788→        <CardContent>
   789→          <div className="max-h-[60vh] overflow-auto rounded-md border">
   790→            <Table>
   791→              <TableHeader className="sticky top-0 bg-background z-10">
   792→                <TableRow>
   793→                  <TableHead>Dispatch No.</TableHead>
   794→                  <TableHead>Customer</TableHead>
   795→                  <TableHead>Brick Type</TableHead>
   796→                  <TableHead className="text-right">Quantity</TableHead>
   797→                  <TableHead>Truck No.</TableHead>
   798→                  <TableHead className="hidden md:table-cell">Driver</TableHead>
   799→                  <TableHead>Date</TableHead>
   800→                  <TableHead className="text-right">Actions</TableHead>
   801→                </TableRow>
   802→              </TableHeader>
   803→              <TableBody>
   804→                {loading ? (
   805→                  renderSkeletons()
   806→                ) : filteredDispatches.length === 0 ? (
   807→                  <TableRow>
   808→                    <TableCell colSpan={8} className="h-32 text-center text-muted-foreground">
   809→                      <div className="flex flex-col items-center gap-2">
   810→                        <Truck className="size-8 text-muted-foreground/40" />
   811→                        <span>No dispatch entries yet. Click &quot;Create Dispatch&quot; to get started.</span>
   812→                      </div>
   813→                    </TableCell>
   814→                  </TableRow>
   815→                ) : (
   816→                  filteredDispatches.map((dispatch) => (
   817→                    <TableRow key={dispatch.id}>
   818→                      <TableCell className="font-medium font-mono text-xs">
   819→                        {dispatch.dispatchNumber}
   820→                      </TableCell>
   821→                      <TableCell>
   822→                        <div>
   823→                          <span className="font-medium">{dispatch.customer?.name || '—'}</span>
   824→                          {dispatch.order && (
   825→                            <span className="block text-xs text-muted-foreground">
   826→                              Ref: {dispatch.order.orderNumber}
   827→                            </span>
   828→                          )}
   829→                        </div>
   830→                      </TableCell>
   831→                      <TableCell>{dispatch.brickType}</TableCell>
   832→                      <TableCell className="text-right font-mono">
   833→                        {dispatch.quantity.toLocaleString()}
   834→                      </TableCell>
   835→                      <TableCell>
   836→                        <Badge
   837→                          variant="outline"
   838→                          className="font-mono text-xs bg-amber-50 text-amber-700 border-amber-200"
   839→                        >
   840→                          {dispatch.truckNumber}
   841→                        </Badge>
   842→                      </TableCell>
   843→                      <TableCell className="hidden md:table-cell text-muted-foreground">
   844→                        {dispatch.driverName}
   845→                      </TableCell>
   846→                      <TableCell className="whitespace-nowrap">
   847→                        {formatDate(dispatch.date)}
   848→                      </TableCell>
   849→                      <TableCell className="text-right">
   850→                        <div className="flex items-center justify-end gap-1">
   851→                          <Button
   852→                            variant="ghost"
   853→                            size="icon"
   854→                            onClick={() => setChallanDispatch(dispatch)}
   855→                            title="View Challan"
   856→                            className="text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   857→                          >
   858→                            <Eye className="size-4" />
   859→                            <span className="sr-only">View Challan</span>
   860→                          </Button>
   861→                          <Button
   862→                            variant="ghost"
   863→                            size="icon"
   864→                            onClick={() => setDeleteTarget(dispatch)}
   865→                            title="Delete Dispatch"
   866→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   867→                          >
   868→                            <Trash2 className="size-4" />
   869→                            <span className="sr-only">Delete</span>
   870→                          </Button>
   871→                        </div>
   872→                      </TableCell>
   873→                    </TableRow>
   874→                  ))
   875→                )}
   876→              </TableBody>
   877→            </Table>
   878→          </div>
   879→        </CardContent>
   880→      </Card>
   881→
   882→      {/* Dialogs */}
   883→      {renderCreateDialog()}

   885→      {renderDeleteDialog()}
   886→
   887→      <ExcelImport module="dispatch" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchDispatches} />
   888→    </div>
   889→  )
   890→}
