     1→'use client'
     2→
     3→import * as React from 'react'
     4→import { api } from '@/lib/api'
     5→import { toast } from '@/hooks/use-toast'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Textarea } from '@/components/ui/textarea'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Skeleton } from '@/components/ui/skeleton'
    12→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    13→import {
    14→  Table,
    15→  TableBody,
    16→  TableCell,
    17→  TableHead,
    18→  TableHeader,
    19→  TableRow,
    20→} from '@/components/ui/table'
    21→import {
    22→  Dialog,
    23→  DialogContent,
    24→  DialogDescription,
    25→  DialogFooter,
    26→  DialogHeader,
    27→  DialogTitle,
    28→} from '@/components/ui/dialog'
    29→import {
    30→  AlertDialog,
    31→  AlertDialogAction,
    32→  AlertDialogCancel,
    33→  AlertDialogContent,
    34→  AlertDialogDescription,
    35→  AlertDialogFooter,
    36→  AlertDialogHeader,
    37→  AlertDialogTitle,
    38→} from '@/components/ui/alert-dialog'
    39→import { Truck, Plus, Trash2, Pencil, Loader2, IndianRupee, Upload , Search} from 'lucide-react'
    40→import ExcelImport from '@/components/erp/excel-import'
    41→import { AiFillButton } from '@/components/ui/ai-fill-button'
    42→import { FieldVoiceInput } from '@/components/ui/field-voice-input'
    43→import { consumePendingAiResult } from '@/components/ui/ai-chat-widget'
    44→import { isFormEmpty, showPleaseFillDataToast } from '@/lib/form-validation'
    45→import { Checkbox } from '@/components/ui/checkbox'
    46→
    47→interface TractorPayment {
    48→  id: string
    49→  date: string
    50→  vendorName: string
    51→  quantityTon: number
    52→  rate: number
    53→  totalAmount: number
    54→  paidAmount: number
    55→  remainingAmount: number
    56→  remarks: string
    57→  createdAt: string
    58→  updatedAt: string
    59→}
    60→
    61→interface TractorPaymentFormData {
    62→  date: string
    63→  vendorName: string
    64→  quantityTon: string
    65→  rate: string
    66→  paidAmount: string
    67→  remarks: string
    68→}
    69→
    70→const formatCurrency = (amount: number): string =>
    71→  new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', minimumFractionDigits: 0, maximumFractionDigits: 2 }).format(amount)
    72→
    73→const formatDate = (dateStr: string): string =>
    74→  new Date(dateStr + 'T00:00:00').toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    75→
    76→const emptyForm: TractorPaymentFormData = { date: '', vendorName: '', quantityTon: '', rate: '', paidAmount: '', remarks: '' }
    77→
    78→export function TractorPaymentModule() {
    79→  const [items, setItems] = React.useState<TractorPayment[]>([])
    80→  const [search, setSearch] = React.useState('')
    81→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
    82→  const [loading, setLoading] = React.useState(true)
    83→  const [formOpen, setFormOpen] = React.useState(false)
    84→  const [editingItem, setEditingItem] = React.useState<TractorPayment | null>(null)
    85→  const [formData, setFormData] = React.useState<TractorPaymentFormData>(emptyForm)
    86→  const [formSubmitting, setFormSubmitting] = React.useState(false)
    87→  const [deleteTarget, setDeleteTarget] = React.useState<TractorPayment | null>(null)
    88→  const [deleting, setDeleting] = React.useState(false)
    89→
    90→  // ── Multi-select state ──────────────────────────────────────────────────
    91→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
    92→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
    93→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
    94→
    95→  const toggleSelect = (id: string) => {
    96→    setSelectedIds((prev) => {
    97→      const next = new Set(prev)
    98→      if (next.has(id)) next.delete(id)
    99→      else next.add(id)
   100→      return next
   101→    })
   102→  }
   103→
   104→  const toggleSelectAll = () => {
   105→    setSelectedIds((prev) => {
   106→      if (prev.size === filteredItems.length && filteredItems.length > 0) {
   107→        return new Set()
   108→      }
   109→      return new Set(filteredItems.map((i) => i.id))
   110→    })
   111→  }
   112→
   113→  const clearSelection = () => setSelectedIds(new Set())
   114→
   115→  const handleBulkDelete = async () => {
   116→    const ids = Array.from(selectedIds)
   117→    if (ids.length === 0) return
   118→    setBulkDeleting(true)
   119→    try {
   120→      const res = await api.bulkDeleteTractorPayments(ids)
   121→      toast({
   122→        title: 'Success',
   123→        description: `${res.deletedCount} of ${ids.length} tractor payment entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   124→      })
   125→      setBulkDeleteOpen(false)
   126→      clearSelection()
   127→      fetchData()
   128→    } catch (err) {
   129→      toast({
   130→        title: 'Error',
   131→        description: err instanceof Error ? err.message : 'Failed to delete selected entries',
   132→        variant: 'destructive',
   133→      })
   134→    } finally {
   135→      setBulkDeleting(false)
   136→    }
   137→  }
   138→
   139→  const fetchData = React.useCallback(async () => {
   140→    setLoading(true)
   141→    try {
   142→      const res = await api.getTractorPayments()
   143→      const data = (res.tractorPayments as TractorPayment[]).sort(
   144→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   145→      )
   146→      setItems(data)
   147→    } catch (err) {
   148→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to fetch data', variant: 'destructive' })
   149→    } finally { setLoading(false) }
   150→  }, [])
   151→
   152→  // Debounced search
   153→  React.useEffect(() => {
   154→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   155→    return () => clearTimeout(t)
   156→  }, [search])
   157→
   158→  // Client-side filter
   159→  const filteredItems = React.useMemo(() => {
   160→    if (!debouncedSearch.trim()) return items
   161→    const q = debouncedSearch.toLowerCase()
   162→    return items.filter((item: any) =>
   163→      ['date', 'vendorName', 'remarks'].some((f) =>
   164→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   165→      )
   166→    )
   167→  }, [items, debouncedSearch])
   168→
   169→  React.useEffect(() => { fetchData() }, [fetchData])
   170→
   171→  // Excel import
   172→
   173→  const [importOpen, setImportOpen] = React.useState(false)
   174→
   175→
   176→  const openAddDialog = () => {
   177→    setEditingItem(null)
   178→    setFormData(emptyForm)
   179→    const pending = consumePendingAiResult('tractorPayment')
   180→    if (pending) {
   181→      setFormData((prev) => ({
   182→        ...prev,
   183→        date: pending.date ? String(pending.date).slice(0, 10) : prev.date,
   184→        vendorName: pending.vendorName != null ? String(pending.vendorName) : prev.vendorName,
   185→        quantityTon: pending.quantityTon != null ? String(pending.quantityTon) : prev.quantityTon,
   186→        rate: pending.rate != null ? String(pending.rate) : prev.rate,
   187→        paidAmount: pending.paidAmount != null ? String(pending.paidAmount) : prev.paidAmount,
   188→        remarks: pending.remarks != null ? String(pending.remarks) : prev.remarks,
   189→      }))
   190→    }
   191→    setFormOpen(true)
   192→  }
   193→  const openEditDialog = (item: TractorPayment) => {
   194→    setEditingItem(item)
   195→    setFormData({
   196→      date: item.date ? item.date.split('T')[0] : '',
   197→      vendorName: item.vendorName || '',
   198→      quantityTon: String(item.quantityTon || ''),
   199→      rate: String(item.rate || ''),
   200→      paidAmount: String(item.paidAmount || ''),
   201→      remarks: item.remarks || '',
   202→    })
   203→    setFormOpen(true)
   204→  }
   205→
   206→  const handleFormChange = (field: keyof TractorPaymentFormData, value: string) => {
   207→    setFormData((prev) => ({ ...prev, [field]: value }))
   208→  }
   209→
   210→  // Auto-calculated values
   211→  const qty = Number(formData.quantityTon) || 0
   212→  const rate = Number(formData.rate) || 0
   213→  const totalAmount = qty * rate
   214→  const paidAmount = Number(formData.paidAmount) || 0
   215→  const remainingAmount = totalAmount - paidAmount
   216→
   217→  const handleSubmit = async () => {
   218→    // Unified empty-form check — show ONE popup instead of cascading errors
   219→    if (isFormEmpty([formData.date, formData.vendorName, formData.quantityTon, formData.rate, formData.paidAmount, formData.remarks])) {
   220→      toast(showPleaseFillDataToast())
   221→      return
   222→    }
   223→    if (!formData.date) { toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' }); return }
   224→    if (!formData.vendorName.trim()) { toast({ title: 'Validation Error', description: 'Vendor name is required', variant: 'destructive' }); return }
   225→    setFormSubmitting(true)
   226→    try {
   227→      const payload = {
   228→        date: formData.date,
   229→        vendorName: formData.vendorName.trim(),
   230→        quantityTon: qty,
   231→        rate,
   232→        totalAmount,
   233→        paidAmount,
   234→        remainingAmount,
   235→        remarks: formData.remarks.trim(),
   236→      }
   237→      if (editingItem) {
   238→        await api.updateTractorPayment(editingItem.id, payload)
   239→        toast({ title: 'Success', description: 'Tractor payment updated successfully' })
   240→      } else {
   241→        await api.createTractorPayment(payload)
   242→        toast({ title: 'Success', description: 'Tractor payment created successfully' })
   243→      }
   244→      setFormOpen(false); setFormData(emptyForm); setEditingItem(null); fetchData()
   245→    } catch (err) {
   246→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to save', variant: 'destructive' })
   247→    } finally { setFormSubmitting(false) }
   248→  }
   249→
   250→  const handleDelete = async () => {
   251→    if (!deleteTarget) return
   252→    setDeleting(true)
   253→    try {
   254→      await api.deleteTractorPayment(deleteTarget.id)
   255→      toast({ title: 'Success', description: 'Tractor payment deleted successfully' })
   256→      setDeleteTarget(null); fetchData()
   257→    } catch (err) {
   258→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to delete', variant: 'destructive' })
   259→    } finally { setDeleting(false) }
   260→  }
   261→
   262→  const grandTotal = items.reduce((sum, i) => sum + (i.totalAmount || 0), 0)
   263→  const grandPaid = items.reduce((sum, i) => sum + (i.paidAmount || 0), 0)
   264→  const grandRemaining = items.reduce((sum, i) => sum + (i.remainingAmount || 0), 0)
   265→
   266→  return (
   267→    <div className="space-y-6">
   268→      {/* Full-screen loading overlay during bulk delete */}
   269→      {bulkDeleting && (
   270→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   271→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   272→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   273→            <div className="text-center">
   274→              <p className="text-lg font-semibold">Deleting {selectedIds.size} entr{selectedIds.size === 1 ? 'y' : 'ies'}...</p>
   275→              <p className="text-sm text-muted-foreground mt-1">Please wait while records are removed.</p>
   276→            </div>
   277→          </div>
   278→        </div>
   279→      )}
   280→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   281→        <div className="flex items-center gap-3">
   282→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700"><Truck className="size-5" /></div>
   283→          <div>
   284→            <h2 className="text-2xl font-bold tracking-tight">Tractor Payment</h2>
   285→            <p className="text-sm text-muted-foreground">Track tractor vendor payments and dues</p>
   286→          </div>
   287→        </div>
   288→        <div className="flex flex-col gap-2 w-full sm:hidden">
   289→          <Button variant="outline" onClick={() => setImportOpen(true)} className="w-full"><Upload className="size-4 mr-2" />Import Excel</Button>
   290→          {selectedIds.size > 0 && (
   291→            <Button
   292→              variant="outline"
   293→              onClick={() => setBulkDeleteOpen(true)}
   294→              disabled={bulkDeleting || loading}
   295→              className="w-full text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   296→            >
   297→              <Trash2 className="size-4 mr-2" />Delete Selected
   298→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">{selectedIds.size}</Badge>
   299→            </Button>
   300→          )}
   301→          {selectedIds.size > 0 && (
   302→            <Button variant="ghost" onClick={clearSelection} disabled={bulkDeleting} className="w-full">Clear Selection</Button>
   303→          )}
   304→          <Button onClick={openAddDialog} className="bg-emerald-600 hover:bg-emerald-700 text-white w-full"><Plus className="size-4" />Add Payment</Button>
   305→        </div>
   306→        <div className="hidden sm:flex gap-2">
   307→          <Button variant="outline" onClick={() => setImportOpen(true)}><Upload className="size-4 mr-2" />Import Excel</Button>
   308→          {selectedIds.size > 0 && (
   309→            <Button
   310→              variant="outline"
   311→              onClick={() => setBulkDeleteOpen(true)}
   312→              disabled={bulkDeleting || loading}
   313→              className="text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   314→            >
   315→              <Trash2 className="size-4 mr-2" />Delete Selected
   316→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">{selectedIds.size}</Badge>
   317→            </Button>
   318→          )}
   319→          {selectedIds.size > 0 && (
   320→            <Button variant="ghost" onClick={clearSelection} disabled={bulkDeleting}>Clear Selection</Button>
   321→          )}
   322→          <Button onClick={openAddDialog} className="bg-emerald-600 hover:bg-emerald-700 text-white"><Plus className="size-4" />Add Payment</Button>
   323→        </div>
   324→      </div>
   325→
   326→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   327→        <Card className="border-l-4 border-l-emerald-500"><CardContent className="p-4"><p className="text-xs text-muted-foreground">Total Amount</p>{loading ? <Skeleton className="h-6 w-32 mt-1" /> : <p className="text-xl font-bold text-emerald-700">{formatCurrency(grandTotal)}</p>}</CardContent></Card>
   328→        <Card className="border-l-4 border-l-amber-500"><CardContent className="p-4"><p className="text-xs text-muted-foreground">Total Paid</p>{loading ? <Skeleton className="h-6 w-32 mt-1" /> : <p className="text-xl font-bold text-amber-700">{formatCurrency(grandPaid)}</p>}</CardContent></Card>
   329→        <Card className="border-l-4 border-l-rose-500"><CardContent className="p-4"><p className="text-xs text-muted-foreground">Total Remaining</p>{loading ? <Skeleton className="h-6 w-32 mt-1" /> : <p className="text-xl font-bold text-rose-700">{formatCurrency(grandRemaining)}</p>}</CardContent></Card>
   330→      </div>
   331→
   332→      {/* Search */}
   333→      <Card>
   334→        <CardContent className="pt-6">
   335→          <div className="relative">
   336→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   337→            <Input
   338→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   339→              value={search}
   340→              onChange={(e) => setSearch(e.target.value)}
   341→              className="pl-9"
   342→            />
   343→          </div>
   344→        </CardContent>
   345→      </Card>
   346→
   347→      {/* Desktop: Table view */}
   348→      <Card className="hidden sm:block">
   349→        <CardHeader><CardTitle className="flex items-center justify-between"><span>Tractor Payment Records</span><div className="flex items-center gap-2">{selectedIds.size > 0 && <Badge variant="secondary" className="bg-destructive/10 text-destructive border-destructive/30">{selectedIds.size} selected</Badge>}<Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">{filteredItems.length} of {items.length} record{items.length !== 1 ? 's' : ''}</Badge></div></CardTitle></CardHeader>
   350→        <CardContent>
   351→          <div className="max-h-[60vh] overflow-auto rounded-md border">
   352→            <Table>
   353→              <TableHeader className="sticky top-0 bg-background z-10"><TableRow><TableHead className="w-10"><Checkbox checked={filteredItems.length > 0 && selectedIds.size === filteredItems.length} onCheckedChange={toggleSelectAll} aria-label="Select all rows" /></TableHead><TableHead>Date</TableHead><TableHead>Vendor Name</TableHead><TableHead className="text-right">Qty (Ton)</TableHead><TableHead className="text-right">Rate</TableHead><TableHead className="text-right">Total (₹)</TableHead><TableHead className="text-right">Paid (₹)</TableHead><TableHead className="text-right">Remaining (₹)</TableHead><TableHead>Remarks</TableHead><TableHead className="text-right">Actions</TableHead></TableRow></TableHeader>
   354→              <TableBody>
   355→                {loading ? Array.from({ length: 5 }).map((_, i) => <TableRow key={i}><TableCell className="w-10"><Skeleton className="h-4 w-4" /></TableCell>{Array.from({ length: 9 }).map((_, j) => <TableCell key={j}><Skeleton className="h-4 w-20" /></TableCell>)}</TableRow>)
   356→                : filteredItems.length === 0 ? <TableRow><TableCell colSpan={10} className="h-32 text-center text-muted-foreground">No tractor payments yet. Click &quot;Add Payment&quot; to get started.</TableCell></TableRow>
   357→                : filteredItems.map((item) => (
   358→                  <TableRow key={item.id} data-state={selectedIds.has(item.id) ? 'selected' : undefined} className={selectedIds.has(item.id) ? 'bg-emerald-50/60 dark:bg-emerald-900/15' : ''}>
   359→                    <TableCell className="w-10"><Checkbox checked={selectedIds.has(item.id)} onCheckedChange={() => toggleSelect(item.id)} aria-label={`Select row for ${item.vendorName}`} /></TableCell>
   360→                    <TableCell className="font-medium whitespace-nowrap">{formatDate(item.date)}</TableCell>
   361→                    <TableCell className="font-medium">{item.vendorName}</TableCell>
   362→                    <TableCell className="text-right font-mono">{item.quantityTon}</TableCell>
   363→                    <TableCell className="text-right font-mono">{formatCurrency(item.rate)}</TableCell>
   364→                    <TableCell className="text-right font-medium whitespace-nowrap">{formatCurrency(item.totalAmount)}</TableCell>
   365→                    <TableCell className="text-right whitespace-nowrap text-emerald-700">{formatCurrency(item.paidAmount)}</TableCell>
   366→                    <TableCell className="text-right whitespace-nowrap font-medium text-rose-700">{formatCurrency(item.remainingAmount)}</TableCell>
   367→                    <TableCell className="max-w-[120px] truncate text-muted-foreground">{item.remarks || '—'}</TableCell>
   368→                    <TableCell className="text-right"><div className="flex items-center justify-end gap-1"><Button variant="ghost" size="icon" onClick={() => openEditDialog(item)} title="Edit"><Pencil className="size-4" /></Button><Button variant="ghost" size="icon" onClick={() => setDeleteTarget(item)} title="Delete" className="text-destructive hover:text-destructive hover:bg-destructive/10"><Trash2 className="size-4" /></Button></div></TableCell>
   369→                  </TableRow>
   370→                ))}
   371→              </TableBody>
   372→            </Table>
   373→          </div>
   374→        </CardContent>
   375→      </Card>
   376→
   377→      {/* Mobile: Card list view */}
   378→      <div className="sm:hidden space-y-3">
   379→        <div className="flex items-center justify-between px-1">
   380→          <div className="flex items-center gap-2">
   381→            <Checkbox
   382→              checked={filteredItems.length > 0 && selectedIds.size === filteredItems.length}
   383→              onCheckedChange={toggleSelectAll}
   384→              aria-label="Select all"
   385→            />
   386→            <h3 className="text-base font-semibold">Records</h3>
   387→          </div>
   388→          <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">{filteredItems.length} of {items.length}</Badge>
   389→        </div>
   390→        {loading ? Array.from({ length: 4 }).map((_, i) => <Card key={i}><CardContent className="p-4 space-y-2"><Skeleton className="h-4 w-3/4" /><Skeleton className="h-4 w-1/2" /><Skeleton className="h-4 w-2/3" /></CardContent></Card>)
   391→        : filteredItems.length === 0 ? <Card><CardContent className="p-8 text-center text-muted-foreground">No tractor payments yet. Tap &quot;Add Payment&quot; to get started.</CardContent></Card>
   392→        : filteredItems.map((item) => (
   393→          <Card key={item.id} data-state={selectedIds.has(item.id) ? 'selected' : undefined} className={selectedIds.has(item.id) ? 'border-destructive/40 bg-destructive/5' : ''}>
   394→            <CardContent className="p-4 space-y-2">
   395→              <div className="flex items-start justify-between gap-2">
   396→                <div className="flex items-start gap-2 min-w-0 flex-1">
   397→                  <Checkbox checked={selectedIds.has(item.id)} onCheckedChange={() => toggleSelect(item.id)} aria-label={`Select ${item.vendorName}`} className="mt-1" />
   398→                  <div className="min-w-0 flex-1">
   399→                    <p className="font-semibold truncate">{item.vendorName}</p>
   400→                    <p className="text-xs text-muted-foreground">{formatDate(item.date)}</p>
   401→                  </div>
   402→                </div>
   403→                <p className="font-bold whitespace-nowrap">{formatCurrency(item.totalAmount)}</p>
   404→              </div>
   405→              <div className="grid grid-cols-2 gap-2 text-sm">
   406→                <div><p className="text-xs text-muted-foreground">Qty (Ton)</p><p className="font-medium">{item.quantityTon}</p></div>
   407→                <div><p className="text-xs text-muted-foreground">Rate</p><p className="font-medium">{formatCurrency(item.rate)}</p></div>
   408→                <div><p className="text-xs text-muted-foreground">Paid</p><p className="font-medium text-emerald-700">{formatCurrency(item.paidAmount)}</p></div>
   409→                <div><p className="text-xs text-muted-foreground">Remaining</p><p className="font-medium text-rose-700">{formatCurrency(item.remainingAmount)}</p></div>
   410→              </div>
   411→              {item.remarks && <p className="text-sm text-muted-foreground truncate"><span className="font-medium text-foreground">Remarks:</span> {item.remarks}</p>}
   412→              <div className="flex items-center justify-end gap-2 pt-2 border-t">
   413→                <Button variant="outline" size="sm" onClick={() => openEditDialog(item)} className="h-8"><Pencil className="size-3.5 mr-1" />Edit</Button>
   414→                <Button variant="outline" size="sm" onClick={() => setDeleteTarget(item)} className="h-8 text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"><Trash2 className="size-3.5 mr-1" />Delete</Button>
   415→              </div>
   416→            </CardContent>
   417→          </Card>
   418→        ))}
   419→      </div>
   420→
   421→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   422→        <DialogContent className="sm:max-w-md">
   423→          <DialogHeader><DialogTitle className="text-base sm:text-lg">{editingItem ? 'Edit Tractor Payment' : 'Add Tractor Payment'}</DialogTitle><DialogDescription className="text-xs sm:text-sm">{editingItem ? 'Update the payment details.' : 'Fill in the details to create a new tractor payment.'}</DialogDescription></DialogHeader>
   424→          <div className="grid gap-3 sm:gap-4">
   425→            <div className="grid gap-1.5 sm:gap-2"><Label htmlFor="tp-date" className="text-xs sm:text-sm">Date <span className="text-destructive">*</span></Label><Input id="tp-date" type="date" value={formData.date} onChange={(e) => handleFormChange('date', e.target.value)} className="h-10" /></div>
   426→            {!editingItem && (
   427→              <div className="flex justify-end -mt-1">
   428→                <AiFillButton module="tractorPayment" onApply={(fields) => setFormData((prev) => ({
   429→                  ...prev,
   430→                  date: fields.date ? String(fields.date).slice(0, 10) : prev.date,
   431→                  vendorName: fields.vendorName != null ? String(fields.vendorName) : prev.vendorName,
   432→                  quantityTon: fields.quantityTon != null ? String(fields.quantityTon) : prev.quantityTon,
   433→                  rate: fields.rate != null ? String(fields.rate) : prev.rate,
   434→                  paidAmount: fields.paidAmount != null ? String(fields.paidAmount) : prev.paidAmount,
   435→                  remarks: fields.remarks != null ? String(fields.remarks) : prev.remarks,
   436→                }))} />
   437→              </div>
   438→            )}
   439→            <div className="grid gap-1.5 sm:gap-2"><Label htmlFor="tp-vendor" className="text-xs sm:text-sm">Vendor Name <span className="text-destructive">*</span></Label>
   440→              <div className="relative">
   441→                <Input id="tp-vendor" placeholder="Enter vendor name" value={formData.vendorName} onChange={(e) => handleFormChange('vendorName', e.target.value)} className="pr-10 h-10" />
   442→                <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   443→                  <FieldVoiceInput fieldLabel="vendor name" onChange={(text) => handleFormChange('vendorName', text)} />
   444→                </div>
   445→              </div>
   446→            </div>
   447→            <div className="grid grid-cols-2 gap-3 sm:gap-4">
   448→              <div className="grid gap-1.5 sm:gap-2"><Label htmlFor="tp-qty" className="text-xs sm:text-sm">Quantity (Ton)</Label><Input id="tp-qty" type="number" min="0" step="0.01" placeholder="0" value={formData.quantityTon} onChange={(e) => handleFormChange('quantityTon', e.target.value)} className="h-10" /></div>
   449→              <div className="grid gap-1.5 sm:gap-2"><Label htmlFor="tp-rate" className="text-xs sm:text-sm">Rate</Label><div className="relative"><IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" /><Input id="tp-rate" type="number" min="0" placeholder="0" className="pl-9 h-10" value={formData.rate} onChange={(e) => handleFormChange('rate', e.target.value)} /></div></div>
   450→            </div>
   451→            <div className="grid grid-cols-2 gap-3 sm:gap-4">
   452→              <div className="grid gap-1.5 sm:gap-2"><Label className="text-xs sm:text-sm">Total Amount (₹)</Label><div className="h-10 flex items-center px-3 rounded-md border bg-muted/50 font-medium text-sm">{formatCurrency(totalAmount)}</div></div>
   453→              <div className="grid gap-1.5 sm:gap-2"><Label className="text-xs sm:text-sm">Remaining (₹)</Label><div className={`h-10 flex items-center px-3 rounded-md border font-medium text-sm ${remainingAmount > 0 ? 'bg-rose-50 text-rose-700 border-rose-200' : 'bg-emerald-50 text-emerald-700 border-emerald-200'}`}>{formatCurrency(remainingAmount)}</div></div>
   454→            </div>
   455→            <div className="grid gap-1.5 sm:gap-2"><Label htmlFor="tp-paid" className="text-xs sm:text-sm">Paid Amount (₹)</Label><div className="relative"><IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" /><Input id="tp-paid" type="number" min="0" placeholder="0" className="pl-9 h-10" value={formData.paidAmount} onChange={(e) => handleFormChange('paidAmount', e.target.value)} /></div></div>
   456→            <div className="grid gap-1.5 sm:gap-2"><Label htmlFor="tp-remarks" className="text-xs sm:text-sm">Remarks</Label>
   457→              <div className="relative">
   458→                <Textarea id="tp-remarks" placeholder="Optional remarks..." value={formData.remarks} onChange={(e) => handleFormChange('remarks', e.target.value)} className="min-h-[72px] sm:min-h-[80px] pr-10 text-sm" />
   459→                <div className="absolute right-1.5 top-2">
   460→                  <FieldVoiceInput fieldLabel="remarks" onChange={(text) => handleFormChange('remarks', text)} />
   461→                </div>
   462→              </div>
   463→            </div>
   464→          </div>
   465→          <DialogFooter className="pt-2"><Button variant="outline" onClick={() => setFormOpen(false)} disabled={formSubmitting} className="h-10">Cancel</Button><Button onClick={handleSubmit} disabled={formSubmitting} className="bg-emerald-600 hover:bg-emerald-700 text-white h-10">{formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}{editingItem ? 'Update' : 'Create'}</Button></DialogFooter>
   466→        </DialogContent>
   467→      </Dialog>
   468→
   469→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   470→        <AlertDialogContent><AlertDialogHeader><AlertDialogTitle>Delete Tractor Payment</AlertDialogTitle><AlertDialogDescription>Are you sure you want to delete this payment entry? This action cannot be undone.</AlertDialogDescription></AlertDialogHeader><AlertDialogFooter><AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel><AlertDialogAction onClick={handleDelete} disabled={deleting} className="bg-destructive text-white hover:bg-destructive/90">{deleting && <Loader2 className="mr-2 size-4 animate-spin" />}Delete</AlertDialogAction></AlertDialogFooter></AlertDialogContent>
   471→      </AlertDialog>
   472→
   473→      {/* Bulk Delete Selected confirmation */}
   474→      <AlertDialog open={bulkDeleteOpen} onOpenChange={(open) => { if (!open && !bulkDeleting) setBulkDeleteOpen(false) }}>
   475→        <AlertDialogContent>
   476→          <AlertDialogHeader>
   477→            <AlertDialogTitle className="text-destructive">Delete {selectedIds.size} Selected Tractor Payment {selectedIds.size === 1 ? 'Entry' : 'Entries'}?</AlertDialogTitle>
   478→            <AlertDialogDescription className="space-y-2 text-sm">
   479→              <span className="block">You are about to permanently delete <strong className="text-destructive">{selectedIds.size} tractor payment {selectedIds.size === 1 ? 'entry' : 'entries'}</strong>. This action <strong>cannot be undone</strong>.</span>
   480→            </AlertDialogDescription>
   481→          </AlertDialogHeader>
   482→          <AlertDialogFooter>
   483→            <AlertDialogCancel disabled={bulkDeleting}>Cancel</AlertDialogCancel>
   484→            <AlertDialogAction onClick={handleBulkDelete} disabled={bulkDeleting || selectedIds.size === 0} className="bg-destructive text-white hover:bg-destructive/90">{bulkDeleting && <Loader2 className="mr-2 size-4 animate-spin" />}Delete {selectedIds.size} {selectedIds.size === 1 ? 'Entry' : 'Entries'}</AlertDialogAction>
   485→          </AlertDialogFooter>
   486→        </AlertDialogContent>
   487→      </AlertDialog>
   488→
   489→      <ExcelImport module="tractorPayment" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchData} />
   490→    </div>
   491→  )
   492→}
