    12→// ─── Duplicate detection ────────────────────────────────────────────────────
    13→//
    14→// Per user request: "jo duplicate ho use skip kre or uska mess de but jo
    15→// duplicate nhi h usko import kre record me" — when an Excel row matches an
    16→// existing record in the DB (by a natural key), SKIP it with a clear message
    17→// but still IMPORT every non-duplicate row.
    18→//
    19→// To keep this fast, for each module we run ONE batched `find()` that
    20→// returns every existing row whose date falls within the dates seen in
    21→// the import payload, then build a Set of natural-key strings. Per-row
    22→// lookup then becomes an O(1) Set.has() check instead of a findOne().
    23→//
    24→// APPEND-ONLY semantics are preserved — we never wipe existing rows.
    25→
    26→// Build a stable natural-key string for a row, per module.
    27→function rowKey(module: string, row: Record<string, unknown>): string {
    28→  const s = (v: unknown) => String(v ?? '').trim().toLowerCase()
    29→  switch (module) {
    30→    case 'customers':
    31→      return s(row.mobile)
    32→    case 'production':
    33→      return `${s(row.date)}|${s(row.customerName || row.customer)}|${s(row.address)}`
    34→    case 'stock':
    35→      // One stock entry per date — same date = duplicate
    36→      return s(row.date)
    37→    case 'dailySell':
    38→      return `${s(row.date)}|${s(row.customerName)}|${s(row.amount)}`
    39→    case 'customerPayment':
    40→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
    41→    case 'labourPayment':
    42→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
    43→    case 'tractorPayment':
    44→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantityTon)}|${s(row.rate)}`
    45→    case 'dustPurchase':
    46→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantity)}|${s(row.rate)}`
    47→    case 'cementPurchase':
    48→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.itemName)}|${s(row.quantity)}|${s(row.rate)}`
    49→    case 'hardner':
    50→      // Only date+amount available — same date AND same amount = duplicate
    51→      return `${s(row.date)}|${s(row.amount)}`
    52→    case 'electricity':
    53→      return `${s(row.date)}|${s(row.name)}|${s(row.work)}|${s(row.amount)}`
    54→    case 'factoryStuff':
    55→      return `${s(row.date)}|${s(row.itemName)}|${s(row.amount)}`
    56→    case 'orders':
    57→      return `${s(row.customerId)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.rate)}|${s(row.deliveryDate || row.date)}`
    58→    case 'dispatch':
    59→      return `${s(row.customerId)}|${s(row.truckNumber)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.date)}`
    60→    case 'payments':
    61→      return `${s(row.customerId)}|${s(row.paymentType)}|${s(row.amount)}|${s(row.date)}`
    62→    case 'expenses':
    63→      return `${s(row.category)}|${s(row.amount)}|${s(row.date)}|${s(row.description)}`
    64→    default:
    65→      return ''
    66→  }
    67→}

   110→// Pre-fetch existing DB records that could collide with any row in `data`.
   111→// Returns a Set of natural-key strings.
   112→async function buildExistingKeys(module: string, data: Record<string, unknown>[]): Promise<Set<string>> {
   113→  const keys = new Set<string>()
   114→  if (data.length === 0) return keys
   115→
   116→  // Collect unique dates and customerIds from the import payload so we can
   117→  // narrow the DB query. This keeps the result set small.
   118→  const dates = new Set<string>()
   119→  const mobiles = new Set<string>()
   120→  const customerIds = new Set<string>()
   121→  for (const row of data) {
   122→    const d = String(row.date ?? '').split('T')[0]
   123→    if (d) dates.add(d)
   124→    const dd = String(row.deliveryDate ?? '').split('T')[0]
   125→    if (dd) dates.add(dd)
   126→    const m = String(row.mobile ?? '').trim()
   127→    if (m) mobiles.add(m)
   128→    const cid = String(row.customerId ?? '').trim()
   129→    if (cid) customerIds.add(cid)
   130→  }
   131→
   132→  let docs: Record<string, unknown>[] = []
   133→
   134→  switch (module) {
   135→    case 'customers':
   136→      docs = mobiles.size > 0
   137→        ? await Customer.find({ mobile: { $in: Array.from(mobiles) } }).lean() as any
   138→        : []
   139→      break
   140→    case 'production':
   141→      docs = dates.size > 0
   142→        ? await Production.find({ date: { $in: Array.from(dates) } }).lean() as any
   143→        : []
   144→      break
   145→    case 'stock':
   146→      docs = dates.size > 0
   147→        ? await Stock.find({ date: { $in: Array.from(dates) } }).lean() as any
   148→        : []
   149→      break
   150→    case 'dailySell':
   151→      docs = dates.size > 0
   152→        ? await DailySell.find({ date: { $in: Array.from(dates) } }).lean() as any
   153→        : []
   154→      break
   155→    case 'customerPayment':
   156→      docs = dates.size > 0
   157→        ? await CustomerPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   158→        : []
   159→      break
   160→    case 'labourPayment':
   161→      docs = dates.size > 0
   162→        ? await LabourPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   163→        : []
   164→      break
   165→    case 'tractorPayment':
   166→      docs = dates.size > 0
   167→        ? await TractorPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   168→        : []
   169→      break
   170→    case 'dustPurchase':
   171→      docs = dates.size > 0
   172→        ? await DustPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   173→        : []
   174→      break
   175→    case 'cementPurchase':
   176→      docs = dates.size > 0
   177→        ? await CementPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   178→        : []
   179→      break
   180→    case 'hardner':
   181→      docs = dates.size > 0
   182→        ? await Hardner.find({ date: { $in: Array.from(dates) } }).lean() as any
   183→        : []
   184→      break
   185→    case 'electricity':
   186→      docs = dates.size > 0
   187→        ? await Electricity.find({ date: { $in: Array.from(dates) } }).lean() as any
   188→        : []
   189→      break
   190→    case 'factoryStuff':
   191→      docs = dates.size > 0
   192→        ? await FactoryStuff.find({ date: { $in: Array.from(dates) } }).lean() as any
   193→        : []
   194→      break
   195→    case 'orders':
   196→      docs = customerIds.size > 0
   197→        ? await Order.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   198→        : []
   199→      break
   200→    case 'dispatch':
   201→      docs = customerIds.size > 0
   202→        ? await Dispatch.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   203→        : []
   204→      break
   205→    case 'payments':
   206→      docs = customerIds.size > 0
   207→        ? await Payment.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   208→        : []
   209→      break
   210→    case 'expenses':
   211→      // Expenses dedupe by category+amount+date — narrow by date
   212→      docs = dates.size > 0
   213→        ? await Expense.find({ date: { $in: Array.from(dates) } }).lean() as any
   214→        : []
   215→      break
   216→  }
   217→
   218→  for (const d of docs) {
   219→    const k = dbKey(module, d as Record<string, unknown>)
   220→    if (k) keys.add(k)
   221→  }
   222→  return keys
   223→}

   225→export async function POST(request: Request) {
   226→  try {
   227→    await connectDB()
   228→    const body = await request.json()
   229→    const { module, data } = body
   230→
   231→    if (!module || !data || !Array.isArray(data)) {
   232→      return NextResponse.json({ error: 'Module and data array are required' }, { status: 400 })
   233→    }
   234→
   235→    // ── Pre-fetch existing keys (1 DB query for the whole batch) ──────────
   236→    const existingKeys = await buildExistingKeys(module, data as Record<string, unknown>[])
   237→    // Track keys we've already seen WITHIN this batch so a duplicate Excel
   238→    // row doesn't get inserted twice in the same import run.
   239→    const seenInBatch = new Set<string>()
   240→
   241→    let imported = 0
   242→    let skipped = 0
   243→    let duplicatesSkipped = 0
   244→    const errors: string[] = []
   245→    const skippedReasons: string[] = []
   246→
   247→    for (let i = 0; i < data.length; i++) {
   248→      try {
   249→        const row = data[i]
   250→
   251→        // ── Duplicate check (against DB AND within this batch) ──────────
   252→        const key = rowKey(module, row as Record<string, unknown>)
   253→        if (key) {
   254→          if (existingKeys.has(key)) {
   255→            skipped++
   256→            duplicatesSkipped++
   257→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   258→            skippedReasons.push(`Row ${i + 1}: Duplicate — ${label} already exists in records, skipped`)
   259→            continue
   260→          }
   261→          if (seenInBatch.has(key)) {
   262→            skipped++
   263→            duplicatesSkipped++
   264→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   265→            skippedReasons.push(`Row ${i + 1}: Duplicate — ${label} appears more than once in this Excel file, only first occurrence imported`)
   266→            continue
   267→          }
   268→          seenInBatch.add(key)
   269→        }
   270→
   271→        switch (module) {
   272→          // ─── Customers ──────────────────────────────────────────────
   273→          case 'customers': {
   274→            if (!row.name || !row.mobile) {
   275→              errors.push(`Row ${i + 1}: Name and mobile are required`)
   276→              skipped++
   277→              continue
   278→            }
   279→            await Customer.create({
   280→              name: String(row.name).trim(),
   281→              mobile: String(row.mobile).trim(),
   282→              gstNumber: row.gstNumber || '',
   283→              address: row.address || '',
   284→              creditLimit: Number(row.creditLimit) || 0,
   285→            })
   286→            imported++
   287→            break
   288→          }
   289→
   290→          // ─── Production ─────────────────────────────────────────────
   291→          case 'production': {
   292→            if (!row.date) {
   293→              errors.push(`Row ${i + 1}: Date is required`)
   294→              skipped++
   295→              continue
   296→            }
   297→            await Production.create({
   298→              date: String(row.date),
   299→              customerName: String(row.customerName || row.customer || ''),
   300→              address: String(row.address || ''),
   301→              zigZagWhite80: Number(row.zigZagWhite80) || 0,
   302→              zigZagRed80: Number(row.zigZagRed80) || 0,
   303→              zigZagYellow80: Number(row.zigZagYellow80) || 0,
   304→              zigZagWhite60: Number(row.zigZagWhite60) || 0,
   305→              zigZagRed60: Number(row.zigZagRed60) || 0,
   306→              zigZagYellow60: Number(row.zigZagYellow60) || 0,
   307→              curveStone: Number(row.curveStone) || 0,
   308→              chequreTile: Number(row.chequreTile) || 0,
   309→              transportationCharge: Number(row.transportationCharge) || 0,
   310→              remarks: String(row.remarks || ''),
   311→            })
   312→            imported++
   313→            break
   314→          }
   315→
   316→          // ─── Stock ─────────────────────────────────────────────────
   317→          case 'stock': {
   318→            if (!row.date) {
   319→              errors.push(`Row ${i + 1}: Date is required`)
   320→              skipped++
   321→              continue
   322→            }
   323→            await Stock.create({
   324→              date: String(row.date),
   325→              cement: Number(row.cement) || 0,
   326→              zigZagGrey80: Number(row.zigZagGrey80) || 0,
   327→              zigZagRed80: Number(row.zigZagRed80) || 0,
   328→              zigZagYellow80: Number(row.zigZagYellow80) || 0,
   329→              zigZagGrey60: Number(row.zigZagGrey60) || 0,
   330→              zigZagRed60: Number(row.zigZagRed60) || 0,
   331→              zigZagYellow60: Number(row.zigZagYellow60) || 0,
   332→              chequreTile: Number(row.chequreTile) || 0,
   333→              curveStone: Number(row.curveStone) || 0,
   334→              dumbleGrey80: Number(row.dumbleGrey80) || 0,
   335→              dumbleRed80: Number(row.dumbleRed80) || 0,
   336→              dumbleYellow80: Number(row.dumbleYellow80) || 0,

   369→              date: String(row.date),

   380→          case 'labourPayment': {

   382→              errors.push(`Row ${i + 1}: Date, name, and amount are required`)

   385→            }

   387→              date: String(row.date),

   530→              errors.push(`Row ${i + 1}: Customer, brick type, quantity, and rate are required`)
   531→              skipped++
   532→              continue
   533→            }
   534→            const count = await Order.countDocuments({})
   535→            const { Company } = await import('@/lib/models')
   536→            const company = await Company.findOne({})
   537→            const prefix = company?.orderPrefix || 'ORD'
   538→            const orderNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   539→
   540→            await Order.create({
   541→              orderNumber,
   542→              customerId: row.customerId,
   543→              brickType: row.brickType,
   544→              quantity: Number(row.quantity),
   545→              rate: Number(row.rate),
   546→              amount: Number(row.amount) || Number(row.quantity) * Number(row.rate),
   547→              deliveryDate: row.deliveryDate || row.date || new Date().toISOString().split('T')[0],
   548→              status: row.status || 'Pending',
   549→            })
   550→            imported++
   551→            break
   552→          }
   553→
   554→          case 'dispatch': {
   555→            if (!row.customerId || !row.truckNumber || !row.quantity || !row.brickType || !row.date) {
   556→              errors.push(`Row ${i + 1}: Customer, truck, quantity, brick type, and date are required`)
   557→              skipped++
   558→              continue
   559→            }
   560→            const count = await Dispatch.countDocuments({})
   561→            const { Company } = await import('@/lib/models')
   562→            const company = await Company.findOne({})
   563→            const prefix = company?.dispatchPrefix || 'DSP'
   564→            const dispatchNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   565→
   566→            await Dispatch.create({
   567→              dispatchNumber,
   568→              customerId: row.customerId,
   569→              orderId: row.orderId || null,
   570→              truckNumber: row.truckNumber,
   571→              driverName: row.driverName || '',
   572→              quantity: Number(row.quantity),
   573→              brickType: row.brickType,
   574→              date: row.date,
   575→            })
   576→            imported++
   577→            break
   578→          }
   579→
   580→          case 'payments': {
   581→            if (!row.customerId || !row.paymentType || !row.amount || !row.date) {
   582→              errors.push(`Row ${i + 1}: Customer, payment type, amount, and date are required`)
   583→              skipped++
   584→              continue
   585→            }
   586→            await Payment.create({
   587→              customerId: row.customerId,
   588→              paymentType: row.paymentType,
   589→              amount: Number(row.amount),
   590→              date: row.date,
   591→              remarks: row.remarks || '',
   592→            })
   593→            imported++
   594→            break
   595→          }
   596→
   597→          case 'expenses': {
   598→            if (!row.category || !row.amount || !row.date) {
   599→              errors.push(`Row ${i + 1}: Category, amount, and date are required`)
   600→              skipped++
   601→              continue
   602→            }
   603→            await Expense.create({
   604→              category: row.category,
   605→              amount: Number(row.amount),
   606→              date: row.date,
   607→              description: row.description || '',
   608→            })
   609→            imported++
   610→            break
   611→          }
   612→
   613→          default:
   614→            errors.push(`Unknown module: ${module}`)
   615→        }
   616→      } catch (err) {
   617→        errors.push(`Row ${i + 1}: ${err instanceof Error ? err.message : 'Import failed'}`)
   618→        skipped++
   619→      }
   620→    }
   621→
   622→    // Merge errors and skippedReasons so the UI can show every reason a row
   623→    // was not imported (validation error OR duplicate OR anything else).
   624→    const allReasons = [...errors, ...skippedReasons]
   625→
   626→    return NextResponse.json({
   627→      success: true,
   628→      imported,
   629→      skipped,
   630→      duplicatesSkipped,
   631→      total: data.length,
   632→      errors: allReasons.length > 0 ? allReasons.slice(0, 50) : undefined,
   633→    })
   634→  } catch (error) {
   635→    console.error('Error importing data:', error)
   636→    return NextResponse.json({ error: 'Failed to import data' }, { status: 500 })
   637→  }
   638→}

   640→// Human-readable label for a duplicate row, used in the skip reason.
   641→function duplicateRowLabel(module: string, row: Record<string, unknown>): string {
   642→  switch (module) {
   643→    case 'customers':
   644→      return `customer "${row.name}" (mobile ${row.mobile})`
   645→    case 'production':
   646→      return `production entry on ${row.date}`
   647→    case 'stock':
   648→      return `stock entry for ${row.date}`
   649→    case 'dailySell':
   650→      return `daily sell for "${row.customerName}" on ${row.date} (₹${row.amount})`
   651→    case 'customerPayment':
   652→      return `payment by "${row.name}" on ${row.date} (₹${row.amount})`
   653→    case 'labourPayment':
   654→      return `payment to "${row.name}" on ${row.date} (₹${row.amount})`
   655→    case 'tractorPayment':
   656→      return `tractor payment to "${row.vendorName}" on ${row.date}`
   657→    case 'dustPurchase':
   658→      return `dust purchase from "${row.vendorName}" on ${row.date}`
   659→    case 'cementPurchase':
   660→      return `cement purchase from "${row.vendorName}" on ${row.date}`
   661→    case 'hardner':
   662→      return `hardner entry on ${row.date} (₹${row.amount})`
   663→    case 'electricity':
   664→      return `electricity entry on ${row.date} (₹${row.amount})`
   665→    case 'factoryStuff':
   666→      return `factory stuff "${row.itemName}" on ${row.date} (₹${row.amount})`
   667→    case 'orders':
   668→      return `order for customer ${row.customerId} on ${row.deliveryDate || row.date}`
   669→    case 'dispatch':
   670→      return `dispatch to customer ${row.customerId} on ${row.date}`
   671→    case 'payments':
   672→      return `payment for customer ${row.customerId} on ${row.date}`
   673→    case 'expenses':
   674→      return `expense "${row.category}" on ${row.date} (₹${row.amount})`
   675→    default:
   676→      return 'record'
   677→  }
   678→}
