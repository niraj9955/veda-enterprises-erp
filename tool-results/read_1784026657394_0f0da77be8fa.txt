   367→   367→// Format a 10-digit Indian mobile number as "+91 XXXXX XXXXX" for display.
   368→   368→// Falls back to the original string if it doesn't look like a 10-digit number.
   369→   369→function formatPhone(phone: string): string {
   370→   370→  const digits = phone.replace(/\D/g, '')
   371→   371→  // Handle "9572831213" or "919572831213" or "+919572831213"
   372→   372→  if (digits.length === 10) {
   373→   373→    return `+91 ${digits.slice(0, 5)} ${digits.slice(5)}`
   374→   374→  }
   375→   375→  if (digits.length === 12 && digits.startsWith('91')) {
   376→   376→    return `+91 ${digits.slice(2, 7)} ${digits.slice(7)}`
   377→   377→  }
   378→   378→  return phone
   379→   379→}
