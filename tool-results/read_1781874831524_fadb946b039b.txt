     1→'use client'
     2→
     3→import * as React from 'react'
     4→import { api } from '@/lib/api'
     5→import { toast } from '@/hooks/use-toast'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Badge } from '@/components/ui/badge'
    10→import { Skeleton } from '@/components/ui/skeleton'
    11→import { Textarea } from '@/components/ui/textarea'
    12→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    13→import {
    14→  Select,
    15→  SelectContent,
    16→  SelectItem,
    17→  SelectTrigger,
    18→  SelectValue,
    19→} from '@/components/ui/select'
    20→import {
    21→  Table,
    22→  TableBody,
    23→  TableCell,
    24→  TableHead,
    25→  TableHeader,
    26→  TableRow,
    27→} from '@/components/ui/table'
    28→import {
    29→  Dialog,
    30→  DialogContent,
    31→  DialogDescription,
    32→  DialogFooter,
    33→  DialogHeader,
    34→  DialogTitle,
    35→} from '@/components/ui/dialog'
    36→import {
    37→  AlertDialog,
    38→  AlertDialogAction,
    39→  AlertDialogCancel,
    40→  AlertDialogContent,
    41→  AlertDialogDescription,
    42→  AlertDialogFooter,
    43→  AlertDialogHeader,
    44→  AlertDialogTitle,
    45→} from '@/components/ui/alert-dialog'
    46→import {
    47→  Wallet,
    48→  Plus,
    49→  Pencil,
    50→  Trash2,
    51→  Loader2,
    52→  IndianRupee,
    53→  Banknote,
    54→  Smartphone,
    55→  Landmark,
    56→  Users,
    57→  Upload,
    58→  Search,
    59→} from 'lucide-react'
    60→import ExcelImport from '@/components/erp/excel-import'
    61→import CustomerSearchInput from '@/components/erp/customer-search-input'
    62→
    63→// ── Types ───────────────────────────────────────────────────────────────────
    64→
    65→interface Customer {
    66→  id: string
    67→  name: string
    68→}
    69→
    70→interface Payment {
    71→  id: string
    72→  customerId: string
    73→  paymentType: string
    74→  amount: number
    75→  date: string
    76→  remarks: string
    77→  createdAt: string
    78→  billId?: string | null
    79→  billNumber?: string
    80→  customer: { id: string; name: string }
    81→}
    82→
    83→interface Order {
    84→  id: string
    85→  customerId: string
    86→  amount: number
    87→  status: string
    88→  createdAt: string
    89→  customer: { id: string; name: string }
    90→}
    91→
    92→interface PaymentFormData {
    93→  customerId: string
    94→  paymentType: string
    95→  amount: number | string
    96→  date: string
    97→  remarks: string
    98→}
    99→
   100→interface OutstandingEntry {
   101→  customerId: string
   102→  customerName: string
   103→  totalOrderAmount: number
   104→  totalPayments: number
   105→  outstanding: number
   106→}
   107→
   108→// ── Constants ───────────────────────────────────────────────────────────────
   109→
   110→const PAYMENT_TYPES = ['Cash', 'UPI', 'Bank Transfer'] as const
   111→
   112→const PAYMENT_TYPE_COLORS: Record<string, string> = {
   113→  Cash: 'bg-emerald-100 text-emerald-700 border-emerald-200',
   114→  UPI: 'bg-sky-100 text-sky-700 border-sky-200',
   115→  'Bank Transfer': 'bg-amber-100 text-amber-700 border-amber-200',
   116→}
   117→
   118→const PAYMENT_TYPE_ICONS: Record<string, React.ReactNode> = {
   119→  Cash: <Banknote className="size-3" />,
   120→  UPI: <Smartphone className="size-3" />,
   121→  'Bank Transfer': <Landmark className="size-3" />,
   122→}
   123→
   124→// ── Helpers ─────────────────────────────────────────────────────────────────
   125→
   126→const formatCurrency = (amount: number): string =>
   127→  new Intl.NumberFormat('en-IN', {
   128→    style: 'currency',
   129→    currency: 'INR',
   130→    minimumFractionDigits: 0,
   131→    maximumFractionDigits: 2,
   132→  }).format(amount)
   133→
   134→const formatDate = (dateStr: string): string =>
   135→  new Date(dateStr).toLocaleDateString('en-IN', {
   136→    day: '2-digit',
   137→    month: 'short',
   138→    year: 'numeric',
   139→  })
   140→
   141→const emptyForm: PaymentFormData = {
   142→  customerId: '',
   143→  paymentType: '',
   144→  amount: '',
   145→  date: '',
   146→  remarks: '',
   147→}
   148→
   149→// ── Component ───────────────────────────────────────────────────────────────
   150→
   151→export function PaymentModule() {
   152→  // State
   153→  const [payments, setPayments] = React.useState<Payment[]>([])
   154→  const [search, setSearch] = React.useState('')
   155→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   156→  const [customers, setCustomers] = React.useState<Customer[]>([])
   157→  const [orders, setOrders] = React.useState<Order[]>([])
   158→  const [loading, setLoading] = React.useState(true)
   159→
   160→  // Dialog states
   161→  const [dialogOpen, setDialogOpen] = React.useState(false)
   162→  const [editingPayment, setEditingPayment] = React.useState<Payment | null>(null)
   163→  const [formData, setFormData] = React.useState<PaymentFormData>(emptyForm)
   164→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   165→
   166→  // Excel import
   167→  const [importOpen, setImportOpen] = React.useState(false)
   168→
   169→  // Delete confirmation
   170→  const [deleteTarget, setDeleteTarget] = React.useState<Payment | null>(null)
   171→  const [deleting, setDeleting] = React.useState(false)
   172→
   173→  // ── Computed: Summary values ─────────────────────────────────────────────
   174→  const totalReceived = React.useMemo(
   175→    () => payments.reduce((sum, p) => sum + Number(p.amount), 0),
   176→    [payments]
   177→  )
   178→
   179→  const cashPayments = React.useMemo(
   180→    () =>
   181→      payments
   182→        .filter((p) => p.paymentType === 'Cash')
   183→        .reduce((sum, p) => sum + Number(p.amount), 0),
   184→    [payments]
   185→  )
   186→
   187→  const upiBankPayments = React.useMemo(
   188→    () =>
   189→      payments
   190→        .filter((p) => p.paymentType === 'UPI' || p.paymentType === 'Bank Transfer')
   191→        .reduce((sum, p) => sum + Number(p.amount), 0),
   192→    [payments]
   193→  )
   194→
   195→  // ── Computed: Customer outstanding ───────────────────────────────────────
   196→  const outstandingEntries = React.useMemo<OutstandingEntry[]>(() => {
   197→    const map = new Map<string, OutstandingEntry>()
   198→
   199→    // Sum order amounts per customer (exclude Cancelled)
   200→    for (const order of orders) {
   201→      if (order.status === 'Cancelled') continue
   202→      const existing = map.get(order.customerId)
   203→      if (existing) {
   204→        existing.totalOrderAmount += Number(order.amount)
   205→      } else {
   206→        map.set(order.customerId, {
   207→          customerId: order.customerId,
   208→          customerName: order.customer?.name || 'Unknown',
   209→          totalOrderAmount: Number(order.amount),
   210→          totalPayments: 0,
   211→          outstanding: 0,
   212→        })
   213→      }
   214→    }
   215→
   216→    // Sum payments per customer
   217→    for (const payment of payments) {
   218→      const existing = map.get(payment.customerId)
   219→      if (existing) {
   220→        existing.totalPayments += Number(payment.amount)
   221→      } else {
   222→        map.set(payment.customerId, {
   223→          customerId: payment.customerId,
   224→          customerName: payment.customer?.name || 'Unknown',
   225→          totalOrderAmount: 0,
   226→          totalPayments: Number(payment.amount),
   227→          outstanding: 0,
   228→        })
   229→      }
   230→    }
   231→
   232→    // Calculate outstanding
   233→    for (const entry of map.values()) {
   234→      entry.outstanding = entry.totalOrderAmount - entry.totalPayments
   235→    }
   236→
   237→    // Return only customers with outstanding > 0, sorted by outstanding descending
   238→    return Array.from(map.values())
   239→      .filter((e) => e.outstanding > 0.01)
   240→      .sort((a, b) => b.outstanding - a.outstanding)
   241→  }, [orders, payments])
   242→
   243→  // ── Fetch data ───────────────────────────────────────────────────────────
   244→  const fetchPayments = React.useCallback(async () => {
   245→    setLoading(true)
   246→    try {
   247→      const res = await api.getPayments()
   248→      const paymentList = (res.payments as Payment[]).sort(
   249→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   250→      )
   251→      setPayments(paymentList)
   252→    } catch (err) {
   253→      toast({
   254→        title: 'Error',
   255→        description: err instanceof Error ? err.message : 'Failed to fetch payments',
   256→        variant: 'destructive',
   257→      })
   258→    } finally {
   259→      setLoading(false)
   260→    }
   261→  }, [])
   262→
   263→  const fetchCustomers = React.useCallback(async () => {
   264→    try {
   265→      const res = await api.getCustomers()
   266→      setCustomers(res.customers as Customer[])
   267→    } catch {
   268→      // Silently fail — customer dropdown will just be empty
   269→    }
   270→  }, [])
   271→
   272→  const fetchOrders = React.useCallback(async () => {
   273→    try {
   274→      const res = await api.getOrders()
   275→      setOrders(res.orders as Order[])
   276→    } catch {
   277→      // Silently fail — outstanding section will just be empty
   278→    }
   279→  }, [])
   280→
   281→  // Debounced search
   282→  React.useEffect(() => {
   283→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   284→    return () => clearTimeout(t)
   285→  }, [search])
   286→
   287→  // Client-side filter
   288→  const filteredPayments = React.useMemo(() => {
   289→    if (!debouncedSearch.trim()) return payments
   290→    const q = debouncedSearch.toLowerCase()
   291→    return payments.filter((item: any) =>
   292→      ['date', 'paymentType', 'remarks'].some((f) =>
   293→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   294→      )
   295→    )
   296→  }, [payments, debouncedSearch])
   297→
   298→  React.useEffect(() => {
   299→    fetchPayments()
   300→    fetchCustomers()
   301→    fetchOrders()
   302→  }, [fetchPayments, fetchCustomers, fetchOrders])
   303→
   304→  // ── Form handlers ────────────────────────────────────────────────────────
   305→  const openCreateDialog = () => {
   306→    setEditingPayment(null)
   307→    setFormData(emptyForm)
   308→    setDialogOpen(true)
   309→  }
   310→
   311→  const openEditDialog = (payment: Payment) => {
   312→    setEditingPayment(payment)
   313→    setFormData({
   314→      customerId: payment.customerId,
   315→      paymentType: payment.paymentType,
   316→      amount: payment.amount,
   317→      date: payment.date,
   318→      remarks: payment.remarks,
   319→    })
   320→    setDialogOpen(true)
   321→  }
   322→
   323→  const handleFormChange = (field: keyof PaymentFormData, value: string) => {
   324→    setFormData((prev) => ({ ...prev, [field]: value }))
   325→  }
   326→
   327→  const handleSubmit = async () => {
   328→    // Validation
   329→    if (!formData.customerId) {
   330→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   331→      return
   332→    }
   333→    if (!formData.paymentType) {
   334→      toast({ title: 'Validation Error', description: 'Please select a payment type', variant: 'destructive' })
   335→      return
   336→    }
   337→    if (!formData.amount || Number(formData.amount) <= 0) {
   338→      toast({ title: 'Validation Error', description: 'Amount must be greater than 0', variant: 'destructive' })
   339→      return
   340→    }
   341→    if (!formData.date) {
   342→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   343→      return
   344→    }
   345→
   346→    setFormSubmitting(true)
   347→    try {
   348→      const payload = {
   349→        customerId: formData.customerId,
   350→        paymentType: formData.paymentType,
   351→        amount: Number(formData.amount),
   352→        date: formData.date,
   353→        remarks: formData.remarks,
   354→      }
   355→
   356→      if (editingPayment) {
   357→        await api.updatePayment(editingPayment.id, payload)
   358→        toast({ title: 'Success', description: 'Payment updated successfully' })
   359→      } else {
   360→        await api.createPayment(payload)
   361→        toast({ title: 'Success', description: 'Payment recorded successfully' })
   362→      }
   363→
   364→      setDialogOpen(false)
   365→      setEditingPayment(null)
   366→      setFormData(emptyForm)
   367→      fetchPayments()
   368→      fetchOrders() // Refresh for outstanding calculations
   369→    } catch (err) {
   370→      toast({
   371→        title: 'Error',
   372→        description: err instanceof Error ? err.message : `Failed to ${editingPayment ? 'update' : 'record'} payment`,
   373→        variant: 'destructive',
   374→      })
   375→    } finally {
   376→      setFormSubmitting(false)
   377→    }
   378→  }
   379→
   380→  // ── Delete handler ──────────────────────────────────────────────────────
   381→  const handleDelete = async () => {
   382→    if (!deleteTarget) return
   383→    setDeleting(true)
   384→    try {
   385→      await api.deletePayment(deleteTarget.id)
   386→      toast({ title: 'Success', description: 'Payment deleted successfully' })
   387→      setDeleteTarget(null)
   388→      fetchPayments()
   389→      fetchOrders()
   390→    } catch (err) {
   391→      toast({
   392→        title: 'Error',
   393→        description: err instanceof Error ? err.message : 'Failed to delete payment',
   394→        variant: 'destructive',
   395→      })
   396→    } finally {
   397→      setDeleting(false)
   398→    }
   399→  }
   400→
   401→  // ── Render: Loading skeletons ───────────────────────────────────────────
   402→  const renderSkeletons = () =>
   403→    Array.from({ length: 5 }).map((_, i) => (
   404→      <TableRow key={i}>
   405→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   406→        <TableCell><Skeleton className="h-5 w-16" /></TableCell>
   407→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   408→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   409→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   410→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   411→      </TableRow>
   412→    ))
   413→
   414→  // ── Render: Payment type badge ──────────────────────────────────────────
   415→  const renderPaymentTypeBadge = (type: string) => (
   416→    <Badge className={`${PAYMENT_TYPE_COLORS[type] || 'bg-gray-100 text-gray-700 border-gray-200'} gap-1`}>
   417→      {PAYMENT_TYPE_ICONS[type]}
   418→      {type}
   419→    </Badge>
   420→  )
   421→
   422→  // ── Render: Payment dialog ──────────────────────────────────────────────
   423→  const renderPaymentDialog = () => (
   424→    <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   425→      <DialogContent className="sm:max-w-lg">
   426→        <DialogHeader>
   427→          <DialogTitle>
   428→            {editingPayment ? 'Edit Payment' : 'Receive Payment'}
   429→          </DialogTitle>
   430→          <DialogDescription>
   431→            {editingPayment
   432→              ? 'Update the payment details below.'
   433→              : 'Fill in the details to record a new payment.'}
   434→          </DialogDescription>
   435→        </DialogHeader>
   436→
   437→        <div className="grid gap-4 py-2">
   438→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   439→          <CustomerSearchInput
   440→            value={formData.customerId}
   441→            onSelect={(c) => handleFormChange('customerId', c.id)}
   442→            onClear={() => handleFormChange('customerId', '')}
   443→            required
   444→            label="Customer"
   445→            placeholder="Type customer name or mobile to search..."
   446→          />
   447→
   448→          {/* Payment Type */}
   449→          <div className="grid gap-2">
   450→            <Label htmlFor="payment-type">
   451→              Payment Type <span className="text-destructive">*</span>
   452→            </Label>
   453→            <Select
   454→              value={formData.paymentType}
   455→              onValueChange={(val) => handleFormChange('paymentType', val)}
   456→            >
   457→              <SelectTrigger className="w-full">
   458→                <SelectValue placeholder="Select payment type" />
   459→              </SelectTrigger>
   460→              <SelectContent>
   461→                {PAYMENT_TYPES.map((type) => (
   462→                  <SelectItem key={type} value={type}>
   463→                    {type}
   464→                  </SelectItem>
   465→                ))}
   466→              </SelectContent>
   467→            </Select>
   468→          </div>
   469→
   470→          {/* Amount */}
   471→          <div className="grid gap-2">
   472→            <Label htmlFor="payment-amount">
   473→              Amount (₹) <span className="text-destructive">*</span>
   474→            </Label>
   475→            <div className="relative">
   476→              <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   477→              <Input
   478→                id="payment-amount"
   479→                type="number"
   480→                placeholder="0"
   481→                min="1"
   482→                className="pl-9"
   483→                value={formData.amount}
   484→                onChange={(e) => handleFormChange('amount', e.target.value)}
   485→              />
   486→            </div>
   487→          </div>
   488→
   489→          {/* Date */}
   490→          <div className="grid gap-2">
   491→            <Label htmlFor="payment-date">
   492→              Date <span className="text-destructive">*</span>
   493→            </Label>
   494→            <Input
   495→              id="payment-date"
   496→              type="date"
   497→              value={formData.date}
   498→              onChange={(e) => handleFormChange('date', e.target.value)}
   499→            />
   500→          </div>
   501→
   502→          {/* Remarks */}
   503→          <div className="grid gap-2">
   504→            <Label htmlFor="payment-remarks">Remarks</Label>
   505→            <Textarea
   506→              id="payment-remarks"
   507→              placeholder="Optional remarks about this payment..."
   508→              value={formData.remarks}
   509→              onChange={(e) => handleFormChange('remarks', e.target.value)}
   510→              className="min-h-[80px] resize-none"
   511→            />
   512→          </div>
   513→        </div>
   514→
   515→        <DialogFooter>
   516→          <Button
   517→            variant="outline"
   518→            onClick={() => setDialogOpen(false)}
   519→            disabled={formSubmitting}
   520→          >
   521→            Cancel
   522→          </Button>
   523→          <Button
   524→            onClick={handleSubmit}
   525→            disabled={formSubmitting}
   526→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   527→          >
   528→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   529→            {editingPayment ? 'Update Payment' : 'Record Payment'}
   530→          </Button>
   531→        </DialogFooter>
   532→      </DialogContent>
   533→    </Dialog>
   534→  )
   535→
   536→  // ── Render: Delete confirmation ─────────────────────────────────────────
   537→  const renderDeleteDialog = () => (
   538→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   539→      <AlertDialogContent>
   540→        <AlertDialogHeader>
   541→          <AlertDialogTitle>Delete Payment</AlertDialogTitle>
   542→          <AlertDialogDescription>
   543→            Are you sure you want to delete this payment of{' '}
   544→            <strong>{deleteTarget ? formatCurrency(Number(deleteTarget.amount)) : ''}</strong>{' '}
   545→            from <strong>{deleteTarget?.customer?.name || 'this customer'}</strong>? This action
   546→            cannot be undone.
   547→          </AlertDialogDescription>
   548→        </AlertDialogHeader>
   549→        <AlertDialogFooter>
   550→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   551→          <AlertDialogAction
   552→            onClick={handleDelete}
   553→            disabled={deleting}
   554→            className="bg-destructive text-white hover:bg-destructive/90"
   555→          >
   556→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   557→            Delete
   558→          </AlertDialogAction>
   559→        </AlertDialogFooter>
   560→      </AlertDialogContent>
   561→    </AlertDialog>
   562→  )
   563→
   564→  // ── Render: Main ────────────────────────────────────────────────────────
   565→  return (
   566→    <div className="space-y-6">
   567→      {/* Header */}
   568→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   569→        <div className="flex items-center gap-3">
   570→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   571→            <Wallet className="size-5" />
   572→          </div>
   573→          <div>
   574→            <h2 className="text-2xl font-bold tracking-tight">Payment Management</h2>
   575→            <p className="text-sm text-muted-foreground">
   576→              Track and manage customer payments
   577→            </p>
   578→          </div>
   579→        </div>
   580→        <div className="flex gap-2 w-full sm:w-auto">
   581→          <Button
   582→            variant="outline"
   583→            onClick={() => setImportOpen(true)}
   584→            className="w-full sm:w-auto"
   585→          >
   586→            <Upload className="size-4 mr-2" />
   587→            Import Excel
   588→          </Button>
   589→          <Button
   590→            onClick={openCreateDialog}
   591→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   592→          >
   593→            <Plus className="size-4" />
   594→            Receive Payment
   595→          </Button>
   596→        </div>
   597→      </div>
   598→
   599→      {/* Summary Cards */}
   600→      <div className="grid gap-4 sm:grid-cols-3">
   601→        <Card>
   602→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   603→            <CardTitle className="text-sm font-medium text-muted-foreground">
   604→              Total Received
   605→            </CardTitle>
   606→            <IndianRupee className="size-4 text-emerald-600" />
   607→          </CardHeader>
   608→          <CardContent>
   609→            {loading ? (
   610→              <Skeleton className="h-8 w-32" />
   611→            ) : (
   612→              <p className="text-2xl font-bold">{formatCurrency(totalReceived)}</p>
   613→            )}
   614→          </CardContent>
   615→        </Card>
   616→
   617→        <Card>
   618→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   619→            <CardTitle className="text-sm font-medium text-muted-foreground">
   620→              Cash Payments
   621→            </CardTitle>
   622→            <Banknote className="size-4 text-emerald-600" />
   623→          </CardHeader>
   624→          <CardContent>
   625→            {loading ? (
   626→              <Skeleton className="h-8 w-32" />
   627→            ) : (

   632→
   633→        <Card>
   634→          <CardHeader className="flex flex-row items-center justify-between pb-2">

   637→            </CardTitle>
   638→            <Landmark className="size-4 text-sky-600" />
   639→          </CardHeader>
   640→          <CardContent>
   641→            {loading ? (
   642→              <Skeleton className="h-8 w-32" />
   643→            ) : (

   645→            )}

   731→                            title="Edit Payment"

   733→                          >
   734→                            <Pencil className="size-4" />
   735→                          </Button>

   737→                            variant="ghost"
   738→                            size="icon"
   739→                            onClick={() => setDeleteTarget(payment)}
   740→                            title="Delete Payment"
   741→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   742→                          >
   743→                            <Trash2 className="size-4" />
   744→                          </Button>
   745→                        </div>
   746→                      </TableCell>
   747→                    </TableRow>
   748→                  ))
   749→                )}
   750→              </TableBody>
   751→            </Table>
   752→          </div>
   753→        </CardContent>
   754→      </Card>
   755→
   756→      {/* Customer Outstanding Section */}
   757→      <Card>
   758→        <CardHeader>
   759→          <CardTitle className="flex items-center gap-2">
   760→            <Users className="size-5" />
   761→            <span>Customer Outstanding Balances</span>
   762→          </CardTitle>
   763→        </CardHeader>
   764→        <CardContent>
   765→          {loading ? (
   766→            <div className="space-y-3">
   767→              {Array.from({ length: 3 }).map((_, i) => (
   768→                <Skeleton key={i} className="h-10 w-full" />
   769→              ))}
   770→            </div>
   771→          ) : outstandingEntries.length === 0 ? (
   772→            <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   773→              <Users className="size-10 mb-2 opacity-40" />
   774→              <p className="text-sm">No outstanding balances. All customers are paid up!</p>
   775→            </div>
   776→          ) : (
   777→            <div className="max-h-[60vh] overflow-auto rounded-md border">
   778→              <Table>
   779→                <TableHeader className="sticky top-0 bg-background z-10">
   780→                  <TableRow>
   781→                    <TableHead>Customer</TableHead>
   782→                    <TableHead className="text-right">Total Orders (₹)</TableHead>
   783→                    <TableHead className="text-right">Total Paid (₹)</TableHead>
   784→                    <TableHead className="text-right">Outstanding (₹)</TableHead>
   785→                  </TableRow>
   786→                </TableHeader>
   787→                <TableBody>
   788→                  {outstandingEntries.map((entry) => (

   790→                      <TableCell className="font-medium whitespace-nowrap">
   791→                        {entry.customerName}

   802→                    </TableRow>
   803→                  ))}
   804→                </TableBody>
   805→              </Table>
   806→            </div>
   807→          )}
   808→        </CardContent>
   809→      </Card>
   810→
   811→      {/* Dialogs */}
   812→      {renderPaymentDialog()}
   813→      {renderDeleteDialog()}
   814→
   815→      <ExcelImport module="payments" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchPayments} />
   816→    </div>
   817→  )
   818→}
