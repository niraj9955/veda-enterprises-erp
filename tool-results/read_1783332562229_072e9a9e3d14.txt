    13→// ─── Server-side date normalization ─────────────────────────────────────────
    14→//
    15→// The client (excel-import.tsx) already normalizes dates to YYYY-MM-DD, but
    16→// we re-normalize here as defense-in-depth so a direct API call (or a future
    17→// caller that bypasses the Excel wizard) is still safe.
    18→//
    19→// Supported input formats (any separator among - / . or space):
    20→//   • YYYY-MM-DD          (already canonical)
    21→//   • DD-MM-YYYY  DD/MM/YYYY  DD.MM.YYYY   (day-first, Indian format)
    22→//   • DD-MM-YY    DD/MM/YY    DD.MM.YY     (short year, prefixed with 20)
    23→//   • MM/DD/YYYY  (only when first number > 12, e.g. 13/01/2024 -> Jan 13)
    24→//   • Datetime strings like "2024-01-15 10:30:00" (time part stripped)
    25→//   • Excel serial numbers (e.g. 46178)
    26→//   • Fallback: native Date parsing, then the raw string unchanged.
    27→function normalizeDate(value: unknown): string {
    28→  if (value == null) return ''
    29→  if (value instanceof Date && !isNaN(value.getTime())) {
    30→    return value.toISOString().split('T')[0]
    31→  }
    32→  if (typeof value === 'number' && Number.isFinite(value) && value > 59 && value < 60000) {
    33→    // Excel serial date: days since 1899-12-30
    34→    const ms = Math.round((value - 25569) * 86400 * 1000)
    35→    const d = new Date(ms)
    36→    if (!isNaN(d.getTime())) return d.toISOString().split('T')[0]
    37→  }
    38→  const raw = String(value ?? '').trim()
    39→  if (raw === '') return ''
    40→
    41→  // Strip time portion
    42→  const trimmed = raw
    43→    .replace(/[Tt]\s*\d{1,2}:\d{2}.*$/, '')
    44→    .replace(/\s+\d{1,2}:\d{2}.*$/, '')
    45→    .trim()
    46→  if (trimmed === '') return raw
    47→
    48→  // YYYY-MM-DD
    49→  if (/^\d{4}-\d{1,2}-\d{1,2}$/.test(trimmed)) {
    50→    const [y, m, d] = trimmed.split('-')
    51→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    52→  }
    53→  // YYYY/MM/DD or YYYY.MM.DD
    54→  const ymdMatch = trimmed.match(/^(\d{4})[/.\s](\d{1,2})[/.\s](\d{1,2})$/)
    55→  if (ymdMatch) {
    56→    const [, y, m, d] = ymdMatch
    57→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    58→  }
    59→  // DD-MM-YYYY / DD/MM/YYYY / DD.MM.YYYY (with US fallback if second > 12)
    60→  const dmyMatch = trimmed.match(/^(\d{1,2})[/.\s-](\d{1,2})[/.\s-](\d{4})$/)
    61→  if (dmyMatch) {
    62→    const [, a, b, y] = dmyMatch
    63→    const d = Number(b) > 12 && Number(a) <= 12 ? b : a
    64→    const m = Number(b) > 12 && Number(a) <= 12 ? a : b
    65→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    66→  }
    67→  // DD-MM-YY / DD/MM/YY / DD.MM.YY
    68→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/.\s-](\d{1,2})[/.\s-](\d{2})$/)
    69→  if (dmyShortMatch) {
    70→    const [, a, b, y] = dmyShortMatch
    71→    const d = Number(b) > 12 && Number(a) <= 12 ? b : a
    72→    const m = Number(b) > 12 && Number(a) <= 12 ? a : b
    73→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    74→  }
    75→  // Native Date fallback
    76→  try {
    77→    const date = new Date(trimmed)
    78→    if (!isNaN(date.getTime())) return date.toISOString().split('T')[0]
    79→  } catch {}
    80→  return raw
    81→}

    83→// Normalize every date-like field on a row before validation/insert.
    84→// This keeps the per-module switch statement below simple.
    85→function normalizeRowDates(row: Record<string, unknown>): void {
    86→  if ('date' in row) row.date = normalizeDate(row.date)
    87→  if ('deliveryDate' in row) row.deliveryDate = normalizeDate(row.deliveryDate)
    88→}

    90→// ─── Duplicate detection ────────────────────────────────────────────────────
    91→//
    92→// Per user request: "jo duplicate ho use skip kre or uska mess de but jo
    93→// duplicate nhi h usko import kre record me" — when an Excel row matches an
    94→// existing record in the DB (by a natural key), SKIP it with a clear message
    95→// but still IMPORT every non-duplicate row.
    96→//
    97→// To keep this fast, for each module we run ONE batched `find()` that
    98→// returns every existing row whose date falls within the dates seen in
    99→// the import payload, then build a Set of natural-key strings. Per-row
   100→// lookup then becomes an O(1) Set.has() check instead of a findOne().
   101→//
   102→// APPEND-ONLY semantics are preserved — we never wipe existing rows.
   103→
   104→// Build a stable natural-key string for a row, per module.
   105→function rowKey(module: string, row: Record<string, unknown>): string {
   106→  const s = (v: unknown) => String(v ?? '').trim().toLowerCase()
   107→  switch (module) {
   108→    case 'customers':
   109→      return s(row.mobile)
   110→    case 'production':
   111→      // One production entry per date — same date = duplicate (matches stock semantics)
   112→      return s(row.date)
   113→    case 'stock':
   114→      // One stock entry per date — same date = duplicate
   115→      return s(row.date)
   116→    case 'dailySell':
   117→      return `${s(row.date)}|${s(row.customerName)}|${s(row.amount)}`
   118→    case 'customerPayment':
   119→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
   120→    case 'labourPayment':
   121→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
   122→    case 'tractorPayment':
   123→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantityTon)}|${s(row.rate)}`
   124→    case 'dustPurchase':
   125→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantity)}|${s(row.rate)}`
   126→    case 'cementPurchase':
   127→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.itemName)}|${s(row.quantity)}|${s(row.rate)}`
   128→    case 'hardner':
   129→      // Only date+amount available — same date AND same amount = duplicate
   130→      return `${s(row.date)}|${s(row.amount)}`
   131→    case 'electricity':
   132→      return `${s(row.date)}|${s(row.name)}|${s(row.work)}|${s(row.amount)}`
   133→    case 'factoryStuff':
   134→      return `${s(row.date)}|${s(row.itemName)}|${s(row.amount)}`
   135→    case 'orders':
   136→      return `${s(row.customerId)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.rate)}|${s(row.deliveryDate || row.date)}`
   137→    case 'dispatch':
   138→      return `${s(row.customerId)}|${s(row.truckNumber)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.date)}`
   139→    case 'payments':
   140→      return `${s(row.customerId)}|${s(row.paymentType)}|${s(row.amount)}|${s(row.date)}`
   141→    case 'expenses':
   142→      return `${s(row.category)}|${s(row.amount)}|${s(row.date)}|${s(row.description)}`
   143→    default:
   144→      return ''
   145→  }
   146→}

   189→// Pre-fetch existing DB records that could collide with any row in `data`.
   190→// Returns a Set of natural-key strings.
   191→async function buildExistingKeys(module: string, data: Record<string, unknown>[]): Promise<Set<string>> {
   192→  const keys = new Set<string>()
   193→  if (data.length === 0) return keys
   194→
   195→  // Collect unique dates and customerIds from the import payload so we can
   196→  // narrow the DB query. This keeps the result set small.
   197→  const dates = new Set<string>()
   198→  const mobiles = new Set<string>()
   199→  const customerIds = new Set<string>()
   200→  for (const row of data) {
   201→    const d = String(row.date ?? '').split('T')[0]
   202→    if (d) dates.add(d)
   203→    const dd = String(row.deliveryDate ?? '').split('T')[0]
   204→    if (dd) dates.add(dd)
   205→    const m = String(row.mobile ?? '').trim()
   206→    if (m) mobiles.add(m)
   207→    const cid = String(row.customerId ?? '').trim()
   208→    if (cid) customerIds.add(cid)
   209→  }
   210→
   211→  let docs: Record<string, unknown>[] = []
   212→
   213→  switch (module) {
   214→    case 'customers':
   215→      docs = mobiles.size > 0
   216→        ? await Customer.find({ mobile: { $in: Array.from(mobiles) } }).lean() as any
   217→        : []
   218→      break
   219→    case 'production':
   220→      docs = dates.size > 0
   221→        ? await Production.find({ date: { $in: Array.from(dates) } }).lean() as any
   222→        : []
   223→      break
   224→    case 'stock':
   225→      docs = dates.size > 0
   226→        ? await Stock.find({ date: { $in: Array.from(dates) } }).lean() as any
   227→        : []
   228→      break
   229→    case 'dailySell':
   230→      docs = dates.size > 0
   231→        ? await DailySell.find({ date: { $in: Array.from(dates) } }).lean() as any
   232→        : []
   233→      break
   234→    case 'customerPayment':
   235→      docs = dates.size > 0
   236→        ? await CustomerPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   237→        : []
   238→      break
   239→    case 'labourPayment':
   240→      docs = dates.size > 0
   241→        ? await LabourPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   242→        : []
   243→      break
   244→    case 'tractorPayment':
   245→      docs = dates.size > 0
   246→        ? await TractorPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   247→        : []
   248→      break
   249→    case 'dustPurchase':
   250→      docs = dates.size > 0
   251→        ? await DustPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   252→        : []
   253→      break
   254→    case 'cementPurchase':
   255→      docs = dates.size > 0
   256→        ? await CementPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   257→        : []
   258→      break
   259→    case 'hardner':
   260→      docs = dates.size > 0
   261→        ? await Hardner.find({ date: { $in: Array.from(dates) } }).lean() as any
   262→        : []
   263→      break
   264→    case 'electricity':
   265→      docs = dates.size > 0
   266→        ? await Electricity.find({ date: { $in: Array.from(dates) } }).lean() as any
   267→        : []
   268→      break
   269→    case 'factoryStuff':
   270→      docs = dates.size > 0
   271→        ? await FactoryStuff.find({ date: { $in: Array.from(dates) } }).lean() as any
   272→        : []
   273→      break
   274→    case 'orders':
   275→      docs = customerIds.size > 0
   276→        ? await Order.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   277→        : []
   278→      break
   279→    case 'dispatch':
   280→      docs = customerIds.size > 0
   281→        ? await Dispatch.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   282→        : []
   283→      break
   284→    case 'payments':
   285→      docs = customerIds.size > 0
   286→        ? await Payment.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   287→        : []
   288→      break
   289→    case 'expenses':
   290→      // Expenses dedupe by category+amount+date — narrow by date
   291→      docs = dates.size > 0
   292→        ? await Expense.find({ date: { $in: Array.from(dates) } }).lean() as any
   293→        : []
   294→      break
   295→  }
   296→
   297→  for (const d of docs) {
   298→    const k = dbKey(module, d as Record<string, unknown>)
   299→    if (k) keys.add(k)
   300→  }
   301→  return keys
   302→}

   304→export async function POST(request: Request) {
   305→  try {
   306→    await connectDB()
   307→    const body = await request.json()
   308→    const { module, data } = body
   309→
   310→    if (!module || !data || !Array.isArray(data)) {
   311→      return NextResponse.json({ error: 'Module and data array are required' }, { status: 400 })
   312→    }
   313→
   314→    // ── Pre-fetch existing keys (1 DB query for the whole batch) ──────────
   315→    const existingKeys = await buildExistingKeys(module, data as Record<string, unknown>[])
   316→    // Track keys we've already seen WITHIN this batch so a duplicate Excel
   317→    // row doesn't get inserted twice in the same import run.
   318→    const seenInBatch = new Set<string>()
   319→
   320→    let imported = 0
   321→    let skipped = 0
   322→    let duplicatesSkipped = 0
   323→    const errors: string[] = []
   324→    const skippedReasons: string[] = []
   325→
   326→    for (let i = 0; i < data.length; i++) {
   327→      try {
   328→        const row = data[i]
   329→
   330→        // Normalize date fields BEFORE duplicate check / validation / insert
   331→        // so DD-MM-YYYY, DD/MM/YYYY, DD.MM.YYYY, datetime strings, and Excel
   332→        // serial numbers are all converted to canonical YYYY-MM-DD.
   333→        normalizeRowDates(row as Record<string, unknown>)
   334→
   335→        // ── Duplicate check (against DB AND within this batch) ──────────
   336→        const key = rowKey(module, row as Record<string, unknown>)
   337→        if (key) {
   338→          if (existingKeys.has(key)) {
   339→            skipped++
   340→            duplicatesSkipped++
   341→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   342→            skippedReasons.push(`Row ${i + 1}: Duplicate — ${label} already exists in records, skipped`)
   343→            continue
   344→          }
   345→          if (seenInBatch.has(key)) {
   346→            skipped++
   347→            duplicatesSkipped++
   348→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   349→            skippedReasons.push(`Row ${i + 1}: Duplicate — ${label} appears more than once in this Excel file, only first occurrence imported`)
   350→            continue
   351→          }
   352→          seenInBatch.add(key)
   353→        }
   354→
   355→        switch (module) {
   356→          // ─── Customers ──────────────────────────────────────────────
   357→          case 'customers': {
   358→            if (!row.name || !row.mobile) {
   359→              errors.push(`Row ${i + 1}: Name and mobile are required`)
   360→              skipped++
   361→              continue
   362→            }
   363→            await Customer.create({
   364→              name: String(row.name).trim(),
   365→              mobile: String(row.mobile).trim(),
   366→              gstNumber: row.gstNumber || '',
   367→              address: row.address || '',
   368→              creditLimit: Number(row.creditLimit) || 0,
   369→            })
   370→            imported++
   371→            break
   372→          }
   373→
   374→          // ─── Production ─────────────────────────────────────────────
   375→          case 'production': {
   376→            if (!row.date) {
   377→              errors.push(`Row ${i + 1}: Date is required`)
   378→              skipped++
   379→              continue
   380→            }
   381→            await Production.create({
   382→              date: String(row.date),
   383→              customerId: row.customerId || null,
   384→              cement: Number(row.cement) || 0,
   385→              zigZagGrey80: Number(row.zigZagGrey80) || 0,
   386→              zigZagRed80: Number(row.zigZagRed80) || 0,
   387→              zigZagYellow80: Number(row.zigZagYellow80) || 0,
   388→              zigZagGrey60: Number(row.zigZagGrey60) || 0,
   389→              zigZagRed60: Number(row.zigZagRed60) || 0,
   390→              zigZagYellow60: Number(row.zigZagYellow60) || 0,
   391→              curveStone: Number(row.curveStone) || 0,
   392→              chequreTile: Number(row.chequreTile) || 0,
   393→              dumbleGrey80: Number(row.dumbleGrey80) || 0,
   394→              dumbleRed80: Number(row.dumbleRed80) || 0,
   395→              dumbleYellow80: Number(row.dumbleYellow80) || 0,
   396→              transportationCharge: Number(row.transportationCharge) || 0,
   397→              remarks: String(row.remarks || ''),
   398→            })
   399→            imported++
   400→            break
   401→          }
   402→
   403→          // ─── Stock ─────────────────────────────────────────────────
   404→          case 'stock': {
   405→            if (!row.date) {
   406→              errors.push(`Row ${i + 1}: Date is required`)
   407→              skipped++
   408→              continue
   409→            }
   410→            await Stock.create({
   411→              date: String(row.date),
   412→              cement: Number(row.cement) || 0,
   413→              zigZagGrey80: Number(row.zigZagGrey80) || 0,
   414→              zigZagRed80: Number(row.zigZagRed80) || 0,
   415→              zigZagYellow80: Number(row.zigZagYellow80) || 0,
   416→              zigZagGrey60: Number(row.zigZagGrey60) || 0,
   417→              zigZagRed60: Number(row.zigZagRed60) || 0,
   418→              zigZagYellow60: Number(row.zigZagYellow60) || 0,

   421→              dumbleGrey80: Number(row.dumbleGrey80) || 0,
   422→              dumbleRed80: Number(row.dumbleRed80) || 0,
   423→              dumbleYellow80: Number(row.dumbleYellow80) || 0,
   424→            })
   425→            imported++
   426→            break
   427→          }
   428→
   429→          // ─── Daily Sell ────────────────────────────────────────────
   430→          case 'dailySell': {
   431→            if (!row.date || !row.customerName || !row.amount) {
   432→              errors.push(`Row ${i + 1}: Date, customer name, and amount are required`)
   433→              skipped++

   442→              amount: Number(row.amount),
   443→              remarks: String(row.remarks || ''),
   444→            })
   445→            imported++
   446→            break
   447→          }
   448→
   449→          // ─── Customer Payment ──────────────────────────────────────
   450→          case 'customerPayment': {
   451→            if (!row.date || !row.name || !row.amount) {
   452→              errors.push(`Row ${i + 1}: Date, name, and amount are required`)
   453→              skipped++
   454→              continue
   455→            }
   456→            await CustomerPayment.create({
   457→              date: String(row.date),
   458→              name: String(row.name),
   459→              address: String(row.address || ''),
   460→              amount: Number(row.amount),

   470→              errors.push(`Row ${i + 1}: Date, name, and amount are required`)
   471→              skipped++
   472→              continue
   473→            }
   474→            await LabourPayment.create({
   475→              date: String(row.date),
   476→              name: String(row.name),
   477→              address: String(row.address || ''),
   478→              amount: Number(row.amount),
   479→              remarks: String(row.remarks || ''),
   480→            })
   481→            imported++
   482→            break
   483→          }
   484→
   485→          // ─── Tractor Payment ───────────────────────────────────────
   486→          case 'tractorPayment': {
   487→            if (!row.date || !row.vendorName || !row.quantityTon || !row.rate) {
   488→              errors.push(`Row ${i + 1}: Date, vendor, quantity, and rate are required`)
   489→              skipped++
   490→              continue
   491→            }
   492→            const qty = Number(row.quantityTon)
   493→            const rate = Number(row.rate)
   494→            const totalAmount = Number(row.totalAmount) || qty * rate
   495→            const paidAmount = Number(row.paidAmount) || 0
   496→            await TractorPayment.create({
   497→              date: String(row.date),
   498→              vendorName: String(row.vendorName),
   499→              quantityTon: qty,
   500→              rate,
   501→              totalAmount,
   502→              paidAmount,
   503→              remainingAmount: Number(row.remainingAmount) || (totalAmount - paidAmount),
   504→              remarks: String(row.remarks || ''),
   505→            })
   506→            imported++
   507→            break
   508→          }
   509→
   510→          // ─── Dust Purchase ─────────────────────────────────────────
   511→          case 'dustPurchase': {
   512→            if (!row.date || !row.vendorName || !row.quantity || !row.rate) {
   513→              errors.push(`Row ${i + 1}: Date, vendor, quantity, and rate are required`)
   514→              skipped++
   515→              continue
   516→            }
   517→            const qty = Number(row.quantity)
   518→            const rate = Number(row.rate)
   519→            const totalAmount = Number(row.totalAmount) || qty * rate
   520→            const paidAmount = Number(row.paidAmount) || 0
   521→            await DustPurchase.create({
   522→              date: String(row.date),
   523→              vendorName: String(row.vendorName),
   524→              cementName: String(row.cementName || ''),
   525→              quantity: qty,
   526→              rate,
   527→              totalAmount,
   528→              paidAmount,
   529→              transportationCharge: Number(row.transportationCharge) || 0,
   530→              gst: Number(row.gst) || 0,
   531→              remarks: String(row.remarks || ''),
   532→            })
   533→            imported++
   534→            break
   535→          }
   536→
   537→          // ─── Cement Purchase ───────────────────────────────────────
   538→          case 'cementPurchase': {
   539→            if (!row.date || !row.vendorName || !row.quantity || !row.rate) {
   540→              errors.push(`Row ${i + 1}: Date, vendor, quantity, and rate are required`)
   541→              skipped++
   542→              continue
   543→            }
   544→            const qty = Number(row.quantity)
   545→            const rate = Number(row.rate)
   546→            const totalAmount = Number(row.totalAmount) || qty * rate
   547→            const paidAmount = Number(row.paidAmount) || 0
   548→            await CementPurchase.create({
   549→              date: String(row.date),
   550→              vendorName: String(row.vendorName),
   551→              itemName: String(row.itemName || ''),
   552→              quantity: qty,
   553→              rate,
   554→              totalAmount,
   555→              paidAmount,
   556→              transportationCharge: Number(row.transportationCharge) || 0,
   557→              gst: Number(row.gst) || 0,
   558→              remarks: String(row.remarks || ''),
   559→            })
   560→            imported++
   561→            break
   562→          }
   563→
   564→          // ─── Hardner ───────────────────────────────────────────────
   565→          case 'hardner': {
   566→            if (!row.date || !row.amount) {
   567→              errors.push(`Row ${i + 1}: Date and amount are required`)
   568→              skipped++
   569→              continue
   570→            }
   571→            await Hardner.create({
   572→              date: String(row.date),
   573→              amount: Number(row.amount),
   574→            })
   575→            imported++
   576→            break
   577→          }
   578→
   579→          // ─── Electricity ───────────────────────────────────────────
   580→          case 'electricity': {
   581→            if (!row.date || !row.amount) {
   582→              errors.push(`Row ${i + 1}: Date and amount are required`)
   583→              skipped++
   584→              continue
   585→            }
   586→            await Electricity.create({
   587→              date: String(row.date),
   588→              name: String(row.name || ''),
   589→              work: String(row.work || ''),
   590→              amount: Number(row.amount),
   591→              remarks: String(row.remarks || ''),
   592→            })
   593→            imported++
   594→            break
   595→          }
   596→
   597→          // ─── Factory Stuff ─────────────────────────────────────────
   598→          case 'factoryStuff': {
   599→            if (!row.date || !row.itemName || !row.amount) {
   600→              errors.push(`Row ${i + 1}: Date, item name, and amount are required`)
   601→              skipped++
   602→              continue
   603→            }
   604→            await FactoryStuff.create({
   605→              date: String(row.date),
   606→              itemName: String(row.itemName),
   607→              quantity: Number(row.quantity) || 0,
   608→              amount: Number(row.amount),
   609→              remarks: String(row.remarks || ''),
   610→            })
   611→            imported++
   612→            break
   613→          }
   614→
   615→          // ─── Legacy modules (Orders/Dispatch/Payments/Expenses) ────
   616→          case 'orders': {
   617→            if (!row.customerId || !row.brickType || !row.quantity || !row.rate) {
   618→              errors.push(`Row ${i + 1}: Customer, brick type, quantity, and rate are required`)
   619→              skipped++
   620→              continue
   621→            }
   622→            const count = await Order.countDocuments({})
   623→            const { Company } = await import('@/lib/models')
   624→            const company = await Company.findOne({})
   625→            const prefix = company?.orderPrefix || 'ORD'
   626→            const orderNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   627→
   628→            await Order.create({
   629→              orderNumber,
   630→              customerId: row.customerId,
   631→              brickType: row.brickType,
   632→              quantity: Number(row.quantity),
   633→              rate: Number(row.rate),
   634→              amount: Number(row.amount) || Number(row.quantity) * Number(row.rate),
   635→              deliveryDate: row.deliveryDate || row.date || new Date().toISOString().split('T')[0],
   636→              status: row.status || 'Pending',
   637→            })
   638→            imported++
   639→            break
   640→          }
   641→
   642→          case 'dispatch': {
   643→            if (!row.customerId || !row.truckNumber || !row.quantity || !row.brickType || !row.date) {
   644→              errors.push(`Row ${i + 1}: Customer, truck, quantity, brick type, and date are required`)
   645→              skipped++
   646→              continue
   647→            }
   648→            const count = await Dispatch.countDocuments({})
   649→            const { Company } = await import('@/lib/models')
   650→            const company = await Company.findOne({})
   651→            const prefix = company?.dispatchPrefix || 'DSP'
   652→            const dispatchNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   653→
   654→            await Dispatch.create({
   655→              dispatchNumber,
   656→              customerId: row.customerId,
   657→              orderId: row.orderId || null,
   658→              truckNumber: row.truckNumber,
   659→              driverName: row.driverName || '',
   660→              quantity: Number(row.quantity),
   661→              brickType: row.brickType,
   662→              date: row.date,
   663→            })
   664→            imported++
   665→            break
   666→          }
   667→
   668→          case 'payments': {
   669→            if (!row.customerId || !row.paymentType || !row.amount || !row.date) {
   670→              errors.push(`Row ${i + 1}: Customer, payment type, amount, and date are required`)
   671→              skipped++
   672→              continue
   673→            }
   674→            await Payment.create({
   675→              customerId: row.customerId,
   676→              paymentType: row.paymentType,
   677→              amount: Number(row.amount),
   678→              date: row.date,
   679→              remarks: row.remarks || '',
   680→            })
   681→            imported++
   682→            break
   683→          }
   684→
   685→          case 'expenses': {
   686→            if (!row.category || !row.amount || !row.date) {
   687→              errors.push(`Row ${i + 1}: Category, amount, and date are required`)
   688→              skipped++
   689→              continue
   690→            }
   691→            await Expense.create({
   692→              category: row.category,
   693→              amount: Number(row.amount),
   694→              date: row.date,
   695→              description: row.description || '',
   696→            })
   697→            imported++
   698→            break
   699→          }
   700→
   701→          default:
   702→            errors.push(`Unknown module: ${module}`)
   703→        }
   704→      } catch (err) {
   705→        errors.push(`Row ${i + 1}: ${err instanceof Error ? err.message : 'Import failed'}`)
   706→        skipped++
   707→      }
   708→    }
   709→
   710→    // Merge errors and skippedReasons so the UI can show every reason a row
   711→    // was not imported (validation error OR duplicate OR anything else).
   712→    const allReasons = [...errors, ...skippedReasons]
   713→
   714→    // Auto-sync Stock Overview from Production — when production rows are
   715→    // imported, the Stock module must reflect the latest daily totals
   716→    // automatically. We collect every date touched by THIS import and
   717→    // re-aggregate the matching Stock snapshot for each.
   718→    if (module === 'production' && imported > 0) {
   719→      const touchedDates = data
   720→        .map((row) => String(row.date || '').split('T')[0])
   721→        .filter(Boolean)
   722→      if (touchedDates.length > 0) {
   723→        try {
   724→          await syncStockForDates(touchedDates)
   725→        } catch (err) {
   726→          console.error('[import] Stock sync failed:', err)
   727→          // Don't fail the import — production rows were already saved.
   728→          // Just append a soft warning to the error list.
   729→          allReasons.push('Warning: production rows imported, but Stock Overview auto-sync failed. Check server logs.')
   730→        }
   731→      }
   732→    }
   733→
   734→    return NextResponse.json({
   735→      success: true,
   736→      imported,
   737→      skipped,
   738→      duplicatesSkipped,
   739→      total: data.length,
   740→      errors: allReasons.length > 0 ? allReasons.slice(0, 50) : undefined,
   741→    })
   742→  } catch (error) {
   743→    console.error('Error importing data:', error)
   744→    return NextResponse.json({ error: 'Failed to import data' }, { status: 500 })
   745→  }
   746→}

   748→// Human-readable label for a duplicate row, used in the skip reason.
   749→function duplicateRowLabel(module: string, row: Record<string, unknown>): string {
   750→  switch (module) {
   751→    case 'customers':
   752→      return `customer "${row.name}" (mobile ${row.mobile})`
   753→    case 'production':
   754→      return `production entry on ${row.date}`
   755→    case 'stock':
   756→      return `stock entry for ${row.date}`
   757→    case 'dailySell':
   758→      return `daily sell for "${row.customerName}" on ${row.date} (₹${row.amount})`
   759→    case 'customerPayment':
   760→      return `payment by "${row.name}" on ${row.date} (₹${row.amount})`
   761→    case 'labourPayment':
   762→      return `payment to "${row.name}" on ${row.date} (₹${row.amount})`
   763→    case 'tractorPayment':
   764→      return `tractor payment to "${row.vendorName}" on ${row.date}`
   765→    case 'dustPurchase':
   766→      return `dust purchase from "${row.vendorName}" on ${row.date}`
   767→    case 'cementPurchase':
   768→      return `cement purchase from "${row.vendorName}" on ${row.date}`
   769→    case 'hardner':
   770→      return `hardner entry on ${row.date} (₹${row.amount})`
   771→    case 'electricity':
   772→      return `electricity entry on ${row.date} (₹${row.amount})`
   773→    case 'factoryStuff':
   774→      return `factory stuff "${row.itemName}" on ${row.date} (₹${row.amount})`
   775→    case 'orders':
   776→      return `order for customer ${row.customerId} on ${row.deliveryDate || row.date}`
   777→    case 'dispatch':
   778→      return `dispatch to customer ${row.customerId} on ${row.date}`
   779→    case 'payments':
   780→      return `payment for customer ${row.customerId} on ${row.date}`
   781→    case 'expenses':
   782→      return `expense "${row.category}" on ${row.date} (₹${row.amount})`
   783→    default:
   784→      return 'record'
   785→  }
