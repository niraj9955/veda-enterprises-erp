    57→// ── Types ───────────────────────────────────────────────────────────────────
    58→
    59→interface Customer {
    60→  id: string
    61→  name: string
    62→  mobile: string
    63→  gstNumber: string
    64→  address: string
    65→  creditLimit: number
    66→  createdAt: string
    67→  updatedAt: string
    68→}
    69→
    70→interface Payment {
    71→  id: string
    72→  customerId: string
    73→  paymentType: string
    74→  amount: number
    75→  date: string
    76→  remarks: string
    77→  customer: { name: string }
    78→}
    79→
    80→interface CustomerFormData {
    81→  name: string
    82→  mobile: string
    83→  gstNumber: string
    84→  address: string
    85→  creditLimit: number | string
    86→}
    87→
    88→// ── Helpers ─────────────────────────────────────────────────────────────────
    89→
    90→const formatCurrency = (amount: number): string =>
    91→  new Intl.NumberFormat('en-IN', {
    92→    style: 'currency',
    93→    currency: 'INR',
    94→    minimumFractionDigits: 0,
    95→    maximumFractionDigits: 2,
    96→  }).format(amount)
    97→
    98→const formatDate = (dateStr: string): string =>
    99→  new Date(dateStr).toLocaleDateString('en-IN', {
   100→    day: '2-digit',
   101→    month: 'short',
   102→    year: 'numeric',
   103→  })
   104→
   105→const emptyForm: CustomerFormData = {
   106→  name: '',
   107→  mobile: '',
   108→  gstNumber: '',
   109→  address: '',
   110→  creditLimit: '',
   111→}
   112→
   113→// ── Component ───────────────────────────────────────────────────────────────
   114→
   115→export function CustomerModule() {
   116→  // State
   117→  const [customers, setCustomers] = React.useState<Customer[]>([])
   118→  const [totalCount, setTotalCount] = React.useState(0)
   119→  const [loading, setLoading] = React.useState(true)
   120→  const [search, setSearch] = React.useState('')
   121→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   122→
   123→  // Dialog states
   124→  const [formOpen, setFormOpen] = React.useState(false)
   125→  const [editingCustomer, setEditingCustomer] = React.useState<Customer | null>(null)
   126→  const [formData, setFormData] = React.useState<CustomerFormData>(emptyForm)
   127→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   128→
   129→  // Excel import
   130→  const [importOpen, setImportOpen] = React.useState(false)
   131→
   132→  // Delete confirmation
   133→  const [deleteTarget, setDeleteTarget] = React.useState<Customer | null>(null)
   134→  const [deleting, setDeleting] = React.useState(false)
   135→
   136→  // Ledger
   137→  const [ledgerCustomer, setLedgerCustomer] = React.useState<Customer | null>(null)
   138→  const [ledgerPayments, setLedgerPayments] = React.useState<Payment[]>([])
   139→  const [ledgerLoading, setLedgerLoading] = React.useState(false)
   140→
   141→  // Full customer history PAGE (full-screen, not modal)
   142→  const [historyCustomerId, setHistoryCustomerId] = React.useState<string | null>(null)
   143→
   144→  // ── Debounced search ────────────────────────────────────────────────────
   145→  React.useEffect(() => {
   146→    const timer = setTimeout(() => {
   147→      setDebouncedSearch(search)
   148→    }, 300)
   149→    return () => clearTimeout(timer)
   150→  }, [search])
   151→
   152→  // ── Fetch customers ─────────────────────────────────────────────────────
   153→  const fetchCustomers = React.useCallback(async () => {
   154→    setLoading(true)
   155→    try {
   156→      const res = await api.getCustomers(debouncedSearch || undefined)
   157→      setCustomers(res.customers as Customer[])
   158→      // Track total count separately so the badge can show "X of Y" when
   159→      // pagination is in effect (the API now returns `total` alongside
   160→      // the page slice). If `total` is missing (older deployments), fall
   161→      // back to the slice length.
   162→      const r = res as { total?: number; customers: Customer[] }
   163→      setTotalCount(r.total ?? r.customers.length)
   164→    } catch (err) {
   165→      toast({
   166→        title: 'Error',
   167→        description: err instanceof Error ? err.message : 'Failed to fetch customers',
   168→        variant: 'destructive',
   169→      })
   170→    } finally {
   171→      setLoading(false)
   172→    }
   173→  }, [debouncedSearch])
   174→
   175→  React.useEffect(() => {
   176→    fetchCustomers()
   177→  }, [fetchCustomers])
   178→
   179→  // ── Form handlers ───────────────────────────────────────────────────────
   180→  const openAddDialog = () => {
   181→    setEditingCustomer(null)
   182→    const pending = consumePendingAiResult('customer')
   183→    if (pending) {
   184→      const cl = pending.creditLimit
   185→      setFormData({
   186→        name: String(pending.name || ''),
   187→        mobile: String(pending.mobile || pending.contactNumber || ''),
   188→        gstNumber: String(pending.gstNumber || ''),
   189→        address: String(pending.address || ''),
   190→        creditLimit: cl != null && cl !== '' ? String(cl) : '',
   191→      })
   192→      toast({ title: 'AI auto-fill applied', description: 'Edit & verify before saving.' })
   193→    } else {
   194→      setFormData(emptyForm)
   195→    }
   196→    setFormOpen(true)
   197→  }
   198→
   199→  const openEditDialog = (customer: Customer) => {
   200→    setEditingCustomer(customer)
   201→    setFormData({
   202→      name: customer.name,
   203→      mobile: customer.mobile,
   204→      gstNumber: customer.gstNumber,
   205→      address: customer.address,
   206→      creditLimit: customer.creditLimit,
   207→    })
   208→    setFormOpen(true)
   209→  }
   210→
   211→  const handleFormChange = (field: keyof CustomerFormData, value: string) => {
   212→    setFormData((prev) => ({ ...prev, [field]: value }))
   213→  }
   214→
   215→  const handleSubmit = async () => {
   216→    if (!formData.name.trim()) {
   217→      toast({ title: 'Validation Error', description: 'Customer name is required', variant: 'destructive' })
   218→      return
   219→    }
   220→    if (!formData.mobile.trim()) {
   221→      toast({ title: 'Validation Error', description: 'Mobile number is required', variant: 'destructive' })
   222→      return
   223→    }
   224→
   225→    setFormSubmitting(true)
   226→    try {
   227→      const payload = {
   228→        name: formData.name.trim(),
   229→        mobile: formData.mobile.trim(),
   230→        gstNumber: formData.gstNumber.trim(),
   231→        address: formData.address.trim(),
   232→        creditLimit: Number(formData.creditLimit) || 0,
   233→      }
   234→
   235→      if (editingCustomer) {
   236→        await api.updateCustomer(editingCustomer.id, payload)
   237→        toast({ title: 'Success', description: 'Customer updated successfully' })
   238→      } else {
   239→        await api.createCustomer(payload)
   240→        toast({ title: 'Success', description: 'Customer created successfully' })
   241→      }
   242→
   243→      setFormOpen(false)
   244→      setFormData(emptyForm)
   245→      setEditingCustomer(null)
   246→      fetchCustomers()
   247→    } catch (err) {
   248→      toast({
   249→        title: 'Error',
   250→        description: err instanceof Error ? err.message : 'Failed to save customer',
   251→        variant: 'destructive',
   252→      })
   253→    } finally {
   254→      setFormSubmitting(false)
   255→    }
   256→  }
   257→
   258→  // ── Delete handler ──────────────────────────────────────────────────────
   259→  const handleDelete = async () => {
   260→    if (!deleteTarget) return
   261→    setDeleting(true)
   262→    try {
   263→      await api.deleteCustomer(deleteTarget.id)
   264→      toast({ title: 'Success', description: 'Customer deleted successfully' })
   265→      setDeleteTarget(null)
   266→      fetchCustomers()
   267→    } catch (err) {
   268→      toast({
   269→        title: 'Error',
   270→        description: err instanceof Error ? err.message : 'Failed to delete customer',
   271→        variant: 'destructive',
   272→      })
   273→    } finally {
   274→      setDeleting(false)
   275→    }
   276→  }
   277→
   278→  // ── Ledger handler ──────────────────────────────────────────────────────
   279→  const openLedger = async (customer: Customer) => {
   280→    setLedgerCustomer(customer)
   281→    setLedgerLoading(true)
   282→    try {
   283→      const res = await api.getPayments()
   284→      const allPayments = res.payments as Payment[]
   285→      const filtered = allPayments.filter((p) => p.customerId === customer.id)
   286→      // Sort by date descending
   287→      filtered.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   288→      setLedgerPayments(filtered)
   289→    } catch (err) {
   290→      toast({
   291→        title: 'Error',
   292→        description: err instanceof Error ? err.message : 'Failed to fetch payment history',
   293→        variant: 'destructive',
   294→      })
   295→      setLedgerPayments([])
   296→    } finally {
   297→      setLedgerLoading(false)
   298→    }
   299→  }
   300→
   301→  // ── Render: Loading skeletons ───────────────────────────────────────────
   302→  const renderSkeletons = () =>
   303→    Array.from({ length: 5 }).map((_, i) => (
   304→      <TableRow key={i}>
   305→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   306→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   307→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   308→        <TableCell><Skeleton className="h-4 w-40" /></TableCell>
   309→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   310→        <TableCell><Skeleton className="h-8 w-28" /></TableCell>
   311→      </TableRow>
   312→    ))
   313→
   314→  // ── Render: Form dialog ─────────────────────────────────────────────────
   315→  const renderFormDialog = () => (
   316→    <Dialog open={formOpen} onOpenChange={setFormOpen}>
   317→      <DialogContent className="sm:max-w-md">
   318→        <DialogHeader>
   319→          <DialogTitle>
   320→            {editingCustomer ? 'Edit Customer' : 'Add Customer'}
   321→          </DialogTitle>
   322→          <DialogDescription>
   323→            {editingCustomer
   324→              ? 'Update the customer details below.'
   325→              : 'Fill in the details to create a new customer.'}
   326→          </DialogDescription>
   327→        </DialogHeader>
   328→        {!editingCustomer && (
   329→          <div className="flex justify-end">
   330→            <AiFillButton
   331→              module="customer"
   332→              onApply={(fields) => setFormData((prev) => ({
   333→                name: fields.name != null ? String(fields.name) : prev.name,
   334→                mobile: fields.mobile != null ? String(fields.mobile) : (fields.contactNumber != null ? String(fields.contactNumber) : prev.mobile),
   335→                gstNumber: fields.gstNumber != null ? String(fields.gstNumber) : prev.gstNumber,
   336→                address: fields.address != null ? String(fields.address) : prev.address,
   337→                creditLimit: fields.creditLimit != null && fields.creditLimit !== '' ? String(fields.creditLimit) : prev.creditLimit,
   338→              }))}
   339→            />
   340→          </div>
   341→        )}
   342→
   343→        <div className="grid gap-4 py-2">
   344→          {/* Customer Name */}
   345→          <div className="grid gap-2">
   346→            <Label htmlFor="cust-name">
   347→              Customer Name <span className="text-destructive">*</span>
   348→            </Label>
   349→            <div className="relative">
   350→              <Input
   351→                id="cust-name"
   352→                placeholder="Enter customer name"
   353→                value={formData.name}
   354→                onChange={(e) => handleFormChange('name', e.target.value)}
   355→                className="pr-9"
   356→              />
   357→              <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   358→                <FieldVoiceInput
   359→                  fieldLabel="customer name"
   360→                  onChange={(text) => handleFormChange('name', text)}
   361→                />
   362→              </div>
   363→            </div>
   364→          </div>
   365→
   366→          {/* Mobile Number */}
   367→          <div className="grid gap-2">
   368→            <Label htmlFor="cust-mobile">
   369→              Mobile Number <span className="text-destructive">*</span>
   370→            </Label>
   371→            <div className="relative">
   372→              <Input
   373→                id="cust-mobile"
   374→                placeholder="Enter mobile number"
   375→                value={formData.mobile}
   376→                onChange={(e) => handleFormChange('mobile', e.target.value)}
   377→                className="pr-9"
   378→              />
   379→              <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   380→                <FieldVoiceInput
   381→                  fieldLabel="mobile number"
   382→                  onChange={(text) => handleFormChange('mobile', text.replace(/[^0-9+\-\s]/g, '').trim())}
   383→                />
   384→              </div>
   385→            </div>
   386→          </div>
   387→
   388→          {/* GST Number */}
   389→          <div className="grid gap-2">
   390→            <Label htmlFor="cust-gst">GST Number</Label>
   391→            <div className="relative">
   392→              <Input
   393→                id="cust-gst"
   394→                placeholder="Enter GST number (optional)"
   395→                value={formData.gstNumber}
   396→                onChange={(e) => handleFormChange('gstNumber', e.target.value)}
   397→                className="pr-9"
   398→              />
   399→              <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   400→                <FieldVoiceInput
   401→                  fieldLabel="GST number"
   402→                  onChange={(text) => handleFormChange('gstNumber', text.toUpperCase().replace(/[^A-Z0-9]/g, ''))}
   403→                />
   404→              </div>
   405→            </div>
   406→          </div>
   407→
   408→          {/* Address */}
   409→          <div className="grid gap-2">
   410→            <Label htmlFor="cust-address">Address</Label>
   411→            <div className="relative">
   412→              <Textarea
   413→                id="cust-address"
   414→                placeholder="Enter address"
   415→                value={formData.address}
   416→                onChange={(e) => handleFormChange('address', e.target.value)}
   417→                className="min-h-[80px] pr-9"
   418→              />
   419→              <div className="absolute right-1.5 top-2">
   420→                <FieldVoiceInput
   421→                  fieldLabel="address"
   422→                  onChange={(text) => handleFormChange('address', text)}
   423→                />
   424→              </div>
   425→            </div>
   426→          </div>
   427→
   428→          {/* Credit Limit */}
   429→          <div className="grid gap-2">
   430→            <Label htmlFor="cust-credit">Credit Limit (₹)</Label>
   431→            <div className="relative">
   432→              <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   433→              <Input
   434→                id="cust-credit"
   435→                type="number"
   436→                placeholder="0"
   437→                className="pl-9"
   438→                value={formData.creditLimit}
   439→                onChange={(e) => handleFormChange('creditLimit', e.target.value)}
   440→              />
   441→            </div>
   442→          </div>
   443→        </div>
   444→
   445→        <DialogFooter>
   446→          <Button
   447→            variant="outline"
   448→            onClick={() => setFormOpen(false)}
   449→            disabled={formSubmitting}
   450→          >
   451→            Cancel
   452→          </Button>
   453→          <Button
   454→            onClick={handleSubmit}
   455→            disabled={formSubmitting}
   456→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   457→          >
   458→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   459→            {editingCustomer ? 'Update Customer' : 'Create Customer'}
   460→          </Button>
   461→        </DialogFooter>
   462→      </DialogContent>
   463→    </Dialog>
   464→  )
   465→
   466→  // ── Render: Delete confirmation ─────────────────────────────────────────
   467→  const renderDeleteDialog = () => (
   468→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   469→      <AlertDialogContent>
   470→        <AlertDialogHeader>
   471→          <AlertDialogTitle>Delete Customer</AlertDialogTitle>
   472→          <AlertDialogDescription>
   473→            Are you sure you want to delete <strong>{deleteTarget?.name}</strong>? This action
   474→            cannot be undone.
   475→          </AlertDialogDescription>
   476→        </AlertDialogHeader>
   477→        <AlertDialogFooter>
   478→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   479→          <AlertDialogAction
   480→            onClick={handleDelete}
   481→            disabled={deleting}
   482→            className="bg-destructive text-white hover:bg-destructive/90"
   483→          >
   484→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   485→            Delete
   486→          </AlertDialogAction>
   487→        </AlertDialogFooter>
   488→      </AlertDialogContent>
   489→    </AlertDialog>
   490→  )
   491→
   492→  // ── Render: Ledger dialog ───────────────────────────────────────────────
   493→  const renderLedgerDialog = () => (
   494→    <Dialog open={!!ledgerCustomer} onOpenChange={(open) => !open && setLedgerCustomer(null)}>
   495→      <DialogContent className="sm:max-w-2xl">
   496→        <DialogHeader>
   497→          <DialogTitle className="flex items-center gap-2">
   498→            <BookOpen className="size-5 text-emerald-600" />
   499→            Payment Ledger — {ledgerCustomer?.name}
   500→          </DialogTitle>
   501→          <DialogDescription>
   502→            Payment history for {ledgerCustomer?.name}
   503→          </DialogDescription>
   504→        </DialogHeader>
   505→
   506→        {ledgerLoading ? (
   507→          <div className="space-y-3 py-4">
   508→            {Array.from({ length: 3 }).map((_, i) => (
   509→              <Skeleton key={i} className="h-10 w-full" />
   510→            ))}
   511→          </div>
   512→        ) : ledgerPayments.length === 0 ? (
   513→          <div className="py-8 text-center text-muted-foreground">
   514→            No payment records found for this customer.
   515→          </div>
   516→        ) : (
   517→          <div className="max-h-96 overflow-y-auto">
   518→            <Table>
   519→              <TableHeader>
   520→                <TableRow>
   521→                  <TableHead>Date</TableHead>
   522→                  <TableHead>Type</TableHead>
   523→                  <TableHead className="text-right">Amount</TableHead>
   524→                  <TableHead>Remarks</TableHead>
   525→                </TableRow>
   526→              </TableHeader>
   527→              <TableBody>
   528→                {ledgerPayments.map((payment) => (
   529→                  <TableRow key={payment.id}>
   530→                    <TableCell className="whitespace-nowrap">
   531→                      {formatDate(payment.date)}
   532→                    </TableCell>
   533→                    <TableCell>
   534→                      <Badge
   535→                        variant={
   536→                          payment.paymentType?.toLowerCase() === 'credit' ? 'secondary' : 'outline'
   537→                        }
   538→                        className={
   539→                          payment.paymentType?.toLowerCase() === 'credit'
   540→                            ? 'bg-emerald-100 text-emerald-700 border-emerald-200'
   541→                            : 'bg-orange-100 text-orange-700 border-orange-200'
   542→                        }
   543→                      >
   544→                        {payment.paymentType}
   545→                      </Badge>
   546→                    </TableCell>
   547→                    <TableCell className="text-right font-medium whitespace-nowrap">
   548→                      {formatCurrency(payment.amount)}
   549→                    </TableCell>
   550→                    <TableCell className="max-w-[200px] truncate text-muted-foreground">
   551→                      {payment.remarks || '—'}
   552→                    </TableCell>
   553→                  </TableRow>
   554→                ))}
   555→              </TableBody>
   556→            </Table>
   557→          </div>
   558→        )}
   559→
   560→        <DialogFooter>
   561→          <Button variant="outline" onClick={() => setLedgerCustomer(null)}>
   562→            Close
   563→          </Button>
   564→        </DialogFooter>
   565→      </DialogContent>
   566→    </Dialog>
   567→  )
   568→
   569→  // ── Render: Main ────────────────────────────────────────────────────────
   570→  // If a customer is selected for history view, render the full-screen history page
   571→  // INSTEAD of the customer list — user requested this be a separate page, not a modal.
   572→  if (historyCustomerId) {
   573→    return (
   574→      <CustomerHistoryPage
   575→        customerId={historyCustomerId}
   576→        onBack={() => setHistoryCustomerId(null)}
   577→      />
   578→    )
   579→  }
   580→
   581→  return (
   582→    <div className="space-y-4 md:space-y-6">
   583→      {/* Header */}
   584→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   585→        <div className="flex items-center gap-2.5 sm:gap-3 min-w-0">
   586→          <div className="flex size-9 sm:size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700 shrink-0">
   587→            <Users className="size-4 sm:size-5" />
   588→          </div>
   589→          <div className="min-w-0">
   590→            <h2 className="text-lg sm:text-2xl font-bold tracking-tight truncate">Customer Management</h2>
   591→            <p className="text-xs sm:text-sm text-muted-foreground truncate">
   592→              Manage your customer database and ledgers
   593→            </p>
   594→          </div>
   595→        </div>
   596→        <div className="flex gap-2 w-full sm:w-auto">
   597→          <Button
   598→            variant="outline"
   599→            onClick={() => setImportOpen(true)}
   600→            className="flex-1 sm:flex-initial px-3 sm:px-4 text-xs sm:text-sm"
   601→          >
   602→            <Upload className="size-3.5 sm:size-4 mr-1.5 sm:mr-2" />
   603→            <span className="truncate">Import Excel</span>
   604→          </Button>
   605→          <Button
   606→            onClick={openAddDialog}
   607→            className="bg-emerald-600 hover:bg-emerald-700 text-white flex-1 sm:flex-initial px-3 sm:px-4 text-xs sm:text-sm"
   608→          >
   609→            <Plus className="size-3.5 sm:size-4 mr-1.5 sm:mr-2" />
   610→            <span className="truncate">Add Customer</span>
   611→          </Button>
   612→        </div>
   613→      </div>
   614→
   615→      {/* Search */}
   616→      <Card>
   617→        <CardContent className="pt-6">
   618→          <div className="relative">
   619→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   620→            <Input
   621→              placeholder="Search customers by name or mobile..."
   622→              value={search}
   623→              onChange={(e) => setSearch(e.target.value)}
   624→              className="pl-9"
   625→            />
   626→          </div>
   627→        </CardContent>
   628→      </Card>
   629→
   630→      {/* Table */}
   631→      <Card>
   632→        <CardHeader>
   633→          <CardTitle className="flex items-center justify-between">
   634→            <span>Customers</span>
   635→            <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   636→              {loading
   637→                ? 'Loading…'
   638→                : totalCount > customers.length
   639→                  ? `Showing ${customers.length} of ${totalCount} records`
   640→                  : `${customers.length} record${customers.length !== 1 ? 's' : ''}`}
   641→            </Badge>
   642→          </CardTitle>
   643→        </CardHeader>
   644→        <CardContent>
   645→          <ScrollableTable maxHeight="max-h-[60vh]">
   646→            <Table>
   647→              <TableHeader>
   648→                <TableRow>
   649→                  <TableHead className="sticky left-0 bg-background z-20 min-w-[140px]">Name</TableHead>
   650→                  <TableHead className="whitespace-nowrap">Mobile</TableHead>
   651→                  <TableHead className="whitespace-nowrap">GST Number</TableHead>
   652→                  <TableHead className="min-w-[180px]">Address</TableHead>
   653→                  <TableHead className="text-right whitespace-nowrap">Credit Limit (₹)</TableHead>
   654→                  <TableHead className="text-right whitespace-nowrap">Actions</TableHead>
   655→                </TableRow>
   656→              </TableHeader>
   657→              <TableBody>
   658→                {loading ? (
   659→                  renderSkeletons()
   660→                ) : customers.length === 0 ? (
   661→                  <TableRow>
   662→                    <TableCell colSpan={6} className="h-32 text-center text-muted-foreground">
   663→                      {debouncedSearch
   664→                        ? 'No customers found matching your search.'
   665→                        : 'No customers yet. Click "Add Customer" to get started.'}
   666→                    </TableCell>
   667→                  </TableRow>
   668→                ) : (
   669→                  customers.map((customer) => (
   670→                    <TableRow key={customer.id} className="cursor-pointer hover:bg-emerald-50/40" onClick={() => setHistoryCustomerId(customer.id)}>
   671→                      <TableCell className="font-medium sticky left-0 bg-background z-10 min-w-[140px]">
   672→                        <button
   673→                          type="button"
   674→                          className="text-emerald-700 hover:text-emerald-900 hover:underline text-left"
   675→                          onClick={(e) => {
   676→                            e.stopPropagation()
   677→                            setHistoryCustomerId(customer.id)
   678→                          }}
   679→                          title="Click to view full customer history"
   680→                        >
   681→                          {customer.name}
   682→                        </button>
   683→                      </TableCell>
   684→                      <TableCell className="whitespace-nowrap">{customer.mobile}</TableCell>
   685→                      <TableCell>
   686→                        {customer.gstNumber ? (
   687→                          <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">
   688→                            {customer.gstNumber}
   689→                          </Badge>
   690→                        ) : (
   691→                          <Badge variant="outline" className="text-muted-foreground">
   692→                            No GST
   693→                          </Badge>
   694→                        )}
   695→                      </TableCell>
   696→                      <TableCell className="max-w-[200px] truncate text-muted-foreground">
   697→                        {customer.address || '—'}
   698→                      </TableCell>
   699→                      <TableCell className="text-right font-medium whitespace-nowrap">
   700→                        {formatCurrency(customer.creditLimit)}
   701→                      </TableCell>
   702→                      <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   703→                        <div className="flex items-center justify-end gap-1">
   704→                          <Button
   705→                            variant="ghost"
   706→                            size="icon"
   707→                            onClick={() => setHistoryCustomerId(customer.id)}
   708→                            title="View Full History"
   709→                            className="text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   710→                          >
   711→                            <BookOpen className="size-4" />
   712→                          </Button>
   713→                          <Button
   714→                            variant="ghost"
   715→                            size="icon"
   716→                            onClick={() => openEditDialog(customer)}
   717→                            title="Edit Customer"
   718→                          >
   719→                            <Pencil className="size-4" />
   720→                          </Button>
   721→                          <Button
   722→                            variant="ghost"
   723→                            size="icon"
   724→                            onClick={() => setDeleteTarget(customer)}
   725→                            title="Delete Customer"
   726→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   727→                          >
   728→                            <Trash2 className="size-4" />
   729→                          </Button>
   730→                        </div>
   731→                      </TableCell>
   732→                    </TableRow>
   733→                  ))
   734→                )}
   735→              </TableBody>
   736→            </Table>
   737→          </ScrollableTable>
   738→          <p className="text-xs text-muted-foreground mt-2">
   739→            💡 Tip: Click on a customer's name or the book icon to view their complete transaction history (orders, dispatch, payments, sells, production, balance).
   740→          </p>
   741→        </CardContent>
   742→      </Card>
   743→
   744→      {/* Dialogs */}
   745→      {renderFormDialog()}
   746→      {renderDeleteDialog()}
   747→      {renderLedgerDialog()}
   748→
   749→      <ExcelImport module="customers" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchCustomers} />
   750→    </div>
   751→  )
   752→}
   753→
   754→export default CustomerModule
