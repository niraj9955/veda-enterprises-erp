   455→  const handleRestoreBackup = async () => {
   456→    const input = document.createElement('input')
   457→    input.type = 'file'
   458→    input.accept = '.json'
   459→    input.onchange = async (e) => {
   460→      const file = (e.target as HTMLInputElement).files?.[0]
   461→      if (!file) return
   462→      setDbLoading(true)
   463→      try {
   464→        const text = await file.text()
   465→        const parsed = JSON.parse(text)
   466→        // Backup files exported by the v2 export route are wrapped as
   467→        // { version, exportedAt, data: { ...collections... }, counts }.
   468→        // Older / hand-crafted backups may pass the collections map directly.
   469→        // Normalise both shapes here so the backend always sees the inner
   470→        // collections object regardless of file format.
   471→        const payload = parsed?.data && typeof parsed.data === 'object' && !Array.isArray(parsed.data)
   472→          ? parsed.data
   473→          : parsed
   474→
   475→        const result = await api.restoreBackup(payload)
   476→        // Result shape (MERGE mode):
   477→        //   { mode: 'merge', counts: { inserted, replaced }, perCollection: { customers: {inserted, replaced, skipped}, ... } }
   478→        // Show a clear breakdown so the user can verify their data was restored.
   479→        const inserted = (result as any)?.counts?.inserted ?? 0
   480→        const replaced = (result as any)?.counts?.replaced ?? 0
   481→        const perCollection = (result as any)?.perCollection || {}
   482→        const summary = Object.entries(perCollection)
   483→          .filter(([, v]: any) => (v?.inserted ?? 0) + (v?.replaced ?? 0) > 0)
   484→          .map(([k, v]: any) => `${k}: +${v.inserted ?? 0} new / ~${v.replaced ?? 0} updated`)
   485→          .join(' • ')
   486→        toast({
   487→          title: 'Backup restored (merge mode)',
   488→          description: summary
   489→            ? `${inserted + replaced} docs affected (${inserted} new + ${replaced} updated). Current data NOT in backup is preserved. ${summary}`
   490→            : `${inserted + replaced} docs affected. Current data NOT in backup is preserved.`,
   491→        })
   492→      } catch (err) {
   493→        console.error('Restore failed:', err)
   494→        toast({
   495→          title: 'Restore failed',
   496→          description: err instanceof Error ? err.message : 'Could not restore backup',
   497→          variant: 'destructive',
   498→        })
   499→      } finally {
   500→        setDbLoading(false)
   501→      }
   502→    }
   503→    input.click()
   504→  }
