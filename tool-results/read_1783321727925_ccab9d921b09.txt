   239→// Normalize a string for comparison: lowercase, remove special chars, spaces
   240→function normalize(s: string): string {
   241→  return s.toLowerCase().replace(/[^a-z0-9\u0900-\u097F]/g, '')
   242→}

   244→// Smart auto-detect mapping between Excel columns and module fields
   245→function autoMapColumns(excelColumns: string[], moduleFields: { key: string; label: string; required: boolean; aliases: string[] }[]): Record<string, string> {
   246→  const mapping: Record<string, string> = {} // excelColumn -> fieldKey
   247→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   248→
   249→  for (const field of moduleFields) {
   250→    let bestMatch = ''
   251→    let bestScore = 0
   252→
   253→    for (const excelCol of normalizedExcel) {
   254→      // Check exact match with key
   255→      if (excelCol.normalized === normalize(field.key)) {
   256→        bestMatch = excelCol.original
   257→        bestScore = 100
   258→        break
   259→      }
   260→
   261→      // Check aliases
   262→      for (const alias of field.aliases) {
   263→        const normAlias = normalize(alias)
   264→        if (excelCol.normalized === normAlias) {
   265→          if (100 > bestScore) {
   266→            bestMatch = excelCol.original
   267→            bestScore = 100
   268→          }
   269→          break
   270→        }
   271→      }
   272→
   273→      if (bestScore >= 100) break
   274→
   275→      // Fuzzy match: check if alias is contained in column name or vice versa
   276→      for (const alias of field.aliases) {
   277→        const normAlias = normalize(alias)
   278→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   279→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   280→          if (score > bestScore) {
   281→            bestMatch = excelCol.original
   282→            bestScore = score
   283→          }
   284→        }
   285→      }
   286→    }
   287→
   288→    if (bestMatch && bestScore >= 50) {
   289→      mapping[bestMatch] = field.key
   290→    }
   291→  }
   292→
   293→  return mapping
   294→}

   296→// Auto-detect the module from Excel column headers
   297→function autoDetectModule(columns: string[]): string | null {
   298→  const normalized = columns.map(normalize)
   299→  const scores: Record<string, number> = {}
   300→
   301→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   302→    scores[moduleKey] = 0
   303→    for (const field of template.fields) {
   304→      for (const alias of field.aliases) {
   305→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   306→          scores[moduleKey] += field.required ? 3 : 1
   307→        }
   308→      }
   309→    }
   310→  }
   311→
   312→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   313→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   314→}

   316→// Transform data using mapping, also handle customer name -> ID resolution
   317→function transformRow(
   318→  row: Record<string, unknown>,
   319→  mapping: Record<string, string>,
   320→  customers?: { id: string; name: string }[]
   321→): Record<string, unknown> {
   322→  const result: Record<string, unknown> = {}
   323→
   324→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   325→    let value = row[excelCol]
   326→
   327→    // Handle customer name -> ID resolution
   328→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   329→      const nameStr = String(value || '').trim().toLowerCase()
   330→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   331→      if (matched) {
   332→        result['customerId'] = matched.id
   333→      } else {
   334→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   335→      }
   336→      continue
   337→    }
   338→
   339→    // Handle numeric fields
   340→    if ([
   341→      'quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate',
   342→      'zigZagWhite80', 'zigZagRed80', 'zigZagYellow80',
   343→      'zigZagWhite60', 'zigZagRed60', 'zigZagYellow60',
   344→      'curveStone', 'chequreTile', 'transportationCharge',
   345→      'cement', 'zigZagGrey80', 'zigZagGrey60',
   346→      'dumbleGrey80', 'dumbleRed80', 'dumbleYellow80',
   347→      'quantityTon', 'totalAmount', 'paidAmount', 'remainingAmount',
   348→      'gst',
   349→    ].includes(fieldKey)) {
   350→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   351→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   352→      continue
   353→    }
   354→
   355→    // Handle date fields - try to parse various formats
   356→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   357→      result[fieldKey] = parseDate(String(value || ''))
   358→      continue
   359→    }
   360→
   361→    // Handle shift field - normalize
   362→    if (fieldKey === 'shift') {
   363→      const shiftStr = String(value || 'Morning').trim()
   364→      const lower = shiftStr.toLowerCase()
   365→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   366→        result[fieldKey] = 'Morning'
   367→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   368→        result[fieldKey] = 'Evening'
   369→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   370→        result[fieldKey] = 'Night'
   371→      } else {
   372→        result[fieldKey] = shiftStr
   373→      }
   374→      continue
   375→    }
   376→
   377→    // Handle payment type
   378→    if (fieldKey === 'paymentType') {
   379→      const ptStr = String(value || 'Cash').trim()
   380→      const lower = ptStr.toLowerCase()
   381→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   382→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   383→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   384→      else result[fieldKey] = ptStr
   385→      continue
   386→    }
   387→
   388→    // Handle expense category
   389→    if (fieldKey === 'category') {
   390→      const catStr = String(value || '').trim()
   391→      const lower = catStr.toLowerCase()
   392→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   393→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   394→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   395→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   396→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   397→      else result[fieldKey] = catStr
   398→      continue
   399→    }
   400→
   401→    // Handle status
   402→    if (fieldKey === 'status') {
   403→      const stStr = String(value || 'Pending').trim()
   404→      const lower = stStr.toLowerCase()
   405→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   406→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   407→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   408→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   409→      else result[fieldKey] = stStr
   410→      continue
   411→    }
   412→
   413→    result[fieldKey] = String(value || '')
   414→  }
   415→
   416→  // For orders/dispatch: calculate amount if not mapped
   417→  if (mapping) {
   418→    const targetModule = Object.values(mapping)
   419→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   420→      const qty = Number(result.quantity || 0)
   421→      const rate = Number(result.rate || 0)
   422→      result.amount = qty * rate
   423→    }
   424→  }
   425→
   426→  return result
   427→}

   429→// Parse various date formats to YYYY-MM-DD
   430→function parseDate(value: string): string {
   431→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   432→
   433→  const trimmed = value.trim()
   434→
   435→  // Already in YYYY-MM-DD format
   436→  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed
   437→
   438→  // DD-MM-YYYY or DD/MM/YYYY
   439→  const dmyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   440→  if (dmyMatch) {
   441→    const [, d, m, y] = dmyMatch
   442→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   443→  }
   444→
   445→  // DD-MM-YY or DD/MM/YY
   446→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2})$/)
   447→  if (dmyShortMatch) {
   448→    const [, d, m, y] = dmyShortMatch
   449→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   450→  }
   451→
   452→  // MM/DD/YYYY (US format) - only if first number > 12
   453→  const mdyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   454→  if (mdyMatch) {
   455→    const [, m, d, y] = mdyMatch
   456→    if (Number(m) > 12) {
   457→      return `${y}-${d.padStart(2, '0')}-${m.padStart(2, '0')}`
   458→    }
   459→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   460→  }
   461→
   462→  // Try native Date parsing as fallback
   463→  try {
   464→    const date = new Date(trimmed)
   465→    if (!isNaN(date.getTime())) {
   466→      return date.toISOString().split('T')[0]
   467→    }
   468→  } catch {}
   469→
   470→  return trimmed
   471→}
