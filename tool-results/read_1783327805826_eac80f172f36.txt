   247→// Normalize a string for comparison: lowercase, remove special chars, spaces
   248→function normalize(s: string): string {
   249→  return s.toLowerCase().replace(/[^a-z0-9\u0900-\u097F]/g, '')
   250→}

   252→// Smart auto-detect mapping between Excel columns and module fields
   253→function autoMapColumns(excelColumns: string[], moduleFields: { key: string; label: string; required: boolean; aliases: string[] }[]): Record<string, string> {
   254→  const mapping: Record<string, string> = {} // excelColumn -> fieldKey
   255→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   256→
   257→  for (const field of moduleFields) {
   258→    let bestMatch = ''
   259→    let bestScore = 0
   260→
   261→    for (const excelCol of normalizedExcel) {
   262→      // Check exact match with key
   263→      if (excelCol.normalized === normalize(field.key)) {
   264→        bestMatch = excelCol.original
   265→        bestScore = 100
   266→        break
   267→      }
   268→
   269→      // Check aliases
   270→      for (const alias of field.aliases) {
   271→        const normAlias = normalize(alias)
   272→        if (excelCol.normalized === normAlias) {
   273→          if (100 > bestScore) {
   274→            bestMatch = excelCol.original
   275→            bestScore = 100
   276→          }
   277→          break
   278→        }
   279→      }
   280→
   281→      if (bestScore >= 100) break
   282→
   283→      // Fuzzy match: check if alias is contained in column name or vice versa
   284→      for (const alias of field.aliases) {
   285→        const normAlias = normalize(alias)
   286→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   287→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   288→          if (score > bestScore) {
   289→            bestMatch = excelCol.original
   290→            bestScore = score
   291→          }
   292→        }
   293→      }
   294→    }
   295→
   296→    if (bestMatch && bestScore >= 50) {
   297→      mapping[bestMatch] = field.key
   298→    }
   299→  }
   300→
   301→  return mapping
   302→}

   304→// Auto-detect the module from Excel column headers
   305→function autoDetectModule(columns: string[]): string | null {
   306→  const normalized = columns.map(normalize)
   307→  const scores: Record<string, number> = {}
   308→
   309→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   310→    scores[moduleKey] = 0
   311→    for (const field of template.fields) {
   312→      for (const alias of field.aliases) {
   313→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   314→          scores[moduleKey] += field.required ? 3 : 1
   315→        }
   316→      }
   317→    }
   318→  }
   319→
   320→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   321→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   322→}

   324→// Transform data using mapping, also handle customer name -> ID resolution
   325→function transformRow(
   326→  row: Record<string, unknown>,
   327→  mapping: Record<string, string>,
   328→  customers?: { id: string; name: string }[]
   329→): Record<string, unknown> {
   330→  const result: Record<string, unknown> = {}
   331→
   332→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   333→    let value = row[excelCol]
   334→
   335→    // Handle customer name -> ID resolution
   336→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   337→      const nameStr = String(value || '').trim().toLowerCase()
   338→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   339→      if (matched) {
   340→        result['customerId'] = matched.id
   341→      } else {
   342→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   343→      }
   344→      continue
   345→    }
   346→
   347→    // Handle numeric fields
   348→    if ([
   349→      'quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate',
   350→      'zigZagGrey80', 'zigZagRed80', 'zigZagYellow80',
   351→      'zigZagGrey60', 'zigZagRed60', 'zigZagYellow60',
   352→      'curveStone', 'chequreTile', 'transportationCharge',
   353→      'cement', 'dumbleGrey80', 'dumbleRed80', 'dumbleYellow80',
   354→      'quantityTon', 'totalAmount', 'paidAmount', 'remainingAmount',
   355→      'gst',
   356→    ].includes(fieldKey)) {
   357→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   358→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   359→      continue
   360→    }
   361→
   362→    // Handle date fields - try to parse various formats
   363→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   364→      // Excel may return dates as numeric serial numbers (e.g. 46178).
   365→      // Detect that case and convert to YYYY-MM-DD before string parsing.
   366→      if (typeof value === 'number' && Number.isFinite(value)) {
   367→        // Excel serial date: days since 1899-12-30
   368→        const ms = Math.round((value - 25569) * 86400 * 1000)
   369→        const d = new Date(ms)
   370→        if (!isNaN(d.getTime())) {
   371→          result[fieldKey] = d.toISOString().split('T')[0]
   372→          continue
   373→        }
   374→      }
   375→      // Excel may also return a Date object if cellDates:true was used.
   376→      if (value instanceof Date && !isNaN(value.getTime())) {
   377→        result[fieldKey] = value.toISOString().split('T')[0]
   378→        continue
   379→      }
   380→      result[fieldKey] = parseDate(String(value || ''))
   381→      continue
   382→    }
   383→
   384→    // Handle shift field - normalize
   385→    if (fieldKey === 'shift') {
   386→      const shiftStr = String(value || 'Morning').trim()
   387→      const lower = shiftStr.toLowerCase()
   388→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   389→        result[fieldKey] = 'Morning'
   390→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   391→        result[fieldKey] = 'Evening'
   392→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   393→        result[fieldKey] = 'Night'
   394→      } else {
   395→        result[fieldKey] = shiftStr
   396→      }
   397→      continue
   398→    }
   399→
   400→    // Handle payment type
   401→    if (fieldKey === 'paymentType') {
   402→      const ptStr = String(value || 'Cash').trim()
   403→      const lower = ptStr.toLowerCase()
   404→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   405→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   406→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   407→      else result[fieldKey] = ptStr
   408→      continue
   409→    }
   410→
   411→    // Handle expense category
   412→    if (fieldKey === 'category') {
   413→      const catStr = String(value || '').trim()
   414→      const lower = catStr.toLowerCase()
   415→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   416→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   417→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   418→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   419→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   420→      else result[fieldKey] = catStr
   421→      continue
   422→    }
   423→
   424→    // Handle status
   425→    if (fieldKey === 'status') {
   426→      const stStr = String(value || 'Pending').trim()
   427→      const lower = stStr.toLowerCase()
   428→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   429→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   430→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   431→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   432→      else result[fieldKey] = stStr
   433→      continue
   434→    }
   435→
   436→    result[fieldKey] = String(value || '')
   437→  }
   438→
   439→  // For orders/dispatch: calculate amount if not mapped
   440→  if (mapping) {
   441→    const targetModule = Object.values(mapping)
   442→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   443→      const qty = Number(result.quantity || 0)
   444→      const rate = Number(result.rate || 0)
   445→      result.amount = qty * rate
   446→    }
   447→  }
   448→
   449→  return result
   450→}

   452→// Parse various date formats to YYYY-MM-DD
   453→function parseDate(value: string): string {
   454→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   455→
   456→  const trimmed = value.trim()
   457→
   458→  // Already in YYYY-MM-DD format
   459→  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed
   460→
   461→  // DD-MM-YYYY or DD/MM/YYYY
   462→  const dmyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   463→  if (dmyMatch) {
   464→    const [, d, m, y] = dmyMatch
   465→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   466→  }
   467→
   468→  // DD-MM-YY or DD/MM/YY
   469→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2})$/)
   470→  if (dmyShortMatch) {
   471→    const [, d, m, y] = dmyShortMatch
   472→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   473→  }
   474→
   475→  // MM/DD/YYYY (US format) - only if first number > 12
   476→  const mdyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   477→  if (mdyMatch) {
   478→    const [, m, d, y] = mdyMatch
   479→    if (Number(m) > 12) {
   480→      return `${y}-${d.padStart(2, '0')}-${m.padStart(2, '0')}`
   481→    }
   482→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   483→  }
   484→
   485→  // Try native Date parsing as fallback
   486→  try {
   487→    const date = new Date(trimmed)
   488→    if (!isNaN(date.getTime())) {
   489→      return date.toISOString().split('T')[0]
   490→    }
   491→  } catch {}
   492→
   493→  return trimmed
   494→}

   825→                          <TableCell colSpan={template.fields.filter((f) => mappedFieldKeys.has(f.key)).length + 1} className="text-xs text-center text-muted-foreground">
   826→                            ... and {transformedData.length - 10} more rows
   827→                          </TableCell>
   828→                        </TableRow>
   829→                      )}
   830→                    </TableBody>
   831→                  </Table>
   832→                </div>
   833→              </ScrollArea>
   834→            </div>
   835→          )}
   836→
   837→          {/* Import result is now shown in a dedicated popup after the
   838→              import dialog closes — see the result Dialog below. */}
   839→        </div>
   840→
   841→        <DialogFooter>
   842→          <Button variant="outline" onClick={() => handleClose()}>Cancel</Button>
   843→          <Button
   844→            onClick={handleImport}
   845→            disabled={transformedData.length === 0 || importing || unmappedRequired.length > 0}
   846→            className="bg-emerald-600 hover:bg-emerald-700"
   847→          >
   848→            {importing ? (
   849→              <>
   850→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   851→                Importing...
   852→              </>
   853→            ) : (
   854→              <>
   855→                <Upload className="h-4 w-4 mr-2" />
   856→                Import {transformedData.length} Rows
   857→              </>
   858→            )}
   859→          </Button>
   860→        </DialogFooter>
   861→      </DialogContent>
   862→    </Dialog>
   863→
   864→    {/* Result popup — opens after import finishes (success, partial, or error). */}
   865→    <Dialog open={resultOpen} onOpenChange={(open) => { if (!open) closeResultPopup() }}>
   866→      <DialogContent className="max-w-2xl max-h-[85vh] flex flex-col">
   867→        <DialogHeader>
   868→          <DialogTitle className="flex items-center gap-2">
   869→            {result && result.errors && result.errors.length > 0 ? (
   870→              <>
   871→                <AlertCircle className="h-5 w-5 text-destructive" />
   872→                <span className="text-destructive">
   873→                  {result.imported > 0 ? 'Partial Import' : 'Import Failed'}
   874→                </span>
   875→              </>
   876→            ) : (
   877→              <>
   878→                <CheckCircle2 className="h-5 w-5 text-emerald-600" />
   879→                <span className="text-emerald-700 dark:text-emerald-400">Import Successful</span>
   880→              </>
   881→            )}
   882→          </DialogTitle>
   883→          <DialogDescription>
   884→            {template.label} import result
   885→          </DialogDescription>
   886→        </DialogHeader>
   887→
   888→        {result && (
   889→          <div className="space-y-4 py-2">
   890→            {/* Summary numbers */}
   891→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   892→              <div className="rounded-lg border bg-emerald-50 dark:bg-emerald-900/20 p-3 text-center">
   893→                <div className="text-2xl font-bold text-emerald-700 dark:text-emerald-400">{result.imported}</div>
   894→                <div className="text-xs text-muted-foreground mt-1">Imported</div>
   895→              </div>
   896→              <div className="rounded-lg border bg-muted/50 p-3 text-center">
   897→                <div className="text-2xl font-bold">{result.total}</div>
   898→                <div className="text-xs text-muted-foreground mt-1">Total Rows</div>
   899→              </div>
   900→              <div className="rounded-lg border bg-amber-50 dark:bg-amber-900/20 p-3 text-center">
   901→                <div className="text-2xl font-bold text-amber-700 dark:text-amber-400">
   902→                  {result.total - result.imported}
   903→                </div>
   904→                <div className="text-xs text-muted-foreground mt-1">Skipped</div>
   905→              </div>
   906→              <div className="rounded-lg border bg-muted/50 p-3 text-center">
   907→                <div className="text-2xl font-bold">
   908→                  {result.duplicatesSkipped || 0}
   909→                </div>
   910→                <div className="text-xs text-muted-foreground mt-1">Duplicates</div>
   911→              </div>
   912→            </div>
   913→
   914→            {/* Status message */}
   915→            <Alert variant={result.errors && result.errors.length > 0 ? 'destructive' : 'default'}>
   916→              <AlertDescription>
   917→                {result.imported === 0 ? (
   918→                  <span>
   919→                    No rows were imported.{' '}
   920→                    {result.errors && result.errors.length > 0
   921→                      ? `Reasons are listed below.`
   922→                      : 'Please check your file and column mapping, then try again.'}
   923→                  </span>
   924→                ) : result.imported === result.total ? (
   925→                  <span>All {result.imported} row(s) imported successfully. The list has been refreshed.</span>
   926→                ) : (
   927→                  <span>
   928→                    {result.imported} of {result.total} row(s) imported.{' '}
   929→                    {result.duplicatesSkipped && result.duplicatesSkipped > 0
   930→                      ? `${result.duplicatesSkipped} duplicate(s) skipped. `
   931→                      : ''}
   932→                    {result.errors && result.errors.length > 0
   933→                      ? `${result.errors.length} error(s) listed below.`
   934→                      : ''}
   935→                  </span>
   936→                )}
   937→              </AlertDescription>
   938→            </Alert>
   939→
   940→            {/* Error list */}
   941→            {result.errors && result.errors.length > 0 && (
   942→              <div className="space-y-2">
   943→                <h4 className="text-sm font-medium flex items-center gap-2">
   944→                  <AlertCircle className="h-4 w-4 text-destructive" />
   945→                  Errors ({result.errors.length})
   946→                </h4>
   947→                <ScrollArea className="max-h-64 rounded-md border">
   948→                  <ul className="text-xs space-y-1 p-3">
   949→                    {result.errors.map((err, i) => (
   950→                      <li key={i} className="text-destructive flex gap-2">
   951→                        <span className="font-mono text-muted-foreground shrink-0">#{i + 1}</span>
   952→                        <span>{err}</span>
   953→                      </li>
   954→                    ))}
   955→                  </ul>
   956→                </ScrollArea>
   957→              </div>
   958→            )}
   959→          </div>
   960→        )}
   961→
   962→        <DialogFooter>
   963→          <Button onClick={closeResultPopup} className="bg-emerald-600 hover:bg-emerald-700">
   964→            Close
   965→          </Button>
   966→        </DialogFooter>
   967→      </DialogContent>
   968→    </Dialog>
   969→    </>
   970→  )
   971→}
