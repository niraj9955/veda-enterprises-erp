     1→'use client'
     2→
     3→import * as React from 'react'
     4→import { api } from '@/lib/api'
     5→import { toast } from '@/hooks/use-toast'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Badge } from '@/components/ui/badge'
    10→import { Skeleton } from '@/components/ui/skeleton'
    11→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from '@/components/ui/select'
    19→import {
    20→  Table,
    21→  TableBody,
    22→  TableCell,
    23→  TableHead,
    24→  TableHeader,
    25→  TableRow,
    26→} from '@/components/ui/table'
    27→import {
    28→  Dialog,
    29→  DialogContent,

    31→  DialogFooter,
    32→  DialogHeader,
    33→  DialogTitle,
    34→} from '@/components/ui/dialog'
    35→import {
    36→  AlertDialog,
    37→  AlertDialogAction,
    38→  AlertDialogCancel,
    39→  AlertDialogContent,
    40→  AlertDialogDescription,
    41→  AlertDialogFooter,
    42→  AlertDialogHeader,
    43→  AlertDialogTitle,
    44→} from '@/components/ui/alert-dialog'
    45→import {
    46→  ClipboardList,
    47→  Plus,
    48→  Pencil,
    49→  Trash2,
    50→  Loader2,
    51→  IndianRupee,
    52→  Upload,
    53→  Search,
    54→  X,
    55→} from 'lucide-react'

    58→
    59→// ── Types ───────────────────────────────────────────────────────────────────
    60→
    61→interface Customer {
    62→  id: string
    63→  name: string
    64→}
    65→
    66→interface OrderItem {
    67→  description: string
    68→  hsn: string
    69→  quantity: number
    70→  unit: string
    71→  rate: number
    72→  amount: number
    73→}
    74→
    75→interface Order {
    76→  id: string
    77→  orderNumber: string
    78→  customerId: string
    79→  brickType: string
    80→  quantity: number
    81→  rate: number
    82→  amount: number
    83→  items?: OrderItem[]
    84→  deliveryDate: string
    85→  status: string
    86→  createdAt: string
    87→  customer: { id: string; name: string }
    88→}
    89→
    90→interface OrderFormData {
    91→  customerId: string
    92→  brickType: string
    93→  quantity: number | string
    94→  rate: number | string
    95→  deliveryDate: string
    96→  status: string
    97→  items: OrderItem[]
    98→}
    99→
   100→// ── Constants ───────────────────────────────────────────────────────────────
   101→
   102→const BRICK_TYPES = ['Red Brick', 'Fly Ash Brick', 'Cement Brick', 'Hollow Block'] as const
   103→
   104→const ORDER_STATUSES = ['Pending', 'Processing', 'Delivered', 'Cancelled'] as const
   105→
   106→const STATUS_COLORS: Record<string, string> = {
   107→  Pending: 'bg-amber-100 text-amber-700 border-amber-200',
   108→  Processing: 'bg-sky-100 text-sky-700 border-sky-200',
   109→  Delivered: 'bg-emerald-100 text-emerald-700 border-emerald-200',
   110→  Cancelled: 'bg-rose-100 text-rose-700 border-rose-200',
   111→}
   112→
   113→// ── Helpers ─────────────────────────────────────────────────────────────────
   114→
   115→const formatCurrency = (amount: number): string =>
   116→  new Intl.NumberFormat('en-IN', {
   117→    style: 'currency',
   118→    currency: 'INR',
   119→    minimumFractionDigits: 0,
   120→    maximumFractionDigits: 2,
   121→  }).format(amount)
   122→
   123→const formatDate = (dateStr: string): string =>
   124→  new Date(dateStr).toLocaleDateString('en-IN', {
   125→    day: '2-digit',
   126→    month: 'short',
   127→    year: 'numeric',
   128→  })
   129→
   130→const emptyForm: OrderFormData = {
   131→  customerId: '',
   132→  brickType: '',
   133→  quantity: '',
   134→  rate: '',
   135→  deliveryDate: '',
   136→  status: 'Pending',
   137→  items: [],
   138→}
   139→
   140→// Default new line item — used when user clicks "Add Item"
   141→const newItem = (): OrderItem => ({
   142→  description: '', hsn: '', quantity: 1, unit: 'pcs', rate: 0, amount: 0,
   143→})
   144→
   145→// ── Component ───────────────────────────────────────────────────────────────
   146→
   147→export function OrderModule() {
   148→  // State
   149→  const [orders, setOrders] = React.useState<Order[]>([])
   150→  const [search, setSearch] = React.useState('')
   151→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   152→  const [customers, setCustomers] = React.useState<Customer[]>([])
   153→  const [loading, setLoading] = React.useState(true)
   154→
   155→  // Create dialog states
   156→  const [createOpen, setCreateOpen] = React.useState(false)
   157→  const [formData, setFormData] = React.useState<OrderFormData>(emptyForm)
   158→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   159→
   160→  // Edit status dialog states
   161→  const [editStatusOpen, setEditStatusOpen] = React.useState(false)
   162→  const [editTarget, setEditTarget] = React.useState<Order | null>(null)
   163→  const [editStatus, setEditStatus] = React.useState('')
   164→  const [editSubmitting, setEditSubmitting] = React.useState(false)
   165→
   166→  // Excel import
   167→  const [importOpen, setImportOpen] = React.useState(false)
   168→
   169→  // Delete confirmation
   170→  const [deleteTarget, setDeleteTarget] = React.useState<Order | null>(null)
   171→  const [deleting, setDeleting] = React.useState(false)
   172→
   173→  // ── Computed amount ─────────────────────────────────────────────────────
   174→  // If items[] are present, total amount = sum of item amounts.
   175→  // Otherwise fall back to top-level qty * rate.
   176→  const computedAmount = React.useMemo(() => {
   177→    if (formData.items.length > 0) {
   178→      return formData.items.reduce((s, it) => s + (Number(it.amount) || 0), 0)
   179→    }
   180→    const qty = Number(formData.quantity) || 0
   181→    const rate = Number(formData.rate) || 0
   182→    return qty * rate
   183→  }, [formData.quantity, formData.rate, formData.items])
   184→
   185→  // ── Item management ──────────────────────────────────────────────────────
   186→  const updateItem = (idx: number, field: keyof OrderItem, value: string | number) => {
   187→    setFormData((prev) => {
   188→      const items = [...prev.items]
   189→      const item = { ...items[idx] }
   190→      if (field === 'description' || field === 'hsn' || field === 'unit') {
   191→        (item as any)[field] = value as string
   192→      } else {
   193→        (item as any)[field] = Number(value) || 0
   194→      }
   195→      item.amount = item.quantity * item.rate
   196→      items[idx] = item
   197→      return { ...prev, items }
   198→    })
   199→  }
   200→  const addItem = () => setFormData((prev) => ({ ...prev, items: [...prev.items, newItem()] }))
   201→  const removeItem = (idx: number) =>
   202→    setFormData((prev) => ({ ...prev, items: prev.items.filter((_, i) => i !== idx) }))
   203→
   204→  // ── Fetch orders ────────────────────────────────────────────────────────
   205→  const fetchOrders = React.useCallback(async () => {
   206→    setLoading(true)
   207→    try {
   208→      const res = await api.getOrders()
   209→      const orderList = (res.orders as Order[]).sort(
   210→        (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   211→      )
   212→      setOrders(orderList)
   213→    } catch (err) {
   214→      toast({
   215→        title: 'Error',
   216→        description: err instanceof Error ? err.message : 'Failed to fetch orders',
   217→        variant: 'destructive',
   218→      })
   219→    } finally {
   220→      setLoading(false)
   221→    }
   222→  }, [])
   223→
   224→  // ── Fetch customers (for dropdown) ─────────────────────────────────────
   225→  const fetchCustomers = React.useCallback(async () => {
   226→    try {
   227→      const res = await api.getCustomers()
   228→      setCustomers(res.customers as Customer[])
   229→    } catch {
   230→      // Silently fail — customer dropdown will just be empty
   231→    }
   232→  }, [])
   233→
   234→  // Debounced search
   235→  React.useEffect(() => {
   236→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   237→    return () => clearTimeout(t)
   238→  }, [search])
   239→
   240→  // Client-side filter
   241→  const filteredOrders = React.useMemo(() => {
   242→    if (!debouncedSearch.trim()) return orders
   243→    const q = debouncedSearch.toLowerCase()
   244→    return orders.filter((item: any) =>
   245→      ['orderNumber', 'brickType', 'status', 'deliveryDate'].some((f) =>
   246→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   247→      )
   248→    )
   249→  }, [orders, debouncedSearch])
   250→
   251→  React.useEffect(() => {
   252→    fetchOrders()
   253→    fetchCustomers()
   254→  }, [fetchOrders, fetchCustomers])
   255→
   256→  // ── Create dialog handlers ──────────────────────────────────────────────
   257→  const openCreateDialog = () => {
   258→    setFormData(emptyForm)
   259→    setCreateOpen(true)
   260→  }
   261→
   262→  const handleFormChange = (field: keyof OrderFormData, value: string) => {
   263→    setFormData((prev) => ({ ...prev, [field]: value }))
   264→  }
   265→
   266→  const handleCreateSubmit = async () => {
   267→    if (!formData.customerId) {
   268→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   269→      return
   270→    }
   271→    if (!formData.brickType) {
   272→      toast({ title: 'Validation Error', description: 'Please select a brick type', variant: 'destructive' })
   273→      return
   274→    }
   275→    if (!formData.quantity || Number(formData.quantity) <= 0) {
   276→      toast({ title: 'Validation Error', description: 'Quantity must be greater than 0', variant: 'destructive' })
   277→      return
   278→    }
   279→    if (!formData.rate || Number(formData.rate) <= 0) {
   280→      toast({ title: 'Validation Error', description: 'Rate must be greater than 0', variant: 'destructive' })
   281→      return
   282→    }
   283→    if (!formData.deliveryDate) {
   284→      toast({ title: 'Validation Error', description: 'Delivery date is required', variant: 'destructive' })
   285→      return
   286→    }
   287→
   288→    // Either items[] OR brickType+qty+rate must be present
   289→    const hasItems = formData.items.length > 0 && formData.items.some((i) => i.description.trim())
   290→    if (!hasItems && (!formData.brickType || !formData.quantity || !formData.rate)) {
   291→      toast({
   292→        title: 'Validation Error',
   293→        description: 'Add at least one item, OR fill brick type + quantity + rate',
   294→        variant: 'destructive',
   295→      })
   296→      return
   297→    }
   298→
   299→    setFormSubmitting(true)
   300→    try {
   301→      const payload = {
   302→        customerId: formData.customerId,
   303→        brickType: formData.brickType,
   304→        quantity: Number(formData.quantity) || 0,
   305→        rate: Number(formData.rate) || 0,
   306→        amount: computedAmount,
   307→        deliveryDate: formData.deliveryDate,
   308→        status: formData.status,
   309→        items: formData.items.filter((i) => i.description.trim()),
   310→      }
   311→
   312→      await api.createOrder(payload)
   313→      toast({ title: 'Success', description: 'Order created successfully' })
   314→      setCreateOpen(false)
   315→      setFormData(emptyForm)
   316→      fetchOrders()
   317→    } catch (err) {
   318→      toast({
   319→        title: 'Error',
   320→        description: err instanceof Error ? err.message : 'Failed to create order',
   321→        variant: 'destructive',
   322→      })
   323→    } finally {
   324→      setFormSubmitting(false)
   325→    }
   326→  }
   327→
   328→  // ── Edit status handlers ────────────────────────────────────────────────
   329→  const openEditStatusDialog = (order: Order) => {
   330→    setEditTarget(order)
   331→    setEditStatus(order.status)
   332→    setEditStatusOpen(true)
   333→  }
   334→
   335→  const handleEditStatusSubmit = async () => {
   336→    if (!editTarget) return
   337→    setEditSubmitting(true)
   338→    try {
   339→      await api.updateOrder(editTarget.id, { status: editStatus })
   340→      toast({ title: 'Success', description: 'Order status updated successfully' })
   341→      setEditStatusOpen(false)
   342→      setEditTarget(null)
   343→      setEditStatus('')
   344→      fetchOrders()
   345→    } catch (err) {
   346→      toast({
   347→        title: 'Error',
   348→        description: err instanceof Error ? err.message : 'Failed to update order status',
   349→        variant: 'destructive',
   350→      })
   351→    } finally {
   352→      setEditSubmitting(false)
   353→    }
   354→  }
   355→
   356→  // ── Delete handler ──────────────────────────────────────────────────────
   357→  const handleDelete = async () => {
   358→    if (!deleteTarget) return
   359→    setDeleting(true)
   360→    try {
   361→      await api.deleteOrder(deleteTarget.id)
   362→      toast({ title: 'Success', description: 'Order deleted successfully' })
   363→      setDeleteTarget(null)
   364→      fetchOrders()
   365→    } catch (err) {
   366→      toast({
   367→        title: 'Error',
   368→        description: err instanceof Error ? err.message : 'Failed to delete order',
   369→        variant: 'destructive',
   370→      })
   371→    } finally {
   372→      setDeleting(false)
   373→    }
   374→  }
   375→
   376→  // ── Render: Loading skeletons ───────────────────────────────────────────
   377→  const renderSkeletons = () =>
   378→    Array.from({ length: 5 }).map((_, i) => (
   379→      <TableRow key={i}>
   380→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   381→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   382→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   383→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   384→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   385→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   386→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   387→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   388→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   389→      </TableRow>
   390→    ))
   391→
   392→  // ── Render: Status badge ────────────────────────────────────────────────
   393→  const renderStatusBadge = (status: string) => (
   394→    <Badge className={STATUS_COLORS[status] || 'bg-gray-100 text-gray-700 border-gray-200'}>
   395→      {status}
   396→    </Badge>
   397→  )
   398→
   399→  // ── Render: Create Order dialog ─────────────────────────────────────────
   400→  const renderCreateDialog = () => (
   401→    <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   402→      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto">
   403→        <DialogHeader>
   404→          <DialogTitle>Create Order</DialogTitle>
   405→          <DialogDescription>
   406→            Fill in the details to create a new order. You can add multiple items below the brick type.
   407→          </DialogDescription>
   408→        </DialogHeader>
   409→
   410→        <div className="grid gap-4 py-2">
   411→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   412→          <CustomerSearchInput
   413→            value={formData.customerId}
   414→            onSelect={(c) => handleFormChange('customerId', c.id)}
   415→            onClear={() => handleFormChange('customerId', '')}
   416→            required
   417→            label="Customer"
   418→            placeholder="Type customer name or mobile to search..."
   419→          />
   420→
   421→          {/* Brick Type — now optional since items[] can replace it */}
   422→          <div className="grid gap-2">
   423→            <Label htmlFor="order-brick">
   424→              Brick Type <span className="text-muted-foreground text-xs font-normal">(optional when items are added below)</span>
   425→            </Label>
   426→            <Select
   427→              value={formData.brickType}
   428→              onValueChange={(val) => handleFormChange('brickType', val)}
   429→            >
   430→              <SelectTrigger className="w-full">
   431→                <SelectValue placeholder="Select brick type (or skip if using items)" />
   432→              </SelectTrigger>
   433→              <SelectContent>
   434→                {BRICK_TYPES.map((type) => (
   435→                  <SelectItem key={type} value={type}>
   436→                    {type}
   437→                  </SelectItem>
   438→                ))}
   439→              </SelectContent>
   440→            </Select>
   441→          </div>
   442→
   443→          {/* Items section — multi-line item entry */}
   444→          <div className="grid gap-2">
   445→            <div className="flex items-center justify-between">
   446→              <Label>Items <span className="text-muted-foreground text-xs font-normal">(add multiple products / brick types)</span></Label>
   447→              <Button type="button" size="sm" variant="outline" onClick={addItem}>
   448→                <Plus className="h-3 w-3 mr-1" /> Add Item
   449→              </Button>
   450→            </div>
   451→            <div className="border rounded-md p-2 space-y-2 bg-muted/20">
   452→              {formData.items.length === 0 ? (
   453→                <p className="text-xs text-muted-foreground text-center py-2">
   454→                  No items added. Use "Add Item" to add multiple products, OR skip this section and use brick type + qty + rate above.
   455→                </p>
   456→              ) : (
   457→                <>
   458→                  <div className="hidden md:grid grid-cols-12 gap-2 text-xs font-semibold text-muted-foreground px-1">
   459→                    <div className="col-span-5">Description</div>
   460→                    <div className="col-span-2">Qty</div>
   461→                    <div className="col-span-2">Unit</div>
   462→                    <div className="col-span-2">Rate (₹)</div>
   463→                    <div className="col-span-1">Amount</div>
   464→                  </div>
   465→                  {formData.items.map((item, idx) => (
   466→                    <div key={idx} className="grid grid-cols-12 gap-2 items-center">
   467→                      <Input
   468→                        className="col-span-12 md:col-span-5"
   469→                        placeholder="Item description"
   470→                        value={item.description}
   471→                        onChange={(e) => updateItem(idx, 'description', e.target.value)}
   472→                      />
   473→                      <Input
   474→                        type="number"
   475→                        className="col-span-4 md:col-span-2"
   476→                        placeholder="Qty"
   477→                        value={item.quantity}
   478→                        onChange={(e) => updateItem(idx, 'quantity', e.target.value)}
   479→                      />
   480→                      <Input
   481→                        className="col-span-4 md:col-span-2"
   482→                        placeholder="Unit"
   483→                        value={item.unit}
   484→                        onChange={(e) => updateItem(idx, 'unit', e.target.value)}
   485→                      />
   486→                      <Input
   487→                        type="number"
   488→                        className="col-span-4 md:col-span-2"
   489→                        placeholder="Rate"
   490→                        value={item.rate}
   491→                        onChange={(e) => updateItem(idx, 'rate', e.target.value)}
   492→                      />
   493→                      <div className="col-span-11 md:col-span-1 flex items-center gap-1">
   494→                        <span className="text-xs font-medium flex-1">₹{(item.amount || 0).toLocaleString('en-IN')}</span>
   495→                        <Button
   496→                          type="button"
   497→                          size="icon"
   498→                          variant="ghost"
   499→                          className="h-7 w-7 text-destructive shrink-0"
   500→                          onClick={() => removeItem(idx)}
   501→                        >
   502→                          <X className="h-3 w-3" />
   503→                        </Button>
   504→                      </div>
   505→                    </div>
   506→                  ))}
   507→                </>
   508→              )}
   509→              {formData.items.length > 0 && (
   510→                <div className="flex justify-end pt-1 border-t text-sm font-medium">
   511→                  Total: ₹{computedAmount.toLocaleString('en-IN')}
   512→                </div>
   513→              )}
   514→            </div>
   515→          </div>
   516→
   517→          {/* Quantity & Rate — kept for backward compat / quick single-item orders */}
   518→          <div className="grid grid-cols-2 gap-4">
   519→            <div className="grid gap-2">
   520→              <Label htmlFor="order-qty">
   521→                Quantity <span className="text-muted-foreground text-xs font-normal">(if no items)</span>
   522→              </Label>
   523→              <Input
   524→                id="order-qty"
   525→                type="number"
   526→                placeholder="0"
   527→                min="1"
   528→                value={formData.quantity}
   529→                onChange={(e) => handleFormChange('quantity', e.target.value)}
   530→              />
   531→            </div>
   532→            <div className="grid gap-2">
   533→              <Label htmlFor="order-rate">
   534→                Rate (₹) <span className="text-muted-foreground text-xs font-normal">(if no items)</span>
   535→              </Label>
   536→              <div className="relative">
   537→                <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   538→                <Input
   539→                  id="order-rate"
   540→                  type="number"
   541→                  placeholder="0"
   542→                  min="1"
   543→                  className="pl-9"
   544→                  value={formData.rate}
   545→                  onChange={(e) => handleFormChange('rate', e.target.value)}
   546→                />
   547→              </div>
   548→            </div>
   549→          </div>
   550→
   551→          {/* Amount (auto-calculated) */}
   552→          <div className="grid gap-2">
   553→            <Label>Amount (₹)</Label>
   554→            <div className="flex h-9 items-center rounded-md border bg-muted/50 px-3 text-sm font-medium">
   555→              {formatCurrency(computedAmount)}
   556→            </div>
   557→          </div>
   558→
   559→          {/* Delivery Date */}
   560→          <div className="grid gap-2">
   561→            <Label htmlFor="order-date">
   562→              Delivery Date <span className="text-destructive">*</span>
   563→            </Label>
   564→            <Input
   565→              id="order-date"
   566→              type="date"
   567→              value={formData.deliveryDate}
   568→              onChange={(e) => handleFormChange('deliveryDate', e.target.value)}
   569→            />
   570→          </div>
   571→
   572→          {/* Status */}
   573→          <div className="grid gap-2">
   574→            <Label htmlFor="order-status">Status</Label>
   575→            <Select
   576→              value={formData.status}
   577→              onValueChange={(val) => handleFormChange('status', val)}
   578→            >
   579→              <SelectTrigger className="w-full">
   580→                <SelectValue placeholder="Select status" />
   581→              </SelectTrigger>
   582→              <SelectContent>
   583→                {ORDER_STATUSES.map((status) => (
   584→                  <SelectItem key={status} value={status}>
   585→                    {status}
   586→                  </SelectItem>
   587→                ))}
   588→              </SelectContent>
   589→            </Select>
   590→          </div>
   591→        </div>
   592→
   593→        <DialogFooter>
   594→          <Button
   595→            variant="outline"
   596→            onClick={() => setCreateOpen(false)}
   597→            disabled={formSubmitting}
   598→          >
   599→            Cancel
   600→          </Button>
   601→          <Button
   602→            onClick={handleCreateSubmit}
   603→            disabled={formSubmitting}
   604→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   605→          >
   606→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   607→            Create Order
   608→          </Button>
   609→        </DialogFooter>
   610→      </DialogContent>
   611→    </Dialog>
   612→  )
   613→
   614→  // ── Render: Edit Status dialog ──────────────────────────────────────────
   615→  const renderEditStatusDialog = () => (
   616→    <Dialog open={editStatusOpen} onOpenChange={setEditStatusOpen}>
   617→      <DialogContent className="sm:max-w-md">
   618→        <DialogHeader>
   619→          <DialogTitle>Update Order Status</DialogTitle>
   620→          <DialogDescription>
   621→            Change the status for order <strong>{editTarget?.orderNumber}</strong>
   622→          </DialogDescription>
   623→        </DialogHeader>
   624→
   625→        <div className="grid gap-4 py-2">
   626→          <div className="grid gap-2">
   627→            <Label>Current Status</Label>
   628→            <div>{editTarget ? renderStatusBadge(editTarget.status) : null}</div>
   629→          </div>
   630→          <div className="grid gap-2">
   631→            <Label htmlFor="edit-status">New Status</Label>
   632→            <Select value={editStatus} onValueChange={setEditStatus}>
   633→              <SelectTrigger className="w-full">
   634→                <SelectValue placeholder="Select status" />
   635→              </SelectTrigger>
   636→              <SelectContent>
   637→                {ORDER_STATUSES.map((status) => (
   638→                  <SelectItem key={status} value={status}>
   639→                    {status}
   640→                  </SelectItem>
   641→                ))}
   642→              </SelectContent>
   643→            </Select>
   644→          </div>
   645→        </div>
   646→
   647→        <DialogFooter>
   648→          <Button
   649→            variant="outline"
   650→            onClick={() => setEditStatusOpen(false)}
   651→            disabled={editSubmitting}
   652→          >
   653→            Cancel
   654→          </Button>
   655→          <Button
   656→            onClick={handleEditStatusSubmit}
   657→            disabled={editSubmitting}
   658→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   659→          >
   660→            {editSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   661→            Update Status
   662→          </Button>
   663→        </DialogFooter>
   664→      </DialogContent>
   665→    </Dialog>
   666→  )
   667→
   668→  // ── Render: Delete confirmation ─────────────────────────────────────────
   669→  const renderDeleteDialog = () => (
   670→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   671→      <AlertDialogContent>
   672→        <AlertDialogHeader>
   673→          <AlertDialogTitle>Delete Order</AlertDialogTitle>
   674→          <AlertDialogDescription>
   675→            Are you sure you want to delete order <strong>{deleteTarget?.orderNumber}</strong>? This
   676→            action cannot be undone.
   677→          </AlertDialogDescription>
   678→        </AlertDialogHeader>
   679→        <AlertDialogFooter>
   680→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   681→          <AlertDialogAction
   682→            onClick={handleDelete}
   683→            disabled={deleting}
   684→            className="bg-destructive text-white hover:bg-destructive/90"
   685→          >
   686→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   687→            Delete
   688→          </AlertDialogAction>
   689→        </AlertDialogFooter>
   690→      </AlertDialogContent>
   691→    </AlertDialog>
   692→  )
   693→
   694→  // ── Render: Main ────────────────────────────────────────────────────────
   695→  return (
   696→    <div className="space-y-6">
   697→      {/* Header */}
   698→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   699→        <div className="flex items-center gap-3">
   700→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   701→            <ClipboardList className="size-5" />
   702→          </div>
   703→          <div>
   704→            <h2 className="text-2xl font-bold tracking-tight">Order Management</h2>
   705→            <p className="text-sm text-muted-foreground">
   706→              Track and manage customer orders
   707→            </p>
   708→          </div>
   709→        </div>
   710→        <div className="flex gap-2 w-full sm:w-auto">
   711→          <Button
   712→            variant="outline"
   713→            onClick={() => setImportOpen(true)}
   714→            className="w-full sm:w-auto"
   715→          >
   716→            <Upload className="size-4 mr-2" />
   717→            Import Excel
   718→          </Button>
   719→          <Button
   720→            onClick={openCreateDialog}
   721→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   722→          >
   723→            <Plus className="size-4" />
   724→            Create Order
   725→          </Button>
   726→        </div>
   727→      </div>
   728→
   729→      {/* Search */}
   730→      <Card>
   731→        <CardContent className="pt-6">
   732→          <div className="relative">
   733→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   734→            <Input
   735→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   736→              value={search}
   737→              onChange={(e) => setSearch(e.target.value)}
   738→              className="pl-9"
   739→            />
   740→          </div>
   741→        </CardContent>
   742→      </Card>
   743→
   744→      {/* Table */}
   745→      <Card>
   746→        <CardHeader>
   747→          <CardTitle className="flex items-center justify-between">
   748→            <span>Orders</span>
   749→            <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   750→              {filteredOrders.length} of {orders.length} record{orders.length !== 1 ? 's' : ''}
   751→            </Badge>
   752→          </CardTitle>
   753→        </CardHeader>
   754→        <CardContent>
   755→          <div className="max-h-[60vh] overflow-auto rounded-md border">
   756→            <Table>
   757→              <TableHeader className="sticky top-0 bg-background z-10">
   758→                <TableRow>
   759→                  <TableHead>Order No.</TableHead>
   760→                  <TableHead>Customer</TableHead>
   761→                  <TableHead>Brick Type</TableHead>
   762→                  <TableHead className="text-right">Quantity</TableHead>
   763→                  <TableHead className="text-right">Rate (₹)</TableHead>
   764→                  <TableHead className="text-right">Amount (₹)</TableHead>
   765→                  <TableHead>Delivery Date</TableHead>
   766→                  <TableHead>Status</TableHead>
   767→                  <TableHead className="text-right">Actions</TableHead>
   768→                </TableRow>
   769→              </TableHeader>
   770→              <TableBody>
   771→                {loading ? (
   772→                  renderSkeletons()
   773→                ) : filteredOrders.length === 0 ? (
   774→                  <TableRow>
   775→                    <TableCell colSpan={9} className="h-32 text-center text-muted-foreground">
   776→                      No orders yet. Click &quot;Create Order&quot; to get started.
   777→                    </TableCell>
   778→                  </TableRow>
   779→                ) : (
   780→                  filteredOrders.map((order) => (
   781→                    <TableRow key={order.id}>
   782→                      <TableCell className="font-medium whitespace-nowrap">
   783→                        {order.orderNumber}
   784→                      </TableCell>
   785→                      <TableCell>{order.customer?.name || '—'}</TableCell>
   786→                      <TableCell>
   787→                        <Badge variant="outline">{order.brickType}</Badge>
   788→                        {order.items && order.items.length > 0 && (
   789→                          <Badge variant="secondary" className="ml-1 text-[10px]">
   790→                            {order.items.length} item{order.items.length > 1 ? 's' : ''}
   791→                          </Badge>
   792→                        )}
   793→                      </TableCell>
   794→                      <TableCell className="text-right">
   795→                        {new Intl.NumberFormat('en-IN').format(order.quantity)}
   796→                      </TableCell>
   797→                      <TableCell className="text-right whitespace-nowrap">
   798→                        {formatCurrency(order.rate)}
   799→                      </TableCell>
   800→                      <TableCell className="text-right font-medium whitespace-nowrap">
   801→                        {formatCurrency(order.amount)}
   802→                      </TableCell>
   803→                      <TableCell className="whitespace-nowrap">
   804→                        {formatDate(order.deliveryDate)}
   805→                      </TableCell>
   806→                      <TableCell>{renderStatusBadge(order.status)}</TableCell>
   807→                      <TableCell className="text-right">
   808→                        <div className="flex items-center justify-end gap-1">
   809→                          <Button
   810→                            variant="ghost"
   811→                            size="icon"
   812→                            onClick={() => openEditStatusDialog(order)}
   813→                            title="Edit Status"
   814→                            className="text-sky-600 hover:text-sky-700 hover:bg-sky-50"
   815→                          >
   816→                            <Pencil className="size-4" />
   817→                          </Button>
   818→                          <Button
   819→                            variant="ghost"
   820→                            size="icon"
   821→                            onClick={() => setDeleteTarget(order)}
   822→                            title="Delete Order"
   823→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   824→                          >
   825→                            <Trash2 className="size-4" />
   826→                          </Button>
   827→                        </div>
   828→                      </TableCell>
   829→                    </TableRow>
   830→                  ))
   831→                )}
   832→              </TableBody>
   833→            </Table>
   834→          </div>
   835→        </CardContent>
   836→      </Card>
   837→
   838→      {/* Dialogs */}
   839→      {renderCreateDialog()}
   840→      {renderEditStatusDialog()}
   841→      {renderDeleteDialog()}
   842→
   843→      <ExcelImport module="orders" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchOrders} />
   844→    </div>
   845→  )
   846→}
   847→
   848→export default OrderModule
