   128→// ── Helpers ─────────────────────────────────────────────────────────────────
   129→
   130→const formatDate = (dateStr: string): string => {
   131→  if (!dateStr) return '—'
   132→  const date = new Date(dateStr + 'T00:00:00')
   133→  return date.toLocaleDateString('en-IN', {
   134→    day: '2-digit',
   135→    month: 'short',
   136→    year: 'numeric',
   137→  })
   138→}

   140→// ── Component ───────────────────────────────────────────────────────────────
   141→
   142→export function DispatchModule() {
   143→  // ── State ─────────────────────────────────────────────────────────────
   144→  const [dispatches, setDispatches] = React.useState<Dispatch[]>([])
   145→  const [loading, setLoading] = React.useState(true)
   146→
   147→  // Create dialog
   148→  const [createOpen, setCreateOpen] = React.useState(false)
   149→  const [formData, setFormData] = React.useState<DispatchFormData>(emptyForm)
   150→  const [submitting, setSubmitting] = React.useState(false)
   151→
   152→  // Dropdown data
   153→  const [customers, setCustomers] = React.useState<Customer[]>([])
   154→  const [orders, setOrders] = React.useState<Order[]>([])
   155→  const [dropdownsLoading, setDropdownsLoading] = React.useState(false)
   156→
   157→  // Challan dialog
   158→  const [challanDispatch, setChallanDispatch] = React.useState<Dispatch | null>(null)
   159→
   160→  // Excel import
   161→  const [importOpen, setImportOpen] = React.useState(false)
   162→
   163→  // Delete confirmation
   164→  const [deleteTarget, setDeleteTarget] = React.useState<Dispatch | null>(null)
   165→  const [deleting, setDeleting] = React.useState(false)
   166→
   167→  // ── Fetch dispatches ──────────────────────────────────────────────────
   168→  const fetchDispatches = React.useCallback(async () => {
   169→    setLoading(true)
   170→    try {
   171→      const res = await api.getDispatches()
   172→      setDispatches(res.dispatches as Dispatch[])
   173→    } catch (err) {
   174→      toast({
   175→        title: 'Error',
   176→        description: err instanceof Error ? err.message : 'Failed to fetch dispatches',
   177→        variant: 'destructive',
   178→      })
   179→    } finally {
   180→      setLoading(false)
   181→    }
   182→  }, [])
   183→
   184→  React.useEffect(() => {
   185→    fetchDispatches()
   186→  }, [fetchDispatches])
   187→
   188→  // ── Fetch dropdown data when create dialog opens ──────────────────────
   189→  React.useEffect(() => {
   190→    if (!createOpen) return
   191→
   192→    const fetchDropdownData = async () => {
   193→      setDropdownsLoading(true)
   194→      try {
   195→        const [custRes, ordRes] = await Promise.all([
   196→          api.getCustomers(),
   197→          api.getOrders(),
   198→        ])
   199→        setCustomers(custRes.customers as Customer[])
   200→        setOrders(ordRes.orders as Order[])
   201→      } catch (err) {
   202→        toast({
   203→          title: 'Error',
   204→          description: 'Failed to load dropdown data',
   205→          variant: 'destructive',
   206→        })
   207→      } finally {
   208→        setDropdownsLoading(false)
   209→      }
   210→    }
   211→
   212→    fetchDropdownData()
   213→  }, [createOpen])
   214→
   215→  // ── Filter orders by selected customer ────────────────────────────────
   216→  const pendingOrdersForCustomer = React.useMemo(() => {
   217→    if (!formData.customerId) return []
   218→    return orders.filter(
   219→      (o) =>
   220→        o.customerId === formData.customerId &&
   221→        (o.status === 'Pending' || o.status === 'Processing')
   222→    )
   223→  }, [orders, formData.customerId])
   224→
   225→  // ── Auto-fill brick type when order is selected ───────────────────────
   226→  const handleOrderChange = (orderId: string) => {
   227→    setFormData((prev) => {
   228→      if (!orderId) return { ...prev, orderId: '' }
   229→      const selectedOrder = orders.find((o) => o.id === orderId)
   230→      return {
   231→        ...prev,
   232→        orderId,
   233→        brickType: selectedOrder?.brickType || prev.brickType,
   234→        quantity: selectedOrder ? String(selectedOrder.quantity) : prev.quantity,
   235→      }
   236→    })
   237→  }
   238→
   239→  // ── Create dispatch ───────────────────────────────────────────────────
   240→  const handleCreate = async () => {
   241→    if (!formData.customerId) {
   242→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   243→      return
   244→    }
   245→    if (!formData.brickType) {
   246→      toast({ title: 'Validation Error', description: 'Please select a brick type', variant: 'destructive' })
   247→      return
   248→    }
   249→    if (!formData.quantity || Number(formData.quantity) <= 0) {
   250→      toast({ title: 'Validation Error', description: 'Please enter a valid quantity', variant: 'destructive' })
   251→      return
   252→    }
   253→    if (!formData.truckNumber.trim()) {
   254→      toast({ title: 'Validation Error', description: 'Please enter the truck number', variant: 'destructive' })
   255→      return
   256→    }
   257→    if (!formData.driverName.trim()) {
   258→      toast({ title: 'Validation Error', description: 'Please enter the driver name', variant: 'destructive' })
   259→      return
   260→    }
   261→    if (!formData.date) {
   262→      toast({ title: 'Validation Error', description: 'Please select a date', variant: 'destructive' })
   263→      return
   264→    }
   265→
   266→    setSubmitting(true)
   267→    try {
   268→      const payload = {
   269→        customerId: formData.customerId,
   270→        orderId: formData.orderId || null,
   271→        brickType: formData.brickType,
   272→        quantity: Number(formData.quantity),
   273→        truckNumber: formData.truckNumber.trim(),
   274→        driverName: formData.driverName.trim(),
   275→        date: formData.date,
   276→      }
   277→
   278→      await api.createDispatch(payload)
   279→      toast({ title: 'Success', description: 'Dispatch created successfully' })
   280→      setCreateOpen(false)
   281→      setFormData(emptyForm)
   282→      fetchDispatches()
   283→    } catch (err) {
   284→      toast({
   285→        title: 'Error',
   286→        description: err instanceof Error ? err.message : 'Failed to create dispatch',
   287→        variant: 'destructive',
   288→      })
   289→    } finally {
   290→      setSubmitting(false)
   291→    }
   292→  }
   293→
   294→  // ── Delete dispatch ───────────────────────────────────────────────────
   295→  const handleDelete = async () => {
   296→    if (!deleteTarget) return
   297→    setDeleting(true)
   298→    try {
   299→      await api.deleteDispatch(deleteTarget.id)
   300→      toast({ title: 'Success', description: 'Dispatch deleted successfully' })
   301→      setDeleteTarget(null)
   302→      fetchDispatches()
   303→    } catch (err) {
   304→      toast({
   305→        title: 'Error',
   306→        description: err instanceof Error ? err.message : 'Failed to delete dispatch',
   307→        variant: 'destructive',
   308→      })
   309→    } finally {
   310→      setDeleting(false)
   311→    }
   312→  }
   313→
   314→  // ── Open create dialog ────────────────────────────────────────────────
   315→  const openCreateDialog = () => {
   316→    setFormData(emptyForm)
   317→    setCreateOpen(true)
   318→  }
   319→
   320→  // ── Print challan ─────────────────────────────────────────────────────
   321→  const handlePrint = () => {
   322→    window.print()
   323→  }
   324→
   325→  // ── Render: Loading skeletons ─────────────────────────────────────────
   326→  const renderSkeletons = () =>
   327→    Array.from({ length: 5 }).map((_, i) => (
   328→      <TableRow key={i}>
   329→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   330→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   331→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   332→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   333→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   334→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   335→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   336→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   337→      </TableRow>
   338→    ))
   339→
   340→  // ── Render: Create dispatch dialog ────────────────────────────────────
   341→  const renderCreateDialog = () => (
   342→    <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   343→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   344→        <DialogHeader>
   345→          <DialogTitle className="flex items-center gap-2">
   346→            <Truck className="size-5 text-emerald-600" />
   347→            Create Dispatch
   348→          </DialogTitle>
   349→          <DialogDescription>
   350→            Fill in the details to create a new dispatch entry.
   351→          </DialogDescription>
   352→        </DialogHeader>
   353→
   354→        <div className="grid gap-4 py-2">
   355→          {/* Customer */}
   356→          <div className="grid gap-2">
   357→            <Label>
   358→              Customer <span className="text-destructive">*</span>
   359→            </Label>
   360→            <Select
   361→              value={formData.customerId}
   362→              onValueChange={(val) =>
   363→                setFormData((prev) => ({ ...prev, customerId: val, orderId: '', brickType: '', quantity: '' }))
   364→              }
   365→              disabled={dropdownsLoading}
   366→            >
   367→              <SelectTrigger className="w-full">
   368→                <SelectValue placeholder={dropdownsLoading ? 'Loading...' : 'Select customer'} />
   369→              </SelectTrigger>
   370→              <SelectContent>
   371→                {customers.map((c) => (
   372→                  <SelectItem key={c.id} value={c.id}>
   373→                    {c.name} — {c.mobile}
   374→                  </SelectItem>
   375→                ))}
   376→              </SelectContent>
   377→            </Select>
   378→          </div>
   379→
   380→          {/* Order (optional) */}
   381→          <div className="grid gap-2">
   382→            <Label>
   383→              Order <span className="text-muted-foreground text-xs font-normal">(optional)</span>
   384→            </Label>
   385→            <Select
   386→              value={formData.orderId}
   387→              onValueChange={handleOrderChange}
   388→              disabled={!formData.customerId || pendingOrdersForCustomer.length === 0}
   389→            >
   390→              <SelectTrigger className="w-full">
   391→                <SelectValue
   392→                  placeholder={
   393→                    !formData.customerId
   394→                      ? 'Select a customer first'
   395→                      : pendingOrdersForCustomer.length === 0
   396→                        ? 'No pending orders'
   397→                        : 'Select order'
   398→                  }
   399→                />
   400→              </SelectTrigger>
   401→              <SelectContent>
   402→                {pendingOrdersForCustomer.map((o) => (
   403→                  <SelectItem key={o.id} value={o.id}>
   404→                    {o.orderNumber} — {o.brickType} ({o.quantity.toLocaleString()})
   405→                  </SelectItem>
   406→                ))}
   407→              </SelectContent>
   408→            </Select>
   409→          </div>
   410→
   411→          {/* Brick Type */}
   412→          <div className="grid gap-2">
   413→            <Label>
   414→              Brick Type <span className="text-destructive">*</span>
   415→            </Label>
   416→            <Select
   417→              value={formData.brickType}
   418→              onValueChange={(val) => setFormData((prev) => ({ ...prev, brickType: val }))}
   419→            >
   420→              <SelectTrigger className="w-full">
   421→                <SelectValue placeholder="Select brick type" />
   422→              </SelectTrigger>
   423→              <SelectContent>
   424→                {BRICK_TYPES.map((type) => (
   425→                  <SelectItem key={type} value={type}>
   426→                    {type}
   427→                  </SelectItem>
   428→                ))}
   429→              </SelectContent>
   430→            </Select>
   431→          </div>
   432→
   433→          {/* Quantity */}
   434→          <div className="grid gap-2">
   435→            <Label htmlFor="dispatch-quantity">
   436→              Quantity <span className="text-destructive">*</span>
   437→            </Label>
   438→            <Input
   439→              id="dispatch-quantity"
   440→              type="number"
   441→              min="1"
   442→              placeholder="Enter quantity"
   443→              value={formData.quantity}
   444→              onChange={(e) => setFormData((prev) => ({ ...prev, quantity: e.target.value }))}
   445→            />
   446→          </div>
   447→
   448→          {/* Truck Number */}
   449→          <div className="grid gap-2">
   450→            <Label htmlFor="dispatch-truck">
   451→              Truck Number <span className="text-destructive">*</span>
   452→            </Label>
   453→            <Input
   454→              id="dispatch-truck"
   455→              placeholder="e.g. RJ-14-AB-1234"
   456→              value={formData.truckNumber}
   457→              onChange={(e) => setFormData((prev) => ({ ...prev, truckNumber: e.target.value }))}
   458→            />
   459→          </div>
   460→
   461→          {/* Driver Name */}
   462→          <div className="grid gap-2">
   463→            <Label htmlFor="dispatch-driver">
   464→              Driver Name <span className="text-destructive">*</span>
   465→            </Label>
   466→            <Input
   467→              id="dispatch-driver"
   468→              placeholder="Enter driver name"
   469→              value={formData.driverName}
   470→              onChange={(e) => setFormData((prev) => ({ ...prev, driverName: e.target.value }))}
   471→            />
   472→          </div>
   473→
   474→          {/* Date */}
   475→          <div className="grid gap-2">
   476→            <Label htmlFor="dispatch-date">
   477→              Date <span className="text-destructive">*</span>
   478→            </Label>
   479→            <Input
   480→              id="dispatch-date"
   481→              type="date"
   482→              value={formData.date}
   483→              onChange={(e) => setFormData((prev) => ({ ...prev, date: e.target.value }))}
   484→            />
   485→          </div>
   486→        </div>
   487→
   488→        <DialogFooter>
   489→          <Button
   490→            variant="outline"
   491→            onClick={() => setCreateOpen(false)}
   492→            disabled={submitting}
   493→          >
   494→            Cancel
   495→          </Button>
   496→          <Button
   497→            onClick={handleCreate}
   498→            disabled={submitting}
   499→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   500→          >
   501→            {submitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   502→            Create Dispatch
   503→          </Button>
   504→        </DialogFooter>
   505→      </DialogContent>
   506→    </Dialog>
   507→  )
   508→
   509→  // ── Render: Challan / Print View dialog ───────────────────────────────
   510→  const renderChallanDialog = () => {
   511→    if (!challanDispatch) return null
   512→    const d = challanDispatch
   513→
   514→    return (
   515→      <Dialog open={!!challanDispatch} onOpenChange={(open) => !open && setChallanDispatch(null)}>
   516→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto print:max-w-none print:max-h-none print:p-0 print:shadow-none print:border-none">
   517→          {/* Screen-only close button header */}
   518→          <DialogHeader className="print:hidden">
   519→            <DialogTitle className="flex items-center gap-2">
   520→              <Eye className="size-5 text-emerald-600" />
   521→              Dispatch Challan
   522→            </DialogTitle>
   523→            <DialogDescription>
   524→              Preview and print the dispatch slip
   525→            </DialogDescription>
   526→          </DialogHeader>
   527→
   528→          {/* ── Challan Content (Print-friendly) ─────────────────────── */}
   529→          <div id="challan-content" className="print:p-6">
   530→            {/* Company Header */}
   531→            <div className="text-center mb-6 print:mb-8">
   532→              <div className="flex items-center justify-center gap-3 mb-2">
   533→                <Building2 className="size-8 text-emerald-600 print:text-black" />
   534→                <h1 className="text-2xl font-bold tracking-wide text-emerald-800 print:text-black">
   535→                  Veda Enterprises
   536→                </h1>
   537→              </div>
   538→              <p className="text-sm text-muted-foreground print:text-gray-600">
   539→                Brick Manufacturing &amp; Supply
   540→              </p>
   541→              <Separator className="mt-4 print:mt-6" />
   542→            </div>
   543→
   544→            {/* Title */}
   545→            <div className="text-center mb-6">
   546→              <h2 className="text-lg font-semibold underline underline-offset-4 print:text-black">
   547→                DISPATCH SLIP
   548→              </h2>
   549→            </div>
   550→
   551→            {/* Dispatch & Customer Info Grid */}
   552→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-6 text-sm print:text-black">
   553→              {/* Left column: Dispatch info */}
   554→              <div className="space-y-2">
   555→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   556→                  <span className="text-muted-foreground print:text-gray-500">Dispatch No:</span>
   557→                  <span className="font-semibold">{d.dispatchNumber}</span>
   558→                </div>
   559→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   560→                  <span className="text-muted-foreground print:text-gray-500">Date:</span>
   561→                  <span className="font-semibold">{formatDate(d.date)}</span>
   562→                </div>
   563→                {d.order && (
   564→                  <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   565→                    <span className="text-muted-foreground print:text-gray-500">Order Ref:</span>
   566→                    <span className="font-semibold">{d.order.orderNumber}</span>
   567→                  </div>
   568→                )}
   569→              </div>
   570→
   571→              {/* Right column: Customer info */}
   572→              <div className="space-y-2">
   573→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   574→                  <span className="text-muted-foreground print:text-gray-500">Customer:</span>
   575→                  <span className="font-semibold">{d.customer?.name || '—'}</span>
   576→                </div>
   577→                {d.customer?.mobile && (
   578→                  <div className="flex items-center justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   579→                    <span className="text-muted-foreground print:text-gray-500 flex items-center gap-1">
   580→                      <Phone className="size-3" /> Phone:
   581→                    </span>
   582→                    <span>{d.customer.mobile}</span>
   583→                  </div>
   584→                )}
   585→                {d.customer?.address && (
   586→                  <div className="flex items-start justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   587→                    <span className="text-muted-foreground print:text-gray-500 flex items-center gap-1 shrink-0">
   588→                      <MapPin className="size-3" /> Address:
   589→                    </span>
   590→                    <span className="text-right max-w-[60%]">{d.customer.address}</span>

   592→                )}
   593→              </div>
   594→            </div>
   595→
   596→            <Separator className="mb-6 print:border-gray-400" />

   598→            {/* Transport Info */}

   600→              <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   601→                <span className="text-muted-foreground print:text-gray-500">Truck Number:</span>
   602→                <span className="font-semibold">
   603→                  <Badge
   604→                    variant="outline"
   605→                    className="font-mono text-xs print:border-black print:text-black print:bg-transparent"
   606→                  >
   607→                    {d.truckNumber}
   608→                  </Badge>
   609→                </span>
   610→              </div>
   611→              <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   612→                <span className="text-muted-foreground print:text-gray-500">Driver Name:</span>
   613→                <span className="font-semibold">{d.driverName}</span>
   614→              </div>
   615→            </div>
   616→
   617→            <Separator className="mb-6 print:border-gray-400" />
   618→
   619→            {/* Material Details */}
   620→            <div className="mb-8 print:text-black">
   621→              <h3 className="text-sm font-semibold mb-3 text-muted-foreground print:text-gray-500 uppercase tracking-wider">
   622→                Material Details
   623→              </h3>
   624→              <Table>
   625→                <TableHeader>
   626→                  <TableRow className="print:border-gray-400">
   627→                    <TableHead className="print:text-black print:border-gray-400">Brick Type</TableHead>
   628→                    <TableHead className="text-right print:text-black print:border-gray-400">Quantity</TableHead>
   629→                  </TableRow>
   630→                </TableHeader>
   631→                <TableBody>
   632→                  <TableRow className="print:border-gray-400">
   633→                    <TableCell className="font-medium print:text-black print:border-gray-400">
   634→                      {d.brickType}
   635→                    </TableCell>
   636→                    <TableCell className="text-right font-mono font-semibold print:text-black print:border-gray-400">
   637→                      {d.quantity.toLocaleString()}
   638→                    </TableCell>
   639→                  </TableRow>
   640→                </TableBody>
   641→              </Table>
   642→            </div>
   643→
   644→            {/* Signature area */}
   645→            <div className="grid grid-cols-2 gap-8 mt-12 print:text-black">
   646→              <div className="text-center">
   647→                <div className="border-t border-gray-400 pt-2 text-sm text-muted-foreground print:text-gray-500">
   648→                  Driver Signature
   649→                </div>
   650→              </div>
   651→              <div className="text-center">
   652→                <div className="border-t border-gray-400 pt-2 text-sm text-muted-foreground print:text-gray-500">
   653→                  Authorized Signature
   654→                </div>
   655→              </div>
   656→            </div>
   657→
   658→            {/* Footer */}
   659→            <div className="mt-8 text-center text-xs text-muted-foreground print:text-gray-400 print:mt-12">

   661→            </div>
   662→          </div>
   663→
   664→          {/* Screen-only footer with print button */}
   665→          <DialogFooter className="print:hidden">
   666→            <Button
   667→              variant="outline"
   668→              onClick={() => setChallanDispatch(null)}
   669→            >
   670→              Close
   671→            </Button>
   672→            <Button
   673→              onClick={handlePrint}
   674→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   675→            >
   676→              <Printer className="size-4" />
   677→              Print Challan
   678→            </Button>
   679→          </DialogFooter>
   680→        </DialogContent>
   681→      </Dialog>
   682→    )
   683→  }
   684→
   685→  // ── Render: Delete confirmation ───────────────────────────────────────
   686→  const renderDeleteDialog = () => (
   687→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   688→      <AlertDialogContent>
   689→        <AlertDialogHeader>
   690→          <AlertDialogTitle>Delete Dispatch</AlertDialogTitle>
   691→          <AlertDialogDescription>
   692→            Are you sure you want to delete dispatch{' '}
   693→            <strong>{deleteTarget?.dispatchNumber}</strong>? This action cannot
   694→            be undone and the stock will not be restored.
   695→          </AlertDialogDescription>
   696→        </AlertDialogHeader>
   697→        <AlertDialogFooter>
   698→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   699→          <AlertDialogAction
   700→            onClick={handleDelete}
   701→            disabled={deleting}
   702→            className="bg-destructive text-white hover:bg-destructive/90"
   703→          >
   704→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   705→            Delete
   706→          </AlertDialogAction>
   707→        </AlertDialogFooter>
   708→      </AlertDialogContent>
   709→    </AlertDialog>
   710→  )
   711→
   712→  // ── Render: Main ──────────────────────────────────────────────────────
   713→  return (
   714→    <div className="space-y-6 print:hidden">
   715→      {/* Header */}
   716→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   717→        <div className="flex items-center gap-3">
   718→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   719→            <Truck className="size-5" />
   720→          </div>
   721→          <div>
   722→            <h2 className="text-2xl font-bold tracking-tight">Dispatch Management</h2>
   723→            <p className="text-sm text-muted-foreground">
   724→              Manage dispatch entries and generate challans
   725→            </p>
   726→          </div>
   727→        </div>
   728→        <div className="flex gap-2 w-full sm:w-auto">
   729→          <Button
   730→            variant="outline"
   731→            onClick={() => setImportOpen(true)}
   732→            className="w-full sm:w-auto"
   733→          >
   734→            <Upload className="size-4 mr-2" />
   735→            Import Excel
   736→          </Button>
   737→          <Button
   738→            onClick={openCreateDialog}
   739→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   740→          >
   741→            <Plus className="size-4" />
   742→            Create Dispatch
   743→          </Button>
   744→        </div>
   745→      </div>
   746→
   747→      {/* Table */}
   748→      <Card>
   749→        <CardHeader>
   750→          <CardTitle className="flex items-center justify-between">
   751→            <span>Dispatches</span>
   752→            <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   753→              {dispatches.length} record{dispatches.length !== 1 ? 's' : ''}
   754→            </Badge>
   755→          </CardTitle>
   756→        </CardHeader>
   757→        <CardContent>
   758→          <div className="overflow-x-auto">
   759→            <Table>
   760→              <TableHeader>
   761→                <TableRow>
   762→                  <TableHead>Dispatch No.</TableHead>
   763→                  <TableHead>Customer</TableHead>
   764→                  <TableHead>Brick Type</TableHead>
   765→                  <TableHead className="text-right">Quantity</TableHead>
   766→                  <TableHead>Truck No.</TableHead>
   767→                  <TableHead className="hidden md:table-cell">Driver</TableHead>
   768→                  <TableHead>Date</TableHead>
   769→                  <TableHead className="text-right">Actions</TableHead>
   770→                </TableRow>
   771→              </TableHeader>
   772→              <TableBody>
   773→                {loading ? (
   774→                  renderSkeletons()
   775→                ) : dispatches.length === 0 ? (
   776→                  <TableRow>
   777→                    <TableCell colSpan={8} className="h-32 text-center text-muted-foreground">
   778→                      <div className="flex flex-col items-center gap-2">
   779→                        <Truck className="size-8 text-muted-foreground/40" />
   780→                        <span>No dispatch entries yet. Click &quot;Create Dispatch&quot; to get started.</span>
   781→                      </div>
   782→                    </TableCell>
   783→                  </TableRow>
   784→                ) : (
   785→                  dispatches.map((dispatch) => (
   786→                    <TableRow key={dispatch.id}>
   787→                      <TableCell className="font-medium font-mono text-xs">
   788→                        {dispatch.dispatchNumber}
   789→                      </TableCell>
   790→                      <TableCell>
   791→                        <div>
   792→                          <span className="font-medium">{dispatch.customer?.name || '—'}</span>
   793→                          {dispatch.order && (
   794→                            <span className="block text-xs text-muted-foreground">
   795→                              Ref: {dispatch.order.orderNumber}
   796→                            </span>
   797→                          )}
   798→                        </div>
   799→                      </TableCell>
   800→                      <TableCell>{dispatch.brickType}</TableCell>
   801→                      <TableCell className="text-right font-mono">
   802→                        {dispatch.quantity.toLocaleString()}
   803→                      </TableCell>
   804→                      <TableCell>
   805→                        <Badge
   806→                          variant="outline"
   807→                          className="font-mono text-xs bg-amber-50 text-amber-700 border-amber-200"
   808→                        >
   809→                          {dispatch.truckNumber}
   810→                        </Badge>
   811→                      </TableCell>
   812→                      <TableCell className="hidden md:table-cell text-muted-foreground">
   813→                        {dispatch.driverName}
   814→                      </TableCell>
   815→                      <TableCell className="whitespace-nowrap">
   816→                        {formatDate(dispatch.date)}
   817→                      </TableCell>
   818→                      <TableCell className="text-right">
   819→                        <div className="flex items-center justify-end gap-1">
   820→                          <Button
   821→                            variant="ghost"
   822→                            size="icon"
   823→                            onClick={() => setChallanDispatch(dispatch)}
   824→                            title="View Challan"
   825→                            className="text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   826→                          >
   827→                            <Eye className="size-4" />
   828→                            <span className="sr-only">View Challan</span>
   829→                          </Button>
   830→                          <Button
   831→                            variant="ghost"
   832→                            size="icon"
   833→                            onClick={() => setDeleteTarget(dispatch)}
   834→                            title="Delete Dispatch"
   835→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   836→                          >
   837→                            <Trash2 className="size-4" />
   838→                            <span className="sr-only">Delete</span>
   839→                          </Button>
   840→                        </div>
   841→                      </TableCell>
   842→                    </TableRow>
   843→                  ))
   844→                )}
   845→              </TableBody>
   846→            </Table>
   847→          </div>
   848→        </CardContent>
   849→      </Card>
   850→
   851→      {/* Dialogs */}
   852→      {renderCreateDialog()}
   853→      {renderChallanDialog()}
   854→      {renderDeleteDialog()}
   855→
   856→      <ExcelImport module="dispatch" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchDispatches} />
   857→    </div>
   858→  )
   859→}
