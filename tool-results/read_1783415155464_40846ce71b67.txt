   148→   148→  const [formData, setFormData] = React.useState<ProductionFormData>(emptyForm)
   149→   149→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   150→   150→  const [deleteTarget, setDeleteTarget] = React.useState<Production | null>(null)
   151→   151→  const [deleting, setDeleting] = React.useState(false)
   152→   152→  const [deleteAllOpen, setDeleteAllOpen] = React.useState(false)
   153→   153→  const [deletingAll, setDeletingAll] = React.useState(false)
   154→   154→  const [deleteAllConfirm, setDeleteAllConfirm] = React.useState('')
   155→   155→
   156→   156→  // ── Multi-select state ──────────────────────────────────────────────────

   158→   158→  // "Delete Selected" button appears in the header whenever this set is
   159→   159→  // non-empty. We use a Set for O(1) toggle/lookup, then convert to an
   160→   160→  // array when sending to the API.

   163→   163→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   164→   164→
   165→   165→  const toggleSelect = (id: string) => {
   166→   166→    setSelectedIds((prev) => {
   167→   167→      const next = new Set(prev)
   168→   168→      if (next.has(id)) next.delete(id)
   169→   169→      else next.add(id)
   170→   170→      return next
   171→   171→    })
   172→   172→  }
   173→   173→
   174→   174→  const toggleSelectAll = () => {
   175→   175→    setSelectedIds((prev) => {
   176→   176→      // If everything is already selected, clear. Otherwise select all
   177→   177→      // currently-filtered rows (so the user can select-all within a search
   178→   178→      // result without picking rows they can't see).
   179→   179→      if (prev.size === filteredProductions.length && filteredProductions.length > 0) {
   180→   180→        return new Set()
   181→   181→      }
   182→   182→      return new Set(filteredProductions.map((p) => p.id))
   183→   183→    })
   184→   184→  }
   185→   185→
   186→   186→  const clearSelection = () => setSelectedIds(new Set())
   187→   187→
   188→   188→  const handleBulkDelete = async () => {
   189→   189→    const ids = Array.from(selectedIds)
   190→   190→    if (ids.length === 0) return
   191→   191→    setBulkDeleting(true)
   192→   192→    try {
   193→   193→      const res = await api.bulkDeleteProductions(ids)
   194→   194→      toast({
   195→   195→        title: 'Success',
   196→   196→        description: `${res.deletedCount} of ${ids.length} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   197→   197→      })
   198→   198→      setBulkDeleteOpen(false)
   199→   199→      clearSelection()
   200→   200→      fetchProductions()
   201→   201→    } catch (err) {
   202→   202→      toast({
   203→   203→        title: 'Error',
   204→   204→        description: err instanceof Error ? err.message : 'Failed to delete selected production entries',
   205→   205→        variant: 'destructive',
   206→   206→      })
   207→   207→    } finally {
   208→   208→      setBulkDeleting(false)
   209→   209→    }
   210→   210→  }
   211→   211→
   212→   212→  const fetchProductions = React.useCallback(async () => {
   213→   213→    setLoading(true)
   214→   214→    try {
   215→   215→      const res = await api.getProduction()
   216→   216→      const data = (res.productions as Production[]).sort(
   217→   217→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   218→   218→      )
   219→   219→      setProductions(data)
   220→   220→    } catch (err) {
   221→   221→      toast({
   222→   222→        title: 'Error',
   223→   223→        description: err instanceof Error ? err.message : 'Failed to fetch production data',
   224→   224→        variant: 'destructive',
   225→   225→      })
   226→   226→    } finally {
   227→   227→      setLoading(false)
   228→   228→    }
   229→   229→  }, [])
   230→   230→
   231→   231→  // Debounced search
   232→   232→  React.useEffect(() => {
   233→   233→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   234→   234→    return () => clearTimeout(t)
   235→   235→  }, [search])
   236→   236→
   237→   237→  // Client-side filter
   238→   238→  const filteredProductions = React.useMemo(() => {
   239→   239→    if (!debouncedSearch.trim()) return productions
   240→   240→    const q = debouncedSearch.toLowerCase()
   241→   241→    return productions.filter((item: any) =>
   242→   242→      ['date', 'remarks'].some((f) =>
   243→   243→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   244→   244→      )
   245→   245→    )
   246→   246→  }, [productions, debouncedSearch])
   247→   247→
   248→   248→  React.useEffect(() => {
   249→   249→    fetchProductions()
   250→   250→  }, [fetchProductions])
   251→   251→
   252→   252→  // Excel import
   253→   253→  const [importOpen, setImportOpen] = React.useState(false)
   254→   254→
   255→   255→  const openAddDialog = () => {
   256→   256→    setEditingProduction(null)
   257→   257→    // Check for pending AI result from chat widget — auto-fill if present
   258→   258→    const pending = consumePendingAiResult('production')
   259→   259→    if (pending) {
   260→   260→      const str = (v: unknown) => (v == null ? '' : String(v))
   261→   261→      setFormData({
   262→   262→        date: str(pending.date),
   263→   263→        cement: str(pending.cement),
   264→   264→        zigZagGrey80: str(pending.zigZagGrey80),
   265→   265→        zigZagRed80: str(pending.zigZagRed80),
   266→   266→        zigZagYellow80: str(pending.zigZagYellow80),
   267→   267→        zigZagGrey60: str(pending.zigZagGrey60),
   268→   268→        zigZagRed60: str(pending.zigZagRed60),
   269→   269→        zigZagYellow60: str(pending.zigZagYellow60),
   270→   270→        curveStone: str(pending.curveStone),
   271→   271→        chequreTile: str(pending.chequreTile),
   272→   272→        dumbleGrey80: str(pending.dumbleGrey80),
   273→   273→        dumbleRed80: str(pending.dumbleRed80),
   274→   274→        dumbleYellow80: str(pending.dumbleYellow80),
   275→   275→        transportationCharge: str(pending.transportationCharge),
   276→   276→        remarks: str(pending.remarks),
   277→   277→      })
   278→   278→      toast({ title: 'AI auto-fill applied', description: 'Edit & verify before saving.' })
   279→   279→    } else {
   280→   280→      setFormData(emptyForm)
   281→   281→    }
   282→   282→    setFormOpen(true)
   283→   283→  }
   284→   284→
   285→   285→  const openEditDialog = (prod: Production) => {
   286→   286→    setEditingProduction(prod)
   287→   287→    setFormData({
   288→   288→      date: prod.date ? prod.date.split('T')[0] : '',
   289→   289→      cement: String(prod.cement || ''),
   290→   290→      zigZagGrey80: String(prod.zigZagGrey80 || ''),
   291→   291→      zigZagRed80: String(prod.zigZagRed80 || ''),
   292→   292→      zigZagYellow80: String(prod.zigZagYellow80 || ''),
   293→   293→      zigZagGrey60: String(prod.zigZagGrey60 || ''),
   294→   294→      zigZagRed60: String(prod.zigZagRed60 || ''),
   295→   295→      zigZagYellow60: String(prod.zigZagYellow60 || ''),
   296→   296→      curveStone: String(prod.curveStone || ''),
   297→   297→      chequreTile: String(prod.chequreTile || ''),
   298→   298→      dumbleGrey80: String(prod.dumbleGrey80 || ''),
   299→   299→      dumbleRed80: String(prod.dumbleRed80 || ''),
   300→   300→      dumbleYellow80: String(prod.dumbleYellow80 || ''),
   301→   301→      transportationCharge: String(prod.transportationCharge || ''),
   302→   302→      remarks: prod.remarks || '',
   303→   303→    })
   304→   304→    setFormOpen(true)
   305→   305→  }
   306→   306→
   307→   307→  const handleFormChange = (field: keyof ProductionFormData, value: string) => {
   308→   308→    setFormData((prev) => ({ ...prev, [field]: value }))
   309→   309→  }
   310→   310→
   311→   311→  const handleSubmit = async () => {
   312→   312→    if (!formData.date) {
   313→   313→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   314→   314→      return
   315→   315→    }
   316→   316→
   317→   317→    setFormSubmitting(true)
   318→   318→    try {
   319→   319→      const payload = {
   320→   320→        date: formData.date,
   321→   321→        cement: Number(formData.cement) || 0,
   322→   322→        zigZagGrey80: Number(formData.zigZagGrey80) || 0,
   323→   323→        zigZagRed80: Number(formData.zigZagRed80) || 0,
   324→   324→        zigZagYellow80: Number(formData.zigZagYellow80) || 0,
   325→   325→        zigZagGrey60: Number(formData.zigZagGrey60) || 0,
   326→   326→        zigZagRed60: Number(formData.zigZagRed60) || 0,
   327→   327→        zigZagYellow60: Number(formData.zigZagYellow60) || 0,
   328→   328→        curveStone: Number(formData.curveStone) || 0,
   329→   329→        chequreTile: Number(formData.chequreTile) || 0,
   330→   330→        dumbleGrey80: Number(formData.dumbleGrey80) || 0,
   331→   331→        dumbleRed80: Number(formData.dumbleRed80) || 0,
   332→   332→        dumbleYellow80: Number(formData.dumbleYellow80) || 0,
   333→   333→        transportationCharge: Number(formData.transportationCharge) || 0,
   334→   334→        remarks: formData.remarks.trim(),
   335→   335→      }
   336→   336→
   337→   337→      if (editingProduction) {

   342→   342→        toast({ title: 'Success', description: 'Production entry created successfully' })

   345→   345→      setFormOpen(false)
   346→   346→      setFormData(emptyForm)
   347→   347→      setEditingProduction(null)
   348→   348→      fetchProductions()
   349→   349→    } catch (err) {
   350→   350→      toast({
   351→   351→        title: 'Error',
   352→   352→        description: err instanceof Error ? err.message : 'Failed to save production entry',
   353→   353→        variant: 'destructive',
   354→   354→      })
   355→   355→    } finally {
   356→   356→      setFormSubmitting(false)
   357→   357→    }
   358→   358→  }

   362→   362→    setDeleting(true)

   364→   364→      await api.deleteProduction(deleteTarget.id)

   367→   367→      fetchProductions()
   368→   368→    } catch (err) {
   369→   369→      toast({
   370→   370→        title: 'Error',
   371→   371→        description: err instanceof Error ? err.message : 'Failed to delete production entry',
   372→   372→        variant: 'destructive',
   373→   373→      })
   374→   374→    } finally {
   375→   375→      setDeleting(false)
   376→   376→    }
   377→   377→  }
   378→   378→

   380→   380→    // No more 'DELETE ALL' typing requirement — the dialog now uses a
   381→   381→    // simple Yes / No choice. We keep the deleteAllConfirm state around
   382→   382→    // only so existing code that resets it (e.g. on dialog close) doesn't
   383→   383→    // break, but it's no longer gating the delete.
   384→   384→    setDeletingAll(true)
   385→   385→    try {
   386→   386→      const res = await api.deleteAllProductions()
   387→   387→      toast({
   388→   388→        title: 'Success',
   389→   389→        description: `${res.deletedCount} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   390→   390→      })
   391→   391→      setDeleteAllOpen(false)
   392→   392→      setDeleteAllConfirm('')
   393→   393→      fetchProductions()
   394→   394→    } catch (err) {
   395→   395→      toast({
   396→   396→        title: 'Error',
   397→   397→        description: err instanceof Error ? err.message : 'Failed to delete all production entries',
   398→   398→        variant: 'destructive',
   399→   399→      })
   400→   400→    } finally {
   401→   401→      setDeletingAll(false)
   402→   402→    }
   403→   403→  }
   404→   404→
   405→   405→  const renderSkeletons = () =>
   406→   406→    Array.from({ length: 3 }).map((_, i) => (
   407→   407→      <TableRow key={i}>
   408→   408→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   409→   409→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   410→   410→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   411→   411→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   412→   412→        {PRODUCT_FIELDS.map((_, j) => (
   413→   413→          <TableCell key={j}><Skeleton className="h-4 w-16" /></TableCell>
   414→   414→        ))}
   415→   415→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   416→   416→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   417→   417→      </TableRow>
   418→   418→    ))
   419→   419→
   420→   420→  return (
   421→   421→    <div className="space-y-6 relative">
   422→   422→      {/* Full-screen loading overlay shown during any delete operation.
   423→   423→          The spinner + message stay visible until the API call completes and
   424→   424→          the table refreshes, so the user always knows "something is happening". */}
   425→   425→      {(deleting || deletingAll || bulkDeleting) && (
   426→   426→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   427→   427→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   428→   428→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   429→   429→            <div className="text-center">
   430→   430→              <p className="text-lg font-semibold">
   431→   431→                {bulkDeleting
   432→   432→                  ? `Deleting ${selectedIds.size} entries...`
   433→   433→                  : deletingAll
   434→   434→                  ? 'Deleting all entries...'
   435→   435→                  : 'Deleting entry...'}
   436→   436→              </p>
   437→   437→              <p className="text-sm text-muted-foreground mt-1">
   438→   438→                Please wait while records are removed and stock is re-synced.
   439→   439→              </p>
   440→   440→            </div>
   441→   441→          </div>
   442→   442→        </div>
   443→   443→      )}
   444→   444→      {/* Header */}
   445→   445→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   446→   446→        <div className="flex items-center gap-3">
   447→   447→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   448→   448→            <Factory className="size-5" />
   449→   449→          </div>
   450→   450→          <div>
   451→   451→            <h2 className="text-2xl font-bold tracking-tight">Production Management</h2>
