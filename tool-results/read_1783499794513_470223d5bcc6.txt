   108→function formatDate(dateStr: string): string {
   109→  if (!dateStr) return '—'
   110→  try {
   111→    const d = new Date(dateStr)
   112→    return d.toLocaleDateString('en-IN', {
   113→      day: '2-digit',
   114→      month: 'short',
   115→      year: 'numeric',
   116→    })
   117→  } catch {
   118→    return dateStr
   119→  }

   122→// ── CSV Export ───────────────────────────────────────────────────────────────
   123→
   124→function exportCSV(filename: string, headers: string[], rows: string[][]) {
   125→  const csvContent = [
   126→    headers.join(','),
   127→    ...rows.map((row) =>
   128→      row.map((cell) => {
   129→        const str = String(cell ?? '')
   130→        // Escape cells that contain commas, quotes, or newlines
   131→        if (str.includes(',') || str.includes('"') || str.includes('\n')) {
   132→          return `"${str.replace(/"/g, '""')}"`
   133→        }
   134→        return str
   135→      }).join(',')
   136→    ),
   137→  ].join('\n')
   138→
   139→  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   140→  const url = URL.createObjectURL(blob)
   141→  const link = document.createElement('a')
   142→  link.href = url
   143→  link.download = `${filename}.csv`
   144→  document.body.appendChild(link)
   145→  link.click()
   146→  document.body.removeChild(link)
   147→  URL.revokeObjectURL(url)
   148→}

   518→// ── Customer Ledger Report ───────────────────────────────────────────────────
   519→
   520→function CustomerLedgerReport() {
   521→  const [data, setData] = React.useState<CustomerLedgerRow[]>([])
   522→  const [loading, setLoading] = React.useState(true)
   523→
   524→  const loadData = React.useCallback(async () => {
   525→    setLoading(true)
   526→    try {
   527→      // Fetch all customers first
   528→      const [customersRes, ordersRes, paymentsRes] = await Promise.all([
   529→        api.getCustomers() as Promise<{ customers: { id: string; name: string }[] }>,
   530→        api.getOrders() as Promise<{ orders: { customerId: string; amount: number }[] }>,
   531→        api.getPayments() as Promise<{ payments: { customerId: string; amount: number }[] }>,
   532→      ])
   533→
   534→      const customerMap = new Map(
   535→        (customersRes.customers || []).map((c) => [c.id, c])
   536→      )
   537→
   538→      // Aggregate order totals by customer
   539→      const orderTotals = new Map<string, number>()
   540→      for (const order of ordersRes.orders || []) {
   541→        orderTotals.set(order.customerId, (orderTotals.get(order.customerId) || 0) + order.amount)
   542→      }
   543→
   544→      // Aggregate payment totals by customer
   545→      const paymentTotals = new Map<string, number>()
   546→      for (const payment of paymentsRes.payments || []) {
   547→        paymentTotals.set(payment.customerId, (paymentTotals.get(payment.customerId) || 0) + payment.amount)
   548→      }
   549→
   550→      // Build ledger rows
   551→      const allCustomerIds = new Set([
   552→        ...orderTotals.keys(),
   553→        ...paymentTotals.keys(),
   554→      ])
   555→
   556→      const ledgerData: CustomerLedgerRow[] = [...allCustomerIds].map((cId) => {
   557→        const totalOrders = orderTotals.get(cId) || 0
   558→        const totalPayments = paymentTotals.get(cId) || 0
   559→        return {
   560→          customerId: cId,
   561→          customer: customerMap.get(cId)
   562→            ? { id: cId, name: customerMap.get(cId)!.name }
   563→            : null,
   564→          totalOrders,
   565→          totalPayments,
   566→          outstanding: totalOrders - totalPayments,
   567→        }
   568→      })
   569→
   570→      setData(ledgerData)
   571→    } catch (err) {
   572→      toast({
   573→        title: 'Error',
   574→        description: 'Failed to load customer ledger',
   575→        variant: 'destructive',
   576→      })
   577→    } finally {
   578→      setLoading(false)
   579→    }
   580→  }, [])
   581→
   582→  React.useEffect(() => {
   583→    loadData()
   584→  }, [loadData])
   585→
   586→  const handleExportCSV = () => {
   587→    const headers = ['Customer', 'Total Orders', 'Total Payments', 'Outstanding Balance']
   588→    const rows = data.map((row) => [
   589→      row.customer?.name || '—',
   590→      formatCurrency(row.totalOrders),
   591→      formatCurrency(row.totalPayments),
   592→      formatCurrency(row.outstanding),
   593→    ])
   594→    exportCSV('customer-ledger', headers, rows)
   595→  }
   596→
   597→  const totalOrders = data.reduce((sum, r) => sum + r.totalOrders, 0)
