   259→   259→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   260→   260→
   261→   261→  for (const field of moduleFields) {
   262→   262→    let bestMatch = ''
   263→   263→    let bestScore = 0
   264→   264→
   265→   265→    for (const excelCol of normalizedExcel) {
   266→   266→      // Check exact match with key
   267→   267→      if (excelCol.normalized === normalize(field.key)) {
   268→   268→        bestMatch = excelCol.original
   269→   269→        bestScore = 100
   270→   270→        break
   271→   271→      }
   272→   272→
   273→   273→      // Check aliases
   274→   274→      for (const alias of field.aliases) {
   275→   275→        const normAlias = normalize(alias)
   276→   276→        if (excelCol.normalized === normAlias) {
   277→   277→          if (100 > bestScore) {
   278→   278→            bestMatch = excelCol.original
   279→   279→            bestScore = 100
   280→   280→          }
   281→   281→          break
   282→   282→        }
   283→   283→      }
   284→   284→
   285→   285→      if (bestScore >= 100) break
   286→   286→
   287→   287→      // Fuzzy match: check if alias is contained in column name or vice versa
   288→   288→      for (const alias of field.aliases) {
   289→   289→        const normAlias = normalize(alias)
   290→   290→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   291→   291→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   292→   292→          if (score > bestScore) {
   293→   293→            bestMatch = excelCol.original
   294→   294→            bestScore = score
   295→   295→          }
   296→   296→        }
   297→   297→      }
   298→   298→    }
   299→   299→
   300→   300→    if (bestMatch && bestScore >= 50) {
   301→   301→      mapping[bestMatch] = field.key
   302→   302→    }
   303→   303→  }
   304→   304→
   305→   305→  return mapping
   306→   306→}

   308→   308→// Auto-detect the module from Excel column headers
   309→   309→function autoDetectModule(columns: string[]): string | null {
   310→   310→  const normalized = columns.map(normalize)
   311→   311→  const scores: Record<string, number> = {}
   312→   312→
   313→   313→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   314→   314→    scores[moduleKey] = 0
   315→   315→    for (const field of template.fields) {
   316→   316→      for (const alias of field.aliases) {
   317→   317→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   318→   318→          scores[moduleKey] += field.required ? 3 : 1
   319→   319→        }
   320→   320→      }
   321→   321→    }
   322→   322→  }
   323→   323→
   324→   324→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   325→   325→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   326→   326→}

   328→   328→// Transform data using mapping, also handle customer name -> ID resolution
   329→   329→function transformRow(
   330→   330→  row: Record<string, unknown>,
   331→   331→  mapping: Record<string, string>,
   332→   332→  customers?: { id: string; name: string }[]
   333→   333→): Record<string, unknown> {
   334→   334→  const result: Record<string, unknown> = {}
   335→   335→
   336→   336→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   337→   337→    let value = row[excelCol]
   338→   338→
   339→   339→    // Handle customer name -> ID resolution
   340→   340→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   341→   341→      const nameStr = String(value || '').trim().toLowerCase()
   342→   342→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   343→   343→      if (matched) {
   344→   344→        result['customerId'] = matched.id
   345→   345→      } else {
   346→   346→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   347→   347→      }
   348→   348→      continue
   349→   349→    }
   350→   350→
   351→   351→    // Handle numeric fields
   352→   352→    if ([
   353→   353→      'quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate',
   354→   354→      'zigZagGrey80', 'zigZagRed80', 'zigZagYellow80',
   355→   355→      'zigZagGrey60', 'zigZagRed60', 'zigZagYellow60',
   356→   356→      'curveStone', 'chequreTile', 'transportationCharge',
   357→   357→      'cement', 'dumbleGrey80', 'dumbleRed80', 'dumbleYellow80',
   358→   358→      'quantityTon', 'totalAmount', 'paidAmount', 'remainingAmount',
   359→   359→      'gst',
   360→   360→    ].includes(fieldKey)) {
   361→   361→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   362→   362→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   363→   363→      continue
   364→   364→    }
   365→   365→
   366→   366→    // Handle date fields - try to parse various formats
   367→   367→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   368→   368→      // Excel may return dates as numeric serial numbers (e.g. 46178).
   369→   369→      // Detect that case and convert to YYYY-MM-DD before string parsing.
   370→   370→      // Also handle serial numbers passed as strings (e.g. "46178").
   371→   371→      const numericValue = typeof value === 'number'
   372→   372→        ? value
   373→   373→        : (typeof value === 'string' && /^\d{4,6}(\.\d+)?$/.test(value.trim())
   374→   374→            ? Number(value)
   375→   375→            : NaN)
   376→   376→
   377→   377→      if (Number.isFinite(numericValue) && numericValue > 59 && numericValue < 60000) {
   378→   378→        // Excel serial date: days since 1899-12-30
   379→   379→        const ms = Math.round((numericValue - 25569) * 86400 * 1000)
   380→   380→        const d = new Date(ms)
   381→   381→        if (!isNaN(d.getTime())) {
   382→   382→          result[fieldKey] = d.toISOString().split('T')[0]
   383→   383→          continue
   384→   384→        }
   385→   385→      }
   386→   386→      // Excel may also return a Date object if cellDates:true was used.
   387→   387→      if (value instanceof Date && !isNaN(value.getTime())) {
   388→   388→        result[fieldKey] = value.toISOString().split('T')[0]
   389→   389→        continue
   390→   390→      }
   391→   391→      result[fieldKey] = parseDate(String(value || ''))
   392→   392→      continue
   393→   393→    }
   394→   394→
   395→   395→    // Handle shift field - normalize
   396→   396→    if (fieldKey === 'shift') {
   397→   397→      const shiftStr = String(value || 'Morning').trim()
   398→   398→      const lower = shiftStr.toLowerCase()
   399→   399→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   400→   400→        result[fieldKey] = 'Morning'
   401→   401→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   402→   402→        result[fieldKey] = 'Evening'
   403→   403→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   404→   404→        result[fieldKey] = 'Night'
   405→   405→      } else {
   406→   406→        result[fieldKey] = shiftStr
   407→   407→      }
   408→   408→      continue
   409→   409→    }
   410→   410→
   411→   411→    // Handle payment type
   412→   412→    if (fieldKey === 'paymentType') {
   413→   413→      const ptStr = String(value || 'Cash').trim()
   414→   414→      const lower = ptStr.toLowerCase()
   415→   415→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   416→   416→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   417→   417→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   418→   418→      else result[fieldKey] = ptStr
   419→   419→      continue
   420→   420→    }
   421→   421→
   422→   422→    // Handle expense category
   423→   423→    if (fieldKey === 'category') {
   424→   424→      const catStr = String(value || '').trim()
   425→   425→      const lower = catStr.toLowerCase()
   426→   426→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   427→   427→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   428→   428→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   429→   429→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   430→   430→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   431→   431→      else result[fieldKey] = catStr
   432→   432→      continue
   433→   433→    }
   434→   434→
   435→   435→    // Handle status
   436→   436→    if (fieldKey === 'status') {
   437→   437→      const stStr = String(value || 'Pending').trim()
   438→   438→      const lower = stStr.toLowerCase()
   439→   439→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   440→   440→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   441→   441→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   442→   442→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   443→   443→      else result[fieldKey] = stStr
   444→   444→      continue
   445→   445→    }
   446→   446→
   447→   447→    result[fieldKey] = String(value || '')
   448→   448→  }
   449→   449→
   450→   450→  // For orders/dispatch: calculate amount if not mapped
   451→   451→  if (mapping) {
   452→   452→    const targetModule = Object.values(mapping)
   453→   453→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   454→   454→      const qty = Number(result.quantity || 0)
   455→   455→      const rate = Number(result.rate || 0)
   456→   456→      result.amount = qty * rate
   457→   457→    }
   458→   458→  }
   459→   459→
   460→   460→  return result
   461→   461→}

   463→   463→// Parse various date formats to YYYY-MM-DD.
   464→   464→// Supported formats (case-insensitive, any separator among - / . or space):
   465→   465→//   • YYYY-MM-DD          (already canonical, returned as-is)
   466→   466→//   • DD-MM-YYYY  DD/MM/YYYY  DD.MM.YYYY  (full year, day-first — Indian format)
   467→   467→//   • DD-MM-YY    DD/MM/YY    DD.MM.YY    (short year, prefixed with 20)
   468→   468→//   • MM/DD/YYYY  (US format — only used when second number is > 12, so 01/13/2024 -> Jan 13)
   469→   469→//   • Datetime strings like "2024-01-15 10:30:00" or "15-01-2024 10:30" (time part stripped)
   470→   470→//   • Excel serial numbers are handled earlier in transformRow() — not here.
   471→   471→//   • Fallback: native Date parsing, then the raw string unchanged.
   472→   472→function parseDate(value: string): string {
   473→   473→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   474→   474→
   475→   475→  // Strip any time portion (e.g. " 10:30:00", "T10:30", " 10:30 AM")
   476→   476→  // so datetime strings don't confuse the regex below.
   477→   477→  const trimmed = value.trim().replace(/[Tt]\s*\d{1,2}:\d{2}.*$/, '').replace(/\s+\d{1,2}:\d{2}.*$/, '').trim()
   478→   478→  if (trimmed === '') return new Date().toISOString().split('T')[0]
   479→   479→
   480→   480→  // Already in YYYY-MM-DD format
   481→   481→  if (/^\d{4}-\d{1,2}-\d{1,2}$/.test(trimmed)) {
   482→   482→    const [y, m, d] = trimmed.split('-')
   483→   483→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   484→   484→  }
   485→   485→
   486→   486→  // YYYY/MM/DD or YYYY.MM.DD
   487→   487→  const ymdMatch = trimmed.match(/^(\d{4})[/.\s](\d{1,2})[/.\s](\d{1,2})$/)
   488→   488→  if (ymdMatch) {
   489→   489→    const [, y, m, d] = ymdMatch
   490→   490→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   491→   491→  }
   492→   492→
   493→   493→  // DD-MM-YYYY / DD/MM/YYYY / DD.MM.YYYY  (day-first, full year)
   494→   494→  //   — also handles MM/DD/YYYY when first number > 12 (e.g. 13/01/2024)
   495→   495→  const dmyMatch = trimmed.match(/^(\d{1,2})[/.\s-](\d{1,2})[/.\s-](\d{4})$/)
   496→   496→  if (dmyMatch) {
   497→   497→    let [, a, b, y] = dmyMatch
   498→   498→    let d: string, m: string
   499→   499→    // If second number > 12, it MUST be a day -> user wrote MM/DD (US format)
   500→   500→    if (Number(b) > 12 && Number(a) <= 12) {
   501→   501→      m = a; d = b
   502→   502→    } else {
   503→   503→      // Default to DD-MM (Indian format)
   504→   504→      d = a; m = b
   505→   505→    }
   506→   506→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   507→   507→  }
   508→   508→
   509→   509→  // DD-MM-YY / DD/MM/YY / DD.MM.YY (short year)
   510→   510→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/.\s-](\d{1,2})[/.\s-](\d{2})$/)
   511→   511→  if (dmyShortMatch) {
   512→   512→    let [, a, b, y] = dmyShortMatch
   513→   513→    let d: string, m: string
   514→   514→    if (Number(b) > 12 && Number(a) <= 12) {
   515→   515→      m = a; d = b
   516→   516→    } else {
   517→   517→      d = a; m = b
   518→   518→    }
   519→   519→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   520→   520→  }
   521→   521→
   522→   522→  // Try native Date parsing as fallback (handles ISO, RFC2822, etc.)
   523→   523→  try {
   524→   524→    const date = new Date(trimmed)
   525→   525→    if (!isNaN(date.getTime())) {
   526→   526→      return date.toISOString().split('T')[0]
   527→   527→    }
   528→   528→  } catch {}
   529→   529→
   530→   530→  // Last resort — return as-is so server-side validation can flag it
   531→   531→  return trimmed
   532→   532→}
