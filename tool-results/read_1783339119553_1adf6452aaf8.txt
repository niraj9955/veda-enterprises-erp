   136→// ── Component ───────────────────────────────────────────────────────────────
   137→
   138→export function ProductionModule() {
   139→  const [productions, setProductions] = React.useState<Production[]>([])
   140→  const [search, setSearch] = React.useState('')
   141→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   142→  const [loading, setLoading] = React.useState(true)
   143→  const [formOpen, setFormOpen] = React.useState(false)
   144→  const [editingProduction, setEditingProduction] = React.useState<Production | null>(null)
   145→  const [formData, setFormData] = React.useState<ProductionFormData>(emptyForm)
   146→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   147→  const [deleteTarget, setDeleteTarget] = React.useState<Production | null>(null)
   148→  const [deleting, setDeleting] = React.useState(false)
   149→  const [deleteAllOpen, setDeleteAllOpen] = React.useState(false)
   150→  const [deletingAll, setDeletingAll] = React.useState(false)
   151→  const [deleteAllConfirm, setDeleteAllConfirm] = React.useState('')
   152→
   153→  // ── Multi-select state ──────────────────────────────────────────────────
   154→  // Tracks which production rows the user has ticked in the table. The
   155→  // "Delete Selected" button appears in the header whenever this set is
   156→  // non-empty. We use a Set for O(1) toggle/lookup, then convert to an
   157→  // array when sending to the API.
   158→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   159→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   160→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   161→
   162→  const toggleSelect = (id: string) => {
   163→    setSelectedIds((prev) => {
   164→      const next = new Set(prev)
   165→      if (next.has(id)) next.delete(id)
   166→      else next.add(id)
   167→      return next
   168→    })
   169→  }
   170→
   171→  const toggleSelectAll = () => {
   172→    setSelectedIds((prev) => {
   173→      // If everything is already selected, clear. Otherwise select all
   174→      // currently-filtered rows (so the user can select-all within a search
   175→      // result without picking rows they can't see).
   176→      if (prev.size === filteredProductions.length && filteredProductions.length > 0) {
   177→        return new Set()
   178→      }
   179→      return new Set(filteredProductions.map((p) => p.id))
   180→    })
   181→  }
   182→
   183→  const clearSelection = () => setSelectedIds(new Set())
   184→
   185→  const handleBulkDelete = async () => {
   186→    const ids = Array.from(selectedIds)
   187→    if (ids.length === 0) return
   188→    setBulkDeleting(true)
   189→    try {
   190→      const res = await api.bulkDeleteProductions(ids)
   191→      toast({
   192→        title: 'Success',
   193→        description: `${res.deletedCount} of ${ids.length} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   194→      })
   195→      setBulkDeleteOpen(false)
   196→      clearSelection()
   197→      fetchProductions()
   198→    } catch (err) {
   199→      toast({
   200→        title: 'Error',
   201→        description: err instanceof Error ? err.message : 'Failed to delete selected production entries',
   202→        variant: 'destructive',
   203→      })
   204→    } finally {
   205→      setBulkDeleting(false)
   206→    }
   207→  }
   208→
   209→  const fetchProductions = React.useCallback(async () => {
   210→    setLoading(true)
   211→    try {
   212→      const res = await api.getProduction()
   213→      const data = (res.productions as Production[]).sort(
   214→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   215→      )
   216→      setProductions(data)
   217→    } catch (err) {
   218→      toast({
   219→        title: 'Error',
   220→        description: err instanceof Error ? err.message : 'Failed to fetch production data',
   221→        variant: 'destructive',
   222→      })
   223→    } finally {
   224→      setLoading(false)
   225→    }
   226→  }, [])
   227→
   228→  // Debounced search
   229→  React.useEffect(() => {
   230→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   231→    return () => clearTimeout(t)
   232→  }, [search])
   233→
   234→  // Client-side filter
   235→  const filteredProductions = React.useMemo(() => {
   236→    if (!debouncedSearch.trim()) return productions
   237→    const q = debouncedSearch.toLowerCase()
   238→    return productions.filter((item: any) =>
   239→      ['date', 'remarks'].some((f) =>
   240→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   241→      )
   242→    )
   243→  }, [productions, debouncedSearch])
   244→
   245→  React.useEffect(() => {
   246→    fetchProductions()
   247→  }, [fetchProductions])
   248→
   249→  // Excel import
   250→  const [importOpen, setImportOpen] = React.useState(false)
   251→
   252→  const openAddDialog = () => {
   253→    setEditingProduction(null)
   254→    setFormData(emptyForm)
   255→    setFormOpen(true)
   256→  }
   257→
   258→  const openEditDialog = (prod: Production) => {
   259→    setEditingProduction(prod)
   260→    setFormData({
   261→      date: prod.date ? prod.date.split('T')[0] : '',
   262→      cement: String(prod.cement || ''),
   263→      zigZagGrey80: String(prod.zigZagGrey80 || ''),
   264→      zigZagRed80: String(prod.zigZagRed80 || ''),
   265→      zigZagYellow80: String(prod.zigZagYellow80 || ''),
   266→      zigZagGrey60: String(prod.zigZagGrey60 || ''),
   267→      zigZagRed60: String(prod.zigZagRed60 || ''),
   268→      zigZagYellow60: String(prod.zigZagYellow60 || ''),
   269→      curveStone: String(prod.curveStone || ''),
   270→      chequreTile: String(prod.chequreTile || ''),
   271→      dumbleGrey80: String(prod.dumbleGrey80 || ''),
   272→      dumbleRed80: String(prod.dumbleRed80 || ''),
   273→      dumbleYellow80: String(prod.dumbleYellow80 || ''),
   274→      transportationCharge: String(prod.transportationCharge || ''),
   275→      remarks: prod.remarks || '',
   276→    })
   277→    setFormOpen(true)
   278→  }
   279→
   280→  const handleFormChange = (field: keyof ProductionFormData, value: string) => {
   281→    setFormData((prev) => ({ ...prev, [field]: value }))
   282→  }
   283→
   284→  const handleSubmit = async () => {
   285→    if (!formData.date) {
   286→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   287→      return
   288→    }
   289→
   290→    setFormSubmitting(true)
   291→    try {
   292→      const payload = {
   293→        date: formData.date,
   294→        cement: Number(formData.cement) || 0,
   295→        zigZagGrey80: Number(formData.zigZagGrey80) || 0,
   296→        zigZagRed80: Number(formData.zigZagRed80) || 0,
   297→        zigZagYellow80: Number(formData.zigZagYellow80) || 0,
   298→        zigZagGrey60: Number(formData.zigZagGrey60) || 0,
   299→        zigZagRed60: Number(formData.zigZagRed60) || 0,
   300→        zigZagYellow60: Number(formData.zigZagYellow60) || 0,
   301→        curveStone: Number(formData.curveStone) || 0,
   302→        chequreTile: Number(formData.chequreTile) || 0,
   303→        dumbleGrey80: Number(formData.dumbleGrey80) || 0,
   304→        dumbleRed80: Number(formData.dumbleRed80) || 0,
   305→        dumbleYellow80: Number(formData.dumbleYellow80) || 0,
   306→        transportationCharge: Number(formData.transportationCharge) || 0,
   307→        remarks: formData.remarks.trim(),
   308→      }
   309→
   310→      if (editingProduction) {
   311→        await api.updateProduction(editingProduction.id, payload)
   312→        toast({ title: 'Success', description: 'Production entry updated successfully' })
   313→      } else {
   314→        await api.createProduction(payload)
   315→        toast({ title: 'Success', description: 'Production entry created successfully' })
   316→      }
   317→
   318→      setFormOpen(false)
   319→      setFormData(emptyForm)
   320→      setEditingProduction(null)
   321→      fetchProductions()
   322→    } catch (err) {
   323→      toast({
   324→        title: 'Error',
   325→        description: err instanceof Error ? err.message : 'Failed to save production entry',
   326→        variant: 'destructive',
   327→      })
   328→    } finally {
   329→      setFormSubmitting(false)
   330→    }
   331→  }
   332→
   333→  const handleDelete = async () => {
   334→    if (!deleteTarget) return
   335→    setDeleting(true)
   336→    try {
   337→      await api.deleteProduction(deleteTarget.id)
   338→      toast({ title: 'Success', description: 'Production entry deleted successfully' })
   339→      setDeleteTarget(null)
   340→      fetchProductions()
   341→    } catch (err) {
   342→      toast({
   343→        title: 'Error',
   344→        description: err instanceof Error ? err.message : 'Failed to delete production entry',
   345→        variant: 'destructive',
   346→      })
   347→    } finally {
   348→      setDeleting(false)
   349→    }
   350→  }
   351→
   352→  const handleDeleteAll = async () => {
   353→    // No more 'DELETE ALL' typing requirement — the dialog now uses a
   354→    // simple Yes / No choice. We keep the deleteAllConfirm state around
   355→    // only so existing code that resets it (e.g. on dialog close) doesn't
   356→    // break, but it's no longer gating the delete.
   357→    setDeletingAll(true)
   358→    try {
   359→      const res = await api.deleteAllProductions()
   360→      toast({
   361→        title: 'Success',
   362→        description: `${res.deletedCount} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   363→      })
   364→      setDeleteAllOpen(false)
   365→      setDeleteAllConfirm('')
   366→      fetchProductions()
   367→    } catch (err) {
   368→      toast({
   369→        title: 'Error',
   370→        description: err instanceof Error ? err.message : 'Failed to delete all production entries',
   371→        variant: 'destructive',
   372→      })
   373→    } finally {
   374→      setDeletingAll(false)
   375→    }
   376→  }
   377→
   378→  const renderSkeletons = () =>
   379→    Array.from({ length: 3 }).map((_, i) => (
   380→      <TableRow key={i}>
   381→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   382→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   383→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   384→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   385→        {PRODUCT_FIELDS.map((_, j) => (
   386→          <TableCell key={j}><Skeleton className="h-4 w-16" /></TableCell>
   387→        ))}
   388→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   389→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   390→      </TableRow>
   391→    ))
   392→
   393→  return (
   394→    <div className="space-y-6 relative">
   395→      {/* Full-screen loading overlay shown during any delete operation.
   396→          The spinner + message stay visible until the API call completes and
   397→          the table refreshes, so the user always knows "something is happening". */}
   398→      {(deleting || deletingAll || bulkDeleting) && (
   399→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   400→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   401→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   402→            <div className="text-center">
   403→              <p className="text-lg font-semibold">
   404→                {bulkDeleting
   405→                  ? `Deleting ${selectedIds.size} entries...`
   406→                  : deletingAll
   407→                  ? 'Deleting all entries...'
   408→                  : 'Deleting entry...'}
   409→              </p>
   410→              <p className="text-sm text-muted-foreground mt-1">
   411→                Please wait while records are removed and stock is re-synced.
   412→              </p>
   413→            </div>
   414→          </div>
   415→        </div>
   416→      )}
   417→      {/* Header */}
   418→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   419→        <div className="flex items-center gap-3">
   420→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   421→            <Factory className="size-5" />
   422→          </div>
   423→          <div>
   424→            <h2 className="text-2xl font-bold tracking-tight">Production Management</h2>
   425→            <p className="text-sm text-muted-foreground">
   426→              Track daily paver block production entries
   427→            </p>
   428→          </div>
   429→        </div>
   430→        <div className="flex flex-wrap gap-2 w-full sm:w-auto">
   431→          <Button
   432→            variant="outline"
   433→            onClick={() => setImportOpen(true)}
   434→            className="w-full sm:w-auto"
   435→          >
   436→            <Upload className="size-4 mr-2" />
   437→            Import Excel
   438→          </Button>
   439→          {/* Bulk-delete button — only visible when at least one row is selected.
   440→              Clicking it opens the confirmation dialog below. */}
   441→          {selectedIds.size > 0 && (
   442→            <Button
   443→              variant="outline"
   444→              onClick={() => setBulkDeleteOpen(true)}
   445→              disabled={bulkDeleting || loading}
   446→              className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   447→            >
   448→              <Trash2 className="size-4 mr-2" />
   449→              Delete Selected
   450→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">
   451→                {selectedIds.size}
   452→              </Badge>
   453→            </Button>
   454→          )}
   455→          {/* Clear-selection button — small, ghost-styled, only visible when rows are selected */}
   456→          {selectedIds.size > 0 && (
   457→            <Button
   458→              variant="ghost"
   459→              onClick={clearSelection}
   460→              disabled={bulkDeleting}
   461→              className="w-full sm:w-auto"
   462→            >
   463→              Clear Selection
   464→            </Button>
   465→          )}
   466→          <Button
   467→            variant="outline"
   468→            onClick={() => setDeleteAllOpen(true)}
   469→            disabled={productions.length === 0 || loading}
   470→            className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   471→          >
   472→            <Trash className="size-4 mr-2" />
   473→            Delete All
   474→          </Button>
   475→          <Button
   476→            onClick={openAddDialog}
   477→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   478→          >
   479→            <Plus className="size-4" />
   480→            Add Production Entry
   481→          </Button>
   482→        </div>
   483→      </div>
   484→
   485→      {/* Search */}
   486→      <Card>
   487→        <CardContent className="pt-6">
   488→          <div className="relative">
   489→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   490→            <Input
   491→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   492→              value={search}
   493→              onChange={(e) => setSearch(e.target.value)}
   494→              className="pl-9"
   495→            />
   496→          </div>
   497→        </CardContent>
   498→      </Card>
   499→
   500→      {/* Table */}
   501→      <Card>
   502→        <CardHeader>
   503→          <CardTitle className="flex items-center justify-between">
   504→            <span>Production Records</span>
   505→            <div className="flex items-center gap-2">
   506→              {selectedIds.size > 0 && (
   507→                <Badge variant="secondary" className="bg-destructive/10 text-destructive border-destructive/30">
   508→                  {selectedIds.size} selected
   509→                </Badge>
   510→              )}
   511→              <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   512→                {filteredProductions.length} of {productions.length} record{productions.length !== 1 ? 's' : ''}
   513→              </Badge>
   514→            </div>
   515→          </CardTitle>
   516→        </CardHeader>
   517→        <CardContent>
   518→          <ScrollableTable maxHeight="max-h-[60vh]">
   519→            <Table>
   520→              <TableHeader>
   521→                <TableRow>
   522→                  <TableHead className="w-10 sticky left-0 bg-background z-20">
   523→                    <Checkbox
   524→                      checked={
   525→                        filteredProductions.length > 0 &&
   526→                        selectedIds.size === filteredProductions.length
   527→                      }
   528→                      onCheckedChange={toggleSelectAll}
   529→                      aria-label="Select all rows"
   530→                    />
   531→                  </TableHead>
   532→                  <TableHead className="sticky left-10 bg-background z-20">Date</TableHead>
   533→                  <TableHead className="text-right whitespace-nowrap">Cement</TableHead>
   534→                  {PRODUCT_FIELDS.map((f) => (
   535→                    <TableHead key={f.key} className="text-right whitespace-nowrap">{f.label}</TableHead>
   536→                  ))}
   537→                  <TableHead className="text-right whitespace-nowrap">Transport ₹</TableHead>
   538→                  <TableHead>Remarks</TableHead>
   539→                  <TableHead className="text-right">Actions</TableHead>
   540→                </TableRow>
   541→              </TableHeader>
   542→              <TableBody>
   543→                {loading ? (
   544→                  renderSkeletons()
   545→                ) : filteredProductions.length === 0 ? (
   546→                  <TableRow>
   547→                    <TableCell colSpan={PRODUCT_FIELDS.length + 6} className="h-32 text-center text-muted-foreground">
   548→                      No production entries yet. Click &quot;Add Production Entry&quot; to get started.
   549→                    </TableCell>
   550→                  </TableRow>
   551→                ) : (
   552→                  filteredProductions.map((prod) => (
   553→                    <TableRow
   554→                      key={prod.id}
   555→                      data-state={selectedIds.has(prod.id) ? 'selected' : undefined}
   556→                      className={selectedIds.has(prod.id) ? 'bg-emerald-50/60 dark:bg-emerald-900/15' : ''}
   557→                    >
   558→                      <TableCell className="w-10 sticky left-0 bg-background z-10 sticky">
   559→                        <Checkbox
   560→                          checked={selectedIds.has(prod.id)}
   561→                          onCheckedChange={() => toggleSelect(prod.id)}
   562→                          aria-label={`Select row for ${prod.date}`}
   563→                        />
   564→                      </TableCell>
   565→                      <TableCell className="font-medium whitespace-nowrap">
   566→                        {formatDate(prod.date)}
   567→                      </TableCell>
   568→                      <TableCell className="text-right font-mono">
   569→                        {enIN.format(prod.cement || 0)}
   570→                      </TableCell>
   571→                      {PRODUCT_FIELDS.map((f) => (
   572→                        <TableCell key={f.key} className="text-right font-mono">
   573→                          {enIN.format((prod as unknown as Record<string, unknown>)[f.key] as number || 0)}
   574→                        </TableCell>
   575→                      ))}
   576→                      <TableCell className="text-right whitespace-nowrap">
   577→                        {formatCurrency(prod.transportationCharge || 0)}
   578→                      </TableCell>
   579→                      <TableCell className="max-w-[150px] truncate text-muted-foreground">
   580→                        {prod.remarks || '—'}
   581→                      </TableCell>
   582→                      <TableCell className="text-right">
   583→                        <div className="flex items-center justify-end gap-1">
   584→                          <Button
   585→                            variant="ghost"
   586→                            size="icon"
   587→                            onClick={() => openEditDialog(prod)}
   588→                            title="Edit"
   589→                          >
   590→                            <Pencil className="size-4" />
   591→                          </Button>
   592→                          <Button
   593→                            variant="ghost"
   594→                            size="icon"
   595→                            onClick={() => setDeleteTarget(prod)}
   596→                            title="Delete"
   597→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   598→                          >
   599→                            <Trash2 className="size-4" />
   600→                          </Button>
   601→                        </div>
   602→                      </TableCell>
   603→                    </TableRow>
   604→                  ))
   605→                )}
   606→              </TableBody>
   607→            </Table>
   608→          </ScrollableTable>
   609→        </CardContent>
   610→      </Card>
   611→
   612→      {/* Add/Edit Dialog */}
   613→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   614→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   615→          <DialogHeader>
   616→            <DialogTitle>
   617→              {editingProduction ? 'Edit Production Entry' : 'Add Production Entry'}
   618→            </DialogTitle>
   619→            <DialogDescription>
   620→              {editingProduction
   621→                ? 'Update the production entry details below.'
   622→                : 'Fill in the details to create a new production entry.'}
   623→            </DialogDescription>
   624→          </DialogHeader>
   625→          <div className="grid gap-4 py-2">
   626→            <div className="grid gap-2">
   627→              <Label htmlFor="prod-date">
   628→                Date <span className="text-destructive">*</span>
   629→              </Label>
   630→              <Input
   631→                id="prod-date"
   632→                type="date"
   633→                value={formData.date}
   634→                onChange={(e) => handleFormChange('date', e.target.value)}
   635→              />
   636→            </div>
   637→            <div className="grid gap-2">
   638→              <Label htmlFor="prod-cement">Cement (bags)</Label>
   639→              <Input
   640→                id="prod-cement"
   641→                type="number"
   642→                min="0"
   643→                placeholder="0"
   644→                value={formData.cement}
   645→                onChange={(e) => handleFormChange('cement', e.target.value)}
   646→              />
   647→            </div>
   648→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   649→              {PRODUCT_FIELDS.map((f) => (
   650→                <div key={f.key} className="grid gap-2">
   651→                  <Label htmlFor={`prod-${f.key}`}>{f.label}</Label>
   652→                  <Input
   653→                    id={`prod-${f.key}`}
   654→                    type="number"
   655→                    min="0"
   656→                    placeholder="0"
   657→                    value={formData[f.key]}
   658→                    onChange={(e) => handleFormChange(f.key, e.target.value)}
   659→                  />
   660→                </div>
   661→              ))}
   662→            </div>
   663→            <div className="grid gap-2">
   664→              <Label htmlFor="prod-transport">Transportation Charge (₹)</Label>
   665→              <div className="relative">
   666→                <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   667→                <Input
   668→                  id="prod-transport"
   669→                  type="number"
   670→                  min="0"
   671→                  placeholder="0"
   672→                  className="pl-9"
   673→                  value={formData.transportationCharge}
   674→                  onChange={(e) => handleFormChange('transportationCharge', e.target.value)}
   675→                />
   676→              </div>
   677→            </div>
   678→            <div className="grid gap-2">
   679→              <Label htmlFor="prod-remarks">Remarks</Label>
   680→              <Textarea
   681→                id="prod-remarks"
   682→                placeholder="Optional remarks..."
   683→                value={formData.remarks}
   684→                onChange={(e) => handleFormChange('remarks', e.target.value)}
   685→                className="min-h-[80px]"
   686→              />
   687→            </div>
   688→          </div>
   689→          <DialogFooter>
   690→            <Button variant="outline" onClick={() => setFormOpen(false)} disabled={formSubmitting}>
   691→              Cancel
   692→            </Button>
   693→            <Button
   694→              onClick={handleSubmit}
   695→              disabled={formSubmitting}
   696→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   697→            >
   698→              {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   699→              {editingProduction ? 'Update Entry' : 'Create Entry'}
   700→            </Button>
   701→          </DialogFooter>
   702→        </DialogContent>
   703→      </Dialog>
   704→
   705→      {/* Delete confirmation */}
   706→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   707→        <AlertDialogContent>
   708→          <AlertDialogHeader>
   709→            <AlertDialogTitle>Delete Production Entry</AlertDialogTitle>
   710→            <AlertDialogDescription>
   711→              Are you sure you want to delete this production entry? This action cannot be undone.
   712→            </AlertDialogDescription>
   713→          </AlertDialogHeader>
   714→          <AlertDialogFooter>
   715→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   716→            <AlertDialogAction
   717→              onClick={handleDelete}
   718→              disabled={deleting}
   719→              className="bg-destructive text-white hover:bg-destructive/90"
   720→            >
   721→              {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   722→              Delete
   723→            </AlertDialogAction>
   724→          </AlertDialogFooter>
   725→        </AlertDialogContent>
   726→      </AlertDialog>
   727→
   728→      {/* Delete All confirmation — simple Yes / No dialog */}
   729→      <AlertDialog open={deleteAllOpen} onOpenChange={(open) => {
   730→        if (!open && !deletingAll) {
   731→          setDeleteAllOpen(false)
   732→          setDeleteAllConfirm('')
   733→        }
   734→      }}>
   735→        <AlertDialogContent className="max-w-md">
   736→          <AlertDialogHeader>
   737→            <AlertDialogTitle className="text-destructive">Delete ALL Production Entries?</AlertDialogTitle>
   738→            <AlertDialogDescription className="space-y-2 text-sm">
   739→              <span className="block">
   740→                You are about to permanently delete{' '}
   741→                <strong className="text-destructive">{productions.length} production entr{productions.length === 1 ? 'y' : 'ies'}</strong>.
   742→                This action <strong>cannot be undone</strong>.
   743→              </span>
   744→              <span className="block text-muted-foreground">
   745→                Linked customer bill-history aggregations will lose their production totals.
   746→                Customer, Order, Bill, Payment, Stock, and Dispatch records are NOT affected.
   747→              </span>
   748→            </AlertDialogDescription>
   749→          </AlertDialogHeader>
   750→          <AlertDialogFooter>
   751→            <AlertDialogCancel
   752→              disabled={deletingAll}
   753→              onClick={() => { setDeleteAllOpen(false); setDeleteAllConfirm('') }}
   754→              className="border-border"
   755→            >
   756→              No, Cancel
   757→            </AlertDialogCancel>
   758→            <AlertDialogAction
   759→              onClick={handleDeleteAll}
   760→              disabled={deletingAll}
   761→              className="bg-destructive text-white hover:bg-destructive/90"
   762→            >
   763→              {deletingAll && <Loader2 className="mr-2 size-4 animate-spin" />}
   764→              Yes, Delete All
   765→            </AlertDialogAction>
   766→          </AlertDialogFooter>
   767→        </AlertDialogContent>
   768→      </AlertDialog>
   769→
   770→      {/* Bulk Delete Selected confirmation */}
   771→      <AlertDialog open={bulkDeleteOpen} onOpenChange={(open) => {
   772→        if (!open && !bulkDeleting) setBulkDeleteOpen(false)
   773→      }}>
   774→        <AlertDialogContent>
   775→          <AlertDialogHeader>
   776→            <AlertDialogTitle className="text-destructive">
   777→              Delete {selectedIds.size} Selected Production {selectedIds.size === 1 ? 'Entry' : 'Entries'}?
   778→            </AlertDialogTitle>
   779→            <AlertDialogDescription className="space-y-2 text-sm">
   780→              <span className="block">
   781→                You are about to permanently delete{' '}
   782→                <strong className="text-destructive">{selectedIds.size} production {selectedIds.size === 1 ? 'entry' : 'entries'}</strong>.
   783→                This action <strong>cannot be undone</strong>.
   784→              </span>
   785→              <span className="block text-muted-foreground">
   786→                Stock snapshots for the affected dates will be re-aggregated automatically.
   787→                Other records (customers, orders, payments, etc.) are NOT affected.
   788→              </span>
   789→            </AlertDialogDescription>
   790→          </AlertDialogHeader>
   791→          <AlertDialogFooter>
   792→            <AlertDialogCancel disabled={bulkDeleting}>Cancel</AlertDialogCancel>
   793→            <AlertDialogAction
   794→              onClick={handleBulkDelete}
   795→              disabled={bulkDeleting || selectedIds.size === 0}
   796→              className="bg-destructive text-white hover:bg-destructive/90"
   797→            >
   798→              {bulkDeleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   799→              Delete {selectedIds.size} {selectedIds.size === 1 ? 'Entry' : 'Entries'}
   800→            </AlertDialogAction>
   801→          </AlertDialogFooter>
   802→        </AlertDialogContent>
   803→      </AlertDialog>
   804→
   805→      <ExcelImport module="production" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchProductions} />
   806→    </div>
   807→  )
   808→}

   810→export default ProductionModule
