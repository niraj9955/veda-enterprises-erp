   102→   102→// Smart auto-detect mapping between Excel columns and module fields
   103→   103→function autoMapColumns(excelColumns: string[], moduleFields: { key: string; label: string; required: boolean; aliases: string[] }[]): Record<string, string> {
   104→   104→  const mapping: Record<string, string> = {} // excelColumn -> fieldKey
   105→   105→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   106→   106→
   107→   107→  for (const field of moduleFields) {
   108→   108→    let bestMatch = ''
   109→   109→    let bestScore = 0
   110→   110→
   111→   111→    for (const excelCol of normalizedExcel) {
   112→   112→      // Check exact match with key
   113→   113→      if (excelCol.normalized === normalize(field.key)) {
   114→   114→        bestMatch = excelCol.original
   115→   115→        bestScore = 100
   116→   116→        break
   117→   117→      }
   118→   118→
   119→   119→      // Check aliases
   120→   120→      for (const alias of field.aliases) {
   121→   121→        const normAlias = normalize(alias)
   122→   122→        if (excelCol.normalized === normAlias) {
   123→   123→          if (100 > bestScore) {
   124→   124→            bestMatch = excelCol.original
   125→   125→            bestScore = 100
   126→   126→          }
   127→   127→          break
   128→   128→        }
   129→   129→      }
   130→   130→
   131→   131→      if (bestScore >= 100) break
   132→   132→
   133→   133→      // Fuzzy match: check if alias is contained in column name or vice versa
   134→   134→      for (const alias of field.aliases) {
   135→   135→        const normAlias = normalize(alias)
   136→   136→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   137→   137→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   138→   138→          if (score > bestScore) {
   139→   139→            bestMatch = excelCol.original
   140→   140→            bestScore = score
   141→   141→          }
   142→   142→        }
   143→   143→      }
   144→   144→    }
   145→   145→
   146→   146→    if (bestMatch && bestScore >= 50) {
   147→   147→      mapping[bestMatch] = field.key
   148→   148→    }
   149→   149→  }
   150→   150→
   151→   151→  return mapping
   152→   152→}

   154→   154→// Auto-detect the module from Excel column headers
   155→   155→function autoDetectModule(columns: string[]): string | null {
   156→   156→  const normalized = columns.map(normalize)
   157→   157→  const scores: Record<string, number> = {}
   158→   158→
   159→   159→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   160→   160→    scores[moduleKey] = 0
   161→   161→    for (const field of template.fields) {
   162→   162→      for (const alias of field.aliases) {
   163→   163→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   164→   164→          scores[moduleKey] += field.required ? 3 : 1
   165→   165→        }
   166→   166→      }
   167→   167→    }
   168→   168→  }
   169→   169→
   170→   170→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   171→   171→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   172→   172→}

   174→   174→// Transform data using mapping, also handle customer name -> ID resolution
   175→   175→function transformRow(
   176→   176→  row: Record<string, unknown>,
   177→   177→  mapping: Record<string, string>,
   178→   178→  customers?: { id: string; name: string }[]
   179→   179→): Record<string, unknown> {
   180→   180→  const result: Record<string, unknown> = {}
   181→   181→
   182→   182→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   183→   183→    let value = row[excelCol]
   184→   184→
   185→   185→    // Handle customer name -> ID resolution
   186→   186→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   187→   187→      const nameStr = String(value || '').trim().toLowerCase()
   188→   188→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   189→   189→      if (matched) {
   190→   190→        result['customerId'] = matched.id
   191→   191→      } else {
   192→   192→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   193→   193→      }
   194→   194→      continue
   195→   195→    }
   196→   196→
   197→   197→    // Handle numeric fields
   198→   198→    if (['quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate'].includes(fieldKey)) {
   199→   199→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   200→   200→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   201→   201→      continue
   202→   202→    }
   203→   203→
   204→   204→    // Handle date fields - try to parse various formats
   205→   205→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   206→   206→      result[fieldKey] = parseDate(String(value || ''))
   207→   207→      continue
   208→   208→    }
   209→   209→
   210→   210→    // Handle shift field - normalize
   211→   211→    if (fieldKey === 'shift') {
   212→   212→      const shiftStr = String(value || 'Morning').trim()
   213→   213→      const lower = shiftStr.toLowerCase()
   214→   214→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   215→   215→        result[fieldKey] = 'Morning'
   216→   216→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   217→   217→        result[fieldKey] = 'Evening'
   218→   218→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   219→   219→        result[fieldKey] = 'Night'
   220→   220→      } else {
   221→   221→        result[fieldKey] = shiftStr
   222→   222→      }
   223→   223→      continue
   224→   224→    }
   225→   225→
   226→   226→    // Handle payment type
   227→   227→    if (fieldKey === 'paymentType') {
   228→   228→      const ptStr = String(value || 'Cash').trim()
   229→   229→      const lower = ptStr.toLowerCase()
   230→   230→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   231→   231→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   232→   232→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   233→   233→      else result[fieldKey] = ptStr
   234→   234→      continue
   235→   235→    }
   236→   236→
   237→   237→    // Handle expense category
   238→   238→    if (fieldKey === 'category') {
   239→   239→      const catStr = String(value || '').trim()
   240→   240→      const lower = catStr.toLowerCase()
   241→   241→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   242→   242→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   243→   243→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   244→   244→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   245→   245→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   246→   246→      else result[fieldKey] = catStr
   247→   247→      continue
   248→   248→    }
   249→   249→
   250→   250→    // Handle status
   251→   251→    if (fieldKey === 'status') {
   252→   252→      const stStr = String(value || 'Pending').trim()
   253→   253→      const lower = stStr.toLowerCase()
   254→   254→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   255→   255→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   256→   256→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   257→   257→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   258→   258→      else result[fieldKey] = stStr
   259→   259→      continue
   260→   260→    }
   261→   261→
   262→   262→    result[fieldKey] = String(value || '')
   263→   263→  }
   264→   264→
   265→   265→  // For orders/dispatch: calculate amount if not mapped
   266→   266→  if (mapping) {
   267→   267→    const targetModule = Object.values(mapping)
   268→   268→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   269→   269→      const qty = Number(result.quantity || 0)
   270→   270→      const rate = Number(result.rate || 0)
   271→   271→      result.amount = qty * rate
   272→   272→    }
   273→   273→  }
   274→   274→
   275→   275→  return result
   276→   276→}

   278→   278→// Parse various date formats to YYYY-MM-DD
   279→   279→function parseDate(value: string): string {
   280→   280→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   281→   281→
   282→   282→  const trimmed = value.trim()
   283→   283→
   284→   284→  // Already in YYYY-MM-DD format
   285→   285→  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed
   286→   286→
   287→   287→  // DD-MM-YYYY or DD/MM/YYYY
   288→   288→  const dmyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   289→   289→  if (dmyMatch) {
   290→   290→    const [, d, m, y] = dmyMatch
   291→   291→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   292→   292→  }
   293→   293→
   294→   294→  // DD-MM-YY or DD/MM/YY
   295→   295→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2})$/)
   296→   296→  if (dmyShortMatch) {
   297→   297→    const [, d, m, y] = dmyShortMatch
   298→   298→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   299→   299→  }
   300→   300→
   301→   301→  // MM/DD/YYYY (US format) - only if first number > 12
   302→   302→  const mdyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   303→   303→  if (mdyMatch) {
   304→   304→    const [, m, d, y] = mdyMatch
   305→   305→    if (Number(m) > 12) {
   306→   306→      return `${y}-${d.padStart(2, '0')}-${m.padStart(2, '0')}`
   307→   307→    }
   308→   308→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   309→   309→  }
   310→   310→
   311→   311→  // Try native Date parsing as fallback
   312→   312→  try {
   313→   313→    const date = new Date(trimmed)
   314→   314→    if (!isNaN(date.getTime())) {
   315→   315→      return date.toISOString().split('T')[0]
   316→   316→    }
   317→   317→  } catch {}
   318→   318→
   319→   319→  return trimmed
   320→   320→}

   359→   359→        toast({ title: 'Empty file', description: 'No data found in the uploaded file', variant: 'destructive' })
   360→   360→        return
   361→   361→      }
   362→   362→
   363→   363→      const data = jsonData as Record<string, unknown>[]
   364→   364→      const columns = Object.keys(data[0])
   365→   365→
   366→   366→      setRawData(data)
   367→   367→      setExcelColumns(columns)
   368→   368→
   369→   369→      // Auto-detect column mapping
   370→   370→      const autoMapping = autoMapColumns(columns, template.fields)
   371→   371→      setColumnMapping(autoMapping)
   372→   372→
   373→   373→      // Apply transformation immediately
   374→   374→      const transformed = data.map((row) => transformRow(row, autoMapping, customers))
   375→   375→      setTransformedData(transformed)
   376→   376→
   377→   377→      const mappedCount = Object.keys(autoMapping).length
   378→   378→      const totalFields = template.fields.length
   379→   379→      toast({
   380→   380→        title: 'File parsed successfully',
   381→   381→        description: `Found ${data.length} rows. Auto-mapped ${mappedCount} of ${totalFields} fields. Review and adjust if needed.`,
   382→   382→      })
   383→   383→    } catch (err) {
   384→   384→      toast({ title: 'Parse error', description: 'Could not read the Excel file. Make sure it is a valid .xlsx or .csv file.', variant: 'destructive' })
   385→   385→    }
   386→   386→  }
   387→   387→
   388→   388→  const handleMappingChange = (excelCol: string, fieldKey: string) => {
   389→   389→    const newMapping = { ...columnMapping }
   390→   390→
   391→   391→    // Remove old mapping for this excel column
   392→   392→    delete newMapping[excelCol]
   393→   393→
   394→   394→    // Remove any existing mapping to the same field key
   395→   395→    for (const [col, fKey] of Object.entries(newMapping)) {
   396→   396→      if (fKey === fieldKey) {
   397→   397→        delete newMapping[col]
   398→   398→      }
   399→   399→    }
   400→   400→
   401→   401→    // Set new mapping
   402→   402→    if (fieldKey && fieldKey !== 'skip') {
   403→   403→      newMapping[excelCol] = fieldKey
   404→   404→    }
   405→   405→
   406→   406→    setColumnMapping(newMapping)
   407→   407→
   408→   408→    // Re-transform data
   409→   409→    const transformed = rawData.map((row) => transformRow(row, newMapping, customers))
   410→   410→    setTransformedData(transformed)
   411→   411→  }
   412→   412→
   413→   413→  const handleImport = async () => {
   414→   414→    if (transformedData.length === 0) return

   418→   418→      setResult({ imported: res.imported, total: res.total, errors: res.errors })
   419→   419→      if (res.imported > 0) {
   420→   420→        toast({ title: 'Import successful', description: `${res.imported} of ${res.total} rows imported` })
   421→   421→        onSuccess()
   422→   422→      }
   423→   423→    } catch (err) {
   424→   424→      toast({ title: 'Import failed', description: err instanceof Error ? err.message : 'Unknown error', variant: 'destructive' })
   425→   425→    } finally {
   426→   426→      setImporting(false)
   427→   427→    }
   428→   428→  }
   429→   429→
   430→   430→  const downloadTemplate = () => {

   432→   432→    const csvContent = [headers.join(','), ''].join('\n')
   433→   433→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   434→   434→    const url = URL.createObjectURL(blob)
   435→   435→    const a = document.createElement('a')
   436→   436→    a.href = url
   437→   437→    a.download = `${module}_import_template.csv`
   438→   438→    a.click()
   439→   439→    URL.revokeObjectURL(url)
   440→   440→    toast({ title: 'Template downloaded', description: 'Fill in your data and upload the file' })
   441→   441→  }
   442→   442→
   443→   443→  const handleClose = () => {
   444→   444→    setRawData([])
   445→   445→    setTransformedData([])
   446→   446→    setColumnMapping({})
   447→   447→    setExcelColumns([])
   448→   448→    setResult(null)
   449→   449→    setFileName('')
   450→   450→    if (fileInputRef.current) fileInputRef.current.value = ''
   451→   451→    onClose()
   452→   452→  }
   453→   453→
   454→   454→  const mappedFieldKeys = new Set(Object.values(columnMapping))
   455→   455→  const unmappedRequired = template.fields.filter((f) => f.required && !mappedFieldKeys.has(f.key))
   456→   456→
   457→   457→  return (
   458→   458→    <Dialog open={open} onOpenChange={handleClose}>
   459→   459→      <DialogContent className="max-w-4xl max-h-[90vh] flex flex-col">
   460→   460→        <DialogHeader>
   461→   461→          <DialogTitle className="flex items-center gap-2">
   462→   462→            <FileSpreadsheet className="h-5 w-5 text-emerald-600" />
   463→   463→            Import {template.label} from Excel
   464→   464→          </DialogTitle>
   465→   465→          <DialogDescription>
   466→   466→            Upload an Excel or CSV file. Columns are auto-detected and mapped — review and adjust before importing.
   467→   467→          </DialogDescription>
   468→   468→        </DialogHeader>
   469→   469→
   470→   470→        <div className="space-y-4 flex-1 overflow-auto">
   471→   471→          {/* Step 1: Download template */}
   472→   472→          <div className="space-y-2">
   473→   473→            <h4 className="font-medium text-sm flex items-center gap-2">
   474→   474→              <span className="w-6 h-6 bg-emerald-600 text-white rounded-full flex items-center justify-center text-xs font-bold">1</span>
   475→   475→              Template & Format
   476→   476→            </h4>
   477→   477→            <div className="flex flex-wrap gap-2">
   478→   478→              {template.fields.map((field) => (
   479→   479→                <Badge key={field.key} variant={field.required ? 'default' : 'outline'} className="text-xs">
   480→   480→                  {field.label} {field.required && '*'}
   481→   481→                </Badge>
   482→   482→              ))}
   483→   483→            </div>
   484→   484→            <Button variant="outline" size="sm" onClick={downloadTemplate}>
   485→   485→              <Download className="h-4 w-4 mr-2" />
   486→   486→              Download CSV Template
   487→   487→            </Button>
   488→   488→          </div>
   489→   489→
   490→   490→          {/* Step 2: Upload file */}
   491→   491→          <div className="space-y-2">
   492→   492→            <h4 className="font-medium text-sm flex items-center gap-2">
   493→   493→              <span className="w-6 h-6 bg-emerald-600 text-white rounded-full flex items-center justify-center text-xs font-bold">2</span>
   494→   494→              Upload Excel/CSV File
   495→   495→            </h4>
   496→   496→            <div
   497→   497→              className="border-2 border-dashed rounded-lg p-6 text-center hover:border-emerald-400 transition-colors cursor-pointer"
   498→   498→              onClick={() => fileInputRef.current?.click()}
   499→   499→            >
   500→   500→              <Upload className="h-8 w-8 mx-auto mb-2 text-muted-foreground" />
   501→   501→              <p className="text-sm text-muted-foreground">
   502→   502→                {fileName ? fileName : 'Click to upload .xlsx, .xls, or .csv file'}
   503→   503→              </p>
   504→   504→              <input
   505→   505→                ref={fileInputRef}
   506→   506→                type="file"
   507→   507→                accept=".xlsx,.xls,.csv"
   508→   508→                className="hidden"
   509→   509→                onChange={handleFileUpload}
   510→   510→              />
   511→   511→            </div>
   512→   512→          </div>

   514→   514→          {/* Step 3: Column Mapping */}
   515→   515→          {excelColumns.length > 0 && (

   517→   517→              <h4 className="font-medium text-sm flex items-center gap-2">

   524→   524→                <Alert variant="destructive" className="py-2">

   531→   531→
   532→   532→              <div className="border rounded-lg overflow-hidden">
   533→   533→                <Table>
   534→   534→                  <TableHeader>

   541→   541→                  </TableHeader>
   542→   542→                  <TableBody>

   545→   545→                      const field = mappedField ? template.fields.find((f) => f.key === mappedField) : null
   546→   546→                      return (

   570→   570→                              <Badge variant="outline" className="text-xs bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-900/20 dark:text-emerald-400 dark:border-emerald-800">
   571→   571→                                <CheckCircle2 className="h-3 w-3 mr-1" /> Mapped

   579→   579→                        </TableRow>

   595→   595→              <ScrollArea className="max-h-56">
   596→   596→                <div className="overflow-x-auto">

   625→   625→                          </TableCell>
   626→   626→                        </TableRow>

   628→   628→                    </TableBody>
   629→   629→                  </Table>
   630→   630→                </div>
   631→   631→              </ScrollArea>
   632→   632→            </div>

   634→   634→
   635→   635→          {/* Import result */}
   636→   636→          {result && (
   637→   637→            <Alert variant={result.errors && result.errors.length > 0 ? 'destructive' : 'default'}>
   638→   638→              <div className="flex items-start gap-2">
   639→   639→                {result.errors && result.errors.length > 0 ? (
   640→   640→                  <AlertCircle className="h-4 w-4 mt-0.5" />
   641→   641→                ) : (
   642→   642→                  <CheckCircle2 className="h-4 w-4 mt-0.5" />
   643→   643→                )}
   644→   644→                <AlertDescription>
   645→   645→                  <p className="font-medium">
   646→   646→                    Imported {result.imported} of {result.total} rows successfully
   647→   647→                  </p>
   648→   648→                  {result.errors && result.errors.length > 0 && (
   649→   649→                    <ScrollArea className="max-h-32 mt-2">
   650→   650→                      <ul className="text-xs space-y-1">

   654→   654→                        {result.errors.length > 20 && (

   660→   660→                </AlertDescription>

   664→   664→        </div>

   666→   666→        <DialogFooter>
