    83→  address: string
    84→  contactNumber: string
    85→  product: string
    86→  quantity: number
    87→  rate?: number
    88→  amount: number
    89→  transporterName?: string
    90→  transporterFair?: number
    91→  remarks: string
    92→  createdAt: string
    93→  updatedAt: string
    94→}
    95→
    96→interface DailySellFormData {
    97→  date: string
    98→  customerName: string
    99→  address: string
   100→  contactNumber: string
   101→  product: string
   102→  quantity: string
   103→  rate: string
   104→  amount: string
   105→  transporterName: string
   106→  transporterFair: string
   107→  remarks: string
   108→}

   110→// ── Helpers ─────────────────────────────────────────────────────────────────
   111→
   112→const formatCurrency = (amount: number): string =>
   113→  new Intl.NumberFormat('en-IN', {
   114→    style: 'currency',
   115→    currency: 'INR',
   116→    minimumFractionDigits: 0,
   117→    maximumFractionDigits: 2,
   118→  }).format(amount)
   119→
   120→const formatDate = (dateStr: string): string =>
   121→  new Date(dateStr + 'T00:00:00').toLocaleDateString('en-IN', {
   122→    day: '2-digit',
   123→    month: 'short',
   124→    year: 'numeric',
   125→  })
   126→
   127→const emptyForm: DailySellFormData = {
   128→  date: '',
   129→  customerName: '',
   130→  address: '',
   131→  contactNumber: '',
   132→  product: '',
   133→  quantity: '',
   134→  rate: '',
   135→  amount: '',
   136→  transporterName: '',
   137→  transporterFair: '',
   138→  remarks: '',
   139→}

   141→// ── Component ───────────────────────────────────────────────────────────────
   142→
   143→export function DailySellModule() {
   144→  const [dailySells, setDailySells] = React.useState<DailySell[]>([])
   145→  const [search, setSearch] = React.useState('')
   146→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   147→  const [loading, setLoading] = React.useState(true)
   148→  const [formOpen, setFormOpen] = React.useState(false)
   149→  const [editingItem, setEditingItem] = React.useState<DailySell | null>(null)
   150→  const [formData, setFormData] = React.useState<DailySellFormData>(emptyForm)
   151→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   152→  const [deleteTarget, setDeleteTarget] = React.useState<DailySell | null>(null)
   153→  const [deleting, setDeleting] = React.useState(false)
   154→
   155→  // Multi-select state — mirrors the Production module pattern so the user
   156→  // gets identical UX: tick individual rows or use the header checkbox to
   157→  // select all currently-filtered rows, then click "Delete Selected".
   158→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   159→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   160→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   161→
   162→  // Delete All state — simple Yes/No confirmation dialog.
   163→  const [deleteAllOpen, setDeleteAllOpen] = React.useState(false)
   164→  const [deletingAll, setDeletingAll] = React.useState(false)
   165→
   166→  // Excel import
   167→  const [importOpen, setImportOpen] = React.useState(false)
   168→
   169→  const fetchData = React.useCallback(async () => {
   170→    setLoading(true)
   171→    try {
   172→      const res = await api.getDailySells()
   173→      const data = (res.dailySells as DailySell[]).sort(
   174→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   175→      )
   176→      setDailySells(data)
   177→    } catch (err) {
   178→      toast({
   179→        title: 'Error',
   180→        description: err instanceof Error ? err.message : 'Failed to fetch daily sell data',
   181→        variant: 'destructive',
   182→      })
   183→    } finally {
   184→      setLoading(false)
   185→    }
   186→  }, [])
   187→
   188→  // Debounced search
   189→  React.useEffect(() => {
   190→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   191→    return () => clearTimeout(t)
   192→  }, [search])
   193→
   194→  // Client-side filter
   195→  const filteredDailySells = React.useMemo(() => {
   196→    if (!debouncedSearch.trim()) return dailySells
   197→    const q = debouncedSearch.toLowerCase()
   198→    return dailySells.filter((item: any) =>
   199→      ['date', 'customerName', 'address', 'contactNumber', 'product', 'remarks'].some((f) =>
   200→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   201→      )
   202→    )
   203→  }, [dailySells, debouncedSearch])
   204→
   205→  React.useEffect(() => {
   206→    fetchData()
   207→  }, [fetchData])
   208→
   209→  // ── Selection handlers ─────────────────────────────────────────────
   210→  const toggleSelect = (id: string) => {
   211→    setSelectedIds((prev) => {
   212→      const next = new Set(prev)
   213→      if (next.has(id)) next.delete(id)
   214→      else next.add(id)
   215→      return next
   216→    })
   217→  }
   218→
   219→  const toggleSelectAll = () => {
   220→    setSelectedIds((prev) => {
   221→      if (prev.size === filteredDailySells.length && filteredDailySells.length > 0) {
   222→        return new Set()
   223→      }
   224→      return new Set(filteredDailySells.map((s) => s.id))
   225→    })
   226→  }
   227→
   228→  const clearSelection = () => setSelectedIds(new Set())
   229→
   230→  // ── Bulk delete selected ───────────────────────────────────────────
   231→  const handleBulkDelete = async () => {
   232→    const ids = Array.from(selectedIds)
   233→    if (ids.length === 0) return
   234→    setBulkDeleting(true)
   235→    try {
   236→      const res = await api.bulkDeleteDailySells(ids)
   237→      toast({
   238→        title: 'Success',
   239→        description: `${res.deletedCount} of ${ids.length} daily sell entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   240→      })
   241→      setBulkDeleteOpen(false)
   242→      clearSelection()
   243→      fetchData()
   244→    } catch (err) {
   245→      toast({
   246→        title: 'Error',
   247→        description: err instanceof Error ? err.message : 'Failed to delete selected daily sell entries',
   248→        variant: 'destructive',
   249→      })
   250→    } finally {
   251→      setBulkDeleting(false)
   252→    }
   253→  }
   254→
   255→  // ── Delete ALL ─────────────────────────────────────────────────────
   256→  const handleDeleteAll = async () => {
   257→    setDeletingAll(true)
   258→    try {
   259→      const res = await api.deleteAllDailySells()
   260→      toast({
   261→        title: 'Success',
   262→        description: `${res.deletedCount} daily sell entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   263→      })
   264→      setDeleteAllOpen(false)
   265→      clearSelection()
   266→      fetchData()
   267→    } catch (err) {
   268→      toast({
   269→        title: 'Error',
   270→        description: err instanceof Error ? err.message : 'Failed to delete all daily sell entries',
   271→        variant: 'destructive',
   272→      })
   273→    } finally {
   274→      setDeletingAll(false)
   275→    }
   276→  }
   277→
   278→  const openAddDialog = () => {
   279→    setEditingItem(null)
   280→    // Check for pending AI result from chat widget — auto-fill if present
   281→    const pending = consumePendingAiResult('dailySell')
   282→    if (pending) {
   283→      setFormData({
   284→        date: String(pending.date || ''),
   285→        customerName: String(pending.customerName || ''),
   286→        address: String(pending.address || ''),
   287→        contactNumber: String(pending.contactNumber || ''),
   288→        product: String(pending.product || ''),
   289→        quantity: pending.quantity != null ? String(pending.quantity) : '',
   290→        rate: pending.rate != null ? String(pending.rate) : '',
   291→        amount: pending.amount != null ? String(pending.amount) : '',
   292→        transporterName: String(pending.transporterName || ''),
   293→        transporterFair: pending.transporterFair != null ? String(pending.transporterFair) : '',
   294→        remarks: String(pending.remarks || ''),
   295→      })
   296→      toast({ title: 'AI auto-fill applied', description: 'Edit & verify before saving.' })
   297→    } else {
   298→      setFormData(emptyForm)
   299→    }
   300→    setFormOpen(true)
   301→  }
   302→
   303→  const openEditDialog = (item: DailySell) => {
   304→    setEditingItem(item)
   305→    setFormData({
   306→      date: item.date ? item.date.split('T')[0] : '',
   307→      customerName: item.customerName || '',
   308→      address: item.address || '',
   309→      contactNumber: item.contactNumber || '',
   310→      product: item.product || '',
   311→      quantity: String(item.quantity ?? ''),
   312→      rate: String(item.rate ?? ''),
   313→      amount: String(item.amount || ''),
   314→      transporterName: item.transporterName || '',
   315→      transporterFair: String(item.transporterFair ?? ''),
   316→      remarks: item.remarks || '',
   317→    })
   318→    setFormOpen(true)
   319→  }
   320→
   321→  const handleFormChange = (field: keyof DailySellFormData, value: string) => {
   322→    setFormData((prev) => ({ ...prev, [field]: value }))
   323→  }
   324→
   325→  // ── Computed amount (auto = quantity × rate) ───────────────────────────
   326→  const computedAmount = React.useMemo(() => {
   327→    const qty = Number(formData.quantity) || 0
   328→    const rate = Number(formData.rate) || 0
   329→    return qty * rate
   330→  }, [formData.quantity, formData.rate])
   331→
   332→  const handleSubmit = async () => {
   333→    // Unified empty-form check — show ONE popup instead of cascading errors
   334→    if (isFormEmpty([formData.date, formData.customerName, formData.address, formData.contactNumber, formData.product, formData.quantity, formData.rate, formData.amount, formData.transporterName, formData.transporterFair, formData.remarks])) {
   335→      toast(showPleaseFillDataToast())
   336→      return
   337→    }
   338→    if (!formData.date) {
   339→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   340→      return
   341→    }
   342→    if (!formData.customerName.trim()) {
   343→      toast({ title: 'Validation Error', description: 'Customer name is required', variant: 'destructive' })
   344→      return
   345→    }
   346→
   347→    setFormSubmitting(true)
   348→    try {
   349→      const payload = {
   350→        date: formData.date,
   351→        customerName: formData.customerName.trim(),
   352→        address: formData.address.trim(),
   353→        contactNumber: formData.contactNumber.trim(),
   354→        product: formData.product.trim(),
   355→        quantity: Number(formData.quantity) || 0,
   356→        rate: Number(formData.rate) || 0,
   357→        amount: computedAmount,
   358→        transporterName: formData.transporterName.trim(),
   359→        transporterFair: Number(formData.transporterFair) || 0,
   360→        remarks: formData.remarks.trim(),
   361→      }
   362→
   363→      if (editingItem) {
   364→        await api.updateDailySell(editingItem.id, payload)
   365→        toast({ title: 'Success', description: 'Daily sell entry updated successfully' })
   366→      } else {
   367→        await api.createDailySell(payload)
   368→        toast({ title: 'Success', description: 'Daily sell entry created successfully' })
   369→      }
   370→
   371→      setFormOpen(false)
   372→      setFormData(emptyForm)
   373→      setEditingItem(null)
   374→      fetchData()
   375→    } catch (err) {
   376→      toast({
   377→        title: 'Error',
   378→        description: err instanceof Error ? err.message : 'Failed to save daily sell entry',
   379→        variant: 'destructive',
   380→      })
   381→    } finally {
   382→      setFormSubmitting(false)
   383→    }
   384→  }
   385→
   386→  const handleDelete = async () => {
   387→    if (!deleteTarget) return
   388→    setDeleting(true)
   389→    try {
   390→      await api.deleteDailySell(deleteTarget.id)
   391→      toast({ title: 'Success', description: 'Daily sell entry deleted successfully' })
   392→      setDeleteTarget(null)
   393→      fetchData()
   394→    } catch (err) {
   395→      toast({
   396→        title: 'Error',
   397→        description: err instanceof Error ? err.message : 'Failed to delete daily sell entry',
   398→        variant: 'destructive',
   399→      })
   400→    } finally {
   401→      setDeleting(false)
   402→    }
   403→  }
   404→
   405→  const totalAmount = dailySells.reduce((sum, s) => sum + (s.amount || 0), 0)
   406→
   407→  const renderSkeletons = () =>
   408→    Array.from({ length: 5 }).map((_, i) => (
   409→      <TableRow key={i}>
   410→        <TableCell><Skeleton className="h-4 w-6" /></TableCell>
   411→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   412→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   413→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   414→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   415→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   416→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   417→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   418→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   419→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   420→      </TableRow>
   421→    ))
   422→
   423→  return (
   424→    <div className="space-y-6 relative">
   425→      {/* Full-screen loading overlay shown during any delete operation.
   426→          Mirrors the Production module pattern so the user always sees
   427→          that something is happening. */}
   428→      {(deleting || deletingAll || bulkDeleting) && (
   429→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   430→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   431→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   432→            <div className="text-center">
   433→              <p className="text-lg font-semibold">
   434→                {bulkDeleting
   435→                  ? `Deleting ${selectedIds.size} entries...`
   436→                  : deletingAll
   437→                  ? 'Deleting all entries...'
   438→                  : 'Deleting entry...'}
   439→              </p>
   440→              <p className="text-sm text-muted-foreground mt-1">
   441→                Please wait while records are removed.
   442→              </p>
   443→            </div>
   444→          </div>
   445→        </div>
   446→      )}
   447→
   448→      {/* Header */}
   449→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   450→        <div className="flex items-center gap-3">
   451→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   452→            <ShoppingCart className="size-5" />
   453→          </div>
   454→          <div>
   455→            <h2 className="text-2xl font-bold tracking-tight">Daily Sell</h2>
   456→            <p className="text-sm text-muted-foreground">
   457→              Track daily sales transactions
   458→            </p>
   459→          </div>
   460→        </div>
   461→        <div className="flex flex-wrap gap-2 w-full sm:w-auto">
   462→          <Button
   463→            variant="outline"
   464→            onClick={() => setImportOpen(true)}
   465→            className="w-full sm:w-auto"
   466→          >
   467→            <Upload className="size-4 mr-2" />
   468→            Import Excel
   469→          </Button>
   470→          {/* Bulk-delete button — only visible when at least one row is
   471→              selected. Clicking it opens the confirmation dialog below. */}
   472→          {selectedIds.size > 0 && (
   473→            <Button
   474→              variant="outline"
   475→              onClick={() => setBulkDeleteOpen(true)}
   476→              disabled={bulkDeleting || loading}
   477→              className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   478→            >
   479→              <Trash2 className="size-4 mr-2" />
   480→              Delete Selected
   481→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">
   482→                {selectedIds.size}
   483→              </Badge>
   484→            </Button>
   485→          )}
   486→          {/* Clear-selection button — small, ghost-styled, only visible when rows are selected */}
   487→          {selectedIds.size > 0 && (
   488→            <Button
   489→              variant="ghost"
   490→              onClick={clearSelection}
   491→              disabled={bulkDeleting}
   492→              className="w-full sm:w-auto"
   493→            >
   494→              Clear Selection
   495→            </Button>
   496→          )}
   497→          <Button
   498→            variant="outline"
   499→            onClick={() => setDeleteAllOpen(true)}
   500→            disabled={dailySells.length === 0 || loading}
   501→            className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   502→          >
   503→            <Trash className="size-4 mr-2" />
   504→            Delete All
   505→          </Button>
   506→          <Button
   507→            onClick={openAddDialog}
   508→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   509→          >
   510→            <Plus className="size-4" />
   511→            Add Daily Sell
   512→          </Button>
   513→        </div>
   514→      </div>
   515→
   516→      {/* Summary */}
   517→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   518→        <Card className="border-l-4 border-l-emerald-500">
   519→          <CardContent className="p-4">
   520→            <p className="text-xs text-muted-foreground">Total Sales</p>
   521→            {loading ? (
   522→              <Skeleton className="h-6 w-32 mt-1" />
   523→            ) : (
   524→              <p className="text-xl font-bold text-emerald-700">{formatCurrency(totalAmount)}</p>
   525→            )}
   526→          </CardContent>
   527→        </Card>
   528→        <Card className="border-l-4 border-l-amber-500">
   529→          <CardContent className="p-4">
   530→            <p className="text-xs text-muted-foreground">Total Records</p>
   531→            {loading ? (
   532→              <Skeleton className="h-6 w-16 mt-1" />
   533→            ) : (
   534→              <p className="text-xl font-bold text-amber-700">{dailySells.length}</p>
   535→            )}
   536→          </CardContent>
   537→        </Card>
   538→      </div>
   539→
   540→      {/* Search */}
   541→      <Card>
   542→        <CardContent className="pt-6">
   543→          <div className="relative">
   544→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   545→            <Input
   546→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   547→              value={search}
   548→              onChange={(e) => setSearch(e.target.value)}
   549→              className="pl-9"
   550→            />
   551→          </div>
   552→        </CardContent>
   553→      </Card>
   554→
   555→      {/* Table */}
   556→      <Card>
   557→        <CardHeader>
   558→          <CardTitle className="flex items-center justify-between gap-2 flex-wrap">
   559→            <span>Daily Sell Records</span>
   560→            <div className="flex items-center gap-2 flex-wrap">
   561→              {selectedIds.size > 0 && (
   562→                <Badge variant="secondary" className="bg-destructive/10 text-destructive border-destructive/30">
   563→                  {selectedIds.size} selected
   564→                </Badge>
   565→              )}
