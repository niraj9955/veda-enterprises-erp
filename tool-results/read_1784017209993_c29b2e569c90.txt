   103→// Lightweight Bill shape for the "Link to Bill" dropdown in the form.
   104→// We only fetch the fields needed to render each option: id, billNumber,
   105→// grandTotal, paidAmount, balanceAmount, date, status. The dropdown shows
   106→// "BILL-XXXX — ₹balance pending" so the user knows which bill is outstanding.
   107→interface BillOption {
   108→  id: string
   109→  billNumber: string
   110→  grandTotal: number
   111→  paidAmount: number
   112→  balanceAmount: number
   113→  date: string
   114→  status: string
   115→}
   116→
   117→interface OutstandingEntry {
   118→  customerId: string
   119→  customerName: string
   120→  totalOrderAmount: number
   121→  totalPayments: number
   122→  outstanding: number
   123→}

   167→// ── Component ───────────────────────────────────────────────────────────────
   168→
   169→export function PaymentModule() {
   170→  // State
   171→  const [payments, setPayments] = React.useState<Payment[]>([])
   172→  const [search, setSearch] = React.useState('')
   173→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   174→  const [customers, setCustomers] = React.useState<Customer[]>([])
   175→  const [orders, setOrders] = React.useState<Order[]>([])
   176→  const [loading, setLoading] = React.useState(true)
   177→
   178→  // Dialog states
   179→  const [dialogOpen, setDialogOpen] = React.useState(false)
   180→  const [editingPayment, setEditingPayment] = React.useState<Payment | null>(null)
   181→  const [formData, setFormData] = React.useState<PaymentFormData>(emptyForm)
   182→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   183→
   184→  // Bills for the currently-selected customer — used to populate the
   185→  // "Link to Bill" dropdown in the payment form. Fetched whenever the user
   186→  // picks a customer (or opens the edit dialog with a pre-selected customer).
   187→  // Only outstanding bills (status = 'partial' or 'draft' or 'sent') are
   188→  // shown — fully paid bills are hidden to keep the dropdown short.
   189→  const [customerBills, setCustomerBills] = React.useState<BillOption[]>([])
   190→  const [loadingBills, setLoadingBills] = React.useState(false)
   191→
   192→  // Excel import
   193→  const [importOpen, setImportOpen] = React.useState(false)
   194→
   195→  // Delete confirmation
   196→  const [deleteTarget, setDeleteTarget] = React.useState<Payment | null>(null)
   197→  const [deleting, setDeleting] = React.useState(false)
   198→
   199→  // ── Multi-select state ──────────────────────────────────────────────────
   200→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   201→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   202→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   203→
   204→  const toggleSelect = (id: string) => {
   205→    setSelectedIds((prev) => {
   206→      const next = new Set(prev)
   207→      if (next.has(id)) next.delete(id)
   208→      else next.add(id)
   209→      return next
   210→    })
   211→  }
   212→
   213→  const toggleSelectAll = () => {
   214→    setSelectedIds((prev) => {
   215→      if (prev.size === filteredPayments.length && filteredPayments.length > 0) {
   216→        return new Set()
   217→      }
   218→      return new Set(filteredPayments.map((i) => i.id))
   219→    })
   220→  }
   221→
   222→  const clearSelection = () => setSelectedIds(new Set())
   223→
   224→  const handleBulkDelete = async () => {
   225→    const ids = Array.from(selectedIds)
   226→    if (ids.length === 0) return
   227→    setBulkDeleting(true)
   228→    try {
   229→      const res = await api.bulkDeletePayments(ids)
   230→      toast({
   231→        title: 'Success',
   232→        description: `${res.deletedCount} of ${ids.length} payment${res.deletedCount === 1 ? '' : 's'} deleted`,
   233→      })
   234→      setBulkDeleteOpen(false)
   235→      clearSelection()
   236→      fetchPayments()
   237→      fetchOrders()
   238→    } catch (err) {
   239→      toast({
   240→        title: 'Error',
   241→        description: err instanceof Error ? err.message : 'Failed to delete selected payments',
   242→        variant: 'destructive',
   243→      })
   244→    } finally {
   245→      setBulkDeleting(false)
   246→    }
   247→  }
   248→
   249→  // ── Computed: Summary values ─────────────────────────────────────────────
   250→  const totalReceived = React.useMemo(
   251→    () => payments.reduce((sum, p) => sum + Number(p.amount), 0),
   252→    [payments]
   253→  )
   254→
   255→  const cashPayments = React.useMemo(
   256→    () =>
   257→      payments
   258→        .filter((p) => p.paymentType === 'Cash')
   259→        .reduce((sum, p) => sum + Number(p.amount), 0),
   260→    [payments]
   261→  )
   262→
   263→  const upiBankPayments = React.useMemo(
   264→    () =>
   265→      payments
   266→        .filter((p) => p.paymentType === 'UPI' || p.paymentType === 'Bank Transfer')
   267→        .reduce((sum, p) => sum + Number(p.amount), 0),
   268→    [payments]
   269→  )
   270→
   271→  // ── Computed: Customer outstanding ───────────────────────────────────────
   272→  const outstandingEntries = React.useMemo<OutstandingEntry[]>(() => {
   273→    const map = new Map<string, OutstandingEntry>()
   274→
   275→    // Sum order amounts per customer (exclude Cancelled)
   276→    for (const order of orders) {
   277→      if (order.status === 'Cancelled') continue
   278→      const existing = map.get(order.customerId)
   279→      if (existing) {
   280→        existing.totalOrderAmount += Number(order.amount)
   281→      } else {
   282→        map.set(order.customerId, {
   283→          customerId: order.customerId,
   284→          customerName: order.customer?.name || 'Unknown',
   285→          totalOrderAmount: Number(order.amount),
   286→          totalPayments: 0,
   287→          outstanding: 0,
   288→        })
   289→      }
   290→    }
   291→
   292→    // Sum payments per customer
   293→    for (const payment of payments) {
   294→      const existing = map.get(payment.customerId)
   295→      if (existing) {
   296→        existing.totalPayments += Number(payment.amount)
   297→      } else {
   298→        map.set(payment.customerId, {
   299→          customerId: payment.customerId,
   300→          customerName: payment.customer?.name || 'Unknown',
   301→          totalOrderAmount: 0,
   302→          totalPayments: Number(payment.amount),
   303→          outstanding: 0,
   304→        })
   305→      }
   306→    }
   307→
   308→    // Calculate outstanding
   309→    for (const entry of map.values()) {
   310→      entry.outstanding = entry.totalOrderAmount - entry.totalPayments
   311→    }
   312→
   313→    // Return only customers with outstanding > 0, sorted by outstanding descending
   314→    return Array.from(map.values())
   315→      .filter((e) => e.outstanding > 0.01)
   316→      .sort((a, b) => b.outstanding - a.outstanding)
   317→  }, [orders, payments])
   318→
   319→  // ── Fetch data ───────────────────────────────────────────────────────────
   320→  const fetchPayments = React.useCallback(async () => {
   321→    setLoading(true)
   322→    try {
   323→      const res = await api.getPayments()
   324→      const paymentList = (res.payments as Payment[]).sort(
   325→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   326→      )
   327→      setPayments(paymentList)
   328→    } catch (err) {
   329→      toast({
   330→        title: 'Error',
   331→        description: err instanceof Error ? err.message : 'Failed to fetch payments',
   332→        variant: 'destructive',
   333→      })
   334→    } finally {
   335→      setLoading(false)
   336→    }
   337→  }, [])
   338→
   339→  const fetchCustomers = React.useCallback(async () => {
   340→    try {
   341→      const res = await api.getCustomers()
   342→      setCustomers(res.customers as Customer[])
   343→    } catch {
   344→      // Silently fail — customer dropdown will just be empty
   345→    }
   346→  }, [])
   347→
   348→  const fetchOrders = React.useCallback(async () => {
   349→    try {
   350→      const res = await api.getOrders()
   351→      setOrders(res.orders as Order[])
   352→    } catch {
   353→      // Silently fail — outstanding section will just be empty
   354→    }
   355→  }, [])
   356→
   357→  // Debounced search
   358→  React.useEffect(() => {
   359→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   360→    return () => clearTimeout(t)
   361→  }, [search])
   362→
   363→  // Client-side filter
   364→  const filteredPayments = React.useMemo(() => {
   365→    if (!debouncedSearch.trim()) return payments
   366→    const q = debouncedSearch.toLowerCase()
   367→    return payments.filter((item: any) =>
   368→      ['date', 'paymentType', 'remarks'].some((f) =>
   369→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   370→      )
   371→    )
   372→  }, [payments, debouncedSearch])
   373→
   374→  React.useEffect(() => {
   375→    fetchPayments()
   376→    fetchCustomers()
   377→    fetchOrders()
   378→  }, [fetchPayments, fetchCustomers, fetchOrders])
   379→
   380→  // ── Form handlers ────────────────────────────────────────────────────────
   381→  const openCreateDialog = () => {
   382→    setEditingPayment(null)
   383→    setFormData(emptyForm)
   384→    setCustomerBills([])
   385→    setDialogOpen(true)
   386→  }
   387→
   388→  const openEditDialog = (payment: Payment) => {
   389→    setEditingPayment(payment)
   390→    setFormData({
   391→      customerId: payment.customerId,
   392→      paymentType: payment.paymentType,
   393→      amount: payment.amount,
   394→      date: payment.date,
   395→      remarks: payment.remarks,
   396→      billId: payment.billId || '',
   397→    })
   398→    setDialogOpen(true)
   399→    // Fetch this customer's outstanding bills (including the linked one,
   400→    // even if it's already paid, so the user sees the current link).
   401→    fetchCustomerBills(payment.customerId)
   402→  }
   403→
   404→  // Fetch outstanding bills for a customer — populates the "Link to Bill"
   405→  // dropdown. Safe to call repeatedly; debouncing not needed because the
   406→  // customer change is a discrete user action (not a keystroke stream).
   407→  const fetchCustomerBills = async (customerId: string) => {
   408→    if (!customerId) {
   409→      setCustomerBills([])
   410→      return
   411→    }
   412→    setLoadingBills(true)
   413→    try {
   414→      // We use the existing /api/bills endpoint with search filter. The
   415→      // response is the full bill list but we filter client-side by
   416→      // customerId + outstanding status. For large datasets this should
   417→      // eventually become a server-side filter, but for now this is fast
   418→      // enough (typically <50 bills per customer).
   419→      const data = await api.getBills() as { bills: BillOption[] }
   420→      const filtered = (data.bills || [])
   421→        .filter((b) => {
   422→          // Match by customerId. Note: /api/bills returns bills with
   423→          // customerId as a string (toObject renames _id → id).
   424→          const billCustomerId = (b as any).customerId
   425→          return billCustomerId && String(billCustomerId) === String(customerId)
   426→        })
   427→        // Show outstanding + the currently-linked bill (even if 'paid')
   428→        .filter((b) => {
   429→          if (editingPayment && editingPayment.billId && String(b.id) === String(editingPayment.billId)) return true
   430→          return b.status !== 'paid' && b.status !== 'cancelled'
   431→        })
   432→        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   433→      setCustomerBills(filtered)
   434→    } catch (err) {
   435→      console.error('Failed to fetch customer bills:', err)
   436→      setCustomerBills([])
   437→    } finally {
   438→      setLoadingBills(false)
   439→    }
   440→  }
   441→
   442→  // When customer changes in the form, refresh the bills dropdown.
   443→  // Wrapped in useEffect so we don't refetch on every render.
   444→  React.useEffect(() => {
   445→    if (dialogOpen && formData.customerId) {
   446→      fetchCustomerBills(formData.customerId)
   447→    } else if (!formData.customerId) {
   448→      setCustomerBills([])
   449→    }
   450→    // eslint-disable-next-line react-hooks/exhaustive-deps
   451→  }, [formData.customerId, dialogOpen])
   452→
   453→  const handleFormChange = (field: keyof PaymentFormData, value: string) => {
   454→    setFormData((prev) => ({ ...prev, [field]: value }))
   455→  }
   456→
   457→  const handleSubmit = async () => {
   458→    // Unified empty-form check — show ONE popup instead of cascading errors
   459→    if (isFormEmpty([formData.customerId, formData.paymentType, formData.amount, formData.date, formData.remarks, formData.billId])) {
   460→      toast(showPleaseFillDataToast())
   461→      return
   462→    }
   463→    // Validation
   464→    if (!formData.customerId) {
   465→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   466→      return
   467→    }
   468→    if (!formData.paymentType) {
   469→      toast({ title: 'Validation Error', description: 'Please select a payment type', variant: 'destructive' })
   470→      return
   471→    }
   472→    if (!formData.amount || Number(formData.amount) <= 0) {
   473→      toast({ title: 'Validation Error', description: 'Amount must be greater than 0', variant: 'destructive' })
   474→      return
   475→    }
   476→    if (!formData.date) {
   477→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   478→      return
   479→    }
   480→
   481→    setFormSubmitting(true)
   482→    try {
   483→      const payload: Record<string, unknown> = {
   484→        customerId: formData.customerId,
   485→        paymentType: formData.paymentType,
   486→        amount: Number(formData.amount),
   487→        date: formData.date,
   488→        remarks: formData.remarks,
   489→      }
   490→
   491→      // Include billId in the payload so the server can link/unlink the
   492→      // payment to a Bill and auto-sync the Bill's paidAmount.
   493→      // Empty string → unlink (server treats "" as null).
   494→      if (editingPayment) {
   495→        // On edit, always send billId (even if empty) so the server can
   496→        // detect "unlink" intent vs "no change".
   497→        payload.billId = formData.billId || null
   498→      } else {
   499→        // On create, only send billId if user picked one.
   500→        if (formData.billId) payload.billId = formData.billId
   501→      }
   502→
   503→      if (editingPayment) {
   504→        await api.updatePayment(editingPayment.id, payload)
   505→        toast({
   506→          title: 'Success',
   507→          description: formData.billId
   508→            ? 'Payment updated and linked to bill. Bill paid amount auto-synced.'
   509→            : 'Payment updated successfully',
   510→        })
   511→      } else {
   512→        await api.createPayment(payload)
   513→        toast({
   514→          title: 'Success',
   515→          description: formData.billId
   516→            ? 'Payment recorded and linked to bill. Bill paid amount auto-synced.'
   517→            : 'Payment recorded successfully',
   518→        })
   519→      }
   520→
   521→      setDialogOpen(false)
   522→      setEditingPayment(null)
   523→      setFormData(emptyForm)
   524→      setCustomerBills([])
   525→      fetchPayments()
   526→      fetchOrders() // Refresh for outstanding calculations
   527→    } catch (err) {
   528→      toast({
   529→        title: 'Error',
   530→        description: err instanceof Error ? err.message : `Failed to ${editingPayment ? 'update' : 'record'} payment`,
   531→        variant: 'destructive',
   532→      })
   533→    } finally {
   534→      setFormSubmitting(false)
   535→    }
   536→  }
   537→
   538→  // ── Delete handler ──────────────────────────────────────────────────────
   539→  const handleDelete = async () => {
   540→    if (!deleteTarget) return
   541→    setDeleting(true)
   542→    try {
   543→      await api.deletePayment(deleteTarget.id)
   544→      toast({ title: 'Success', description: 'Payment deleted successfully' })
   545→      setDeleteTarget(null)
   546→      fetchPayments()
   547→      fetchOrders()
   548→    } catch (err) {
   549→      toast({
   550→        title: 'Error',
   551→        description: err instanceof Error ? err.message : 'Failed to delete payment',
   552→        variant: 'destructive',
   553→      })
   554→    } finally {
   555→      setDeleting(false)
   556→    }
   557→  }
   558→
   559→  // ── Render: Loading skeletons ───────────────────────────────────────────
   560→  const renderSkeletons = () =>
   561→    Array.from({ length: 5 }).map((_, i) => (
   562→      <TableRow key={i}>
   563→        <TableCell><Skeleton className="h-4 w-6" /></TableCell>
   564→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   565→        <TableCell><Skeleton className="h-5 w-16" /></TableCell>
   566→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   567→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   568→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   569→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   570→      </TableRow>
   571→    ))
   572→
   573→  // ── Render: Payment type badge ──────────────────────────────────────────
   574→  const renderPaymentTypeBadge = (type: string) => (
   575→    <Badge className={`${PAYMENT_TYPE_COLORS[type] || 'bg-gray-100 text-gray-700 border-gray-200'} gap-1`}>
   576→      {PAYMENT_TYPE_ICONS[type]}
   577→      {type}
   578→    </Badge>
   579→  )
   580→
   581→  // ── Render: Payment dialog ──────────────────────────────────────────────
   582→  const renderPaymentDialog = () => (
   583→    <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   584→      <DialogContent className="max-w-[95vw] sm:max-w-lg max-h-[90vh] overflow-y-auto">
   585→        <DialogHeader>
   586→          <DialogTitle>
   587→            {editingPayment ? 'Edit Payment' : 'Receive Payment'}
   588→          </DialogTitle>
   589→          <DialogDescription>
   590→            {editingPayment
   591→              ? 'Update the payment details below.'
   592→              : 'Fill in the details to record a new payment.'}
   593→          </DialogDescription>
   594→        </DialogHeader>
   595→
   596→        <div className="grid gap-4 py-2">
   597→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   598→          <CustomerSearchInput
   599→            value={formData.customerId}
   600→            onSelect={(c) => handleFormChange('customerId', c.id)}

   788→            <Wallet className="size-5" />
   789→          </div>

   791→            <h2 className="text-2xl font-bold tracking-tight">Payment Management</h2>

   793→              Track and manage customer payments
   794→            </p>
   795→          </div>
   796→        </div>
   797→        <div className="flex flex-col gap-2 sm:flex-row sm:w-auto">
   798→          <Button
   799→            variant="outline"
   800→            onClick={() => setImportOpen(true)}
   801→            className="w-full sm:w-auto"
   802→          >

   806→          {selectedIds.size > 0 && (
   807→            <Button
   808→              variant="outline"
   809→              onClick={() => setBulkDeleteOpen(true)}
   810→              disabled={bulkDeleting || loading}
   811→              className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   812→            >
   813→              <Trash2 className="size-4 mr-2" />Delete Selected
   814→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">{selectedIds.size}</Badge>
   815→            </Button>

   818→            <Button variant="ghost" onClick={clearSelection} disabled={bulkDeleting} className="w-full sm:w-auto">Clear Selection</Button>
   819→          )}

   821→            onClick={openCreateDialog}
   822→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   823→          >
   824→            <Plus className="size-4" />

   983→                    </TableRow>
   984→                  ))
   985→                )}
   986→              </TableBody>
   987→            </Table>
   988→          </div>
   989→        </CardContent>
   990→      </Card>
   991→
   992→      {/* Customer Outstanding Section */}
   993→      <Card>

   995→          <CardTitle className="flex items-center gap-2">

   999→        </CardHeader>
  1000→        <CardContent>
  1001→          {loading ? (
  1002→            <div className="space-y-3">
  1003→              {Array.from({ length: 3 }).map((_, i) => (
  1004→                <Skeleton key={i} className="h-10 w-full" />
  1005→              ))}
  1006→            </div>
  1007→          ) : outstandingEntries.length === 0 ? (

  1020→                    <TableHead className="text-right">Outstanding (₹)</TableHead>

  1038→                    </TableRow>

  1044→        </CardContent>
  1045→      </Card>
  1046→
  1047→      {/* Dialogs */}
  1048→      {renderPaymentDialog()}

  1050→
  1051→      {/* Bulk Delete Selected confirmation */}
  1052→      <AlertDialog open={bulkDeleteOpen} onOpenChange={(open) => { if (!open && !bulkDeleting) setBulkDeleteOpen(false) }}>
  1053→        <AlertDialogContent>
  1054→          <AlertDialogHeader>
  1055→            <AlertDialogTitle className="text-destructive">Delete {selectedIds.size} Selected {selectedIds.size === 1 ? 'Payment' : 'Payments'}?</AlertDialogTitle>
  1056→            <AlertDialogDescription className="space-y-2 text-sm">
  1057→              <span className="block">You are about to permanently delete <strong className="text-destructive">{selectedIds.size} payment{selectedIds.size === 1 ? '' : 's'}</strong>. This action <strong>cannot be undone</strong>.</span>
  1058→            </AlertDialogDescription>
  1059→          </AlertDialogHeader>
  1060→          <AlertDialogFooter>
  1061→            <AlertDialogCancel disabled={bulkDeleting}>Cancel</AlertDialogCancel>
  1062→            <AlertDialogAction onClick={handleBulkDelete} disabled={bulkDeleting || selectedIds.size === 0} className="bg-destructive text-white hover:bg-destructive/90">{bulkDeleting && <Loader2 className="mr-2 size-4 animate-spin" />}Delete {selectedIds.size} {selectedIds.size === 1 ? 'Payment' : 'Payments'}</AlertDialogAction>
  1063→          </AlertDialogFooter>
  1064→        </AlertDialogContent>
  1065→      </AlertDialog>
  1066→
  1067→      <ExcelImport module="payments" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchPayments} />
  1068→    </div>
  1069→  )
  1070→}
