    64→// ── Types ───────────────────────────────────────────────────────────────────
    65→
    66→interface Customer {
    67→  id: string
    68→  name: string
    69→}

    71→interface Payment {
    72→  id: string
    73→  customerId: string
    74→  paymentType: string
    75→  amount: number
    76→  date: string
    77→  remarks: string
    78→  createdAt: string
    79→  billId?: string | null
    80→  billNumber?: string
    81→  customer: { id: string; name: string }
    82→}
    83→
    84→interface Order {
    85→  id: string
    86→  customerId: string
    87→  amount: number
    88→  status: string
    89→  createdAt: string
    90→  customer: { id: string; name: string }
    91→}
    92→
    93→interface PaymentFormData {
    94→  customerId: string
    95→  paymentType: string
    96→  amount: number | string
    97→  date: string
    98→  remarks: string
    99→  billId: string  // optional link to a Bill — empty string means "no link"
   100→}
   101→
   102→// Lightweight Bill shape for the "Link to Bill" dropdown in the form.
   103→// We only fetch the fields needed to render each option: id, billNumber,
   104→// grandTotal, paidAmount, balanceAmount, date, status. The dropdown shows
   105→// "BILL-XXXX — ₹balance pending" so the user knows which bill is outstanding.
   106→interface BillOption {
   107→  id: string
   108→  billNumber: string
   109→  grandTotal: number
   110→  paidAmount: number
   111→  balanceAmount: number
   112→  date: string
   113→  status: string
   114→}
   115→
   116→interface OutstandingEntry {
   117→  customerId: string
   118→  customerName: string
   119→  totalOrderAmount: number
   120→  totalPayments: number
   121→  outstanding: number
   122→}

   166→// ── Component ───────────────────────────────────────────────────────────────
   167→
   168→export function PaymentModule() {
   169→  // State
   170→  const [payments, setPayments] = React.useState<Payment[]>([])
   171→  const [search, setSearch] = React.useState('')
   172→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   173→  const [customers, setCustomers] = React.useState<Customer[]>([])
   174→  const [orders, setOrders] = React.useState<Order[]>([])
   175→  const [loading, setLoading] = React.useState(true)
   176→
   177→  // Dialog states
   178→  const [dialogOpen, setDialogOpen] = React.useState(false)
   179→  const [editingPayment, setEditingPayment] = React.useState<Payment | null>(null)
   180→  const [formData, setFormData] = React.useState<PaymentFormData>(emptyForm)
   181→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   182→
   183→  // Bills for the currently-selected customer — used to populate the
   184→  // "Link to Bill" dropdown in the payment form. Fetched whenever the user
   185→  // picks a customer (or opens the edit dialog with a pre-selected customer).
   186→  // Only outstanding bills (status = 'partial' or 'draft' or 'sent') are
   187→  // shown — fully paid bills are hidden to keep the dropdown short.
   188→  const [customerBills, setCustomerBills] = React.useState<BillOption[]>([])
   189→  const [loadingBills, setLoadingBills] = React.useState(false)
   190→
   191→  // Excel import
   192→  const [importOpen, setImportOpen] = React.useState(false)
   193→
   194→  // Delete confirmation
   195→  const [deleteTarget, setDeleteTarget] = React.useState<Payment | null>(null)
   196→  const [deleting, setDeleting] = React.useState(false)
   197→
   198→  // ── Computed: Summary values ─────────────────────────────────────────────
   199→  const totalReceived = React.useMemo(
   200→    () => payments.reduce((sum, p) => sum + Number(p.amount), 0),
   201→    [payments]
   202→  )
   203→
   204→  const cashPayments = React.useMemo(
   205→    () =>
   206→      payments
   207→        .filter((p) => p.paymentType === 'Cash')
   208→        .reduce((sum, p) => sum + Number(p.amount), 0),
   209→    [payments]
   210→  )
   211→
   212→  const upiBankPayments = React.useMemo(
   213→    () =>
   214→      payments
   215→        .filter((p) => p.paymentType === 'UPI' || p.paymentType === 'Bank Transfer')
   216→        .reduce((sum, p) => sum + Number(p.amount), 0),
   217→    [payments]
   218→  )
   219→
   220→  // ── Computed: Customer outstanding ───────────────────────────────────────
   221→  const outstandingEntries = React.useMemo<OutstandingEntry[]>(() => {
   222→    const map = new Map<string, OutstandingEntry>()
   223→
   224→    // Sum order amounts per customer (exclude Cancelled)
   225→    for (const order of orders) {
   226→      if (order.status === 'Cancelled') continue
   227→      const existing = map.get(order.customerId)
   228→      if (existing) {
   229→        existing.totalOrderAmount += Number(order.amount)
   230→      } else {
   231→        map.set(order.customerId, {
   232→          customerId: order.customerId,
   233→          customerName: order.customer?.name || 'Unknown',
   234→          totalOrderAmount: Number(order.amount),
   235→          totalPayments: 0,
   236→          outstanding: 0,
   237→        })
   238→      }
   239→    }
   240→
   241→    // Sum payments per customer
   242→    for (const payment of payments) {
   243→      const existing = map.get(payment.customerId)
   244→      if (existing) {
   245→        existing.totalPayments += Number(payment.amount)
   246→      } else {
   247→        map.set(payment.customerId, {
   248→          customerId: payment.customerId,
   249→          customerName: payment.customer?.name || 'Unknown',
   250→          totalOrderAmount: 0,
   251→          totalPayments: Number(payment.amount),
   252→          outstanding: 0,
   253→        })
   254→      }
   255→    }
   256→
   257→    // Calculate outstanding
   258→    for (const entry of map.values()) {
   259→      entry.outstanding = entry.totalOrderAmount - entry.totalPayments
   260→    }
   261→
   262→    // Return only customers with outstanding > 0, sorted by outstanding descending
   263→    return Array.from(map.values())
   264→      .filter((e) => e.outstanding > 0.01)
   265→      .sort((a, b) => b.outstanding - a.outstanding)
   266→  }, [orders, payments])
   267→
   268→  // ── Fetch data ───────────────────────────────────────────────────────────
   269→  const fetchPayments = React.useCallback(async () => {
   270→    setLoading(true)
   271→    try {
   272→      const res = await api.getPayments()
   273→      const paymentList = (res.payments as Payment[]).sort(
   274→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   275→      )
   276→      setPayments(paymentList)
   277→    } catch (err) {
   278→      toast({
   279→        title: 'Error',
   280→        description: err instanceof Error ? err.message : 'Failed to fetch payments',
   281→        variant: 'destructive',
   282→      })
   283→    } finally {
   284→      setLoading(false)
   285→    }
   286→  }, [])
   287→
   288→  const fetchCustomers = React.useCallback(async () => {
   289→    try {
   290→      const res = await api.getCustomers()
   291→      setCustomers(res.customers as Customer[])
   292→    } catch {
   293→      // Silently fail — customer dropdown will just be empty
   294→    }
   295→  }, [])
   296→
   297→  const fetchOrders = React.useCallback(async () => {
   298→    try {
   299→      const res = await api.getOrders()
   300→      setOrders(res.orders as Order[])
   301→    } catch {
   302→      // Silently fail — outstanding section will just be empty
   303→    }
   304→  }, [])
   305→
   306→  // Debounced search
   307→  React.useEffect(() => {
   308→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   309→    return () => clearTimeout(t)
   310→  }, [search])
   311→
   312→  // Client-side filter
   313→  const filteredPayments = React.useMemo(() => {
   314→    if (!debouncedSearch.trim()) return payments
   315→    const q = debouncedSearch.toLowerCase()
   316→    return payments.filter((item: any) =>
   317→      ['date', 'paymentType', 'remarks'].some((f) =>
   318→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   319→      )
   320→    )
   321→  }, [payments, debouncedSearch])
   322→
   323→  React.useEffect(() => {
   324→    fetchPayments()
   325→    fetchCustomers()
   326→    fetchOrders()
   327→  }, [fetchPayments, fetchCustomers, fetchOrders])
   328→
   329→  // ── Form handlers ────────────────────────────────────────────────────────
   330→  const openCreateDialog = () => {
   331→    setEditingPayment(null)
   332→    setFormData(emptyForm)
   333→    setCustomerBills([])
   334→    setDialogOpen(true)
   335→  }
   336→
   337→  const openEditDialog = (payment: Payment) => {
   338→    setEditingPayment(payment)
   339→    setFormData({
   340→      customerId: payment.customerId,
   341→      paymentType: payment.paymentType,
   342→      amount: payment.amount,
   343→      date: payment.date,
   344→      remarks: payment.remarks,
   345→      billId: payment.billId || '',
   346→    })
   347→    setDialogOpen(true)
   348→    // Fetch this customer's outstanding bills (including the linked one,
   349→    // even if it's already paid, so the user sees the current link).
   350→    fetchCustomerBills(payment.customerId)
   351→  }
   352→
   353→  // Fetch outstanding bills for a customer — populates the "Link to Bill"
   354→  // dropdown. Safe to call repeatedly; debouncing not needed because the
   355→  // customer change is a discrete user action (not a keystroke stream).
   356→  const fetchCustomerBills = async (customerId: string) => {
   357→    if (!customerId) {
   358→      setCustomerBills([])
   359→      return
   360→    }
   361→    setLoadingBills(true)
   362→    try {
   363→      // We use the existing /api/bills endpoint with search filter. The
   364→      // response is the full bill list but we filter client-side by
   365→      // customerId + outstanding status. For large datasets this should
   366→      // eventually become a server-side filter, but for now this is fast
   367→      // enough (typically <50 bills per customer).
   368→      const data = await api.getBills() as { bills: BillOption[] }
   369→      const filtered = (data.bills || [])
   370→        .filter((b) => {
   371→          // Match by customerId. Note: /api/bills returns bills with
   372→          // customerId as a string (toObject renames _id → id).
   373→          const billCustomerId = (b as any).customerId
   374→          return billCustomerId && String(billCustomerId) === String(customerId)
   375→        })
   376→        // Show outstanding + the currently-linked bill (even if 'paid')
   377→        .filter((b) => {
   378→          if (editingPayment && editingPayment.billId && String(b.id) === String(editingPayment.billId)) return true
   379→          return b.status !== 'paid' && b.status !== 'cancelled'
   380→        })
   381→        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   382→      setCustomerBills(filtered)
   383→    } catch (err) {
   384→      console.error('Failed to fetch customer bills:', err)
   385→      setCustomerBills([])
   386→    } finally {
   387→      setLoadingBills(false)
   388→    }
   389→  }
   390→
   391→  // When customer changes in the form, refresh the bills dropdown.
   392→  // Wrapped in useEffect so we don't refetch on every render.
   393→  React.useEffect(() => {
   394→    if (dialogOpen && formData.customerId) {
   395→      fetchCustomerBills(formData.customerId)
   396→    } else if (!formData.customerId) {
   397→      setCustomerBills([])
   398→    }
   399→    // eslint-disable-next-line react-hooks/exhaustive-deps
   400→  }, [formData.customerId, dialogOpen])
   401→
   402→  const handleFormChange = (field: keyof PaymentFormData, value: string) => {
   403→    setFormData((prev) => ({ ...prev, [field]: value }))
   404→  }
   405→
   406→  const handleSubmit = async () => {
   407→    // Unified empty-form check — show ONE popup instead of cascading errors
   408→    if (isFormEmpty([formData.customerId, formData.paymentType, formData.amount, formData.date, formData.remarks, formData.billId])) {
   409→      toast(showPleaseFillDataToast())
   410→      return
   411→    }
   412→    // Validation
   413→    if (!formData.customerId) {
   414→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   415→      return
   416→    }
   417→    if (!formData.paymentType) {
   418→      toast({ title: 'Validation Error', description: 'Please select a payment type', variant: 'destructive' })
   419→      return
   420→    }
   421→    if (!formData.amount || Number(formData.amount) <= 0) {
   422→      toast({ title: 'Validation Error', description: 'Amount must be greater than 0', variant: 'destructive' })
   423→      return
   424→    }
   425→    if (!formData.date) {
   426→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   427→      return
   428→    }
   429→
   430→    setFormSubmitting(true)
   431→    try {
   432→      const payload: Record<string, unknown> = {
   433→        customerId: formData.customerId,
   434→        paymentType: formData.paymentType,
   435→        amount: Number(formData.amount),
   436→        date: formData.date,
   437→        remarks: formData.remarks,
   438→      }
   439→
   440→      // Include billId in the payload so the server can link/unlink the
   441→      // payment to a Bill and auto-sync the Bill's paidAmount.
   442→      // Empty string → unlink (server treats "" as null).
   443→      if (editingPayment) {
   444→        // On edit, always send billId (even if empty) so the server can
   445→        // detect "unlink" intent vs "no change".
   446→        payload.billId = formData.billId || null
   447→      } else {
   448→        // On create, only send billId if user picked one.
   449→        if (formData.billId) payload.billId = formData.billId
   450→      }
   451→
   452→      if (editingPayment) {
   453→        await api.updatePayment(editingPayment.id, payload)
   454→        toast({
   455→          title: 'Success',
   456→          description: formData.billId
   457→            ? 'Payment updated and linked to bill. Bill paid amount auto-synced.'
   458→            : 'Payment updated successfully',
   459→        })
   460→      } else {
   461→        await api.createPayment(payload)
   462→        toast({
   463→          title: 'Success',
   464→          description: formData.billId
   465→            ? 'Payment recorded and linked to bill. Bill paid amount auto-synced.'
   466→            : 'Payment recorded successfully',
   467→        })
   468→      }
   469→
   470→      setDialogOpen(false)
   471→      setEditingPayment(null)
   472→      setFormData(emptyForm)
   473→      setCustomerBills([])
   474→      fetchPayments()
   475→      fetchOrders() // Refresh for outstanding calculations
   476→    } catch (err) {
   477→      toast({
   478→        title: 'Error',
   479→        description: err instanceof Error ? err.message : `Failed to ${editingPayment ? 'update' : 'record'} payment`,
   480→        variant: 'destructive',
   481→      })
   482→    } finally {
   483→      setFormSubmitting(false)
   484→    }
   485→  }
   486→
   487→  // ── Delete handler ──────────────────────────────────────────────────────
   488→  const handleDelete = async () => {
   489→    if (!deleteTarget) return
   490→    setDeleting(true)
   491→    try {
   492→      await api.deletePayment(deleteTarget.id)
   493→      toast({ title: 'Success', description: 'Payment deleted successfully' })
   494→      setDeleteTarget(null)
   495→      fetchPayments()
   496→      fetchOrders()
   497→    } catch (err) {
   498→      toast({
   499→        title: 'Error',
   500→        description: err instanceof Error ? err.message : 'Failed to delete payment',
   501→        variant: 'destructive',
   502→      })
   503→    } finally {
   504→      setDeleting(false)
   505→    }
   506→  }
   507→
   508→  // ── Render: Loading skeletons ───────────────────────────────────────────
   509→  const renderSkeletons = () =>
   510→    Array.from({ length: 5 }).map((_, i) => (
   511→      <TableRow key={i}>
   512→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   513→        <TableCell><Skeleton className="h-5 w-16" /></TableCell>
   514→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   515→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   516→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   517→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   518→      </TableRow>
   519→    ))
   520→
   521→  // ── Render: Payment type badge ──────────────────────────────────────────
   522→  const renderPaymentTypeBadge = (type: string) => (
   523→    <Badge className={`${PAYMENT_TYPE_COLORS[type] || 'bg-gray-100 text-gray-700 border-gray-200'} gap-1`}>
   524→      {PAYMENT_TYPE_ICONS[type]}
   525→      {type}
   526→    </Badge>
   527→  )
   528→
   529→  // ── Render: Payment dialog ──────────────────────────────────────────────
   530→  const renderPaymentDialog = () => (
   531→    <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   532→      <DialogContent className="max-w-[95vw] sm:max-w-lg max-h-[90vh] overflow-y-auto">
   533→        <DialogHeader>
   534→          <DialogTitle>
   535→            {editingPayment ? 'Edit Payment' : 'Receive Payment'}
   536→          </DialogTitle>
   537→          <DialogDescription>
   538→            {editingPayment
   539→              ? 'Update the payment details below.'
   540→              : 'Fill in the details to record a new payment.'}
   541→          </DialogDescription>
   542→        </DialogHeader>
   543→
   544→        <div className="grid gap-4 py-2">
   545→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   546→          <CustomerSearchInput
   547→            value={formData.customerId}
   548→            onSelect={(c) => handleFormChange('customerId', c.id)}
   549→            onClear={() => handleFormChange('customerId', '')}
   550→            required
   551→            label="Customer"
   552→            placeholder="Type customer name or mobile to search..."
   553→          />
   554→
   555→          {/* Payment Type */}
   556→          <div className="grid gap-2">
   557→            <Label htmlFor="payment-type">
   558→              Payment Type <span className="text-destructive">*</span>
   559→            </Label>
   560→            <Select
   561→              value={formData.paymentType}
   562→              onValueChange={(val) => handleFormChange('paymentType', val)}
   563→            >
   564→              <SelectTrigger className="w-full">
   565→                <SelectValue placeholder="Select payment type" />
   566→              </SelectTrigger>
   567→              <SelectContent>
   568→                {PAYMENT_TYPES.map((type) => (
   569→                  <SelectItem key={type} value={type}>
   570→                    {type}
   571→                  </SelectItem>
   572→                ))}
   573→              </SelectContent>
   574→            </Select>
   575→          </div>
   576→
   577→          {/* Amount */}
   578→          <div className="grid gap-2">
   579→            <Label htmlFor="payment-amount">
   580→              Amount (₹) <span className="text-destructive">*</span>
   581→            </Label>
   582→            <div className="relative">
   583→              <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   584→              <Input
   585→                id="payment-amount"
   586→                type="number"
   587→                placeholder="0"
   588→                min="1"
   589→                className="pl-9"
   590→                value={formData.amount}
   591→                onChange={(e) => handleFormChange('amount', e.target.value)}
   592→              />
   593→            </div>
   594→          </div>
   595→
   596→          {/* Date */}
   597→          <div className="grid gap-2">
   598→            <Label htmlFor="payment-date">
   599→              Date <span className="text-destructive">*</span>
   600→            </Label>
   601→            <Input
   602→              id="payment-date"
   603→              type="date"
   604→              value={formData.date}
   605→              onChange={(e) => handleFormChange('date', e.target.value)}
   606→            />
   607→          </div>
   608→
   609→          {/* Remarks */}
   610→          <div className="grid gap-2">
   611→            <Label htmlFor="payment-remarks">Remarks</Label>
   612→            <Textarea
   613→              id="payment-remarks"
   614→              placeholder="Optional remarks about this payment..."
   615→              value={formData.remarks}
   616→              onChange={(e) => handleFormChange('remarks', e.target.value)}
   617→              className="min-h-[80px] resize-none"
   618→            />
   619→          </div>
   620→
   621→          {/* Link to Bill — optional. When set, the Bill's paidAmount will
   622→              auto-update to reflect this payment (reverse sync). */}
   623→          <div className="grid gap-2">
   624→            <Label htmlFor="payment-bill">
   625→              Link to Bill <span className="text-muted-foreground text-xs font-normal">(optional — auto-syncs Bill's paid amount)</span>
   626→            </Label>
   627→            {!formData.customerId ? (
   628→              <p className="text-xs text-muted-foreground italic py-2">
   629→                Select a customer first to see their outstanding bills.
   630→              </p>
   631→            ) : loadingBills ? (
   632→              <p className="text-xs text-muted-foreground py-2">Loading bills...</p>
   633→            ) : customerBills.length === 0 ? (
   634→              <p className="text-xs text-muted-foreground italic py-2">
   635→                No outstanding bills for this customer. You can still record an
   636→                advance payment without linking it.
   637→              </p>
   638→            ) : (
   639→              <Select
   640→                value={formData.billId || '__none__'}
   641→                onValueChange={(val) => handleFormChange('billId', val === '__none__' ? '' : val)}
   642→              >
   643→                <SelectTrigger className="w-full">
   644→                  <SelectValue placeholder="No link (advance payment)" />
   645→                </SelectTrigger>
   646→                <SelectContent>
   647→                  <SelectItem value="__none__">No link (advance payment)</SelectItem>
   648→                  {customerBills.map((b) => {
   649→                    const balance = Number(b.balanceAmount) || 0
   650→                    const date = new Date(b.date).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   651→                    return (
   652→                      <SelectItem key={b.id} value={b.id}>
   653→                        {b.billNumber} — {formatCurrency(balance)} pending ({date})
   654→                      </SelectItem>
   655→                    )
   656→                  })}
   657→                </SelectContent>
   658→              </Select>
   659→            )}
   660→            {formData.billId && (
   661→              <p className="text-xs text-emerald-600 dark:text-emerald-400">
   662→                ✓ This payment will auto-update the linked bill's paid amount.
   663→              </p>
   664→            )}
   665→          </div>
   666→        </div>
   667→
   668→        <DialogFooter>
   669→          <Button
   670→            variant="outline"
   671→            onClick={() => setDialogOpen(false)}
   672→            disabled={formSubmitting}
   673→          >
   674→            Cancel
   675→          </Button>
   676→          <Button
   677→            onClick={handleSubmit}
   678→            disabled={formSubmitting}
   679→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   680→          >
   681→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   682→            {editingPayment ? 'Update Payment' : 'Record Payment'}
   683→          </Button>
   684→        </DialogFooter>
   685→      </DialogContent>
   686→    </Dialog>
   687→  )
   688→
   689→  // ── Render: Delete confirmation ─────────────────────────────────────────
   690→  const renderDeleteDialog = () => (
   691→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   692→      <AlertDialogContent>
   693→        <AlertDialogHeader>
   694→          <AlertDialogTitle>Delete Payment</AlertDialogTitle>
   695→          <AlertDialogDescription>
   696→            Are you sure you want to delete this payment of{' '}
   697→            <strong>{deleteTarget ? formatCurrency(Number(deleteTarget.amount)) : ''}</strong>{' '}
   698→            from <strong>{deleteTarget?.customer?.name || 'this customer'}</strong>? This action
   699→            cannot be undone.
   700→          </AlertDialogDescription>
   701→        </AlertDialogHeader>
   702→        <AlertDialogFooter>
   703→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   704→          <AlertDialogAction
   705→            onClick={handleDelete}
   706→            disabled={deleting}
   707→            className="bg-destructive text-white hover:bg-destructive/90"
   708→          >
   709→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   710→            Delete
   711→          </AlertDialogAction>
   712→        </AlertDialogFooter>
   713→      </AlertDialogContent>
   714→    </AlertDialog>
   715→  )
   716→
   717→  // ── Render: Main ────────────────────────────────────────────────────────
   718→  return (
   719→    <div className="space-y-6">
   720→      {/* Header */}
   721→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   722→        <div className="flex items-center gap-3">
   723→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   724→            <Wallet className="size-5" />
   725→          </div>
   726→          <div>
   727→            <h2 className="text-2xl font-bold tracking-tight">Payment Management</h2>
   728→            <p className="text-sm text-muted-foreground">
   729→              Track and manage customer payments
   730→            </p>
   731→          </div>
   732→        </div>
   733→        <div className="flex flex-col gap-2 sm:flex-row sm:w-auto">
   734→          <Button
   735→            variant="outline"
   736→            onClick={() => setImportOpen(true)}
   737→            className="w-full sm:w-auto"
   738→          >
   739→            <Upload className="size-4 mr-2" />
   740→            Import Excel
   741→          </Button>
   742→          <Button
   743→            onClick={openCreateDialog}
   744→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   745→          >
   746→            <Plus className="size-4" />
   747→            Receive Payment
   748→          </Button>
   749→        </div>
   750→      </div>
   751→
   752→      {/* Summary Cards */}
   753→      <div className="grid gap-4 sm:grid-cols-3">
   754→        <Card>
   755→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   756→            <CardTitle className="text-sm font-medium text-muted-foreground">
   757→              Total Received
   758→            </CardTitle>
   759→            <IndianRupee className="size-4 text-emerald-600" />
   760→          </CardHeader>
   761→          <CardContent>
   762→            {loading ? (
   763→              <Skeleton className="h-8 w-32" />
   764→            ) : (
   765→              <p className="text-2xl font-bold">{formatCurrency(totalReceived)}</p>
   766→            )}
   767→          </CardContent>
   768→        </Card>
   769→
   770→        <Card>
   771→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   772→            <CardTitle className="text-sm font-medium text-muted-foreground">
   773→              Cash Payments
   774→            </CardTitle>
   775→            <Banknote className="size-4 text-emerald-600" />
   776→          </CardHeader>
   777→          <CardContent>
   778→            {loading ? (
   779→              <Skeleton className="h-8 w-32" />
   780→            ) : (
   781→              <p className="text-2xl font-bold">{formatCurrency(cashPayments)}</p>
   782→            )}
   783→          </CardContent>
   784→        </Card>
   785→
   786→        <Card>
   787→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   788→            <CardTitle className="text-sm font-medium text-muted-foreground">
   789→              UPI + Bank
   790→            </CardTitle>
   791→            <Landmark className="size-4 text-sky-600" />
   792→          </CardHeader>
   793→          <CardContent>
   794→            {loading ? (
   795→              <Skeleton className="h-8 w-32" />
   796→            ) : (
   797→              <p className="text-2xl font-bold">{formatCurrency(upiBankPayments)}</p>
   798→            )}
   799→          </CardContent>
   800→        </Card>
   801→      </div>
   802→
   803→      {/* Payment Table */}
   804→      {/* Search */}
   805→      <Card>
   806→        <CardContent className="pt-6">
   807→          <div className="relative">
   808→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   809→            <Input
   810→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   811→              value={search}
   812→              onChange={(e) => setSearch(e.target.value)}
   813→              className="pl-9"
   814→            />
   815→          </div>
   816→        </CardContent>
   817→      </Card>
   818→
   819→      <Card>
   820→        <CardHeader>
   821→          <CardTitle className="flex items-center justify-between">
   822→            <span>Payments</span>
   823→            <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   824→              {filteredPayments.length} of {payments.length} record{payments.length !== 1 ? 's' : ''}
   825→            </Badge>
   826→          </CardTitle>
   827→        </CardHeader>
   828→        <CardContent>
   829→          <div className="max-h-[60vh] overflow-auto rounded-md border">
   830→            <Table>
   831→              <TableHeader className="sticky top-0 bg-background z-10">
   832→                <TableRow>
   833→                  <TableHead>Customer Name</TableHead>
   834→                  <TableHead>Payment Type</TableHead>
   835→                  <TableHead className="text-right">Amount (₹)</TableHead>
   836→                  <TableHead>Date</TableHead>
   837→                  <TableHead className="hidden sm:table-cell">Source</TableHead>
   838→                  <TableHead className="hidden md:table-cell">Remarks</TableHead>
   839→                  <TableHead className="text-right">Actions</TableHead>
   840→                </TableRow>
   841→              </TableHeader>
   842→              <TableBody>
   843→                {loading ? (
   844→                  renderSkeletons()
   845→                ) : filteredPayments.length === 0 ? (
   846→                  <TableRow>
   847→                    <TableCell colSpan={7} className="h-32 text-center text-muted-foreground">
   848→                      No payments yet. Click &quot;Receive Payment&quot; to get started.
   849→                    </TableCell>
   850→                  </TableRow>
   851→                ) : (
   852→                  filteredPayments.map((payment) => (
   853→                    <TableRow key={payment.id}>
   854→                      <TableCell className="font-medium whitespace-nowrap">
   855→                        {payment.customer?.name || '—'}
   856→                      </TableCell>
   857→                      <TableCell>{renderPaymentTypeBadge(payment.paymentType)}</TableCell>
   858→                      <TableCell className="text-right font-medium whitespace-nowrap">
   859→                        {formatCurrency(Number(payment.amount))}
   860→                      </TableCell>
   861→                      <TableCell className="whitespace-nowrap">
   862→                        {formatDate(payment.date)}
   863→                      </TableCell>
   864→                      <TableCell className="hidden sm:table-cell whitespace-nowrap">
   865→                        {payment.billId ? (
   866→                          <Badge variant="outline" className="bg-violet-50 text-violet-700 border-violet-200 dark:bg-violet-950/30 dark:text-violet-300 dark:border-violet-800">
   867→                            Bill {payment.billNumber || ''}
   868→                          </Badge>
   869→                        ) : (
   870→                          <Badge variant="outline" className="bg-slate-50 text-slate-600 border-slate-200 dark:bg-slate-900/30 dark:text-slate-400 dark:border-slate-700">
   871→                            Manual
   872→                          </Badge>
   873→                        )}
   874→                      </TableCell>
   875→                      <TableCell className="hidden md:table-cell max-w-[200px] truncate text-muted-foreground">
   876→                        {payment.remarks || '—'}
   877→                      </TableCell>
   878→                      <TableCell className="text-right">
   879→                        <div className="flex items-center justify-end gap-1">
   880→                          <Button
   881→                            variant="ghost"
   882→                            size="icon"
   883→                            onClick={() => openEditDialog(payment)}
   884→                            title="Edit Payment"
   885→                            className="text-sky-600 hover:text-sky-700 hover:bg-sky-50"
   886→                          >
   887→                            <Pencil className="size-4" />
   888→                          </Button>
   889→                          <Button
   890→                            variant="ghost"
   891→                            size="icon"
   892→                            onClick={() => setDeleteTarget(payment)}
   893→                            title="Delete Payment"
   894→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   895→                          >
   896→                            <Trash2 className="size-4" />
   897→                          </Button>
   898→                        </div>
   899→                      </TableCell>
   900→                    </TableRow>
   901→                  ))
   902→                )}
   903→              </TableBody>
   904→            </Table>
   905→          </div>
   906→        </CardContent>
   907→      </Card>
   908→
   909→      {/* Customer Outstanding Section */}
   910→      <Card>
   911→        <CardHeader>
   912→          <CardTitle className="flex items-center gap-2">
   913→            <Users className="size-5" />
   914→            <span>Customer Outstanding Balances</span>
   915→          </CardTitle>
   916→        </CardHeader>
   917→        <CardContent>
   918→          {loading ? (
   919→            <div className="space-y-3">
   920→              {Array.from({ length: 3 }).map((_, i) => (
   921→                <Skeleton key={i} className="h-10 w-full" />
   922→              ))}
   923→            </div>
   924→          ) : outstandingEntries.length === 0 ? (
   925→            <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   926→              <Users className="size-10 mb-2 opacity-40" />
   927→              <p className="text-sm">No outstanding balances. All customers are paid up!</p>
   928→            </div>
   929→          ) : (
   930→            <div className="max-h-[60vh] overflow-auto rounded-md border">
   931→              <Table>
   932→                <TableHeader className="sticky top-0 bg-background z-10">
   933→                  <TableRow>
   934→                    <TableHead>Customer</TableHead>
   935→                    <TableHead className="text-right">Total Orders (₹)</TableHead>
   936→                    <TableHead className="text-right">Total Paid (₹)</TableHead>
   937→                    <TableHead className="text-right">Outstanding (₹)</TableHead>
   938→                  </TableRow>
   939→                </TableHeader>
   940→                <TableBody>
   941→                  {outstandingEntries.map((entry) => (
   942→                    <TableRow key={entry.customerId}>
   943→                      <TableCell className="font-medium whitespace-nowrap">
   944→                        {entry.customerName}
   945→                      </TableCell>
   946→                      <TableCell className="text-right whitespace-nowrap">
   947→                        {formatCurrency(entry.totalOrderAmount)}
   948→                      </TableCell>
   949→                      <TableCell className="text-right whitespace-nowrap">
   950→                        {formatCurrency(entry.totalPayments)}
   951→                      </TableCell>
   952→                      <TableCell className="text-right font-medium whitespace-nowrap">
   953→                        <span className="text-rose-600">{formatCurrency(entry.outstanding)}</span>
   954→                      </TableCell>
   955→                    </TableRow>
   956→                  ))}
   957→                </TableBody>
   958→              </Table>
   959→            </div>
   960→          )}
   961→        </CardContent>
   962→      </Card>
   963→
   964→      {/* Dialogs */}
   965→      {renderPaymentDialog()}
   966→      {renderDeleteDialog()}
   967→
   968→      <ExcelImport module="payments" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchPayments} />
   969→    </div>
   970→  )
   971→}
   972→
   973→export default PaymentModule
