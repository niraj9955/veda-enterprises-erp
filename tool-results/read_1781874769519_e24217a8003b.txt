   452→  const handleRestoreBackup = async () => {
   453→    const input = document.createElement('input')
   454→    input.type = 'file'
   455→    input.accept = '.json'

   459→      setDbLoading(true)
   460→      try {
   461→        const text = await file.text()
   462→        const parsed = JSON.parse(text)
   463→        // Backup files exported by the v2 export route are wrapped as
   464→        // { version, exportedAt, data: { ...collections... }, counts }.
   465→        // Older / hand-crafted backups may pass the collections map directly.
   466→        // Normalise both shapes here so the backend always sees the inner
   467→        // collections object regardless of file format.
   468→        const payload = parsed?.data && typeof parsed.data === 'object' && !Array.isArray(parsed.data)
   469→          ? parsed.data
   470→          : parsed
   471→
   472→        const result = await api.restoreBackup(payload)
   473→        // Result shape (MERGE mode):
   474→        //   { mode: 'merge', counts: { inserted, replaced }, perCollection: { customers: {inserted, replaced, skipped}, ... } }
   475→        // Show a clear breakdown so the user can verify their data was restored.
   476→        const inserted = (result as any)?.counts?.inserted ?? 0
   477→        const replaced = (result as any)?.counts?.replaced ?? 0
   478→        const perCollection = (result as any)?.perCollection || {}
   479→        const summary = Object.entries(perCollection)
   480→          .filter(([, v]: any) => (v?.inserted ?? 0) + (v?.replaced ?? 0) > 0)
   481→          .map(([k, v]: any) => `${k}: +${v.inserted ?? 0} new / ~${v.replaced ?? 0} updated`)

   484→          title: 'Backup restored (merge mode)',
   485→          description: summary
   486→            ? `${inserted + replaced} docs affected (${inserted} new + ${replaced} updated). Current data NOT in backup is preserved. ${summary}`
   487→            : `${inserted + replaced} docs affected. Current data NOT in backup is preserved.`,
   488→        })
   489→      } catch (err) {
   490→        console.error('Restore failed:', err)
   491→        toast({
   492→          title: 'Restore failed',
   493→          description: err instanceof Error ? err.message : 'Could not restore backup',
   494→          variant: 'destructive',
   495→        })
   496→      } finally {
   497→        setDbLoading(false)
   498→      }
   499→    }
