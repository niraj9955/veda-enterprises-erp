   259→                filteredBills.map((bill) => (
   260→                  <TableRow key={bill.id}>
   261→                    <TableCell className="font-medium">{bill.billNumber}</TableCell>

   264→                      <Badge variant="outline" className="text-xs">
   265→                        {BILL_TYPES.find((t) => t.value === bill.billType)?.label || bill.billType}

   302→// ─── Bill Form Component ──────────────────────────────────────────────────────
   303→function BillForm({ bill, onSave, onCancel }: { bill: Bill | null; onSave: () => void; onCancel: () => void }) {
   304→  const [billType, setBillType] = useState(bill?.billType || 'sales')
   305→  const [date, setDate] = useState(bill?.date || new Date().toISOString().split('T')[0])
   306→  const [dueDate, setDueDate] = useState(bill?.dueDate || '')
   307→  // Customer link — when set, the bill's paidAmount auto-syncs to a Payment
   308→  // document for this customer. May be null (manual party entry).
   309→  const [customerId, setCustomerId] = useState<string | null>(bill?.customerId || null)
   310→  const [toName, setToName] = useState(bill?.toName || '')
   311→  const [toAddress, setToAddress] = useState(bill?.toAddress || '')
   312→  const [toGst, setToGst] = useState(bill?.toGst || '')
   313→  const [toPhone, setToPhone] = useState(bill?.toPhone || '')
   314→  const [items, setItems] = useState<BillItem[]>(
   315→    bill?.items?.length ? bill.items : [{ description: '', hsn: '', quantity: 1, unit: 'pcs', rate: 0, amount: 0 }]
   316→  )
   317→  const [discountPercent, setDiscountPercent] = useState(bill?.discountPercent || 0)
   318→  const [cgstPercent, setCgstPercent] = useState(bill?.cgstPercent || 0)
   319→  const [sgstPercent, setSgstPercent] = useState(bill?.sgstPercent || 0)
   320→  const [igstPercent, setIgstPercent] = useState(bill?.igstPercent || 0)
   321→  const [paidAmount, setPaidAmount] = useState(bill?.paidAmount || 0)
   322→  const [paymentMode, setPaymentMode] = useState(bill?.paymentMode || 'Cash')
   323→  const [notes, setNotes] = useState(bill?.notes || '')
   324→  const [terms, setTerms] = useState(bill?.terms || '')
   325→  const [saving, setSaving] = useState(false)
   326→
   327→  // Recalculate item amount when qty/rate changes
   328→  const updateItem = (index: number, field: keyof BillItem, value: string | number) => {
   329→    const newItems = [...items]
   330→    if (field === 'description' || field === 'hsn' || field === 'unit') {
   331→      newItems[index][field] = value as string
   332→    } else {
   333→      newItems[index][field] = Number(value) || 0
   334→    }
   335→    newItems[index].amount = newItems[index].quantity * newItems[index].rate
   336→    setItems(newItems)
   337→  }
   338→
   339→  const addItem = () => {
   340→    setItems([...items, { description: '', hsn: '', quantity: 1, unit: 'pcs', rate: 0, amount: 0 }])
   341→  }
   342→
   343→  const removeItem = (index: number) => {
   344→    if (items.length > 1) setItems(items.filter((_, i) => i !== index))
   345→  }
   346→
   347→  // Calculations
   348→  const subTotal = items.reduce((sum, item) => sum + (item.amount || 0), 0)
   349→  const discountAmount = (subTotal * discountPercent) / 100
   350→  const taxableAmount = subTotal - discountAmount
   351→  const cgstAmount = (taxableAmount * cgstPercent) / 100
   352→  const sgstAmount = (taxableAmount * sgstPercent) / 100
   353→  const igstAmount = (taxableAmount * igstPercent) / 100
   354→  const totalBeforeRound = taxableAmount + cgstAmount + sgstAmount + igstAmount
   355→  const grandTotal = Math.round(totalBeforeRound)
   356→  const roundOff = grandTotal - totalBeforeRound
   357→  const balanceAmount = grandTotal - paidAmount
   358→
   359→  const handleSubmit = async (e: React.FormEvent) => {
   360→    e.preventDefault()
   361→    if (!toName.trim()) {
   362→      toast({ title: 'Error', description: 'Party name is required', variant: 'destructive' })
   363→      return
   364→    }
   365→    if (items.length === 0 || items.every((i) => !i.description.trim())) {
   366→      toast({ title: 'Error', description: 'Add at least one item', variant: 'destructive' })
   367→      return
   368→    }
   369→
   370→    setSaving(true)
   371→    try {
   372→      const payload = {
   373→        billType,
   374→        date,
   375→        dueDate,
   376→        customerId: customerId || null,
   377→        toName, toAddress, toGst, toPhone,
   378→        items: items.filter((i) => i.description.trim()),
   379→        discountPercent,
   380→        cgstPercent, sgstPercent, igstPercent,
   381→        paidAmount,
   382→        paymentMode,
   383→        notes,
   384→        terms,
   385→      }
   386→      if (bill) {
   387→        await api.updateBill(bill.id, payload)
   388→        toast({ title: 'Updated', description: 'Bill updated successfully' })
   389→      } else {
   390→        await api.createBill(payload)
   391→        toast({ title: 'Created', description: 'Bill created successfully' })
   392→      }
   393→      onSave()
   394→    } catch (err) {
   395→      toast({ title: 'Error', description: err instanceof Error ? err.message : 'Failed to save bill', variant: 'destructive' })
   396→    } finally {
   397→      setSaving(false)
   398→    }
   399→  }
   400→
   401→  return (
   402→    <form onSubmit={handleSubmit} className="space-y-4">
   403→      {/* Bill Type & Date */}
   404→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   405→        <div>
   406→          <Label className="text-xs">Bill Type</Label>
   407→          <Select value={billType} onValueChange={setBillType}>
   408→            <SelectTrigger><SelectValue /></SelectTrigger>
   409→            <SelectContent>
   410→              {BILL_TYPES.map((t) => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}
   411→            </SelectContent>
   412→          </Select>
   413→        </div>
   414→        <div>
   415→          <Label className="text-xs">Date</Label>
   416→          <Input type="date" value={date} onChange={(e) => setDate(e.target.value)} required />
   417→        </div>
   418→        <div>
   419→          <Label className="text-xs">Due Date (optional)</Label>
   420→          <Input type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} />
   421→        </div>
   422→        <div>
   423→          <Label className="text-xs">Payment Mode</Label>
   424→          <Select value={paymentMode} onValueChange={setPaymentMode}>
   425→            <SelectTrigger><SelectValue /></SelectTrigger>
   426→            <SelectContent>
   427→              <SelectItem value="Cash">Cash</SelectItem>
   428→              <SelectItem value="UPI">UPI</SelectItem>
   429→              <SelectItem value="Bank Transfer">Bank Transfer</SelectItem>
   430→              <SelectItem value="Cheque">Cheque</SelectItem>
   431→              <SelectItem value="Credit">Credit</SelectItem>
   432→            </SelectContent>
   433→          </Select>
   434→        </div>
   435→      </div>
   436→
   437→      {/* Customer + Previous-Bill search & auto-fill */}
   438→      <CustomerSearchCard
   439→        selectedCustomerId={customerId}
   440→        onSelectCustomer={(c) => {
   441→          setCustomerId(c.id)
   442→          setToName(c.name)
   443→          setToPhone(c.mobile || '')
   444→          setToAddress(c.address || '')
   445→          setToGst(c.gstNumber || '')
   446→        }}
   447→        onSelectBill={(b) => {
   448→          // Duplicate-from-previous-bill: pre-fill everything except paid
   449→          // amount and notes (those belong to the new transaction). Link
   450→          // the customer too so the new bill's paidAmount auto-syncs.
   451→          setCustomerId(b.customerId || null)
   452→          setToName(b.toName || '')
   453→          setToPhone(b.toPhone || '')
   454→          setToAddress(b.toAddress || '')
   455→          setToGst(b.toGst || '')
   456→          if (b.items?.length) {
   457→            setItems(b.items.map((it) => ({
   458→              description: it.description || '',
   459→              hsn: it.hsn || '',
   460→              quantity: Number(it.quantity) || 0,
   461→              unit: it.unit || 'pcs',
   462→              rate: Number(it.rate) || 0,
   463→              amount: Number(it.amount) || 0,
   464→            })))
   465→          }
   466→          if (typeof b.discountPercent === 'number') setDiscountPercent(b.discountPercent)
   467→          if (typeof b.cgstPercent === 'number') setCgstPercent(b.cgstPercent)
   468→          if (typeof b.sgstPercent === 'number') setSgstPercent(b.sgstPercent)
   469→          if (typeof b.igstPercent === 'number') setIgstPercent(b.igstPercent)
   470→          if (b.paymentMode) setPaymentMode(b.paymentMode)
   471→          if (b.terms) setTerms(b.terms)
   472→          toast({
   473→            title: 'Bill data loaded',
   474→            description: `Pre-filled from ${b.billNumber}. Review and update the paid amount before saving.`,
   475→          })
   476→        }}
   477→        onClear={() => setCustomerId(null)}
   478→      />
   479→
   480→      {/* Bill To */}
   481→      <Card>
   482→        <CardHeader className="pb-3"><CardTitle className="text-sm">Bill To (Party Details)</CardTitle></CardHeader>
   483→        <CardContent className="space-y-3">
   484→          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   485→            <div>
   486→              <Label className="text-xs">Party Name *</Label>
   487→              <Input value={toName} onChange={(e) => {
   488→                setToName(e.target.value)
   489→                // Manual edits break the customer link so we don't accidentally
   490→                // sync payments to the wrong customer.
   491→                if (customerId) setCustomerId(null)
   492→              }} placeholder="Customer / Vendor name" required />
   493→            </div>
   494→            <div>
   495→              <Label className="text-xs">Phone</Label>
   496→              <Input value={toPhone} onChange={(e) => {
   497→                setToPhone(e.target.value)
   498→                if (customerId) setCustomerId(null)
   499→              }} placeholder="Contact number" />
   500→            </div>
   501→          </div>
   502→          <div>
   503→            <Label className="text-xs">Address</Label>
   504→            <Textarea value={toAddress} onChange={(e) => {
   505→              setToAddress(e.target.value)
   506→              if (customerId) setCustomerId(null)
   507→            }} placeholder="Full address" rows={2} />
   508→          </div>

   790→// ─── Customer + Bill Search Card ─────────────────────────────────────────────
   791→// Dual-source live search:
   792→//   • Section 1: customers (click → fills party details, links customer)
   793→//   • Section 2: previous bills (click → fills party + items + tax + payment
   794→//     mode + terms — like a "duplicate bill" shortcut)
   795→//
   796→// Both sections fire in parallel on the same query so the user sees results
   797→// from either source instantly.
   798→interface CustomerSearchResult {
   799→  id: string
   800→  name: string
   801→  mobile?: string
   802→  address?: string
   803→  gstNumber?: string
   804→}

   806→interface BillSearchResult {
   807→  id: string
   808→  billNumber: string
   809→  billType: string
   810→  date: string
   811→  customerId?: string | null
   812→  toName: string
   813→  toPhone?: string
   814→  toAddress?: string
   815→  toGst?: string
   816→  items?: Array<{
   817→    description: string
   818→    hsn?: string
   819→    quantity: number
   820→    unit?: string
   821→    rate: number
   822→    amount: number
   823→  }>
   824→  discountPercent?: number
   825→  cgstPercent?: number

   827→  igstPercent?: number
   828→  grandTotal?: number
   829→  paymentMode?: string
   830→  terms?: string
   831→}

   833→function CustomerSearchCard({
   834→  selectedCustomerId,
   835→  onSelectCustomer,
   836→  onSelectBill,
   837→  onClear,
   838→}: {
   839→  selectedCustomerId: string | null
   840→  onSelectCustomer: (c: CustomerSearchResult) => void
   841→  onSelectBill: (b: BillSearchResult) => void
   842→  onClear: () => void
   843→}) {
   844→  const [query, setQuery] = useState('')
   845→  const [customers, setCustomers] = useState<CustomerSearchResult[]>([])
   846→  const [bills, setBills] = useState<BillSearchResult[]>([])
   847→  const [loading, setLoading] = useState(false)
   848→  const [open, setOpen] = useState(false)
   849→  const boxRef = useRef<HTMLDivElement>(null)
   850→
   851→  // Debounced dual-source search — fires 350ms after the user stops typing.
   852→  // Runs both API calls in parallel via Promise.allSettled so a failure in
   853→  // one source doesn't hide results from the other.
   854→  useEffect(() => {
   855→    if (!query.trim()) {
   856→      setCustomers([])
   857→      setBills([])
   858→      setOpen(false)
   859→      return
   860→    }
   861→    setLoading(true)
   862→    const t = setTimeout(async () => {
   863→      const [custRes, billRes] = await Promise.allSettled([
   864→        api.getCustomers(query.trim()),
   865→        api.getBills({ search: query.trim() }),
   866→      ])
   867→
   868→      const custList: CustomerSearchResult[] =
   869→        custRes.status === 'fulfilled'
   870→          ? (custRes.value.customers as any[]).map((c) => ({
   871→              id: c.id,
   872→              name: c.name,
   873→              mobile: c.mobile || '',
   874→              address: c.address || '',
   875→              gstNumber: c.gstNumber || '',
   876→            }))
   877→          : []
   878→
   879→      const billList: BillSearchResult[] =
   880→        billRes.status === 'fulfilled'
   881→          ? (billRes.value.bills as any[]).map((b) => ({
   882→              id: b.id,
   883→              billNumber: b.billNumber,
   884→              billType: b.billType,
   885→              date: b.date,
   886→              customerId: b.customerId || null,
   887→              toName: b.toName || '',
   888→              toPhone: b.toPhone || '',
   889→              toAddress: b.toAddress || '',
   890→              toGst: b.toGst || '',
   891→              items: b.items || [],
   892→              discountPercent: b.discountPercent,
   893→              cgstPercent: b.cgstPercent,
   894→              sgstPercent: b.sgstPercent,
   895→              igstPercent: b.igstPercent,
   896→              grandTotal: b.grandTotal,
   897→              paymentMode: b.paymentMode,
   898→              terms: b.terms,
   899→            }))
   900→          : []
   901→
   902→      setCustomers(custList)
   903→      setBills(billList)
   904→      setOpen(custList.length > 0 || billList.length > 0)
   905→      setLoading(false)
   906→    }, 350)
   907→    return () => clearTimeout(t)
   908→  }, [query])
   909→
   910→  // Close the dropdown on outside click
   911→  useEffect(() => {
   912→    const handler = (e: MouseEvent) => {
   913→      if (boxRef.current && !boxRef.current.contains(e.target as Node)) {
   914→        setOpen(false)
   915→      }
   916→    }
   917→    document.addEventListener('mousedown', handler)
   918→    return () => document.removeEventListener('mousedown', handler)
   919→  }, [])
   920→
   921→  const handlePickCustomer = (c: CustomerSearchResult) => {
   922→    onSelectCustomer(c)
   923→    setQuery('')
   924→    setCustomers([])
   925→    setBills([])
   926→    setOpen(false)
   927→  }
   928→
   929→  const handlePickBill = (b: BillSearchResult) => {
   930→    onSelectBill(b)
   931→    setQuery('')
   932→    setCustomers([])
   933→    setBills([])
   934→    setOpen(false)
   935→  }
   936→
   937→  const handleClear = () => {
   938→    onClear()
   939→    setQuery('')
   940→    setCustomers([])
   941→    setBills([])
   942→  }
   943→
   944→  const hasAny = customers.length > 0 || bills.length > 0
   945→
   946→  return (
   947→    <Card className="border-emerald-200 dark:border-emerald-900 bg-emerald-50/40 dark:bg-emerald-950/10">
   948→      <CardHeader className="pb-3">
   949→        <CardTitle className="text-sm flex items-center gap-2">
   950→          <Search className="h-4 w-4 text-emerald-600" />
   951→          Search Customer or Previous Bill (auto-fill)
   952→        </CardTitle>
   953→      </CardHeader>
   954→      <CardContent>
   955→        {selectedCustomerId ? (
   956→          <div className="flex items-center justify-between gap-2 bg-emerald-100 dark:bg-emerald-900/30 rounded-md px-3 py-2">
   957→            <div className="flex items-center gap-2 text-sm">
   958→              <UserCheck className="h-4 w-4 text-emerald-600" />
   959→              <span className="font-medium">Customer linked</span>
   960→              <Badge variant="outline" className="text-xs">{selectedCustomerId.slice(-6)}</Badge>
   961→            </div>
   962→            <Button type="button" size="sm" variant="ghost" onClick={handleClear}>
   963→              <X className="h-3 w-3 mr-1" /> Unlink
   964→            </Button>
   965→          </div>
   966→        ) : (
