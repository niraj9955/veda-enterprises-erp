   240→// Normalize a string for comparison: lowercase, remove special chars, spaces
   241→function normalize(s: string): string {
   242→  return s.toLowerCase().replace(/[^a-z0-9\u0900-\u097F]/g, '')
   243→}

   245→// Smart auto-detect mapping between Excel columns and module fields
   246→function autoMapColumns(excelColumns: string[], moduleFields: { key: string; label: string; required: boolean; aliases: string[] }[]): Record<string, string> {
   247→  const mapping: Record<string, string> = {} // excelColumn -> fieldKey
   248→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   249→
   250→  for (const field of moduleFields) {
   251→    let bestMatch = ''
   252→    let bestScore = 0
   253→
   254→    for (const excelCol of normalizedExcel) {
   255→      // Check exact match with key
   256→      if (excelCol.normalized === normalize(field.key)) {
   257→        bestMatch = excelCol.original
   258→        bestScore = 100
   259→        break
   260→      }
   261→
   262→      // Check aliases
   263→      for (const alias of field.aliases) {
   264→        const normAlias = normalize(alias)
   265→        if (excelCol.normalized === normAlias) {
   266→          if (100 > bestScore) {
   267→            bestMatch = excelCol.original
   268→            bestScore = 100
   269→          }
   270→          break
   271→        }
   272→      }
   273→
   274→      if (bestScore >= 100) break
   275→
   276→      // Fuzzy match: check if alias is contained in column name or vice versa
   277→      for (const alias of field.aliases) {
   278→        const normAlias = normalize(alias)
   279→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   280→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   281→          if (score > bestScore) {
   282→            bestMatch = excelCol.original
   283→            bestScore = score
   284→          }
   285→        }
   286→      }
   287→    }
   288→
   289→    if (bestMatch && bestScore >= 50) {
   290→      mapping[bestMatch] = field.key
   291→    }
   292→  }
   293→
   294→  return mapping
   295→}

   297→// Auto-detect the module from Excel column headers
   298→function autoDetectModule(columns: string[]): string | null {
   299→  const normalized = columns.map(normalize)
   300→  const scores: Record<string, number> = {}
   301→
   302→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   303→    scores[moduleKey] = 0
   304→    for (const field of template.fields) {
   305→      for (const alias of field.aliases) {
   306→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   307→          scores[moduleKey] += field.required ? 3 : 1
   308→        }
   309→      }
   310→    }
   311→  }
   312→
   313→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   314→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   315→}

   317→// Transform data using mapping, also handle customer name -> ID resolution
   318→function transformRow(
   319→  row: Record<string, unknown>,
   320→  mapping: Record<string, string>,
   321→  customers?: { id: string; name: string }[]
   322→): Record<string, unknown> {
   323→  const result: Record<string, unknown> = {}
   324→
   325→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   326→    let value = row[excelCol]
   327→
   328→    // Handle customer name -> ID resolution
   329→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   330→      const nameStr = String(value || '').trim().toLowerCase()
   331→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   332→      if (matched) {
   333→        result['customerId'] = matched.id
   334→      } else {
   335→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   336→      }
   337→      continue
   338→    }
   339→
   340→    // Handle numeric fields
   341→    if ([
   342→      'quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate',
   343→      'zigZagWhite80', 'zigZagRed80', 'zigZagYellow80',
   344→      'zigZagWhite60', 'zigZagRed60', 'zigZagYellow60',
   345→      'curveStone', 'chequreTile', 'transportationCharge',
   346→      'cement', 'zigZagGrey80', 'zigZagGrey60',
   347→      'dumbleGrey80', 'dumbleRed80', 'dumbleYellow80',
   348→      'quantityTon', 'totalAmount', 'paidAmount', 'remainingAmount',
   349→      'gst',
   350→    ].includes(fieldKey)) {
   351→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   352→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   353→      continue
   354→    }
   355→
   356→    // Handle date fields - try to parse various formats
   357→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   358→      result[fieldKey] = parseDate(String(value || ''))
   359→      continue
   360→    }
   361→
   362→    // Handle shift field - normalize
   363→    if (fieldKey === 'shift') {
   364→      const shiftStr = String(value || 'Morning').trim()
   365→      const lower = shiftStr.toLowerCase()
   366→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   367→        result[fieldKey] = 'Morning'
   368→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   369→        result[fieldKey] = 'Evening'
   370→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   371→        result[fieldKey] = 'Night'
   372→      } else {
   373→        result[fieldKey] = shiftStr
   374→      }
   375→      continue
   376→    }
   377→
   378→    // Handle payment type
   379→    if (fieldKey === 'paymentType') {
   380→      const ptStr = String(value || 'Cash').trim()
   381→      const lower = ptStr.toLowerCase()
   382→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   383→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   384→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   385→      else result[fieldKey] = ptStr
   386→      continue
   387→    }
   388→
   389→    // Handle expense category
   390→    if (fieldKey === 'category') {
   391→      const catStr = String(value || '').trim()
   392→      const lower = catStr.toLowerCase()
   393→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   394→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   395→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   396→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   397→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   398→      else result[fieldKey] = catStr
   399→      continue
   400→    }
   401→
   402→    // Handle status
   403→    if (fieldKey === 'status') {
   404→      const stStr = String(value || 'Pending').trim()
   405→      const lower = stStr.toLowerCase()
   406→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   407→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   408→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   409→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   410→      else result[fieldKey] = stStr
   411→      continue
   412→    }
   413→
   414→    result[fieldKey] = String(value || '')
   415→  }
   416→
   417→  // For orders/dispatch: calculate amount if not mapped
   418→  if (mapping) {
   419→    const targetModule = Object.values(mapping)
   420→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   421→      const qty = Number(result.quantity || 0)
   422→      const rate = Number(result.rate || 0)
   423→      result.amount = qty * rate
   424→    }
   425→  }
   426→
   427→  return result
   428→}

   430→// Parse various date formats to YYYY-MM-DD
   431→function parseDate(value: string): string {
   432→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   433→
   434→  const trimmed = value.trim()
   435→
   436→  // Already in YYYY-MM-DD format
   437→  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed
   438→
   439→  // DD-MM-YYYY or DD/MM/YYYY
   440→  const dmyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   441→  if (dmyMatch) {
   442→    const [, d, m, y] = dmyMatch
   443→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   444→  }
   445→
   446→  // DD-MM-YY or DD/MM/YY
   447→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2})$/)
   448→  if (dmyShortMatch) {
   449→    const [, d, m, y] = dmyShortMatch
   450→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   451→  }
   452→
   453→  // MM/DD/YYYY (US format) - only if first number > 12
   454→  const mdyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   455→  if (mdyMatch) {
   456→    const [, m, d, y] = mdyMatch
   457→    if (Number(m) > 12) {
   458→      return `${y}-${d.padStart(2, '0')}-${m.padStart(2, '0')}`
   459→    }
   460→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   461→  }
   462→
   463→  // Try native Date parsing as fallback
   464→  try {
   465→    const date = new Date(trimmed)
   466→    if (!isNaN(date.getTime())) {
   467→      return date.toISOString().split('T')[0]
   468→    }
   469→  } catch {}
   470→
   471→  return trimmed
   472→}
