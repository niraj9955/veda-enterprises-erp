   101→   101→
   102→   102→export function DailySellModule() {
   103→   103→  const [dailySells, setDailySells] = React.useState<DailySell[]>([])
   104→   104→  const [search, setSearch] = React.useState('')
   105→   105→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   106→   106→  const [loading, setLoading] = React.useState(true)
   107→   107→  const [formOpen, setFormOpen] = React.useState(false)
   108→   108→  const [editingItem, setEditingItem] = React.useState<DailySell | null>(null)
   109→   109→  const [formData, setFormData] = React.useState<DailySellFormData>(emptyForm)
   110→   110→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   111→   111→  const [deleteTarget, setDeleteTarget] = React.useState<DailySell | null>(null)
   112→   112→  const [deleting, setDeleting] = React.useState(false)
   113→   113→
   114→   114→  // Multi-select state — mirrors the Production module pattern so the user
   115→   115→  // gets identical UX: tick individual rows or use the header checkbox to
   116→   116→  // select all currently-filtered rows, then click "Delete Selected".
   117→   117→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   118→   118→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   119→   119→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   120→   120→
   121→   121→  // Delete All state — simple Yes/No confirmation dialog.
   122→   122→  const [deleteAllOpen, setDeleteAllOpen] = React.useState(false)
   123→   123→  const [deletingAll, setDeletingAll] = React.useState(false)
   124→   124→
   125→   125→  // Excel import
   126→   126→  const [importOpen, setImportOpen] = React.useState(false)
   127→   127→
   128→   128→  const fetchData = React.useCallback(async () => {
   129→   129→    setLoading(true)
   130→   130→    try {
   131→   131→      const res = await api.getDailySells()
   132→   132→      const data = (res.dailySells as DailySell[]).sort(
   133→   133→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   134→   134→      )
   135→   135→      setDailySells(data)
   136→   136→    } catch (err) {
   137→   137→      toast({
   138→   138→        title: 'Error',
   139→   139→        description: err instanceof Error ? err.message : 'Failed to fetch daily sell data',
   140→   140→        variant: 'destructive',
   141→   141→      })
   142→   142→    } finally {
   143→   143→      setLoading(false)
   144→   144→    }
   145→   145→  }, [])
   146→   146→
   147→   147→  // Debounced search
   148→   148→  React.useEffect(() => {
   149→   149→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   150→   150→    return () => clearTimeout(t)
   151→   151→  }, [search])
   152→   152→
   153→   153→  // Client-side filter
   154→   154→  const filteredDailySells = React.useMemo(() => {
   155→   155→    if (!debouncedSearch.trim()) return dailySells
   156→   156→    const q = debouncedSearch.toLowerCase()
   157→   157→    return dailySells.filter((item: any) =>
   158→   158→      ['date', 'customerName', 'address', 'contactNumber', 'product', 'remarks'].some((f) =>
   159→   159→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   160→   160→      )
   161→   161→    )
   162→   162→  }, [dailySells, debouncedSearch])
   163→   163→
   164→   164→  React.useEffect(() => {
   165→   165→    fetchData()
   166→   166→  }, [fetchData])
   167→   167→
   168→   168→  // ── Selection handlers ─────────────────────────────────────────────
   169→   169→  const toggleSelect = (id: string) => {
   170→   170→    setSelectedIds((prev) => {
   171→   171→      const next = new Set(prev)
   172→   172→      if (next.has(id)) next.delete(id)
   173→   173→      else next.add(id)
   174→   174→      return next
   175→   175→    })
   176→   176→  }
   177→   177→
   178→   178→  const toggleSelectAll = () => {
   179→   179→    setSelectedIds((prev) => {
   180→   180→      if (prev.size === filteredDailySells.length && filteredDailySells.length > 0) {
   181→   181→        return new Set()
   182→   182→      }
   183→   183→      return new Set(filteredDailySells.map((s) => s.id))
   184→   184→    })
   185→   185→  }
   186→   186→
   187→   187→  const clearSelection = () => setSelectedIds(new Set())
   188→   188→
   189→   189→  // ── Bulk delete selected ───────────────────────────────────────────
   190→   190→  const handleBulkDelete = async () => {
   191→   191→    const ids = Array.from(selectedIds)
   192→   192→    if (ids.length === 0) return
   193→   193→    setBulkDeleting(true)
   194→   194→    try {
   195→   195→      const res = await api.bulkDeleteDailySells(ids)
   196→   196→      toast({
   197→   197→        title: 'Success',
   198→   198→        description: `${res.deletedCount} of ${ids.length} daily sell entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   199→   199→      })
   200→   200→      setBulkDeleteOpen(false)
   201→   201→      clearSelection()
   202→   202→      fetchData()
   203→   203→    } catch (err) {
   204→   204→      toast({
   205→   205→        title: 'Error',
   206→   206→        description: err instanceof Error ? err.message : 'Failed to delete selected daily sell entries',
   207→   207→        variant: 'destructive',
   208→   208→      })
   209→   209→    } finally {
   210→   210→      setBulkDeleting(false)
   211→   211→    }
   212→   212→  }
   213→   213→
   214→   214→  // ── Delete ALL ─────────────────────────────────────────────────────
   215→   215→  const handleDeleteAll = async () => {
   216→   216→    setDeletingAll(true)
   217→   217→    try {
   218→   218→      const res = await api.deleteAllDailySells()
   219→   219→      toast({
   220→   220→        title: 'Success',
   221→   221→        description: `${res.deletedCount} daily sell entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   222→   222→      })
   223→   223→      setDeleteAllOpen(false)
   224→   224→      clearSelection()
   225→   225→      fetchData()
   226→   226→    } catch (err) {
   227→   227→      toast({
   228→   228→        title: 'Error',
   229→   229→        description: err instanceof Error ? err.message : 'Failed to delete all daily sell entries',
   230→   230→        variant: 'destructive',
   231→   231→      })
   232→   232→    } finally {
   233→   233→      setDeletingAll(false)
   234→   234→    }
   235→   235→  }
   236→   236→
   237→   237→  const openAddDialog = () => {
   238→   238→    setEditingItem(null)
   239→   239→    // Check for pending AI result from chat widget — auto-fill if present
   240→   240→    const pending = consumePendingAiResult('dailySell')
   241→   241→    if (pending) {
   242→   242→      setFormData({
   243→   243→        date: String(pending.date || ''),
   244→   244→        customerName: String(pending.customerName || ''),
   245→   245→        address: String(pending.address || ''),
   246→   246→        contactNumber: String(pending.contactNumber || ''),
   247→   247→        product: String(pending.product || ''),
   248→   248→        amount: pending.amount != null ? String(pending.amount) : '',
   249→   249→        remarks: String(pending.remarks || ''),
   250→   250→      })
   251→   251→      toast({ title: 'AI auto-fill applied', description: 'Edit & verify before saving.' })
   252→   252→    } else {
   253→   253→      setFormData(emptyForm)
   254→   254→    }
   255→   255→    setFormOpen(true)
   256→   256→  }
   257→   257→
   258→   258→  const openEditDialog = (item: DailySell) => {
   259→   259→    setEditingItem(item)
   260→   260→    setFormData({
   261→   261→      date: item.date ? item.date.split('T')[0] : '',
   262→   262→      customerName: item.customerName || '',
   263→   263→      address: item.address || '',
   264→   264→      contactNumber: item.contactNumber || '',
   265→   265→      product: item.product || '',
   266→   266→      amount: String(item.amount || ''),
   267→   267→      remarks: item.remarks || '',
   268→   268→    })
   269→   269→    setFormOpen(true)
   270→   270→  }
   271→   271→
   272→   272→  const handleFormChange = (field: keyof DailySellFormData, value: string) => {
   273→   273→    setFormData((prev) => ({ ...prev, [field]: value }))
   274→   274→  }
   275→   275→
   276→   276→  const handleSubmit = async () => {
   277→   277→    // Unified empty-form check — show ONE popup instead of cascading errors
   278→   278→    if (isFormEmpty([formData.date, formData.customerName, formData.address, formData.contactNumber, formData.product, formData.amount, formData.remarks])) {
   279→   279→      toast(showPleaseFillDataToast())
   280→   280→      return
   281→   281→    }
   282→   282→    if (!formData.date) {
   283→   283→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   284→   284→      return
   285→   285→    }
   286→   286→    if (!formData.customerName.trim()) {
   287→   287→      toast({ title: 'Validation Error', description: 'Customer name is required', variant: 'destructive' })
   288→   288→      return
   289→   289→    }
   290→   290→
   291→   291→    setFormSubmitting(true)
   292→   292→    try {
   293→   293→      const payload = {
   294→   294→        date: formData.date,
   295→   295→        customerName: formData.customerName.trim(),
   296→   296→        address: formData.address.trim(),
   297→   297→        contactNumber: formData.contactNumber.trim(),
   298→   298→        product: formData.product.trim(),
   299→   299→        amount: Number(formData.amount) || 0,
   300→   300→        remarks: formData.remarks.trim(),
   301→   301→      }
   302→   302→
   303→   303→      if (editingItem) {
   304→   304→        await api.updateDailySell(editingItem.id, payload)
   305→   305→        toast({ title: 'Success', description: 'Daily sell entry updated successfully' })
   306→   306→      } else {
   307→   307→        await api.createDailySell(payload)
   308→   308→        toast({ title: 'Success', description: 'Daily sell entry created successfully' })
   309→   309→      }
   310→   310→
   311→   311→      setFormOpen(false)
   312→   312→      setFormData(emptyForm)
   313→   313→      setEditingItem(null)
   314→   314→      fetchData()
   315→   315→    } catch (err) {
   316→   316→      toast({
   317→   317→        title: 'Error',
   318→   318→        description: err instanceof Error ? err.message : 'Failed to save daily sell entry',
   319→   319→        variant: 'destructive',
   320→   320→      })
   321→   321→    } finally {
   322→   322→      setFormSubmitting(false)
   323→   323→    }
   324→   324→  }
   325→   325→
   326→   326→  const handleDelete = async () => {
   327→   327→    if (!deleteTarget) return
   328→   328→    setDeleting(true)
   329→   329→    try {
   330→   330→      await api.deleteDailySell(deleteTarget.id)
   331→   331→      toast({ title: 'Success', description: 'Daily sell entry deleted successfully' })
   332→   332→      setDeleteTarget(null)
   333→   333→      fetchData()
   334→   334→    } catch (err) {
   335→   335→      toast({
   336→   336→        title: 'Error',
   337→   337→        description: err instanceof Error ? err.message : 'Failed to delete daily sell entry',
   338→   338→        variant: 'destructive',
   339→   339→      })
   340→   340→    } finally {
   341→   341→      setDeleting(false)
   342→   342→    }
   343→   343→  }
   344→   344→
   345→   345→  const totalAmount = dailySells.reduce((sum, s) => sum + (s.amount || 0), 0)
   346→   346→
   347→   347→  const renderSkeletons = () =>
   348→   348→    Array.from({ length: 5 }).map((_, i) => (
   349→   349→      <TableRow key={i}>
   350→   350→        <TableCell><Skeleton className="h-4 w-6" /></TableCell>
   351→   351→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   352→   352→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   353→   353→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   354→   354→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   355→   355→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   356→   356→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   357→   357→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   358→   358→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   359→   359→      </TableRow>
   360→   360→    ))
   361→   361→
   362→   362→  return (
   363→   363→    <div className="space-y-6 relative">
   364→   364→      {/* Full-screen loading overlay shown during any delete operation.
   365→   365→          Mirrors the Production module pattern so the user always sees
   366→   366→          that something is happening. */}
   367→   367→      {(deleting || deletingAll || bulkDeleting) && (
   368→   368→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   369→   369→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   370→   370→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   371→   371→            <div className="text-center">
   372→   372→              <p className="text-lg font-semibold">
   373→   373→                {bulkDeleting
   374→   374→                  ? `Deleting ${selectedIds.size} entries...`
   375→   375→                  : deletingAll
   376→   376→                  ? 'Deleting all entries...'
   377→   377→                  : 'Deleting entry...'}
   378→   378→              </p>
   379→   379→              <p className="text-sm text-muted-foreground mt-1">
   380→   380→                Please wait while records are removed.
   381→   381→              </p>
   382→   382→            </div>
   383→   383→          </div>
   384→   384→        </div>
   385→   385→      )}
   386→   386→
   387→   387→      {/* Header */}
   388→   388→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   389→   389→        <div className="flex items-center gap-3">
   390→   390→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   391→   391→            <ShoppingCart className="size-5" />
   392→   392→          </div>
   393→   393→          <div>
   394→   394→            <h2 className="text-2xl font-bold tracking-tight">Daily Sell</h2>
   395→   395→            <p className="text-sm text-muted-foreground">
   396→   396→              Track daily sales transactions
   397→   397→            </p>
   398→   398→          </div>
   399→   399→        </div>
   400→   400→        <div className="flex flex-wrap gap-2 w-full sm:w-auto">
   401→   401→          <Button
   402→   402→            variant="outline"
   403→   403→            onClick={() => setImportOpen(true)}
   404→   404→            className="w-full sm:w-auto"
   405→   405→          >
   406→   406→            <Upload className="size-4 mr-2" />
   407→   407→            Import Excel
   408→   408→          </Button>
   409→   409→          {/* Bulk-delete button — only visible when at least one row is
   410→   410→              selected. Clicking it opens the confirmation dialog below. */}
   411→   411→          {selectedIds.size > 0 && (
   412→   412→            <Button
   413→   413→              variant="outline"
   414→   414→              onClick={() => setBulkDeleteOpen(true)}
   415→   415→              disabled={bulkDeleting || loading}
   416→   416→              className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   417→   417→            >
   418→   418→              <Trash2 className="size-4 mr-2" />
   419→   419→              Delete Selected
   420→   420→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">
   421→   421→                {selectedIds.size}
   422→   422→              </Badge>
   423→   423→            </Button>
   424→   424→          )}
   425→   425→          {/* Clear-selection button — small, ghost-styled, only visible when rows are selected */}
   426→   426→          {selectedIds.size > 0 && (
   427→   427→            <Button
   428→   428→              variant="ghost"
   429→   429→              onClick={clearSelection}
   430→   430→              disabled={bulkDeleting}
   431→   431→              className="w-full sm:w-auto"
   432→   432→            >
   433→   433→              Clear Selection
   434→   434→            </Button>
   435→   435→          )}
   436→   436→          <Button
   437→   437→            variant="outline"
   438→   438→            onClick={() => setDeleteAllOpen(true)}
   439→   439→            disabled={dailySells.length === 0 || loading}
   440→   440→            className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   441→   441→          >
   442→   442→            <Trash className="size-4 mr-2" />
   443→   443→            Delete All
   444→   444→          </Button>
   445→   445→          <Button
   446→   446→            onClick={openAddDialog}
   447→   447→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   448→   448→          >
   449→   449→            <Plus className="size-4" />
   450→   450→            Add Daily Sell
   451→   451→          </Button>
   452→   452→        </div>
   453→   453→      </div>
   454→   454→
   455→   455→      {/* Summary */}
   456→   456→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   457→   457→        <Card className="border-l-4 border-l-emerald-500">
   458→   458→          <CardContent className="p-4">
   459→   459→            <p className="text-xs text-muted-foreground">Total Sales</p>
   460→   460→            {loading ? (
   461→   461→              <Skeleton className="h-6 w-32 mt-1" />
   462→   462→            ) : (
   463→   463→              <p className="text-xl font-bold text-emerald-700">{formatCurrency(totalAmount)}</p>
   464→   464→            )}
   465→   465→          </CardContent>
   466→   466→        </Card>
   467→   467→        <Card className="border-l-4 border-l-amber-500">
   468→   468→          <CardContent className="p-4">
   469→   469→            <p className="text-xs text-muted-foreground">Total Records</p>
   470→   470→            {loading ? (
   471→   471→              <Skeleton className="h-6 w-16 mt-1" />
   472→   472→            ) : (
   473→   473→              <p className="text-xl font-bold text-amber-700">{dailySells.length}</p>
   474→   474→            )}
