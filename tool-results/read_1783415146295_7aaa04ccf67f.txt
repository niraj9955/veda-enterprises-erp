   139→// ── Component ───────────────────────────────────────────────────────────────
   140→
   141→export function ProductionModule() {
   142→  const [productions, setProductions] = React.useState<Production[]>([])
   143→  const [search, setSearch] = React.useState('')
   144→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   145→  const [loading, setLoading] = React.useState(true)
   146→  const [formOpen, setFormOpen] = React.useState(false)
   147→  const [editingProduction, setEditingProduction] = React.useState<Production | null>(null)
   148→  const [formData, setFormData] = React.useState<ProductionFormData>(emptyForm)
   149→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   150→  const [deleteTarget, setDeleteTarget] = React.useState<Production | null>(null)
   151→  const [deleting, setDeleting] = React.useState(false)
   152→  const [deleteAllOpen, setDeleteAllOpen] = React.useState(false)
   153→  const [deletingAll, setDeletingAll] = React.useState(false)
   154→  const [deleteAllConfirm, setDeleteAllConfirm] = React.useState('')
   155→
   156→  // ── Multi-select state ──────────────────────────────────────────────────
   157→  // Tracks which production rows the user has ticked in the table. The
   158→  // "Delete Selected" button appears in the header whenever this set is
   159→  // non-empty. We use a Set for O(1) toggle/lookup, then convert to an
   160→  // array when sending to the API.
   161→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   162→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   163→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   164→
   165→  const toggleSelect = (id: string) => {
   166→    setSelectedIds((prev) => {
   167→      const next = new Set(prev)
   168→      if (next.has(id)) next.delete(id)
   169→      else next.add(id)
   170→      return next
   171→    })
   172→  }
   173→
   174→  const toggleSelectAll = () => {
   175→    setSelectedIds((prev) => {
   176→      // If everything is already selected, clear. Otherwise select all
   177→      // currently-filtered rows (so the user can select-all within a search
   178→      // result without picking rows they can't see).
   179→      if (prev.size === filteredProductions.length && filteredProductions.length > 0) {
   180→        return new Set()
   181→      }
   182→      return new Set(filteredProductions.map((p) => p.id))
   183→    })
   184→  }
   185→
   186→  const clearSelection = () => setSelectedIds(new Set())
   187→
   188→  const handleBulkDelete = async () => {
   189→    const ids = Array.from(selectedIds)
   190→    if (ids.length === 0) return
   191→    setBulkDeleting(true)
   192→    try {
   193→      const res = await api.bulkDeleteProductions(ids)
   194→      toast({
   195→        title: 'Success',
   196→        description: `${res.deletedCount} of ${ids.length} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   197→      })
   198→      setBulkDeleteOpen(false)
   199→      clearSelection()
   200→      fetchProductions()
   201→    } catch (err) {
   202→      toast({
   203→        title: 'Error',
   204→        description: err instanceof Error ? err.message : 'Failed to delete selected production entries',
   205→        variant: 'destructive',
   206→      })
   207→    } finally {
   208→      setBulkDeleting(false)
   209→    }
   210→  }
   211→
   212→  const fetchProductions = React.useCallback(async () => {
   213→    setLoading(true)
   214→    try {
   215→      const res = await api.getProduction()
   216→      const data = (res.productions as Production[]).sort(
   217→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   218→      )
   219→      setProductions(data)
   220→    } catch (err) {
   221→      toast({
   222→        title: 'Error',
   223→        description: err instanceof Error ? err.message : 'Failed to fetch production data',
   224→        variant: 'destructive',
   225→      })
   226→    } finally {
   227→      setLoading(false)
   228→    }
   229→  }, [])
   230→
   231→  // Debounced search
   232→  React.useEffect(() => {
   233→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   234→    return () => clearTimeout(t)
   235→  }, [search])
   236→
   237→  // Client-side filter
   238→  const filteredProductions = React.useMemo(() => {
   239→    if (!debouncedSearch.trim()) return productions
   240→    const q = debouncedSearch.toLowerCase()
   241→    return productions.filter((item: any) =>
   242→      ['date', 'remarks'].some((f) =>
   243→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   244→      )
   245→    )
   246→  }, [productions, debouncedSearch])
   247→
   248→  React.useEffect(() => {
   249→    fetchProductions()
   250→  }, [fetchProductions])
   251→
   252→  // Excel import
   253→  const [importOpen, setImportOpen] = React.useState(false)
   254→
   255→  const openAddDialog = () => {
   256→    setEditingProduction(null)
   257→    // Check for pending AI result from chat widget — auto-fill if present
   258→    const pending = consumePendingAiResult('production')
   259→    if (pending) {
   260→      const str = (v: unknown) => (v == null ? '' : String(v))
   261→      setFormData({
   262→        date: str(pending.date),
   263→        cement: str(pending.cement),
   264→        zigZagGrey80: str(pending.zigZagGrey80),
   265→        zigZagRed80: str(pending.zigZagRed80),
   266→        zigZagYellow80: str(pending.zigZagYellow80),
   267→        zigZagGrey60: str(pending.zigZagGrey60),
   268→        zigZagRed60: str(pending.zigZagRed60),
   269→        zigZagYellow60: str(pending.zigZagYellow60),
   270→        curveStone: str(pending.curveStone),
   271→        chequreTile: str(pending.chequreTile),
   272→        dumbleGrey80: str(pending.dumbleGrey80),
   273→        dumbleRed80: str(pending.dumbleRed80),
   274→        dumbleYellow80: str(pending.dumbleYellow80),
   275→        transportationCharge: str(pending.transportationCharge),
   276→        remarks: str(pending.remarks),
   277→      })
   278→      toast({ title: 'AI auto-fill applied', description: 'Edit & verify before saving.' })
   279→    } else {
   280→      setFormData(emptyForm)
   281→    }
   282→    setFormOpen(true)
   283→  }
   284→
   285→  const openEditDialog = (prod: Production) => {
   286→    setEditingProduction(prod)
   287→    setFormData({
   288→      date: prod.date ? prod.date.split('T')[0] : '',
   289→      cement: String(prod.cement || ''),
   290→      zigZagGrey80: String(prod.zigZagGrey80 || ''),
   291→      zigZagRed80: String(prod.zigZagRed80 || ''),
   292→      zigZagYellow80: String(prod.zigZagYellow80 || ''),
   293→      zigZagGrey60: String(prod.zigZagGrey60 || ''),
   294→      zigZagRed60: String(prod.zigZagRed60 || ''),
   295→      zigZagYellow60: String(prod.zigZagYellow60 || ''),
   296→      curveStone: String(prod.curveStone || ''),
   297→      chequreTile: String(prod.chequreTile || ''),
   298→      dumbleGrey80: String(prod.dumbleGrey80 || ''),
   299→      dumbleRed80: String(prod.dumbleRed80 || ''),
   300→      dumbleYellow80: String(prod.dumbleYellow80 || ''),
   301→      transportationCharge: String(prod.transportationCharge || ''),
   302→      remarks: prod.remarks || '',
   303→    })
   304→    setFormOpen(true)
   305→  }
   306→
   307→  const handleFormChange = (field: keyof ProductionFormData, value: string) => {
   308→    setFormData((prev) => ({ ...prev, [field]: value }))
   309→  }
   310→
   311→  const handleSubmit = async () => {
   312→    if (!formData.date) {
   313→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   314→      return
   315→    }
   316→
   317→    setFormSubmitting(true)
   318→    try {
   319→      const payload = {
   320→        date: formData.date,
   321→        cement: Number(formData.cement) || 0,
   322→        zigZagGrey80: Number(formData.zigZagGrey80) || 0,
   323→        zigZagRed80: Number(formData.zigZagRed80) || 0,
   324→        zigZagYellow80: Number(formData.zigZagYellow80) || 0,
   325→        zigZagGrey60: Number(formData.zigZagGrey60) || 0,
   326→        zigZagRed60: Number(formData.zigZagRed60) || 0,
   327→        zigZagYellow60: Number(formData.zigZagYellow60) || 0,
   328→        curveStone: Number(formData.curveStone) || 0,
   329→        chequreTile: Number(formData.chequreTile) || 0,
   330→        dumbleGrey80: Number(formData.dumbleGrey80) || 0,
   331→        dumbleRed80: Number(formData.dumbleRed80) || 0,
   332→        dumbleYellow80: Number(formData.dumbleYellow80) || 0,
   333→        transportationCharge: Number(formData.transportationCharge) || 0,
   334→        remarks: formData.remarks.trim(),
   335→      }
   336→
   337→      if (editingProduction) {
   338→        await api.updateProduction(editingProduction.id, payload)
   339→        toast({ title: 'Success', description: 'Production entry updated successfully' })
   340→      } else {
   341→        await api.createProduction(payload)
   342→        toast({ title: 'Success', description: 'Production entry created successfully' })
   343→      }
   344→
   345→      setFormOpen(false)
   346→      setFormData(emptyForm)
   347→      setEditingProduction(null)
   348→      fetchProductions()
   349→    } catch (err) {
   350→      toast({
   351→        title: 'Error',
   352→        description: err instanceof Error ? err.message : 'Failed to save production entry',
   353→        variant: 'destructive',
   354→      })
   355→    } finally {
   356→      setFormSubmitting(false)
   357→    }
   358→  }
   359→
   360→  const handleDelete = async () => {
   361→    if (!deleteTarget) return
   362→    setDeleting(true)
   363→    try {
   364→      await api.deleteProduction(deleteTarget.id)
   365→      toast({ title: 'Success', description: 'Production entry deleted successfully' })
   366→      setDeleteTarget(null)
   367→      fetchProductions()
   368→    } catch (err) {
   369→      toast({
   370→        title: 'Error',
   371→        description: err instanceof Error ? err.message : 'Failed to delete production entry',
   372→        variant: 'destructive',
   373→      })
   374→    } finally {
   375→      setDeleting(false)
   376→    }
   377→  }
   378→
   379→  const handleDeleteAll = async () => {
   380→    // No more 'DELETE ALL' typing requirement — the dialog now uses a
   381→    // simple Yes / No choice. We keep the deleteAllConfirm state around
   382→    // only so existing code that resets it (e.g. on dialog close) doesn't
   383→    // break, but it's no longer gating the delete.
   384→    setDeletingAll(true)
   385→    try {
   386→      const res = await api.deleteAllProductions()
   387→      toast({
   388→        title: 'Success',
   389→        description: `${res.deletedCount} production entr${res.deletedCount === 1 ? 'y' : 'ies'} deleted`,
   390→      })
   391→      setDeleteAllOpen(false)
   392→      setDeleteAllConfirm('')
   393→      fetchProductions()
   394→    } catch (err) {
   395→      toast({
   396→        title: 'Error',
   397→        description: err instanceof Error ? err.message : 'Failed to delete all production entries',
   398→        variant: 'destructive',
   399→      })
   400→    } finally {
   401→      setDeletingAll(false)
   402→    }
   403→  }
   404→
   405→  const renderSkeletons = () =>
   406→    Array.from({ length: 3 }).map((_, i) => (
   407→      <TableRow key={i}>
   408→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   409→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   410→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   411→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   412→        {PRODUCT_FIELDS.map((_, j) => (
   413→          <TableCell key={j}><Skeleton className="h-4 w-16" /></TableCell>
   414→        ))}
   415→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   416→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   417→      </TableRow>
   418→    ))
   419→
   420→  return (
   421→    <div className="space-y-6 relative">
   422→      {/* Full-screen loading overlay shown during any delete operation.
   423→          The spinner + message stay visible until the API call completes and
   424→          the table refreshes, so the user always knows "something is happening". */}
   425→      {(deleting || deletingAll || bulkDeleting) && (
   426→        <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/50 backdrop-blur-sm">
   427→          <div className="bg-white dark:bg-zinc-900 rounded-xl shadow-2xl p-8 flex flex-col items-center gap-4 min-w-[280px]">
   428→            <Loader2 className="size-12 animate-spin text-emerald-600" />
   429→            <div className="text-center">
   430→              <p className="text-lg font-semibold">
   431→                {bulkDeleting
   432→                  ? `Deleting ${selectedIds.size} entries...`
   433→                  : deletingAll
   434→                  ? 'Deleting all entries...'
   435→                  : 'Deleting entry...'}
   436→              </p>
   437→              <p className="text-sm text-muted-foreground mt-1">
   438→                Please wait while records are removed and stock is re-synced.
   439→              </p>
   440→            </div>
   441→          </div>
   442→        </div>
   443→      )}
   444→      {/* Header */}
   445→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   446→        <div className="flex items-center gap-3">
   447→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   448→            <Factory className="size-5" />
   449→          </div>
   450→          <div>
   451→            <h2 className="text-2xl font-bold tracking-tight">Production Management</h2>
   452→            <p className="text-sm text-muted-foreground">
   453→              Track daily paper block production entries
   454→            </p>
   455→          </div>
   456→        </div>
   457→        <div className="flex flex-wrap gap-2 w-full sm:w-auto">
   458→          <Button
   459→            variant="outline"
   460→            onClick={() => setImportOpen(true)}
   461→            className="w-full sm:w-auto"
   462→          >
   463→            <Upload className="size-4 mr-2" />
   464→            Import Excel
   465→          </Button>
   466→          {/* Bulk-delete button — only visible when at least one row is selected.
   467→              Clicking it opens the confirmation dialog below. */}
   468→          {selectedIds.size > 0 && (
   469→            <Button
   470→              variant="outline"
   471→              onClick={() => setBulkDeleteOpen(true)}
   472→              disabled={bulkDeleting || loading}
   473→              className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   474→            >
   475→              <Trash2 className="size-4 mr-2" />
   476→              Delete Selected
   477→              <Badge variant="secondary" className="ml-2 bg-destructive/10 text-destructive border-destructive/30">
   478→                {selectedIds.size}
   479→              </Badge>
   480→            </Button>
   481→          )}
   482→          {/* Clear-selection button — small, ghost-styled, only visible when rows are selected */}
   483→          {selectedIds.size > 0 && (
   484→            <Button
   485→              variant="ghost"
   486→              onClick={clearSelection}
   487→              disabled={bulkDeleting}
   488→              className="w-full sm:w-auto"
   489→            >
   490→              Clear Selection
   491→            </Button>
   492→          )}
   493→          <Button
   494→            variant="outline"
   495→            onClick={() => setDeleteAllOpen(true)}
   496→            disabled={productions.length === 0 || loading}
   497→            className="w-full sm:w-auto text-destructive border-destructive/50 hover:bg-destructive/10 hover:text-destructive"
   498→          >
   499→            <Trash className="size-4 mr-2" />
   500→            Delete All
   501→          </Button>
   502→          <Button
   503→            onClick={openAddDialog}
   504→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   505→          >
   506→            <Plus className="size-4" />
   507→            Add Production Entry
   508→          </Button>
   509→        </div>
   510→      </div>
   511→
   512→      {/* Search */}
   513→      <Card>
   514→        <CardContent className="pt-6">
   515→          <div className="relative">
   516→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   517→            <Input
   518→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   519→              value={search}
   520→              onChange={(e) => setSearch(e.target.value)}
   521→              className="pl-9"
   522→            />
   523→          </div>
   524→        </CardContent>
   525→      </Card>
   526→
   527→      {/* Table */}
   528→      <Card>
   529→        <CardHeader>
   530→          <CardTitle className="flex items-center justify-between">
   531→            <span>Production Records</span>
   532→            <div className="flex items-center gap-2">
   533→              {selectedIds.size > 0 && (
   534→                <Badge variant="secondary" className="bg-destructive/10 text-destructive border-destructive/30">
   535→                  {selectedIds.size} selected
   536→                </Badge>
   537→              )}
   538→              <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   539→                {filteredProductions.length} of {productions.length} record{productions.length !== 1 ? 's' : ''}
   540→              </Badge>
   541→            </div>
   542→          </CardTitle>
   543→        </CardHeader>
   544→        <CardContent>
   545→          <ScrollableTable maxHeight="max-h-[60vh]">
   546→            <Table>
   547→              <TableHeader>
   548→                <TableRow>
   549→                  <TableHead className="w-10 sticky left-0 bg-background z-20">
   550→                    <Checkbox
   551→                      checked={
   552→                        filteredProductions.length > 0 &&
   553→                        selectedIds.size === filteredProductions.length
   554→                      }
   555→                      onCheckedChange={toggleSelectAll}
   556→                      aria-label="Select all rows"
   557→                    />
   558→                  </TableHead>
   559→                  <TableHead className="sticky left-10 bg-background z-20">Date</TableHead>
   560→                  <TableHead className="text-right whitespace-nowrap">Cement</TableHead>
   561→                  {PRODUCT_FIELDS.map((f) => (
   562→                    <TableHead key={f.key} className="text-right whitespace-nowrap">{f.label}</TableHead>
   563→                  ))}
   564→                  <TableHead className="text-right whitespace-nowrap">Transport ₹</TableHead>
   565→                  <TableHead>Remarks</TableHead>
   566→                  <TableHead className="text-right">Actions</TableHead>
   567→                </TableRow>
   568→              </TableHeader>
   569→              <TableBody>
   570→                {loading ? (
   571→                  renderSkeletons()
   572→                ) : filteredProductions.length === 0 ? (
   573→                  <TableRow>
   574→                    <TableCell colSpan={PRODUCT_FIELDS.length + 6} className="h-32 text-center text-muted-foreground">
   575→                      No production entries yet. Click &quot;Add Production Entry&quot; to get started.
   576→                    </TableCell>
   577→                  </TableRow>
   578→                ) : (
   579→                  filteredProductions.map((prod) => (
   580→                    <TableRow
   581→                      key={prod.id}
   582→                      data-state={selectedIds.has(prod.id) ? 'selected' : undefined}
   583→                      className={selectedIds.has(prod.id) ? 'bg-emerald-50/60 dark:bg-emerald-900/15' : ''}
   584→                    >
   585→                      <TableCell className="w-10 sticky left-0 bg-background z-10 sticky">
   586→                        <Checkbox
   587→                          checked={selectedIds.has(prod.id)}
   588→                          onCheckedChange={() => toggleSelect(prod.id)}
   589→                          aria-label={`Select row for ${prod.date}`}
   590→                        />
   591→                      </TableCell>
   592→                      <TableCell className="font-medium whitespace-nowrap">
   593→                        {formatDate(prod.date)}
   594→                      </TableCell>
   595→                      <TableCell className="text-right font-mono">
   596→                        {enIN.format(prod.cement || 0)}
   597→                      </TableCell>
   598→                      {PRODUCT_FIELDS.map((f) => (
   599→                        <TableCell key={f.key} className="text-right font-mono">
   600→                          {enIN.format((prod as unknown as Record<string, unknown>)[f.key] as number || 0)}
   601→                        </TableCell>
   602→                      ))}
   603→                      <TableCell className="text-right whitespace-nowrap">
   604→                        {formatCurrency(prod.transportationCharge || 0)}
   605→                      </TableCell>
   606→                      <TableCell className="max-w-[150px] truncate text-muted-foreground">
   607→                        {prod.remarks || '—'}
   608→                      </TableCell>
   609→                      <TableCell className="text-right">
   610→                        <div className="flex items-center justify-end gap-1">
   611→                          <Button
   612→                            variant="ghost"
   613→                            size="icon"
   614→                            onClick={() => openEditDialog(prod)}
   615→                            title="Edit"
   616→                          >
   617→                            <Pencil className="size-4" />
   618→                          </Button>
   619→                          <Button
   620→                            variant="ghost"
   621→                            size="icon"
   622→                            onClick={() => setDeleteTarget(prod)}
   623→                            title="Delete"
   624→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   625→                          >
   626→                            <Trash2 className="size-4" />
   627→                          </Button>
   628→                        </div>
   629→                      </TableCell>
   630→                    </TableRow>
   631→                  ))
   632→                )}
   633→              </TableBody>
   634→            </Table>
   635→          </ScrollableTable>
   636→        </CardContent>
   637→      </Card>
   638→
   639→      {/* Add/Edit Dialog */}
   640→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   641→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   642→          <DialogHeader>
   643→            <DialogTitle>
   644→              {editingProduction ? 'Edit Production Entry' : 'Add Production Entry'}
   645→            </DialogTitle>
   646→            <DialogDescription>
   647→              {editingProduction
   648→                ? 'Update the production entry details below.'
   649→                : 'Fill in the details to create a new production entry.'}
   650→            </DialogDescription>
   651→          </DialogHeader>
   652→          {!editingProduction && (
   653→            <div className="flex justify-end">
   654→              <AiFillButton
   655→                module="production"
   656→                onApply={(fields) => setFormData((prev) => {
   657→                  const next = { ...prev }
   658→                  const keys: (keyof ProductionFormData)[] = [
   659→                    'date', 'cement', 'zigZagGrey80', 'zigZagRed80', 'zigZagYellow80',
   660→                    'zigZagGrey60', 'zigZagRed60', 'zigZagYellow60', 'curveStone',
   661→                    'chequreTile', 'dumbleGrey80', 'dumbleRed80', 'dumbleYellow80',
   662→                    'transportationCharge', 'remarks',
   663→                  ]
   664→                  for (const k of keys) {
   665→                    if (fields[k as string] != null) {
   666→                      ;(next as Record<string, string>)[k as string] = String(fields[k as string])
   667→                    }
   668→                  }
   669→                  return next
   670→                })}
   671→              />
   672→            </div>
   673→          )}
   674→          <div className="grid gap-4 py-2">
   675→            <div className="grid gap-2">
   676→              <Label htmlFor="prod-date">
   677→                Date <span className="text-destructive">*</span>
   678→              </Label>
   679→              <Input
   680→                id="prod-date"
   681→                type="date"
   682→                value={formData.date}
   683→                onChange={(e) => handleFormChange('date', e.target.value)}
   684→              />
   685→            </div>
   686→            <div className="grid gap-2">
   687→              <Label htmlFor="prod-cement">Cement (bags)</Label>
   688→              <Input
   689→                id="prod-cement"
   690→                type="number"
   691→                min="0"
   692→                placeholder="0"
   693→                value={formData.cement}
   694→                onChange={(e) => handleFormChange('cement', e.target.value)}
   695→              />
   696→            </div>
   697→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   698→              {PRODUCT_FIELDS.map((f) => (
   699→                <div key={f.key} className="grid gap-2">
   700→                  <Label htmlFor={`prod-${f.key}`}>{f.label}</Label>
   701→                  <Input
   702→                    id={`prod-${f.key}`}
   703→                    type="number"
   704→                    min="0"
   705→                    placeholder="0"
   706→                    value={formData[f.key]}
   707→                    onChange={(e) => handleFormChange(f.key, e.target.value)}
   708→                  />
   709→                </div>
   710→              ))}
   711→            </div>
   712→            <div className="grid gap-2">
   713→              <Label htmlFor="prod-transport">Transportation Charge (₹)</Label>
   714→              <div className="relative">
   715→                <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   716→                <Input
   717→                  id="prod-transport"
   718→                  type="number"
   719→                  min="0"
   720→                  placeholder="0"
   721→                  className="pl-9"
   722→                  value={formData.transportationCharge}
   723→                  onChange={(e) => handleFormChange('transportationCharge', e.target.value)}
   724→                />
   725→              </div>
   726→            </div>
   727→            <div className="grid gap-2">
   728→              <Label htmlFor="prod-remarks">Remarks</Label>
   729→              <Textarea
   730→                id="prod-remarks"
   731→                placeholder="Optional remarks..."
   732→                value={formData.remarks}
   733→                onChange={(e) => handleFormChange('remarks', e.target.value)}
   734→                className="min-h-[80px]"
   735→              />
   736→            </div>
   737→          </div>
   738→          <DialogFooter>
   739→            <Button variant="outline" onClick={() => setFormOpen(false)} disabled={formSubmitting}>
   740→              Cancel
   741→            </Button>
   742→            <Button
   743→              onClick={handleSubmit}
   744→              disabled={formSubmitting}
   745→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   746→            >
   747→              {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   748→              {editingProduction ? 'Update Entry' : 'Create Entry'}
   749→            </Button>
   750→          </DialogFooter>
   751→        </DialogContent>
   752→      </Dialog>
   753→
   754→      {/* Delete confirmation */}
   755→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   756→        <AlertDialogContent>
   757→          <AlertDialogHeader>
   758→            <AlertDialogTitle>Delete Production Entry</AlertDialogTitle>
   759→            <AlertDialogDescription>
   760→              Are you sure you want to delete this production entry? This action cannot be undone.
   761→            </AlertDialogDescription>
   762→          </AlertDialogHeader>
   763→          <AlertDialogFooter>
   764→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   765→            <AlertDialogAction
   766→              onClick={handleDelete}
   767→              disabled={deleting}
   768→              className="bg-destructive text-white hover:bg-destructive/90"
   769→            >
   770→              {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   771→              Delete
   772→            </AlertDialogAction>
   773→          </AlertDialogFooter>
   774→        </AlertDialogContent>
   775→      </AlertDialog>
   776→
   777→      {/* Delete All confirmation — simple Yes / No dialog */}
   778→      <AlertDialog open={deleteAllOpen} onOpenChange={(open) => {
   779→        if (!open && !deletingAll) {
   780→          setDeleteAllOpen(false)
   781→          setDeleteAllConfirm('')
   782→        }
   783→      }}>
   784→        <AlertDialogContent className="max-w-md">
   785→          <AlertDialogHeader>
   786→            <AlertDialogTitle className="text-destructive">Delete ALL Production Entries?</AlertDialogTitle>
   787→            <AlertDialogDescription className="space-y-2 text-sm">
   788→              <span className="block">
   789→                You are about to permanently delete{' '}
   790→                <strong className="text-destructive">{productions.length} production entr{productions.length === 1 ? 'y' : 'ies'}</strong>.
   791→                This action <strong>cannot be undone</strong>.
   792→              </span>
   793→              <span className="block text-muted-foreground">
   794→                Linked customer bill-history aggregations will lose their production totals.
   795→                Customer, Order, Bill, Payment, Stock, and Dispatch records are NOT affected.
   796→              </span>
   797→            </AlertDialogDescription>
   798→          </AlertDialogHeader>
   799→          <AlertDialogFooter>
   800→            <AlertDialogCancel
   801→              disabled={deletingAll}
   802→              onClick={() => { setDeleteAllOpen(false); setDeleteAllConfirm('') }}
   803→              className="border-border"
   804→            >
   805→              No, Cancel
   806→            </AlertDialogCancel>
   807→            <AlertDialogAction
   808→              onClick={handleDeleteAll}
   809→              disabled={deletingAll}
   810→              className="bg-destructive text-white hover:bg-destructive/90"
   811→            >
   812→              {deletingAll && <Loader2 className="mr-2 size-4 animate-spin" />}
   813→              Yes, Delete All
   814→            </AlertDialogAction>
   815→          </AlertDialogFooter>
   816→        </AlertDialogContent>
   817→      </AlertDialog>
   818→
   819→      {/* Bulk Delete Selected confirmation */}
   820→      <AlertDialog open={bulkDeleteOpen} onOpenChange={(open) => {
   821→        if (!open && !bulkDeleting) setBulkDeleteOpen(false)
   822→      }}>
   823→        <AlertDialogContent>
   824→          <AlertDialogHeader>
   825→            <AlertDialogTitle className="text-destructive">
   826→              Delete {selectedIds.size} Selected Production {selectedIds.size === 1 ? 'Entry' : 'Entries'}?
   827→            </AlertDialogTitle>
   828→            <AlertDialogDescription className="space-y-2 text-sm">
   829→              <span className="block">
   830→                You are about to permanently delete{' '}
   831→                <strong className="text-destructive">{selectedIds.size} production {selectedIds.size === 1 ? 'entry' : 'entries'}</strong>.
   832→                This action <strong>cannot be undone</strong>.
   833→              </span>
   834→              <span className="block text-muted-foreground">
   835→                Stock snapshots for the affected dates will be re-aggregated automatically.
   836→                Other records (customers, orders, payments, etc.) are NOT affected.
   837→              </span>
   838→            </AlertDialogDescription>
   839→          </AlertDialogHeader>
   840→          <AlertDialogFooter>
   841→            <AlertDialogCancel disabled={bulkDeleting}>Cancel</AlertDialogCancel>
   842→            <AlertDialogAction
   843→              onClick={handleBulkDelete}
   844→              disabled={bulkDeleting || selectedIds.size === 0}
   845→              className="bg-destructive text-white hover:bg-destructive/90"
   846→            >
   847→              {bulkDeleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   848→              Delete {selectedIds.size} {selectedIds.size === 1 ? 'Entry' : 'Entries'}
   849→            </AlertDialogAction>
   850→          </AlertDialogFooter>
   851→        </AlertDialogContent>
   852→      </AlertDialog>
   853→
   854→      <ExcelImport module="production" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchProductions} />
   855→    </div>
   856→  )
   857→}

   859→export default ProductionModule
