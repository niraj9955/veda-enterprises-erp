    97→// Normalize a string for comparison: lowercase, remove special chars, spaces
    98→function normalize(s: string): string {
    99→  return s.toLowerCase().replace(/[^a-z0-9\u0900-\u097F]/g, '')
   100→}

   102→// Smart auto-detect mapping between Excel columns and module fields
   103→function autoMapColumns(excelColumns: string[], moduleFields: { key: string; label: string; required: boolean; aliases: string[] }[]): Record<string, string> {
   104→  const mapping: Record<string, string> = {} // excelColumn -> fieldKey
   105→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   106→
   107→  for (const field of moduleFields) {
   108→    let bestMatch = ''
   109→    let bestScore = 0
   110→
   111→    for (const excelCol of normalizedExcel) {
   112→      // Check exact match with key
   113→      if (excelCol.normalized === normalize(field.key)) {
   114→        bestMatch = excelCol.original
   115→        bestScore = 100
   116→        break
   117→      }
   118→
   119→      // Check aliases
   120→      for (const alias of field.aliases) {
   121→        const normAlias = normalize(alias)
   122→        if (excelCol.normalized === normAlias) {
   123→          if (100 > bestScore) {
   124→            bestMatch = excelCol.original
   125→            bestScore = 100
   126→          }
   127→          break
   128→        }
   129→      }
   130→
   131→      if (bestScore >= 100) break
   132→
   133→      // Fuzzy match: check if alias is contained in column name or vice versa
   134→      for (const alias of field.aliases) {
   135→        const normAlias = normalize(alias)
   136→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   137→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   138→          if (score > bestScore) {
   139→            bestMatch = excelCol.original
   140→            bestScore = score
   141→          }
   142→        }
   143→      }
   144→    }
   145→
   146→    if (bestMatch && bestScore >= 50) {
   147→      mapping[bestMatch] = field.key
   148→    }
   149→  }
   150→
   151→  return mapping
   152→}

   154→// Auto-detect the module from Excel column headers
   155→function autoDetectModule(columns: string[]): string | null {
   156→  const normalized = columns.map(normalize)
   157→  const scores: Record<string, number> = {}
   158→
   159→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   160→    scores[moduleKey] = 0
   161→    for (const field of template.fields) {
   162→      for (const alias of field.aliases) {
   163→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   164→          scores[moduleKey] += field.required ? 3 : 1
   165→        }
   166→      }
   167→    }
   168→  }
   169→
   170→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   171→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   172→}

   174→// Transform data using mapping, also handle customer name -> ID resolution
   175→function transformRow(
   176→  row: Record<string, unknown>,
   177→  mapping: Record<string, string>,
   178→  customers?: { id: string; name: string }[]
   179→): Record<string, unknown> {
   180→  const result: Record<string, unknown> = {}
   181→
   182→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   183→    let value = row[excelCol]
   184→
   185→    // Handle customer name -> ID resolution
   186→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   187→      const nameStr = String(value || '').trim().toLowerCase()
   188→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   189→      if (matched) {
   190→        result['customerId'] = matched.id
   191→      } else {
   192→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   193→      }
   194→      continue
   195→    }
   196→
   197→    // Handle numeric fields
   198→    if (['quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate'].includes(fieldKey)) {
   199→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   200→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   201→      continue
   202→    }
   203→
   204→    // Handle date fields - try to parse various formats
   205→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   206→      result[fieldKey] = parseDate(String(value || ''))
   207→      continue
   208→    }
   209→
   210→    // Handle shift field - normalize
   211→    if (fieldKey === 'shift') {
   212→      const shiftStr = String(value || 'Morning').trim()
   213→      const lower = shiftStr.toLowerCase()
   214→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   215→        result[fieldKey] = 'Morning'
   216→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   217→        result[fieldKey] = 'Evening'
   218→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   219→        result[fieldKey] = 'Night'
   220→      } else {
   221→        result[fieldKey] = shiftStr
   222→      }
   223→      continue
   224→    }
   225→
   226→    // Handle payment type
   227→    if (fieldKey === 'paymentType') {
   228→      const ptStr = String(value || 'Cash').trim()
   229→      const lower = ptStr.toLowerCase()
   230→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   231→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   232→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   233→      else result[fieldKey] = ptStr
   234→      continue
   235→    }
   236→
   237→    // Handle expense category
   238→    if (fieldKey === 'category') {
   239→      const catStr = String(value || '').trim()
   240→      const lower = catStr.toLowerCase()
   241→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   242→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   243→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   244→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   245→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   246→      else result[fieldKey] = catStr
   247→      continue
   248→    }
   249→
   250→    // Handle status
   251→    if (fieldKey === 'status') {
   252→      const stStr = String(value || 'Pending').trim()
   253→      const lower = stStr.toLowerCase()
   254→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   255→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   256→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   257→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   258→      else result[fieldKey] = stStr
   259→      continue
   260→    }
   261→
   262→    result[fieldKey] = String(value || '')
   263→  }
   264→
   265→  // For orders/dispatch: calculate amount if not mapped
   266→  if (mapping) {
   267→    const targetModule = Object.values(mapping)
   268→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   269→      const qty = Number(result.quantity || 0)
   270→      const rate = Number(result.rate || 0)
   271→      result.amount = qty * rate
   272→    }
   273→  }
   274→
   275→  return result
   276→}

   278→// Parse various date formats to YYYY-MM-DD
   279→function parseDate(value: string): string {
   280→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   281→
   282→  const trimmed = value.trim()
   283→
   284→  // Already in YYYY-MM-DD format
   285→  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed
   286→
   287→  // DD-MM-YYYY or DD/MM/YYYY
   288→  const dmyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   289→  if (dmyMatch) {
   290→    const [, d, m, y] = dmyMatch
   291→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   292→  }
   293→
   294→  // DD-MM-YY or DD/MM/YY
   295→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2})$/)
   296→  if (dmyShortMatch) {
   297→    const [, d, m, y] = dmyShortMatch
   298→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   299→  }
   300→
   301→  // MM/DD/YYYY (US format) - only if first number > 12
   302→  const mdyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   303→  if (mdyMatch) {
   304→    const [, m, d, y] = mdyMatch
   305→    if (Number(m) > 12) {
   306→      return `${y}-${d.padStart(2, '0')}-${m.padStart(2, '0')}`
   307→    }
   308→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   309→  }
   310→
   311→  // Try native Date parsing as fallback
   312→  try {
   313→    const date = new Date(trimmed)
   314→    if (!isNaN(date.getTime())) {
   315→      return date.toISOString().split('T')[0]
   316→    }
   317→  } catch {}
   318→
   319→  return trimmed
   320→}

   322→export default function ExcelImport({ module, open, onClose, onSuccess }: ExcelImportProps) {
   323→  const [rawData, setRawData] = useState<Record<string, unknown>[]>([])
   324→  const [transformedData, setTransformedData] = useState<Record<string, unknown>[]>([])
   325→  const [columnMapping, setColumnMapping] = useState<Record<string, string>>({})
   326→  const [excelColumns, setExcelColumns] = useState<string[]>([])
   327→  const [importing, setImporting] = useState(false)
   328→  const [result, setResult] = useState<{ imported: number; total: number; errors?: string[] } | null>(null)
   329→  const [fileName, setFileName] = useState('')
   330→  const [customers, setCustomers] = useState<{ id: string; name: string }[]>([])
   331→  const fileInputRef = useRef<HTMLInputElement>(null)
   332→  const template = moduleTemplates[module]
   333→
   334→  // Fetch customers for name->ID resolution
   335→  React.useEffect(() => {
   336→    if (open && ['orders', 'dispatch', 'payments'].includes(module)) {
   337→      api.getCustomers().then((data) => {
   338→        setCustomers((data.customers as { id: string; name: string }[]).map((c) => ({ id: c.id, name: c.name })))
   339→      }).catch(() => {})
   340→    }
   341→  }, [open, module])
   342→
   343→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   344→    const file = e.target.files?.[0]
   345→    if (!file) return
   346→
   347→    setFileName(file.name)
   348→    setResult(null)
   349→
   350→    try {
   351→      const XLSX = await import('xlsx')
   352→      const arrayBuffer = await file.arrayBuffer()
   353→      const workbook = XLSX.read(arrayBuffer)
   354→      const sheetName = workbook.SheetNames[0]
   355→      const worksheet = workbook.Sheets[sheetName]
   356→      const jsonData = XLSX.utils.sheet_to_json(worksheet, { defval: '' })
   357→
   358→      if (jsonData.length === 0) {
   359→        toast({ title: 'Empty file', description: 'No data found in the uploaded file', variant: 'destructive' })
   360→        return
   361→      }
   362→
   363→      const data = jsonData as Record<string, unknown>[]
   364→      const columns = Object.keys(data[0])
   365→
   366→      setRawData(data)
   367→      setExcelColumns(columns)
   368→
   369→      // Auto-detect column mapping
   370→      const autoMapping = autoMapColumns(columns, template.fields)
   371→      setColumnMapping(autoMapping)
   372→
   373→      // Apply transformation immediately
   374→      const transformed = data.map((row) => transformRow(row, autoMapping, customers))
   375→      setTransformedData(transformed)
   376→
   377→      const mappedCount = Object.keys(autoMapping).length
   378→      const totalFields = template.fields.length
   379→      toast({
   380→        title: 'File parsed successfully',
   381→        description: `Found ${data.length} rows. Auto-mapped ${mappedCount} of ${totalFields} fields. Review and adjust if needed.`,
   382→      })
   383→    } catch (err) {
   384→      toast({ title: 'Parse error', description: 'Could not read the Excel file. Make sure it is a valid .xlsx or .csv file.', variant: 'destructive' })
   385→    }
   386→  }
   387→
   388→  const handleMappingChange = (excelCol: string, fieldKey: string) => {
   389→    const newMapping = { ...columnMapping }
   390→
   391→    // Remove old mapping for this excel column
   392→    delete newMapping[excelCol]
   393→
   394→    // Remove any existing mapping to the same field key
   395→    for (const [col, fKey] of Object.entries(newMapping)) {
   396→      if (fKey === fieldKey) {
   397→        delete newMapping[col]
   398→      }
   399→    }
   400→
   401→    // Set new mapping
   402→    if (fieldKey && fieldKey !== 'skip') {
   403→      newMapping[excelCol] = fieldKey
   404→    }
   405→
   406→    setColumnMapping(newMapping)
   407→
   408→    // Re-transform data
   409→    const transformed = rawData.map((row) => transformRow(row, newMapping, customers))
   410→    setTransformedData(transformed)
   411→  }
   412→
   413→  const handleImport = async () => {
   414→    if (transformedData.length === 0) return
   415→    setImporting(true)
   416→    try {
   417→      const res = await api.importData(module, transformedData)
   418→      setResult({ imported: res.imported, total: res.total, errors: res.errors })
   419→      if (res.imported > 0) {
   420→        toast({ title: 'Import successful', description: `${res.imported} of ${res.total} rows imported` })
   421→        onSuccess()
   422→      }
   423→    } catch (err) {
   424→      toast({ title: 'Import failed', description: err instanceof Error ? err.message : 'Unknown error', variant: 'destructive' })
   425→    } finally {
   426→      setImporting(false)
   427→    }
   428→  }
   429→
   430→  const downloadTemplate = () => {
   431→    const headers = template.fields.map((f) => f.label)
   432→    const csvContent = [headers.join(','), ''].join('\n')
   433→    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' })
   434→    const url = URL.createObjectURL(blob)
   435→    const a = document.createElement('a')
   436→    a.href = url
   437→    a.download = `${module}_import_template.csv`
   438→    a.click()
   439→    URL.revokeObjectURL(url)
   440→    toast({ title: 'Template downloaded', description: 'Fill in your data and upload the file' })
   441→  }
   442→
   443→  const handleClose = () => {
   444→    setRawData([])
   445→    setTransformedData([])
   446→    setColumnMapping({})
   447→    setExcelColumns([])
   448→    setResult(null)
   449→    setFileName('')
   450→    if (fileInputRef.current) fileInputRef.current.value = ''
   451→    onClose()
   452→  }
   453→
   454→  const mappedFieldKeys = new Set(Object.values(columnMapping))
   455→  const unmappedRequired = template.fields.filter((f) => f.required && !mappedFieldKeys.has(f.key))
   456→
   457→  return (
   458→    <Dialog open={open} onOpenChange={handleClose}>
   459→      <DialogContent className="max-w-4xl max-h-[90vh] flex flex-col">
   460→        <DialogHeader>
   461→          <DialogTitle className="flex items-center gap-2">
   462→            <FileSpreadsheet className="h-5 w-5 text-emerald-600" />
   463→            Import {template.label} from Excel
   464→          </DialogTitle>
   465→          <DialogDescription>
   466→            Upload an Excel or CSV file. Columns are auto-detected and mapped — review and adjust before importing.
   467→          </DialogDescription>
   468→        </DialogHeader>
   469→
   470→        <div className="space-y-4 flex-1 overflow-auto">
   471→          {/* Step 1: Download template */}
   472→          <div className="space-y-2">
   473→            <h4 className="font-medium text-sm flex items-center gap-2">
   474→              <span className="w-6 h-6 bg-emerald-600 text-white rounded-full flex items-center justify-center text-xs font-bold">1</span>
   475→              Template & Format
   476→            </h4>
   477→            <div className="flex flex-wrap gap-2">
   478→              {template.fields.map((field) => (
   479→                <Badge key={field.key} variant={field.required ? 'default' : 'outline'} className="text-xs">
   480→                  {field.label} {field.required && '*'}
   481→                </Badge>
   482→              ))}
   483→            </div>
   484→            <Button variant="outline" size="sm" onClick={downloadTemplate}>
   485→              <Download className="h-4 w-4 mr-2" />
   486→              Download CSV Template
   487→            </Button>
   488→          </div>
   489→
   490→          {/* Step 2: Upload file */}
   491→          <div className="space-y-2">
   492→            <h4 className="font-medium text-sm flex items-center gap-2">
   493→              <span className="w-6 h-6 bg-emerald-600 text-white rounded-full flex items-center justify-center text-xs font-bold">2</span>
   494→              Upload Excel/CSV File
   495→            </h4>
   496→            <div
   497→              className="border-2 border-dashed rounded-lg p-6 text-center hover:border-emerald-400 transition-colors cursor-pointer"
   498→              onClick={() => fileInputRef.current?.click()}
   499→            >
   500→              <Upload className="h-8 w-8 mx-auto mb-2 text-muted-foreground" />
   501→              <p className="text-sm text-muted-foreground">
   502→                {fileName ? fileName : 'Click to upload .xlsx, .xls, or .csv file'}
   503→              </p>
   504→              <input
   505→                ref={fileInputRef}
   506→                type="file"
   507→                accept=".xlsx,.xls,.csv"
   508→                className="hidden"
   509→                onChange={handleFileUpload}
   510→              />
   511→            </div>
   512→          </div>
   513→
   514→          {/* Step 3: Column Mapping */}
   515→          {excelColumns.length > 0 && (
   516→            <div className="space-y-3">
   517→              <h4 className="font-medium text-sm flex items-center gap-2">
   518→                <span className="w-6 h-6 bg-emerald-600 text-white rounded-full flex items-center justify-center text-xs font-bold">3</span>
   519→                <Sparkles className="h-4 w-4 text-amber-500" />
   520→                Column Mapping (Auto-detected)
   521→              </h4>
   522→
   523→              {unmappedRequired.length > 0 && (
   524→                <Alert variant="destructive" className="py-2">
   525→                  <AlertCircle className="h-4 w-4" />
   526→                  <AlertDescription className="text-xs">
   527→                    Required fields not mapped: {unmappedRequired.map((f) => f.label).join(', ')}
   528→                  </AlertDescription>
   529→                </Alert>
   530→              )}
   531→
   532→              <div className="border rounded-lg overflow-hidden">
   533→                <Table>
   534→                  <TableHeader>
   535→                    <TableRow>
   536→                      <TableHead className="w-1/3">Excel Column</TableHead>
   537→                      <TableHead className="w-10 text-center"></TableHead>
   538→                      <TableHead className="w-1/3">Maps To</TableHead>
   539→                      <TableHead className="w-1/6">Status</TableHead>
   540→                    </TableRow>
   541→                  </TableHeader>
   542→                  <TableBody>
   543→                    {excelColumns.map((col) => {
   544→                      const mappedField = columnMapping[col]
   545→                      const field = mappedField ? template.fields.find((f) => f.key === mappedField) : null
   546→                      return (
   547→                        <TableRow key={col}>
   548→                          <TableCell className="font-mono text-xs">{col}</TableCell>
   549→                          <TableCell className="text-center"><ArrowRight className="h-4 w-4 text-muted-foreground mx-auto" /></TableCell>
   550→                          <TableCell>
   551→                            <Select
   552→                              value={mappedField || 'skip'}
   553→                              onValueChange={(val) => handleMappingChange(col, val)}
   554→                            >
   555→                              <SelectTrigger className="w-full text-xs">
   556→                                <SelectValue placeholder="Select field..." />
   557→                              </SelectTrigger>
   558→                              <SelectContent>
   559→                                <SelectItem value="skip">-- Skip --</SelectItem>
   560→                                {template.fields.map((f) => (
   561→                                  <SelectItem key={f.key} value={f.key}>
   562→                                    {f.label} {f.required && '*'}
   563→                                  </SelectItem>
   564→                                ))}
   565→                              </SelectContent>
   566→                            </Select>
   567→                          </TableCell>
   568→                          <TableCell>
   569→                            {field ? (
   570→                              <Badge variant="outline" className="text-xs bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-900/20 dark:text-emerald-400 dark:border-emerald-800">
   571→                                <CheckCircle2 className="h-3 w-3 mr-1" /> Mapped
   572→                              </Badge>
   573→                            ) : (
   574→                              <Badge variant="outline" className="text-xs text-muted-foreground">
   575→                                Skipped
   576→                              </Badge>
   577→                            )}
   578→                          </TableCell>
   579→                        </TableRow>
   580→                      )
   581→                    })}
   582→                  </TableBody>
   583→                </Table>
   584→              </div>
   585→            </div>
   586→          )}
   587→
   588→          {/* Step 4: Preview transformed data */}
   589→          {transformedData.length > 0 && (
   590→            <div className="space-y-2">
   591→              <h4 className="font-medium text-sm flex items-center gap-2">
   592→                <span className="w-6 h-6 bg-emerald-600 text-white rounded-full flex items-center justify-center text-xs font-bold">4</span>
   593→                Preview ({transformedData.length} rows)
   594→              </h4>
   595→              <ScrollArea className="max-h-56">
   596→                <div className="overflow-x-auto">
   597→                  <Table>
   598→                    <TableHeader>
   599→                      <TableRow>
   600→                        <TableHead className="text-xs w-8">#</TableHead>
   601→                        {template.fields
   602→                          .filter((f) => mappedFieldKeys.has(f.key))
   603→                          .map((f) => (
   604→                            <TableHead key={f.key} className="text-xs whitespace-nowrap">{f.label}</TableHead>
   605→                          ))}
   606→                      </TableRow>
   607→                    </TableHeader>
   608→                    <TableBody>
   609→                      {transformedData.slice(0, 10).map((row, i) => (
   610→                        <TableRow key={i}>
   611→                          <TableCell className="text-xs text-muted-foreground">{i + 1}</TableCell>
   612→                          {template.fields
   613→                            .filter((f) => mappedFieldKeys.has(f.key))
   614→                            .map((f) => (
   615→                              <TableCell key={f.key} className="text-xs whitespace-nowrap">
   616→                                {String(row[f.key] ?? '')}
   617→                              </TableCell>
   618→                            ))}
   619→                        </TableRow>
   620→                      ))}
   621→                      {transformedData.length > 10 && (
   622→                        <TableRow>
   623→                          <TableCell colSpan={template.fields.filter((f) => mappedFieldKeys.has(f.key)).length + 1} className="text-xs text-center text-muted-foreground">
   624→                            ... and {transformedData.length - 10} more rows
   625→                          </TableCell>
   626→                        </TableRow>
   627→                      )}
   628→                    </TableBody>
   629→                  </Table>
   630→                </div>
   631→              </ScrollArea>
   632→            </div>
   633→          )}
   634→
   635→          {/* Import result */}
   636→          {result && (
   637→            <Alert variant={result.errors && result.errors.length > 0 ? 'destructive' : 'default'}>
   638→              <div className="flex items-start gap-2">
   639→                {result.errors && result.errors.length > 0 ? (
   640→                  <AlertCircle className="h-4 w-4 mt-0.5" />
   641→                ) : (
   642→                  <CheckCircle2 className="h-4 w-4 mt-0.5" />
   643→                )}
   644→                <AlertDescription>
   645→                  <p className="font-medium">
   646→                    Imported {result.imported} of {result.total} rows successfully
   647→                  </p>
   648→                  {result.errors && result.errors.length > 0 && (
   649→                    <ScrollArea className="max-h-32 mt-2">
   650→                      <ul className="text-xs space-y-1">
   651→                        {result.errors.slice(0, 20).map((err, i) => (
   652→                          <li key={i} className="text-destructive">{err}</li>
   653→                        ))}
   654→                        {result.errors.length > 20 && (
   655→                          <li className="text-muted-foreground">... and {result.errors.length - 20} more errors</li>
   656→                        )}
   657→                      </ul>
   658→                    </ScrollArea>
   659→                  )}
   660→                </AlertDescription>
   661→              </div>
   662→            </Alert>
   663→          )}
   664→        </div>
   665→
   666→        <DialogFooter>
   667→          <Button variant="outline" onClick={handleClose}>Cancel</Button>
   668→          <Button
   669→            onClick={handleImport}
   670→            disabled={transformedData.length === 0 || importing || unmappedRequired.length > 0}
   671→            className="bg-emerald-600 hover:bg-emerald-700"
   672→          >
   673→            {importing ? (
   674→              <>
   675→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   676→                Importing...
   677→              </>
   678→            ) : (
   679→              <>
   680→                <Upload className="h-4 w-4 mr-2" />
   681→                Import {transformedData.length} Rows
   682→              </>
   683→            )}
   684→          </Button>
   685→        </DialogFooter>
   686→      </DialogContent>
   687→    </Dialog>
   688→  )
   689→}
