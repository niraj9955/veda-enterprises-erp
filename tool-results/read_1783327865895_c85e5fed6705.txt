    13→// ─── Duplicate detection ────────────────────────────────────────────────────
    14→//
    15→// Per user request: "jo duplicate ho use skip kre or uska mess de but jo
    16→// duplicate nhi h usko import kre record me" — when an Excel row matches an
    17→// existing record in the DB (by a natural key), SKIP it with a clear message
    18→// but still IMPORT every non-duplicate row.
    19→//
    20→// To keep this fast, for each module we run ONE batched `find()` that
    21→// returns every existing row whose date falls within the dates seen in
    22→// the import payload, then build a Set of natural-key strings. Per-row
    23→// lookup then becomes an O(1) Set.has() check instead of a findOne().
    24→//
    25→// APPEND-ONLY semantics are preserved — we never wipe existing rows.
    26→
    27→// Build a stable natural-key string for a row, per module.
    28→function rowKey(module: string, row: Record<string, unknown>): string {
    29→  const s = (v: unknown) => String(v ?? '').trim().toLowerCase()
    30→  switch (module) {
    31→    case 'customers':
    32→      return s(row.mobile)
    33→    case 'production':
    34→      // One production entry per date — same date = duplicate (matches stock semantics)
    35→      return s(row.date)
    36→    case 'stock':
    37→      // One stock entry per date — same date = duplicate
    38→      return s(row.date)
    39→    case 'dailySell':
    40→      return `${s(row.date)}|${s(row.customerName)}|${s(row.amount)}`
    41→    case 'customerPayment':
    42→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
    43→    case 'labourPayment':
    44→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
    45→    case 'tractorPayment':
    46→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantityTon)}|${s(row.rate)}`
    47→    case 'dustPurchase':
    48→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantity)}|${s(row.rate)}`
    49→    case 'cementPurchase':
    50→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.itemName)}|${s(row.quantity)}|${s(row.rate)}`
    51→    case 'hardner':
    52→      // Only date+amount available — same date AND same amount = duplicate
    53→      return `${s(row.date)}|${s(row.amount)}`
    54→    case 'electricity':
    55→      return `${s(row.date)}|${s(row.name)}|${s(row.work)}|${s(row.amount)}`
    56→    case 'factoryStuff':
    57→      return `${s(row.date)}|${s(row.itemName)}|${s(row.amount)}`
    58→    case 'orders':
    59→      return `${s(row.customerId)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.rate)}|${s(row.deliveryDate || row.date)}`
    60→    case 'dispatch':
    61→      return `${s(row.customerId)}|${s(row.truckNumber)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.date)}`
    62→    case 'payments':
    63→      return `${s(row.customerId)}|${s(row.paymentType)}|${s(row.amount)}|${s(row.date)}`
    64→    case 'expenses':
    65→      return `${s(row.category)}|${s(row.amount)}|${s(row.date)}|${s(row.description)}`
    66→    default:
    67→      return ''
    68→  }
    69→}

   112→// Pre-fetch existing DB records that could collide with any row in `data`.
   113→// Returns a Set of natural-key strings.
   114→async function buildExistingKeys(module: string, data: Record<string, unknown>[]): Promise<Set<string>> {
   115→  const keys = new Set<string>()
   116→  if (data.length === 0) return keys
   117→
   118→  // Collect unique dates and customerIds from the import payload so we can
   119→  // narrow the DB query. This keeps the result set small.
   120→  const dates = new Set<string>()
   121→  const mobiles = new Set<string>()
   122→  const customerIds = new Set<string>()
   123→  for (const row of data) {
   124→    const d = String(row.date ?? '').split('T')[0]
   125→    if (d) dates.add(d)
   126→    const dd = String(row.deliveryDate ?? '').split('T')[0]
   127→    if (dd) dates.add(dd)
   128→    const m = String(row.mobile ?? '').trim()
   129→    if (m) mobiles.add(m)
   130→    const cid = String(row.customerId ?? '').trim()
   131→    if (cid) customerIds.add(cid)
   132→  }
   133→
   134→  let docs: Record<string, unknown>[] = []
   135→
   136→  switch (module) {
   137→    case 'customers':
   138→      docs = mobiles.size > 0
   139→        ? await Customer.find({ mobile: { $in: Array.from(mobiles) } }).lean() as any
   140→        : []
   141→      break
   142→    case 'production':
   143→      docs = dates.size > 0
   144→        ? await Production.find({ date: { $in: Array.from(dates) } }).lean() as any
   145→        : []
   146→      break
   147→    case 'stock':
   148→      docs = dates.size > 0
   149→        ? await Stock.find({ date: { $in: Array.from(dates) } }).lean() as any
   150→        : []
   151→      break
   152→    case 'dailySell':
   153→      docs = dates.size > 0
   154→        ? await DailySell.find({ date: { $in: Array.from(dates) } }).lean() as any
   155→        : []
   156→      break
   157→    case 'customerPayment':
   158→      docs = dates.size > 0
   159→        ? await CustomerPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   160→        : []
   161→      break
   162→    case 'labourPayment':
   163→      docs = dates.size > 0
   164→        ? await LabourPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   165→        : []
   166→      break
   167→    case 'tractorPayment':
   168→      docs = dates.size > 0
   169→        ? await TractorPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   170→        : []
   171→      break
   172→    case 'dustPurchase':
   173→      docs = dates.size > 0
   174→        ? await DustPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   175→        : []
   176→      break
   177→    case 'cementPurchase':
   178→      docs = dates.size > 0
   179→        ? await CementPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   180→        : []
   181→      break
   182→    case 'hardner':
   183→      docs = dates.size > 0
   184→        ? await Hardner.find({ date: { $in: Array.from(dates) } }).lean() as any
   185→        : []
   186→      break
   187→    case 'electricity':
   188→      docs = dates.size > 0
   189→        ? await Electricity.find({ date: { $in: Array.from(dates) } }).lean() as any
   190→        : []
   191→      break
   192→    case 'factoryStuff':
   193→      docs = dates.size > 0
   194→        ? await FactoryStuff.find({ date: { $in: Array.from(dates) } }).lean() as any
   195→        : []
   196→      break
   197→    case 'orders':
   198→      docs = customerIds.size > 0
   199→        ? await Order.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   200→        : []
   201→      break
   202→    case 'dispatch':
   203→      docs = customerIds.size > 0
   204→        ? await Dispatch.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   205→        : []
   206→      break
   207→    case 'payments':
   208→      docs = customerIds.size > 0
   209→        ? await Payment.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   210→        : []
   211→      break
   212→    case 'expenses':
   213→      // Expenses dedupe by category+amount+date — narrow by date
   214→      docs = dates.size > 0
   215→        ? await Expense.find({ date: { $in: Array.from(dates) } }).lean() as any
   216→        : []
   217→      break
   218→  }
   219→
   220→  for (const d of docs) {
   221→    const k = dbKey(module, d as Record<string, unknown>)
   222→    if (k) keys.add(k)
   223→  }
   224→  return keys
   225→}

   227→export async function POST(request: Request) {
   228→  try {
   229→    await connectDB()
   230→    const body = await request.json()
   231→    const { module, data } = body
   232→
   233→    if (!module || !data || !Array.isArray(data)) {
   234→      return NextResponse.json({ error: 'Module and data array are required' }, { status: 400 })
   235→    }
   236→
   237→    // ── Pre-fetch existing keys (1 DB query for the whole batch) ──────────
   238→    const existingKeys = await buildExistingKeys(module, data as Record<string, unknown>[])
   239→    // Track keys we've already seen WITHIN this batch so a duplicate Excel
   240→    // row doesn't get inserted twice in the same import run.
   241→    const seenInBatch = new Set<string>()
   242→
   243→    let imported = 0
   244→    let skipped = 0
   245→    let duplicatesSkipped = 0
   246→    const errors: string[] = []
   247→    const skippedReasons: string[] = []
   248→
   249→    for (let i = 0; i < data.length; i++) {
   250→      try {
   251→        const row = data[i]
   252→
   253→        // ── Duplicate check (against DB AND within this batch) ──────────
   254→        const key = rowKey(module, row as Record<string, unknown>)
   255→        if (key) {
   256→          if (existingKeys.has(key)) {
   257→            skipped++
   258→            duplicatesSkipped++
   259→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   260→            skippedReasons.push(`Row ${i + 1}: Duplicate — ${label} already exists in records, skipped`)
   261→            continue
   262→          }
   263→          if (seenInBatch.has(key)) {
   264→            skipped++
   265→            duplicatesSkipped++
   266→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   267→            skippedReasons.push(`Row ${i + 1}: Duplicate — ${label} appears more than once in this Excel file, only first occurrence imported`)
   268→            continue
   269→          }
   270→          seenInBatch.add(key)
   271→        }

   531→          }
   532→
   533→          // ─── Legacy modules (Orders/Dispatch/Payments/Expenses) ────
   534→          case 'orders': {
   535→            if (!row.customerId || !row.brickType || !row.quantity || !row.rate) {
   536→              errors.push(`Row ${i + 1}: Customer, brick type, quantity, and rate are required`)
   537→              skipped++
   538→              continue
   539→            }
   540→            const count = await Order.countDocuments({})
   541→            const { Company } = await import('@/lib/models')
   542→            const company = await Company.findOne({})
   543→            const prefix = company?.orderPrefix || 'ORD'
   544→            const orderNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   545→
   546→            await Order.create({
   547→              orderNumber,
   548→              customerId: row.customerId,
   549→              brickType: row.brickType,
   550→              quantity: Number(row.quantity),
   551→              rate: Number(row.rate),
   552→              amount: Number(row.amount) || Number(row.quantity) * Number(row.rate),
   553→              deliveryDate: row.deliveryDate || row.date || new Date().toISOString().split('T')[0],
   554→              status: row.status || 'Pending',
   555→            })
   556→            imported++
   557→            break
   558→          }
   559→
   560→          case 'dispatch': {
   561→            if (!row.customerId || !row.truckNumber || !row.quantity || !row.brickType || !row.date) {
   562→              errors.push(`Row ${i + 1}: Customer, truck, quantity, brick type, and date are required`)
   563→              skipped++
   564→              continue
   565→            }
   566→            const count = await Dispatch.countDocuments({})
   567→            const { Company } = await import('@/lib/models')
   568→            const company = await Company.findOne({})
   569→            const prefix = company?.dispatchPrefix || 'DSP'
   570→            const dispatchNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   571→
   572→            await Dispatch.create({
   573→              dispatchNumber,
   574→              customerId: row.customerId,
   575→              orderId: row.orderId || null,
   576→              truckNumber: row.truckNumber,
   577→              driverName: row.driverName || '',
   578→              quantity: Number(row.quantity),
   579→              brickType: row.brickType,
   580→              date: row.date,
   581→            })
   582→            imported++
   583→            break
   584→          }
   585→
   586→          case 'payments': {
   587→            if (!row.customerId || !row.paymentType || !row.amount || !row.date) {
   588→              errors.push(`Row ${i + 1}: Customer, payment type, amount, and date are required`)
   589→              skipped++
   590→              continue
   591→            }
   592→            await Payment.create({
   593→              customerId: row.customerId,
   594→              paymentType: row.paymentType,
   595→              amount: Number(row.amount),
   596→              date: row.date,
   597→              remarks: row.remarks || '',
   598→            })
   599→            imported++
   600→            break
   601→          }
   602→
   603→          case 'expenses': {
   604→            if (!row.category || !row.amount || !row.date) {
   605→              errors.push(`Row ${i + 1}: Category, amount, and date are required`)
   606→              skipped++
   607→              continue
   608→            }
   609→            await Expense.create({
   610→              category: row.category,
   611→              amount: Number(row.amount),
   612→              date: row.date,
   613→              description: row.description || '',
   614→            })
   615→            imported++
   616→            break
   617→          }
   618→
   619→          default:
   620→            errors.push(`Unknown module: ${module}`)
   621→        }
   622→      } catch (err) {
   623→        errors.push(`Row ${i + 1}: ${err instanceof Error ? err.message : 'Import failed'}`)
   624→        skipped++
   625→      }
   626→    }
   627→
   628→    // Merge errors and skippedReasons so the UI can show every reason a row
   629→    // was not imported (validation error OR duplicate OR anything else).
   630→    const allReasons = [...errors, ...skippedReasons]
   631→
   632→    // Auto-sync Stock Overview from Production — when production rows are
   633→    // imported, the Stock module must reflect the latest daily totals
   634→    // automatically. We collect every date touched by THIS import and
   635→    // re-aggregate the matching Stock snapshot for each.
   636→    if (module === 'production' && imported > 0) {
   637→      const touchedDates = data
   638→        .map((row) => String(row.date || '').split('T')[0])
   639→        .filter(Boolean)
   640→      if (touchedDates.length > 0) {
   641→        try {
   642→          await syncStockForDates(touchedDates)
   643→        } catch (err) {
   644→          console.error('[import] Stock sync failed:', err)
   645→          // Don't fail the import — production rows were already saved.
   646→          // Just append a soft warning to the error list.
   647→          allReasons.push('Warning: production rows imported, but Stock Overview auto-sync failed. Check server logs.')
   648→        }
   649→      }
   650→    }
   651→
   652→    return NextResponse.json({
   653→      success: true,
   654→      imported,
   655→      skipped,
   656→      duplicatesSkipped,
   657→      total: data.length,
   658→      errors: allReasons.length > 0 ? allReasons.slice(0, 50) : undefined,
   659→    })
   660→  } catch (error) {
   661→    console.error('Error importing data:', error)
   662→    return NextResponse.json({ error: 'Failed to import data' }, { status: 500 })
   663→  }
   664→}

   666→// Human-readable label for a duplicate row, used in the skip reason.
   667→function duplicateRowLabel(module: string, row: Record<string, unknown>): string {
   668→  switch (module) {
   669→    case 'customers':
   670→      return `customer "${row.name}" (mobile ${row.mobile})`
   671→    case 'production':
   672→      return `production entry on ${row.date}`
   673→    case 'stock':
   674→      return `stock entry for ${row.date}`
   675→    case 'dailySell':
   676→      return `daily sell for "${row.customerName}" on ${row.date} (₹${row.amount})`
   677→    case 'customerPayment':
   678→      return `payment by "${row.name}" on ${row.date} (₹${row.amount})`
   679→    case 'labourPayment':
   680→      return `payment to "${row.name}" on ${row.date} (₹${row.amount})`
   681→    case 'tractorPayment':
   682→      return `tractor payment to "${row.vendorName}" on ${row.date}`
   683→    case 'dustPurchase':
   684→      return `dust purchase from "${row.vendorName}" on ${row.date}`
   685→    case 'cementPurchase':
   686→      return `cement purchase from "${row.vendorName}" on ${row.date}`
   687→    case 'hardner':
   688→      return `hardner entry on ${row.date} (₹${row.amount})`
   689→    case 'electricity':
   690→      return `electricity entry on ${row.date} (₹${row.amount})`
   691→    case 'factoryStuff':
   692→      return `factory stuff "${row.itemName}" on ${row.date} (₹${row.amount})`
   693→    case 'orders':
   694→      return `order for customer ${row.customerId} on ${row.deliveryDate || row.date}`
   695→    case 'dispatch':
   696→      return `dispatch to customer ${row.customerId} on ${row.date}`
   697→    case 'payments':
   698→      return `payment for customer ${row.customerId} on ${row.date}`
   699→    case 'expenses':
   700→      return `expense "${row.category}" on ${row.date} (₹${row.amount})`
   701→    default:
   702→      return 'record'
   703→  }
   704→}
