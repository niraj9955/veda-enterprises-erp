   166→   166→// ── Component ───────────────────────────────────────────────────────────────
   167→   167→
   168→   168→export function PaymentModule() {
   169→   169→  // State
   170→   170→  const [payments, setPayments] = React.useState<Payment[]>([])
   171→   171→  const [search, setSearch] = React.useState('')
   172→   172→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   173→   173→  const [customers, setCustomers] = React.useState<Customer[]>([])
   174→   174→  const [orders, setOrders] = React.useState<Order[]>([])
   175→   175→  const [loading, setLoading] = React.useState(true)
   176→   176→
   177→   177→  // Dialog states
   178→   178→  const [dialogOpen, setDialogOpen] = React.useState(false)
   179→   179→  const [editingPayment, setEditingPayment] = React.useState<Payment | null>(null)
   180→   180→  const [formData, setFormData] = React.useState<PaymentFormData>(emptyForm)
   181→   181→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   182→   182→
   183→   183→  // Bills for the currently-selected customer — used to populate the
   184→   184→  // "Link to Bill" dropdown in the payment form. Fetched whenever the user
   185→   185→  // picks a customer (or opens the edit dialog with a pre-selected customer).
   186→   186→  // Only outstanding bills (status = 'partial' or 'draft' or 'sent') are
   187→   187→  // shown — fully paid bills are hidden to keep the dropdown short.
   188→   188→  const [customerBills, setCustomerBills] = React.useState<BillOption[]>([])
   189→   189→  const [loadingBills, setLoadingBills] = React.useState(false)
   190→   190→
   191→   191→  // Excel import
   192→   192→  const [importOpen, setImportOpen] = React.useState(false)
   193→   193→
   194→   194→  // Delete confirmation
   195→   195→  const [deleteTarget, setDeleteTarget] = React.useState<Payment | null>(null)
   196→   196→  const [deleting, setDeleting] = React.useState(false)
   197→   197→
   198→   198→  // ── Computed: Summary values ─────────────────────────────────────────────
   199→   199→  const totalReceived = React.useMemo(
   200→   200→    () => payments.reduce((sum, p) => sum + Number(p.amount), 0),
   201→   201→    [payments]
   202→   202→  )
   203→   203→
   204→   204→  const cashPayments = React.useMemo(
   205→   205→    () =>
   206→   206→      payments
   207→   207→        .filter((p) => p.paymentType === 'Cash')
   208→   208→        .reduce((sum, p) => sum + Number(p.amount), 0),
   209→   209→    [payments]
   210→   210→  )
   211→   211→
   212→   212→  const upiBankPayments = React.useMemo(
   213→   213→    () =>
   214→   214→      payments
   215→   215→        .filter((p) => p.paymentType === 'UPI' || p.paymentType === 'Bank Transfer')
   216→   216→        .reduce((sum, p) => sum + Number(p.amount), 0),
   217→   217→    [payments]
   218→   218→  )
   219→   219→
   220→   220→  // ── Computed: Customer outstanding ───────────────────────────────────────
   221→   221→  const outstandingEntries = React.useMemo<OutstandingEntry[]>(() => {
   222→   222→    const map = new Map<string, OutstandingEntry>()
   223→   223→
   224→   224→    // Sum order amounts per customer (exclude Cancelled)
   225→   225→    for (const order of orders) {
   226→   226→      if (order.status === 'Cancelled') continue
   227→   227→      const existing = map.get(order.customerId)
   228→   228→      if (existing) {
   229→   229→        existing.totalOrderAmount += Number(order.amount)
   230→   230→      } else {
   231→   231→        map.set(order.customerId, {
   232→   232→          customerId: order.customerId,
   233→   233→          customerName: order.customer?.name || 'Unknown',
   234→   234→          totalOrderAmount: Number(order.amount),
   235→   235→          totalPayments: 0,
   236→   236→          outstanding: 0,
   237→   237→        })
   238→   238→      }
   239→   239→    }
   240→   240→
   241→   241→    // Sum payments per customer
   242→   242→    for (const payment of payments) {
   243→   243→      const existing = map.get(payment.customerId)
   244→   244→      if (existing) {
   245→   245→        existing.totalPayments += Number(payment.amount)
   246→   246→      } else {
   247→   247→        map.set(payment.customerId, {
   248→   248→          customerId: payment.customerId,
   249→   249→          customerName: payment.customer?.name || 'Unknown',
   250→   250→          totalOrderAmount: 0,
   251→   251→          totalPayments: Number(payment.amount),
   252→   252→          outstanding: 0,
   253→   253→        })
   254→   254→      }
   255→   255→    }
   256→   256→
   257→   257→    // Calculate outstanding
   258→   258→    for (const entry of map.values()) {
   259→   259→      entry.outstanding = entry.totalOrderAmount - entry.totalPayments
   260→   260→    }
   261→   261→
   262→   262→    // Return only customers with outstanding > 0, sorted by outstanding descending
   263→   263→    return Array.from(map.values())
   264→   264→      .filter((e) => e.outstanding > 0.01)
   265→   265→      .sort((a, b) => b.outstanding - a.outstanding)
   266→   266→  }, [orders, payments])
   267→   267→
   268→   268→  // ── Fetch data ───────────────────────────────────────────────────────────
   269→   269→  const fetchPayments = React.useCallback(async () => {
   270→   270→    setLoading(true)
   271→   271→    try {
   272→   272→      const res = await api.getPayments()
   273→   273→      const paymentList = (res.payments as Payment[]).sort(
   274→   274→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   275→   275→      )
   276→   276→      setPayments(paymentList)
   277→   277→    } catch (err) {
   278→   278→      toast({
   279→   279→        title: 'Error',
   280→   280→        description: err instanceof Error ? err.message : 'Failed to fetch payments',
   281→   281→        variant: 'destructive',
   282→   282→      })
   283→   283→    } finally {
   284→   284→      setLoading(false)
   285→   285→    }
   286→   286→  }, [])
   287→   287→
   288→   288→  const fetchCustomers = React.useCallback(async () => {
   289→   289→    try {
   290→   290→      const res = await api.getCustomers()
   291→   291→      setCustomers(res.customers as Customer[])
   292→   292→    } catch {
   293→   293→      // Silently fail — customer dropdown will just be empty
   294→   294→    }
   295→   295→  }, [])
   296→   296→
   297→   297→  const fetchOrders = React.useCallback(async () => {
   298→   298→    try {
   299→   299→      const res = await api.getOrders()
   300→   300→      setOrders(res.orders as Order[])
   301→   301→    } catch {
   302→   302→      // Silently fail — outstanding section will just be empty
   303→   303→    }
   304→   304→  }, [])
   305→   305→
   306→   306→  // Debounced search
   307→   307→  React.useEffect(() => {
   308→   308→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   309→   309→    return () => clearTimeout(t)
   310→   310→  }, [search])
   311→   311→
   312→   312→  // Client-side filter
   313→   313→  const filteredPayments = React.useMemo(() => {
   314→   314→    if (!debouncedSearch.trim()) return payments
   315→   315→    const q = debouncedSearch.toLowerCase()
   316→   316→    return payments.filter((item: any) =>
   317→   317→      ['date', 'paymentType', 'remarks'].some((f) =>
   318→   318→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   319→   319→      )
   320→   320→    )
   321→   321→  }, [payments, debouncedSearch])
   322→   322→
   323→   323→  React.useEffect(() => {
   324→   324→    fetchPayments()
   325→   325→    fetchCustomers()
   326→   326→    fetchOrders()
   327→   327→  }, [fetchPayments, fetchCustomers, fetchOrders])
   328→   328→
   329→   329→  // ── Form handlers ────────────────────────────────────────────────────────
   330→   330→  const openCreateDialog = () => {
   331→   331→    setEditingPayment(null)
   332→   332→    setFormData(emptyForm)
   333→   333→    setCustomerBills([])
   334→   334→    setDialogOpen(true)
   335→   335→  }
   336→   336→
   337→   337→  const openEditDialog = (payment: Payment) => {
   338→   338→    setEditingPayment(payment)
   339→   339→    setFormData({
   340→   340→      customerId: payment.customerId,
   341→   341→      paymentType: payment.paymentType,
   342→   342→      amount: payment.amount,
   343→   343→      date: payment.date,
   344→   344→      remarks: payment.remarks,
   345→   345→      billId: payment.billId || '',
   346→   346→    })
   347→   347→    setDialogOpen(true)
   348→   348→    // Fetch this customer's outstanding bills (including the linked one,
   349→   349→    // even if it's already paid, so the user sees the current link).
   350→   350→    fetchCustomerBills(payment.customerId)
   351→   351→  }
   352→   352→
   353→   353→  // Fetch outstanding bills for a customer — populates the "Link to Bill"
   354→   354→  // dropdown. Safe to call repeatedly; debouncing not needed because the
   355→   355→  // customer change is a discrete user action (not a keystroke stream).
   356→   356→  const fetchCustomerBills = async (customerId: string) => {
   357→   357→    if (!customerId) {
   358→   358→      setCustomerBills([])
   359→   359→      return
   360→   360→    }
   361→   361→    setLoadingBills(true)
   362→   362→    try {
   363→   363→      // We use the existing /api/bills endpoint with search filter. The
   364→   364→      // response is the full bill list but we filter client-side by
   365→   365→      // customerId + outstanding status. For large datasets this should
   366→   366→      // eventually become a server-side filter, but for now this is fast
   367→   367→      // enough (typically <50 bills per customer).
   368→   368→      const data = await api.getBills() as { bills: BillOption[] }
   369→   369→      const filtered = (data.bills || [])
   370→   370→        .filter((b) => {
   371→   371→          // Match by customerId. Note: /api/bills returns bills with
   372→   372→          // customerId as a string (toObject renames _id → id).
   373→   373→          const billCustomerId = (b as any).customerId
   374→   374→          return billCustomerId && String(billCustomerId) === String(customerId)
   375→   375→        })
   376→   376→        // Show outstanding + the currently-linked bill (even if 'paid')
   377→   377→        .filter((b) => {
   378→   378→          if (editingPayment && editingPayment.billId && String(b.id) === String(editingPayment.billId)) return true
   379→   379→          return b.status !== 'paid' && b.status !== 'cancelled'
   380→   380→        })
   381→   381→        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   382→   382→      setCustomerBills(filtered)
   383→   383→    } catch (err) {
   384→   384→      console.error('Failed to fetch customer bills:', err)
   385→   385→      setCustomerBills([])
   386→   386→    } finally {
   387→   387→      setLoadingBills(false)
   388→   388→    }
   389→   389→  }
   390→   390→
   391→   391→  // When customer changes in the form, refresh the bills dropdown.
   392→   392→  // Wrapped in useEffect so we don't refetch on every render.
   393→   393→  React.useEffect(() => {
   394→   394→    if (dialogOpen && formData.customerId) {
   395→   395→      fetchCustomerBills(formData.customerId)
   396→   396→    } else if (!formData.customerId) {
   397→   397→      setCustomerBills([])
   398→   398→    }
   399→   399→    // eslint-disable-next-line react-hooks/exhaustive-deps
   400→   400→  }, [formData.customerId, dialogOpen])
   401→   401→
   402→   402→  const handleFormChange = (field: keyof PaymentFormData, value: string) => {
   403→   403→    setFormData((prev) => ({ ...prev, [field]: value }))
   404→   404→  }
   405→   405→
   406→   406→  const handleSubmit = async () => {
   407→   407→    // Unified empty-form check — show ONE popup instead of cascading errors
   408→   408→    if (isFormEmpty([formData.customerId, formData.paymentType, formData.amount, formData.date, formData.remarks, formData.billId])) {
   409→   409→      toast(showPleaseFillDataToast())
   410→   410→      return
   411→   411→    }
   412→   412→    // Validation
   413→   413→    if (!formData.customerId) {
   414→   414→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   415→   415→      return
   416→   416→    }
   417→   417→    if (!formData.paymentType) {
   418→   418→      toast({ title: 'Validation Error', description: 'Please select a payment type', variant: 'destructive' })
   419→   419→      return
   420→   420→    }
   421→   421→    if (!formData.amount || Number(formData.amount) <= 0) {
   422→   422→      toast({ title: 'Validation Error', description: 'Amount must be greater than 0', variant: 'destructive' })
   423→   423→      return
   424→   424→    }
   425→   425→    if (!formData.date) {
   426→   426→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   427→   427→      return
   428→   428→    }
   429→   429→
   430→   430→    setFormSubmitting(true)
   431→   431→    try {
   432→   432→      const payload: Record<string, unknown> = {
   433→   433→        customerId: formData.customerId,
   434→   434→        paymentType: formData.paymentType,
   435→   435→        amount: Number(formData.amount),
   436→   436→        date: formData.date,
   437→   437→        remarks: formData.remarks,
   438→   438→      }
   439→   439→
   440→   440→      // Include billId in the payload so the server can link/unlink the
   441→   441→      // payment to a Bill and auto-sync the Bill's paidAmount.
   442→   442→      // Empty string → unlink (server treats "" as null).
   443→   443→      if (editingPayment) {
   444→   444→        // On edit, always send billId (even if empty) so the server can
   445→   445→        // detect "unlink" intent vs "no change".
   446→   446→        payload.billId = formData.billId || null
   447→   447→      } else {
   448→   448→        // On create, only send billId if user picked one.
   449→   449→        if (formData.billId) payload.billId = formData.billId
   450→   450→      }
   451→   451→
   452→   452→      if (editingPayment) {
   453→   453→        await api.updatePayment(editingPayment.id, payload)
   454→   454→        toast({
   455→   455→          title: 'Success',
   456→   456→          description: formData.billId
   457→   457→            ? 'Payment updated and linked to bill. Bill paid amount auto-synced.'
   458→   458→            : 'Payment updated successfully',
   459→   459→        })
   460→   460→      } else {
   461→   461→        await api.createPayment(payload)
   462→   462→        toast({
   463→   463→          title: 'Success',
   464→   464→          description: formData.billId
   465→   465→            ? 'Payment recorded and linked to bill. Bill paid amount auto-synced.'
   466→   466→            : 'Payment recorded successfully',
   467→   467→        })
   468→   468→      }
   469→   469→
   470→   470→      setDialogOpen(false)
   471→   471→      setEditingPayment(null)
   472→   472→      setFormData(emptyForm)
   473→   473→      setCustomerBills([])
   474→   474→      fetchPayments()
   475→   475→      fetchOrders() // Refresh for outstanding calculations
   476→   476→    } catch (err) {
   477→   477→      toast({
   478→   478→        title: 'Error',
   479→   479→        description: err instanceof Error ? err.message : `Failed to ${editingPayment ? 'update' : 'record'} payment`,
   480→   480→        variant: 'destructive',
   481→   481→      })
   482→   482→    } finally {
   483→   483→      setFormSubmitting(false)
   484→   484→    }
   485→   485→  }
   486→   486→
   487→   487→  // ── Delete handler ──────────────────────────────────────────────────────
   488→   488→  const handleDelete = async () => {
   489→   489→    if (!deleteTarget) return
   490→   490→    setDeleting(true)
   491→   491→    try {
   492→   492→      await api.deletePayment(deleteTarget.id)
   493→   493→      toast({ title: 'Success', description: 'Payment deleted successfully' })
   494→   494→      setDeleteTarget(null)
   495→   495→      fetchPayments()
   496→   496→      fetchOrders()
   497→   497→    } catch (err) {
