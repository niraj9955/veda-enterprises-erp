    60→// ── Types ───────────────────────────────────────────────────────────────────
    61→
    62→interface Customer {
    63→  id: string
    64→  name: string
    65→  mobile: string
    66→  gstNumber: string
    67→  address: string
    68→  creditLimit: number
    69→  createdAt: string
    70→  updatedAt: string
    71→}
    72→
    73→interface Payment {
    74→  id: string
    75→  customerId: string
    76→  paymentType: string
    77→  amount: number
    78→  date: string
    79→  remarks: string
    80→  customer: { name: string }
    81→}
    82→
    83→interface CustomerFormData {
    84→  name: string
    85→  mobile: string
    86→  gstNumber: string
    87→  address: string
    88→  creditLimit: number | string
    89→}
    90→
    91→// ── Helpers ─────────────────────────────────────────────────────────────────
    92→
    93→const formatCurrency = (amount: number): string =>
    94→  new Intl.NumberFormat('en-IN', {
    95→    style: 'currency',
    96→    currency: 'INR',
    97→    minimumFractionDigits: 0,
    98→    maximumFractionDigits: 2,
    99→  }).format(amount)
   100→
   101→const formatDate = (dateStr: string): string =>
   102→  new Date(dateStr).toLocaleDateString('en-IN', {
   103→    day: '2-digit',
   104→    month: 'short',
   105→    year: 'numeric',
   106→  })
   107→
   108→const emptyForm: CustomerFormData = {
   109→  name: '',
   110→  mobile: '',
   111→  gstNumber: '',
   112→  address: '',
   113→  creditLimit: '',
   114→}
   115→
   116→// ── Component ───────────────────────────────────────────────────────────────
   117→
   118→export function CustomerModule() {
   119→  // State
   120→  const [customers, setCustomers] = React.useState<Customer[]>([])
   121→  const [totalCount, setTotalCount] = React.useState(0)
   122→  const [loading, setLoading] = React.useState(true)
   123→  const [search, setSearch] = React.useState('')
   124→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   125→
   126→  // Dialog states
   127→  const [formOpen, setFormOpen] = React.useState(false)
   128→  const [editingCustomer, setEditingCustomer] = React.useState<Customer | null>(null)
   129→  const [formData, setFormData] = React.useState<CustomerFormData>(emptyForm)
   130→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   131→
   132→  // Excel import
   133→  const [importOpen, setImportOpen] = React.useState(false)
   134→
   135→  // Delete confirmation
   136→  const [deleteTarget, setDeleteTarget] = React.useState<Customer | null>(null)
   137→  const [deleting, setDeleting] = React.useState(false)
   138→
   139→  // ── Multi-select state ──────────────────────────────────────────────────
   140→  // Tracks which customer rows the user has ticked. The "Delete Selected"
   141→  // button appears in the header whenever this set is non-empty. We use a
   142→  // Set for O(1) toggle/lookup, then convert to an array when sending to
   143→  // the API. Same pattern as production / daily-sell modules.
   144→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set())
   145→  const [bulkDeleteOpen, setBulkDeleteOpen] = React.useState(false)
   146→  const [bulkDeleting, setBulkDeleting] = React.useState(false)
   147→
   148→  const toggleSelect = (id: string) => {
   149→    setSelectedIds((prev) => {
   150→      const next = new Set(prev)
   151→      if (next.has(id)) next.delete(id)
   152→      else next.add(id)
   153→      return next
   154→    })
   155→  }
   156→
   157→  const toggleSelectAll = () => {
   158→    setSelectedIds((prev) => {
   159→      if (prev.size === customers.length && customers.length > 0) {
   160→        return new Set()
   161→      }
   162→      return new Set(customers.map((c) => c.id))
   163→    })
   164→  }
   165→
   166→  const clearSelection = () => setSelectedIds(new Set())
   167→
   168→  const handleBulkDelete = async () => {
   169→    const ids = Array.from(selectedIds)
   170→    if (ids.length === 0) return
   171→    setBulkDeleting(true)
   172→    try {
   173→      const res = await api.bulkDeleteCustomers(ids)
   174→      toast({
   175→        title: 'Success',
   176→        description: `${res.deletedCount} of ${ids.length} customer${res.deletedCount === 1 ? '' : 's'} deleted`,
   177→      })
   178→      setBulkDeleteOpen(false)
   179→      clearSelection()
   180→      fetchCustomers()
   181→    } catch (err) {
   182→      toast({
   183→        title: 'Error',
   184→        description: err instanceof Error ? err.message : 'Failed to delete selected customers',
   185→        variant: 'destructive',
   186→      })
   187→    } finally {
   188→      setBulkDeleting(false)
   189→    }
   190→  }
   191→
   192→  // Ledger
   193→  const [ledgerCustomer, setLedgerCustomer] = React.useState<Customer | null>(null)
   194→  const [ledgerPayments, setLedgerPayments] = React.useState<Payment[]>([])
   195→  const [ledgerLoading, setLedgerLoading] = React.useState(false)
   196→
   197→  // Full customer history PAGE (full-screen, not modal)
   198→  const istoryCustomerId, setHistoryCustomerId] = React.useState<string | null>(null)
   199→
   200→  // ── Debounced search ────────────────────────────────────────────────────
   201→  React.useEffect(() => {
   202→    const timer = setTimeout(() => {
   203→      setDebouncedSearch(search)
   204→    }, 300)
   205→    return () => clearTimeout(timer)
   206→  }, [search])
   207→
   208→  // ── Fetch customers ─────────────────────────────────────────────────────
   209→  const fetchCustomers = React.useCallback(async () => {
   210→    setLoading(true)
   211→    try {
   212→      const res = await api.getCustomers(debouncedSearch || undefined)
   213→      setCustomers(res.customers as Customer[])
   214→      // Track total count separately so the badge can show "X of Y" when
   215→      // pagination is in effect (the API now returns `total` alongside
   216→      // the page slice). If `total` is missing (older deployments), fall
   217→      // back to the slice length.
   218→      const r = res as { total?: number; customers: Customer[] }
   219→      setTotalCount(r.total ?? r.customers.length)
   220→    } catch (err) {
   221→      toast({
   222→        title: 'Error',
   223→        description: err instanceof Error ? err.message : 'Failed to fetch customers',
   224→        variant: 'destructive',
   225→      })
   226→    } finally {
   227→      setLoading(false)
   228→    }
   229→  }, [debouncedSearch])
   230→
   231→  React.useEffect(() => {
   232→    fetchCustomers()
   233→  }, [fetchCustomers])
   234→
   235→  // ── Form handlers ───────────────────────────────────────────────────────
   236→  const openAddDialog = () => {
   237→    setEditingCustomer(null)
   238→    const pending = consumePendingAiResult('customer')
   239→    if (pending) {
   240→      const cl = pending.creditLimit
   241→      setFormData({
   242→        name: String(pending.name || ''),
   243→        mobile: String(pending.mobile || pending.contactNumber || ''),
   244→        gstNumber: String(pending.gstNumber || ''),
   245→        address: String(pending.address || ''),
   246→        creditLimit: cl != null && cl !== '' ? String(cl) : '',
   247→      })
   248→      toast({ title: 'AI auto-fill applied', description: 'Edit & verify before saving.' })
   249→    } else {
   250→      setFormData(emptyForm)
   251→    }
   252→    setFormOpen(true)
   253→  }
   254→
   255→  const openEditDialog = (customer: Customer) => {
   256→    setEditingCustomer(customer)
   257→    setFormData({
   258→      name: customer.name,
   259→      mobile: customer.mobile,
   260→      gstNumber: customer.gstNumber,
   261→      address: customer.address,
   262→      creditLimit: customer.creditLimit,
   263→    })
   264→    setFormOpen(true)
   265→  }
   266→
   267→  const handleFormChange = (field: keyof CustomerFormData, value: string) => {
   268→    setFormData((prev) => ({ ...prev, [field]: value }))
   269→  }
   270→
   271→  const handleSubmit = async () => {
   272→    // Unified empty-form check — show ONE popup instead of cascading errors
   273→    if (isFormEmpty([formData.name, formData.mobile, formData.gstNumber, formData.address, formData.creditLimit])) {
   274→      toast(showPleaseFillDataToast())
   275→      return
   276→    }
   277→    if (!formData.name.trim()) {
   278→      toast({ title: 'Validation Error', description: 'Customer name is required', variant: 'destructive' })
   279→      return
   280→    }
   281→    if (!formData.mobile.trim()) {
   282→      toast({ title: 'Validation Error', description: 'Mobile number is required', variant: 'destructive' })
   283→      return
   284→    }
   285→
   286→    setFormSubmitting(true)
   287→    try {
   288→      const payload = {
   289→        name: formData.name.trim(),
   290→        mobile: formData.mobile.trim(),
   291→        gstNumber: formData.gstNumber.trim(),
   292→        address: formData.address.trim(),
   293→        creditLimit: Number(formData.creditLimit) || 0,
   294→      }
   295→
   296→      if (editingCustomer) {
   297→        await api.updateCustomer(editingCustomer.id, payload)
   298→        toast({ title: 'Success', description: 'Customer updated successfully' })
   299→      } else {
   300→        await api.createCustomer(payload)
   301→        toast({ title: 'Success', description: 'Customer created successfully' })
   302→      }
   303→
   304→      setFormOpen(false)
   305→      setFormData(emptyForm)
   306→      setEditingCustomer(null)
   307→      fetchCustomers()
   308→    } catch (err) {
   309→      toast({
   310→        title: 'Error',
   311→        description: err instanceof Error ? err.message : 'Failed to save customer',
   312→        variant: 'destructive',
   313→      })
   314→    } finally {
   315→      setFormSubmitting(false)
   316→    }
   317→  }
   318→
   319→  // ── Delete handler ──────────────────────────────────────────────────────
   320→  const handleDelete = async () => {
   321→    if (!deleteTarget) return
   322→    setDeleting(true)
   323→    try {
   324→      await api.deleteCustomer(deleteTarget.id)
   325→      toast({ title: 'Success', description: 'Customer deleted successfully' })
   326→      setDeleteTarget(null)
   327→      fetchCustomers()
   328→    } catch (err) {
   329→      toast({
   330→        title: 'Error',
   331→        description: err instanceof Error ? err.message : 'Failed to delete customer',
   332→        variant: 'destructive',
   333→      })
   334→    } finally {
   335→      setDeleting(false)
   336→    }
   337→  }
   338→
   339→  // ── Ledger handler ──────────────────────────────────────────────────────
   340→  const openLedger = async (customer: Customer) => {
   341→    setLedgerCustomer(customer)
   342→    setLedgerLoading(true)
   343→    try {
   344→      const res = await api.getPayments()
   345→      const allPayments = res.payments as Payment[]
   346→      const filtered = allPayments.filter((p) => p.customerId === customer.id)
   347→      // Sort by date descending
   348→      filtered.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   349→      setLedgerPayments(filtered)
   350→    } catch (err) {
   351→      toast({
   352→        title: 'Error',
   353→        description: err instanceof Error ? err.message : 'Failed to fetch payment history',
   354→        variant: 'destructive',
   355→      })
   356→      setLedgerPayments([])
   357→    } finally {
   358→      setLedgerLoading(false)
   359→    }
   360→  }
   361→
   362→  // ── Render: Loading skeletons ───────────────────────────────────────────
   363→  const renderSkeletons = () =>
   364→    Array.from({ length: 5 }).map((_, i) => (
   365→      <TableRow key={i}>
   366→        <TableCell className="w-10"><Skeleton className="h-4 w-4" /></TableCell>
   367→        <TableCell><Skeleton className="h-4 w-32" /></TableCell>
   368→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   369→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   370→        <TableCell><Skeleton className="h-4 w-40" /></TableCell>
   371→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   372→        <TableCell><Skeleton className="h-8 w-28" /></TableCell>
   373→      </TableRow>
   374→    ))
   375→
   376→  // ── Render: Form dialog ─────────────────────────────────────────────────
   377→  const renderFormDialog = () => (
   378→    <Dialog open={formOpen} onOpenChange={setFormOpen}>
   379→      <DialogContent className="sm:max-w-md">
   380→        <DialogHeader>
   381→          <DialogTitle>
   382→            {editingCustomer ? 'Edit Customer' : 'Add Customer'}
   383→          </DialogTitle>
   384→          <DialogDescription>
   385→            {editingCustomer
   386→              ? 'Update the customer details below.'
   387→              : 'Fill in the details to create a new customer.'}
   388→          </DialogDescription>
   389→        </DialogHeader>
   390→        {!editingCustomer && (
   391→          <div className="flex justify-end">
   392→            <AiFillButton
   393→              module="customer"
   394→              onApply={(fields) => setFormData((prev) => ({
   395→                name: fields.name != null ? String(fields.name) : prev.name,
   396→                mobile: fields.mobile != null ? String(fields.mobile) : (fields.contactNumber != null ? String(fields.contactNumber) : prev.mobile),
   397→                gstNumber: fields.gstNumber != null ? String(fields.gstNumber) : prev.gstNumber,
   398→                address: fields.address != null ? String(fields.address) : prev.address,
   399→                creditLimit: fields.creditLimit != null && fields.creditLimit !== '' ? String(fields.creditLimit) : prev.creditLimit,
   400→              }))}
   401→            />
   402→          </div>
   403→        )}
   404→
   405→        <div className="grid gap-4 py-2">
   406→          {/* Customer Name */}
   407→          <div className="grid gap-2">
   408→            <Label htmlFor="cust-name">
   409→              Customer Name <span className="text-destructive">*</span>
   410→            </Label>
   411→            <div className="relative">
   412→              <Input
   413→                id="cust-name"
   414→                placeholder="Enter customer name"
   415→                value={formData.name}
   416→                onChange={(e) => handleFormChange('name', e.target.value)}
   417→                className="pr-9"
   418→              />
   419→              <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   420→                <FieldVoiceInput
   421→                  fieldLabel="customer name"
   422→                  onChange={(text) => handleFormChange('name', text)}
   423→                />
   424→              </div>
   425→            </div>
   426→          </div>
   427→
   428→          {/* Mobile Number */}
   429→          <div className="grid gap-2">
   430→            <Label htmlFor="cust-mobile">
   431→              Mobile Number <span className="text-destructive">*</span>
   432→            </Label>
   433→            <div className="relative">
   434→              <Input
   435→                id="cust-mobile"
   436→                placeholder="Enter mobile number"
   437→                value={formData.mobile}
   438→                onChange={(e) => handleFormChange('mobile', e.target.value)}
   439→                className="pr-9"
   440→              />
   441→              <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   442→                <FieldVoiceInput
   443→                  fieldLabel="mobile number"
   444→                  onChange={(text) => handleFormChange('mobile', text.replace(/[^0-9+\-\s]/g, '').trim())}
   445→                />
   446→              </div>
   447→            </div>
   448→          </div>
   449→
   450→          {/* GST Number */}
   451→          <div className="grid gap-2">
   452→            <Label htmlFor="cust-gst">GST Number</Label>
   453→            <div className="relative">
   454→              <Input
   455→                id="cust-gst"
   456→                placeholder="Enter GST number (optional)"
   457→                value={formData.gstNumber}
   458→                onChange={(e) => handleFormChange('gstNumber', e.target.value)}
   459→                className="pr-9"
   460→              />
   461→              <div className="absolute right-1.5 top-1/2 -translate-y-1/2">
   462→                <FieldVoiceInput
   463→                  fieldLabel="GST number"
   464→                  onChange={(text) => handleFormChange('gstNumber', text.toUpperCase().replace(/[^A-Z0-9]/g, ''))}
   465→                />
   466→              </div>
   467→            </div>
   468→          </div>
   469→
   470→          {/* Address */}
   471→          <div className="grid gap-2">
   472→            <Label htmlFor="cust-address">Address</Label>
   473→            <div className="relative">
   474→              <Textarea
   475→                id="cust-address"
   476→                placeholder="Enter address"
   477→                value={formData.address}
   478→                onChange={(e) => handleFormChange('address', e.target.value)}
   479→                className="min-h-[80px] pr-9"
   480→              />
   481→              <div className="absolute right-1.5 top-2">
   482→                <FieldVoiceInput
   483→                  fieldLabel="address"
   484→                  onChange={(text) => handleFormChange('address', text)}
   485→                />
   486→              </div>
   487→            </div>
   488→          </div>
   489→
   490→          {/* Credit Limit */}
   491→          <div className="grid gap-2">
   492→            <Label htmlFor="cust-credit">Credit Limit (₹)</Label>
   493→            <div className="relative">
   494→              <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   495→              <Input
   496→                id="cust-credit"
   497→                type="number"
   498→                placeholder="0"
   499→                className="pl-9"
   500→                value={formData.creditLimit}
   501→                onChange={(e) => handleFormChange('creditLimit', e.target.value)}
   502→              />
   503→            </div>
   504→          </div>
   505→        </div>
   506→
   507→        <DialogFooter>
   508→          <Button
   509→            variant="outline"
   510→            onClick={() => setFormOpen(false)}
   511→            disabled={formSubmitting}
   512→          >
   513→            Cancel
   514→          </Button>
   515→          <Button
   516→            onClick={handleSubmit}
   517→            disabled={formSubmitting}
   518→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   519→          >
   520→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   521→            {editingCustomer ? 'Update Customer' : 'Create Customer'}
   522→          </Button>
   523→        </DialogFooter>
   524→      </DialogContent>
   525→    </Dialog>
   526→  )
   527→
   528→  // ── Render: Delete confirmation ─────────────────────────────────────────
   529→  const renderDeleteDialog = () => (
   530→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   531→      <AlertDialogContent>
   532→        <AlertDialogHeader>
   533→          <AlertDialogTitle>Delete Customer</AlertDialogTitle>
   534→          <AlertDialogDescription>
   535→            Are you sure you want to delete <strong>{deleteTarget?.name}</strong>? This action
   536→            cannot be undone.
   537→          </AlertDialogDescription>
   538→        </AlertDialogHeader>
   539→        <AlertDialogFooter>
   540→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   541→          <AlertDialogAction
   542→            onClick={handleDelete}
   543→            disabled={deleting}
   544→            className="bg-destructive text-white hover:bg-destructive/90"
   545→          >
   546→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   547→            Delete
   548→          </AlertDialogAction>
   549→        </AlertDialogFooter>
   550→      </AlertDialogContent>
   551→    </AlertDialog>
   552→  )
   553→
   554→  // ── Render: Bulk-delete confirmation ────────────────────────────────────
   555→  const renderBulkDeleteDialog = () => (
   556→    <AlertDialog
   557→      open={bulkDeleteOpen}
   558→      onOpenChange={(open) => {
   559→        if (!open && !bulkDeleting) setBulkDeleteOpen(false)
   560→      }}
   561→    >
   562→      <AlertDialogContent>
   563→        <AlertDialogHeader>
   564→          <AlertDialogTitle className="text-destructive">
   565→            Delete {selectedIds.size} Selected Customer{selectedIds.size === 1 ? '' : 's'}?
   566→          </AlertDialogTitle>
   567→          <AlertDialogDescription className="space-y-2 text-sm">
   568→            <span className="block">
   569→              You are about to permanently delete{' '}
   570→              <strong className="text-destructive">
   571→                {selectedIds.size} customer{selectedIds.size === 1 ? '' : 's'}
   572→              </strong>
   573→              . This action <strong>cannot be undone</strong>.
   574→            </span>
   575→            <span className="block text-muted-foreground">
   576→              Linked orders, payments, and dispatches will be kept for audit but
   577→              disassociated from the deleted customers (their customer field will
   578→              be cleared).
   579→            </span>
   580→          </AlertDialogDescription>
   581→        </AlertDialogHeader>
   582→        <AlertDialogFooter>
   583→          <AlertDialogCancel disabled={bulkDeleting}>Cancel</AlertDialogCancel>
   584→          <AlertDialogAction
   585→            onClick={handleBulkDelete}
   586→            disabled={bulkDeleting || selectedIds.size === 0}
   587→            className="bg-destructive text-white hover:bg-destructive/90"
   588→          >
   589→            {bulkDeleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   590→            Delete {selectedIds.size} Customer{selectedIds.size === 1 ? '' : 's'}
   591→          </AlertDialogAction>
   592→        </AlertDialogFooter>
   593→      </AlertDialogContent>
