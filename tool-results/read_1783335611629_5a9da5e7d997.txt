    13→// ─── Server-side date normalization ─────────────────────────────────────────
    14→//
    15→// The client (excel-import.tsx) already normalizes dates to YYYY-MM-DD, but
    16→// we re-normalize here as defense-in-depth so a direct API call (or a future
    17→// caller that bypasses the Excel wizard) is still safe.
    18→//
    19→// Supported input formats (any separator among - / . or space):
    20→//   • YYYY-MM-DD          (already canonical)
    21→//   • DD-MM-YYYY  DD/MM/YYYY  DD.MM.YYYY   (day-first, Indian format)
    22→//   • DD-MM-YY    DD/MM/YY    DD.MM.YY     (short year, prefixed with 20)
    23→//   • MM/DD/YYYY  (only when first number > 12, e.g. 13/01/2024 -> Jan 13)
    24→//   • Datetime strings like "2024-01-15 10:30:00" (time part stripped)
    25→//   • Excel serial numbers (e.g. 46178)
    26→//   • Fallback: native Date parsing, then the raw string unchanged.
    27→function normalizeDate(value: unknown): string {
    28→  if (value == null) return ''
    29→  if (value instanceof Date && !isNaN(value.getTime())) {
    30→    return value.toISOString().split('T')[0]
    31→  }
    32→  if (typeof value === 'number' && Number.isFinite(value) && value > 59 && value < 60000) {
    33→    // Excel serial date: days since 1899-12-30
    34→    const ms = Math.round((value - 25569) * 86400 * 1000)
    35→    const d = new Date(ms)
    36→    if (!isNaN(d.getTime())) return d.toISOString().split('T')[0]
    37→  }
    38→  const raw = String(value ?? '').trim()
    39→  if (raw === '') return ''
    40→
    41→  // Strip time portion
    42→  const trimmed = raw
    43→    .replace(/[Tt]\s*\d{1,2}:\d{2}.*$/, '')
    44→    .replace(/\s+\d{1,2}:\d{2}.*$/, '')
    45→    .trim()
    46→  if (trimmed === '') return raw
    47→
    48→  // YYYY-MM-DD
    49→  if (/^\d{4}-\d{1,2}-\d{1,2}$/.test(trimmed)) {
    50→    const [y, m, d] = trimmed.split('-')
    51→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    52→  }
    53→  // YYYY/MM/DD or YYYY.MM.DD
    54→  const ymdMatch = trimmed.match(/^(\d{4})[/.\s](\d{1,2})[/.\s](\d{1,2})$/)
    55→  if (ymdMatch) {
    56→    const [, y, m, d] = ymdMatch
    57→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    58→  }
    59→  // DD-MM-YYYY / DD/MM/YYYY / DD.MM.YYYY (with US fallback if second > 12)
    60→  const dmyMatch = trimmed.match(/^(\d{1,2})[/.\s-](\d{1,2})[/.\s-](\d{4})$/)
    61→  if (dmyMatch) {
    62→    const [, a, b, y] = dmyMatch
    63→    const d = Number(b) > 12 && Number(a) <= 12 ? b : a
    64→    const m = Number(b) > 12 && Number(a) <= 12 ? a : b
    65→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    66→  }
    67→  // DD-MM-YY / DD/MM/YY / DD.MM.YY
    68→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/.\s-](\d{1,2})[/.\s-](\d{2})$/)
    69→  if (dmyShortMatch) {
    70→    const [, a, b, y] = dmyShortMatch
    71→    const d = Number(b) > 12 && Number(a) <= 12 ? b : a
    72→    const m = Number(b) > 12 && Number(a) <= 12 ? a : b
    73→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
    74→  }
    75→  // Native Date fallback
    76→  try {
    77→    const date = new Date(trimmed)
    78→    if (!isNaN(date.getTime())) return date.toISOString().split('T')[0]
    79→  } catch {}
    80→  return raw
    81→}

    83→// Normalize every date-like field on a row before validation/insert.
    84→// This keeps the per-module switch statement below simple.
    85→function normalizeRowDates(row: Record<string, unknown>): void {
    86→  if ('date' in row) row.date = normalizeDate(row.date)
    87→  if ('deliveryDate' in row) row.deliveryDate = normalizeDate(row.deliveryDate)
    88→}

    90→// ─── Duplicate detection ────────────────────────────────────────────────────
    91→//
    92→// Per user request: "jo duplicate ho use skip kre or uska mess de but jo
    93→// duplicate nhi h usko import kre record me" — when an Excel row matches an
    94→// existing record in the DB (by a natural key), SKIP it with a clear message
    95→// but still IMPORT every non-duplicate row.
    96→//
    97→// To keep this fast, for each module we run ONE batched `find()` that
    98→// returns every existing row whose date falls within the dates seen in
    99→// the import payload, then build a Set of natural-key strings. Per-row
   100→// lookup then becomes an O(1) Set.has() check instead of a findOne().
   101→//
   102→// APPEND-ONLY semantics are preserved — we never wipe existing rows.
   103→
   104→// Build a stable natural-key string for a row, per module.
   105→function rowKey(module: string, row: Record<string, unknown>): string {
   106→  const s = (v: unknown) => String(v ?? '').trim().toLowerCase()
   107→  switch (module) {
   108→    case 'customers':
   109→      return s(row.mobile)
   110→    case 'production':
   111→      // One production entry per date — same date = duplicate (matches stock semantics)
   112→      return s(row.date)
   113→    case 'stock':
   114→      // One stock entry per date — same date = duplicate
   115→      return s(row.date)
   116→    case 'dailySell':
   117→      return `${s(row.date)}|${s(row.customerName)}|${s(row.amount)}`
   118→    case 'customerPayment':
   119→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
   120→    case 'labourPayment':
   121→      return `${s(row.date)}|${s(row.name)}|${s(row.amount)}`
   122→    case 'tractorPayment':
   123→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantityTon)}|${s(row.rate)}`
   124→    case 'dustPurchase':
   125→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.quantity)}|${s(row.rate)}`
   126→    case 'cementPurchase':
   127→      return `${s(row.date)}|${s(row.vendorName)}|${s(row.itemName)}|${s(row.quantity)}|${s(row.rate)}`
   128→    case 'hardner':
   129→      // Only date+amount available — same date AND same amount = duplicate
   130→      return `${s(row.date)}|${s(row.amount)}`
   131→    case 'electricity':
   132→      return `${s(row.date)}|${s(row.name)}|${s(row.work)}|${s(row.amount)}`
   133→    case 'factoryStuff':
   134→      return `${s(row.date)}|${s(row.itemName)}|${s(row.amount)}`
   135→    case 'orders':
   136→      return `${s(row.customerId)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.rate)}|${s(row.deliveryDate || row.date)}`
   137→    case 'dispatch':
   138→      return `${s(row.customerId)}|${s(row.truckNumber)}|${s(row.brickType)}|${s(row.quantity)}|${s(row.date)}`
   139→    case 'payments':
   140→      return `${s(row.customerId)}|${s(row.paymentType)}|${s(row.amount)}|${s(row.date)}`
   141→    case 'expenses':
   142→      return `${s(row.category)}|${s(row.amount)}|${s(row.date)}|${s(row.description)}`
   143→    default:
   144→      return ''
   145→  }
   146→}

   148→// Map a DB record back into the same natural-key shape as a row.
   149→// IMPORTANT: this MUST mirror rowKey() exactly — if rowKey uses just `date`
   150→// for production, dbKey must also use just `date` for production. A mismatch
   151→// here means existingKeys will contain keys like "2026-06-21||" while the
   152→// incoming row key is "2026-06-21", so the duplicate check NEVER matches and
   153→// duplicates get imported. This was the root cause of duplicate production
   154→// entries being imported.

   196→// Pre-fetch existing DB records that could collide with any row in `data`.
   197→// Returns a Set of natural-key strings.
   198→async function buildExistingKeys(module: string, data: Record<string, unknown>[]): Promise<Set<string>> {
   199→  const keys = new Set<string>()
   200→  if (data.length === 0) return keys
   201→
   202→  // Collect unique dates and customerIds from the import payload so we can
   203→  // narrow the DB query. This keeps the result set small.
   204→  const dates = new Set<string>()
   205→  const mobiles = new Set<string>()
   206→  const customerIds = new Set<string>()
   207→  for (const row of data) {
   208→    const d = String(row.date ?? '').split('T')[0]
   209→    if (d) dates.add(d)
   210→    const dd = String(row.deliveryDate ?? '').split('T')[0]
   211→    if (dd) dates.add(dd)
   212→    const m = String(row.mobile ?? '').trim()
   213→    if (m) mobiles.add(m)
   214→    const cid = String(row.customerId ?? '').trim()
   215→    if (cid) customerIds.add(cid)
   216→  }
   217→
   218→  let docs: Record<string, unknown>[] = []
   219→
   220→  switch (module) {
   221→    case 'customers':
   222→      docs = mobiles.size > 0
   223→        ? await Customer.find({ mobile: { $in: Array.from(mobiles) } }).lean() as any
   224→        : []
   225→      break
   226→    case 'production':
   227→      docs = dates.size > 0
   228→        ? await Production.find({ date: { $in: Array.from(dates) } }).lean() as any
   229→        : []
   230→      break
   231→    case 'stock':
   232→      docs = dates.size > 0
   233→        ? await Stock.find({ date: { $in: Array.from(dates) } }).lean() as any
   234→        : []
   235→      break
   236→    case 'dailySell':
   237→      docs = dates.size > 0
   238→        ? await DailySell.find({ date: { $in: Array.from(dates) } }).lean() as any
   239→        : []
   240→      break
   241→    case 'customerPayment':
   242→      docs = dates.size > 0
   243→        ? await CustomerPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   244→        : []
   245→      break
   246→    case 'labourPayment':
   247→      docs = dates.size > 0
   248→        ? await LabourPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   249→        : []
   250→      break
   251→    case 'tractorPayment':
   252→      docs = dates.size > 0
   253→        ? await TractorPayment.find({ date: { $in: Array.from(dates) } }).lean() as any
   254→        : []
   255→      break
   256→    case 'dustPurchase':
   257→      docs = dates.size > 0
   258→        ? await DustPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   259→        : []
   260→      break
   261→    case 'cementPurchase':
   262→      docs = dates.size > 0
   263→        ? await CementPurchase.find({ date: { $in: Array.from(dates) } }).lean() as any
   264→        : []
   265→      break
   266→    case 'hardner':
   267→      docs = dates.size > 0
   268→        ? await Hardner.find({ date: { $in: Array.from(dates) } }).lean() as any
   269→        : []
   270→      break
   271→    case 'electricity':
   272→      docs = dates.size > 0
   273→        ? await Electricity.find({ date: { $in: Array.from(dates) } }).lean() as any
   274→        : []
   275→      break
   276→    case 'factoryStuff':
   277→      docs = dates.size > 0
   278→        ? await FactoryStuff.find({ date: { $in: Array.from(dates) } }).lean() as any
   279→        : []
   280→      break
   281→    case 'orders':
   282→      docs = customerIds.size > 0
   283→        ? await Order.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   284→        : []
   285→      break
   286→    case 'dispatch':
   287→      docs = customerIds.size > 0
   288→        ? await Dispatch.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   289→        : []
   290→      break
   291→    case 'payments':
   292→      docs = customerIds.size > 0
   293→        ? await Payment.find({ customerId: { $in: Array.from(customerIds) } }).lean() as any
   294→        : []
   295→      break
   296→    case 'expenses':
   297→      // Expenses dedupe by category+amount+date — narrow by date
   298→      docs = dates.size > 0
   299→        ? await Expense.find({ date: { $in: Array.from(dates) } }).lean() as any
   300→        : []
   301→      break
   302→  }
   303→
   304→  for (const d of docs) {
   305→    const k = dbKey(module, d as Record<string, unknown>)
   306→    if (k) keys.add(k)
   307→  }
   308→  return keys
   309→}

   311→export async function POST(request: Request) {
   312→  try {
   313→    await connectDB()
   314→    const body = await request.json()
   315→    const { module, data } = body
   316→
   317→    if (!module || !data || !Array.isArray(data)) {
   318→      return NextResponse.json({ error: 'Module and data array are required' }, { status: 400 })
   319→    }
   320→
   321→    // ── Pre-fetch existing keys (1 DB query for the whole batch) ──────────
   322→    const existingKeys = await buildExistingKeys(module, data as Record<string, unknown>[])
   323→    // Track keys we've already seen WITHIN this batch so a duplicate Excel
   324→    // row doesn't get inserted twice in the same import run.
   325→    const seenInBatch = new Set<string>()
   326→
   327→    let imported = 0
   328→    let skipped = 0
   329→    let duplicatesSkipped = 0
   330→    const errors: string[] = []
   331→    const skippedReasons: string[] = []
   332→
   333→    // ── Bulk insert optimization ──────────────────────────────────────────
   334→    // Instead of calling `await Model.create(doc)` for each row (N DB round
   335→    // trips), we collect ALL valid documents into a `toInsert` array and do a
   336→    // single `Model.insertMany(toInsert)` call after the loop. This makes the
   337→    // import dramatically faster — 48 rows that previously took 8-10 seconds
   338→    // (timing out on Vercel Hobby's 10s limit) now take under 1 second.
   339→    //
   340→    // `rowIndexByDoc` maps each document back to its original Excel row index
   341→    // so we can report per-row errors if insertMany throws.
   342→    const toInsert: any[] = []
   343→    const rowIndexByDoc: number[] = []
   344→
   345→    for (let i = 0; i < data.length; i++) {
   346→      try {
   347→        const row = data[i]
   348→
   349→        // Normalize date fields BEFORE duplicate check / validation / insert
   350→        // so DD-MM-YYYY, DD/MM/YYYY, DD.MM.YYYY, datetime strings, and Excel
   351→        // serial numbers are all converted to canonical YYYY-MM-DD.
   352→        normalizeRowDates(row as Record<string, unknown>)
   353→
   354→        // ── Duplicate check (against DB AND within this batch) ──────────
   355→        // Duplicates are pushed to the `errors` array (not just skippedReasons)
   356→        // so they appear in the RED error popup with a clear "Duplicate data
   357→        // found" message, exactly as the user requested. The row is still
   358→        // counted in duplicatesSkipped for the summary card.
   359→        const key = rowKey(module, row as Record<string, unknown>)
   360→        if (key) {
   361→          if (existingKeys.has(key)) {
   362→            skipped++
   363→            duplicatesSkipped++
   364→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   365→            errors.push(`Row ${i + 1}: Duplicate data found — ${label} already exists in records`)
   366→            continue
   367→          }
   368→          if (seenInBatch.has(key)) {
   369→            skipped++
   370→            duplicatesSkipped++
   371→            const label = duplicateRowLabel(module, row as Record<string, unknown>)
   372→            errors.push(`Row ${i + 1}: Duplicate data found — ${label} appears more than once in this Excel file`)
   373→            continue
   374→          }
   375→          seenInBatch.add(key)
   376→        }
   377→
   378→        switch (module) {
   379→          // ─── Customers ──────────────────────────────────────────────
   380→          case 'customers': {
   381→            if (!row.name || !row.mobile) {
   382→              errors.push(`Row ${i + 1}: Name and mobile are required`)
   383→              skipped++
   384→              continue
   385→            }
   386→            toInsert.push({
   387→              name: String(row.name).trim(),
   388→              mobile: String(row.mobile).trim(),
   389→              gstNumber: row.gstNumber || '',
   390→              address: row.address || '',
   391→              creditLimit: Number(row.creditLimit) || 0,
   392→            })
   393→            rowIndexByDoc.push(i)
   394→            break
   395→          }
   396→
   397→          // ─── Production ─────────────────────────────────────────────
   398→          case 'production': {
   399→            if (!row.date) {
   400→              errors.push(`Row ${i + 1}: Date is required`)
   401→              skipped++
   402→              continue
   403→            }
   404→            toInsert.push({
   405→              date: String(row.date),
   406→              customerId: row.customerId || null,
   407→              cement: Number(row.cement) || 0,
   408→              zigZagGrey80: Number(row.zigZagGrey80) || 0,
   409→              zigZagRed80: Number(row.zigZagRed80) || 0,
   410→              zigZagYellow80: Number(row.zigZagYellow80) || 0,
   411→              zigZagGrey60: Number(row.zigZagGrey60) || 0,
   412→              zigZagRed60: Number(row.zigZagRed60) || 0,
   413→              zigZagYellow60: Number(row.zigZagYellow60) || 0,
   414→              curveStone: Number(row.curveStone) || 0,
   415→              chequreTile: Number(row.chequreTile) || 0,
   416→              dumbleGrey80: Number(row.dumbleGrey80) || 0,
   417→              dumbleRed80: Number(row.dumbleRed80) || 0,
   418→              dumbleYellow80: Number(row.dumbleYellow80) || 0,
   419→              transportationCharge: Number(row.transportationCharge) || 0,
   420→              remarks: String(row.remarks || ''),
   421→            })
   422→            rowIndexByDoc.push(i)
   423→            break
   424→          }
   425→
   426→          // ─── Stock ─────────────────────────────────────────────────
   427→          case 'stock': {
   428→            if (!row.date) {
   429→              errors.push(`Row ${i + 1}: Date is required`)
   430→              skipped++
   431→              continue
   432→            }
   433→            toInsert.push({
   434→              date: String(row.date),
   435→              cement: Number(row.cement) || 0,
   436→              zigZagGrey80: Number(row.zigZagGrey80) || 0,
   437→              zigZagRed80: Number(row.zigZagRed80) || 0,
   438→              zigZagYellow80: Number(row.zigZagYellow80) || 0,
   439→              zigZagGrey60: Number(row.zigZagGrey60) || 0,
   440→              zigZagRed60: Number(row.zigZagRed60) || 0,
   441→              zigZagYellow60: Number(row.zigZagYellow60) || 0,
   442→              chequreTile: Number(row.chequreTile) || 0,
   443→              curveStone: Number(row.curveStone) || 0,
   444→              dumbleGrey80: Number(row.dumbleGrey80) || 0,
   445→              dumbleRed80: Number(row.dumbleRed80) || 0,
   446→              dumbleYellow80: Number(row.dumbleYellow80) || 0,
   447→            })
   448→            rowIndexByDoc.push(i)
   449→            break
   450→          }
   451→
   452→          // ─── Daily Sell ────────────────────────────────────────────
   453→          case 'dailySell': {
   454→            if (!row.date || !row.customerName || !row.amount) {
   455→              errors.push(`Row ${i + 1}: Date, customer name, and amount are required`)
   456→              skipped++
   457→              continue
   458→            }
   459→            toInsert.push({
   460→              date: String(row.date),
   461→              customerName: String(row.customerName),
   462→              address: String(row.address || ''),
   463→              contactNumber: String(row.contactNumber || row.mobile || ''),
   464→              product: String(row.product || ''),
   465→              amount: Number(row.amount),
   466→              remarks: String(row.remarks || ''),
   467→            })
   468→            rowIndexByDoc.push(i)
   469→            break
   470→          }
   471→
   472→          // ─── Customer Payment ──────────────────────────────────────
   473→          case 'customerPayment': {
   474→            if (!row.date || !row.name || !row.amount) {
   475→              errors.push(`Row ${i + 1}: Date, name, and amount are required`)
   476→              skipped++
   477→              continue
   478→            }
   479→            toInsert.push({
   480→              date: String(row.date),
   481→              name: String(row.name),
   482→              address: String(row.address || ''),
   483→              amount: Number(row.amount),
   484→              remarks: String(row.remarks || ''),
   485→            })
   486→            rowIndexByDoc.push(i)
   487→            break
   488→          }
   489→
   490→          // ─── Labour Payment ────────────────────────────────────────
   491→          case 'labourPayment': {
   492→            if (!row.date || !row.name || !row.amount) {
   493→              errors.push(`Row ${i + 1}: Date, name, and amount are required`)
   494→              skipped++
   495→              continue
   496→            }
   497→            toInsert.push({
   498→              date: String(row.date),
   499→              name: String(row.name),
   500→              address: String(row.address || ''),
   501→              amount: Number(row.amount),
   502→              remarks: String(row.remarks || ''),
   503→            })
   504→            rowIndexByDoc.push(i)
   505→            break
   506→          }
   507→
   508→          // ─── Tractor Payment ───────────────────────────────────────
   509→          case 'tractorPayment': {
   510→            if (!row.date || !row.vendorName || !row.quantityTon || !row.rate) {
   511→              errors.push(`Row ${i + 1}: Date, vendor, quantity, and rate are required`)
   512→              skipped++
   513→              continue
   514→            }
   515→            const qty = Number(row.quantityTon)
   516→            const rate = Number(row.rate)
   517→            const totalAmount = Number(row.totalAmount) || qty * rate
   518→            const paidAmount = Number(row.paidAmount) || 0
   519→            toInsert.push({
   520→              date: String(row.date),
   521→              vendorName: String(row.vendorName),
   522→              quantityTon: qty,
   523→              rate,
   524→              totalAmount,
   525→              paidAmount,
   526→              remainingAmount: Number(row.remainingAmount) || (totalAmount - paidAmount),
   527→              remarks: String(row.remarks || ''),
   528→            })
   529→            rowIndexByDoc.push(i)
   530→            break
   531→          }
   532→
   533→          // ─── Dust Purchase ─────────────────────────────────────────
   534→          case 'dustPurchase': {
   535→            if (!row.date || !row.vendorName || !row.quantity || !row.rate) {
   536→              errors.push(`Row ${i + 1}: Date, vendor, quantity, and rate are required`)
   537→              skipped++
   538→              continue
   539→            }
   540→            const qty = Number(row.quantity)
   541→            const rate = Number(row.rate)
   542→            const totalAmount = Number(row.totalAmount) || qty * rate
   543→            const paidAmount = Number(row.paidAmount) || 0
   544→            toInsert.push({
   545→              date: String(row.date),
   546→              vendorName: String(row.vendorName),
   547→              cementName: String(row.cementName || ''),
   548→              quantity: qty,
   549→              rate,
   550→              totalAmount,
   551→              paidAmount,
   552→              transportationCharge: Number(row.transportationCharge) || 0,
   553→              gst: Number(row.gst) || 0,
   554→              remarks: String(row.remarks || ''),
   555→            })
   556→            rowIndexByDoc.push(i)
   557→            break
   558→          }
   559→
   560→          // ─── Cement Purchase ───────────────────────────────────────
   561→          case 'cementPurchase': {
   562→            if (!row.date || !row.vendorName || !row.quantity || !row.rate) {
   563→              errors.push(`Row ${i + 1}: Date, vendor, quantity, and rate are required`)
   564→              skipped++
   565→              continue
   566→            }
   567→            const qty = Number(row.quantity)
   568→            const rate = Number(row.rate)
   569→            const totalAmount = Number(row.totalAmount) || qty * rate
   570→            const paidAmount = Number(row.paidAmount) || 0
   571→            toInsert.push({
   572→              date: String(row.date),
   573→              vendorName: String(row.vendorName),
   574→              itemName: String(row.itemName || ''),
   575→              quantity: qty,
   576→              rate,
   577→              totalAmount,
   578→              paidAmount,
   579→              transportationCharge: Number(row.transportationCharge) || 0,
   580→              gst: Number(row.gst) || 0,
   581→              remarks: String(row.remarks || ''),
   582→            })
   583→            rowIndexByDoc.push(i)
   584→            break
   585→          }
   586→
   587→          // ─── Hardner ───────────────────────────────────────────────
   588→          case 'hardner': {
   589→            if (!row.date || !row.amount) {
   590→              errors.push(`Row ${i + 1}: Date and amount are required`)
   591→              skipped++
   592→              continue
   593→            }
   594→            toInsert.push({
   595→              date: String(row.date),
   596→              amount: Number(row.amount),
   597→            })
   598→            rowIndexByDoc.push(i)
   599→            break
   600→          }
   601→
   602→          // ─── Electricity ───────────────────────────────────────────
   603→          case 'electricity': {
   604→            if (!row.date || !row.amount) {
   605→              errors.push(`Row ${i + 1}: Date and amount are required`)
   606→              skipped++
   607→              continue
   608→            }
   609→            toInsert.push({
   610→              date: String(row.date),
   611→              name: String(row.name || ''),
   612→              work: String(row.work || ''),
   613→              amount: Number(row.amount),
   614→              remarks: String(row.remarks || ''),
   615→            })
   616→            rowIndexByDoc.push(i)
   617→            break
   618→          }
   619→
   620→          // ─── Factory Stuff ─────────────────────────────────────────
   621→          case 'factoryStuff': {
   622→            if (!row.date || !row.itemName || !row.amount) {
   623→              errors.push(`Row ${i + 1}: Date, item name, and amount are required`)
   624→              skipped++
   625→              continue
   626→            }
   627→            toInsert.push({
   628→              date: String(row.date),
   629→              itemName: String(row.itemName),
   630→              quantity: Number(row.quantity) || 0,
   631→              amount: Number(row.amount),
   632→              remarks: String(row.remarks || ''),
   633→            })
   634→            rowIndexByDoc.push(i)
   635→            break
   636→          }
   637→
   638→          // ─── Legacy modules (Orders/Dispatch/Payments/Expenses) ────
   639→          case 'orders': {
   640→            if (!row.customerId || !row.brickType || !row.quantity || !row.rate) {
   641→              errors.push(`Row ${i + 1}: Customer, brick type, quantity, and rate are required`)
   642→              skipped++
   643→              continue
   644→            }
   645→            const count = await Order.countDocuments({})
   646→            const { Company } = await import('@/lib/models')
   647→            const company = await Company.findOne({})
   648→            const prefix = company?.orderPrefix || 'ORD'
   649→            const orderNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   650→
   651→            await Order.create({
   652→              orderNumber,
   653→              customerId: row.customerId,
   654→              brickType: row.brickType,
   655→              quantity: Number(row.quantity),
   656→              rate: Number(row.rate),
   657→              amount: Number(row.amount) || Number(row.quantity) * Number(row.rate),
   658→              deliveryDate: row.deliveryDate || row.date || new Date().toISOString().split('T')[0],
   659→              status: row.status || 'Pending',
   660→            })
   661→            imported++
   662→            break
   663→          }
   664→
   665→          case 'dispatch': {
   666→            if (!row.customerId || !row.truckNumber || !row.quantity || !row.brickType || !row.date) {
   667→              errors.push(`Row ${i + 1}: Customer, truck, quantity, brick type, and date are required`)
   668→              skipped++
   669→              continue
   670→            }
   671→            const count = await Dispatch.countDocuments({})
   672→            const { Company } = await import('@/lib/models')
   673→            const company = await Company.findOne({})
   674→            const prefix = company?.dispatchPrefix || 'DSP'
   675→            const dispatchNumber = `${prefix}-${String(count + imported + 1).padStart(4, '0')}`
   676→
   677→            await Dispatch.create({
   678→              dispatchNumber,
   679→              customerId: row.customerId,
   680→              orderId: row.orderId || null,
   681→              truckNumber: row.truckNumber,
   682→              driverName: row.driverName || '',
   683→              quantity: Number(row.quantity),
   684→              brickType: row.brickType,
   685→              date: row.date,
   686→            })
   687→            imported++
   688→            break
   689→          }
   690→
   691→          case 'payments': {
   692→            if (!row.customerId || !row.paymentType || !row.amount || !row.date) {
   693→              errors.push(`Row ${i + 1}: Customer, payment type, amount, and date are required`)
   694→              skipped++
   695→              continue
   696→            }
   697→            toInsert.push({
   698→              customerId: row.customerId,
   699→              paymentType: row.paymentType,
   700→              amount: Number(row.amount),
   701→              date: row.date,
   702→              remarks: row.remarks || '',
   703→            })
   704→            rowIndexByDoc.push(i)
   705→            break
   706→          }
   707→
   708→          case 'expenses': {
   709→            if (!row.category || !row.amount || !row.date) {
   710→              errors.push(`Row ${i + 1}: Category, amount, and date are required`)
   711→              skipped++
   712→              continue
   713→            }
   714→            toInsert.push({
   715→              category: row.category,
   716→              amount: Number(row.amount),
   717→              date: row.date,
   718→              description: row.description || '',
   719→            })
   720→            rowIndexByDoc.push(i)
   721→            break
   722→          }
   723→
   724→          default:
   725→            errors.push(`Unknown module: ${module}`)
   726→        }
   727→      } catch (err) {
   728→        errors.push(`Row ${i + 1}: ${err instanceof Error ? err.message : 'Import failed'}`)
   729→        skipped++
   730→      }
   731→    }
   732→
   733→    // ── Bulk insert all valid documents in ONE DB call ───────────────────
   734→    // This is the key optimization: instead of N separate `Model.create()`
   735→    // calls (each a DB round trip), we do a single `Model.insertMany()`.
   736→    // Orders and dispatch are excluded because they need sequential numbering
   737→    // (already handled row-by-row above).
   738→    if (toInsert.length > 0 && !['orders', 'dispatch'].includes(module)) {
   739→      try {
   740→        const Model = getModelForModule(module)
   741→        if (Model) {
   742→          // `ordered: false` lets MongoDB insert all valid docs even if some
   743→          // fail validation — we collect per-doc errors from the result.
   744→          const result = await Model.insertMany(toInsert, { ordered: false, rawResult: false })
   745→          imported += Array.isArray(result) ? result.length : 0
   746→        }
   747→      } catch (err: any) {
   748→        // insertMany with `ordered: false` throws a BulkWriteError containing
   749→        // `insertErrors` and `result` with details about which docs failed.
   750→        if (err?.name === 'BulkWriteError' || err?.code === 11000) {
   751→          // Some docs may have been inserted before the error — count them.
   752→          const insertedCount = err?.insertedDocs?.length ?? err?.result?.nInserted ?? 0
   753→          imported += insertedCount
   754→          // Walk the writeErrors to map each failure back to its row index.
   755→          const writeErrors = err?.result?.writeErrors || err?.writeErrors || []
   756→          for (const we of writeErrors) {
   757→            const docIdx = we.index
   758→            const rowIdx = rowIndexByDoc[docIdx]
   759→            if (rowIdx !== undefined) {
   760→              const isDup = we.code === 11000
   761→              const label = duplicateRowLabel(module, data[rowIdx] as Record<string, unknown>)
   762→              if (isDup) {
   763→                errors.push(`Row ${rowIdx + 1}: Duplicate data found — ${label}`)
   764→                duplicatesSkipped++
   765→                skipped++
   766→              } else {
   767→                errors.push(`Row ${rowIdx + 1}: ${we.errmsg || we.message || 'Insert failed'}`)
   768→                skipped++
   769→              }
   770→            }
   771→          }
   772→        } else {
   773→          // Generic error — record it once with the count.
   774→          errors.push(`Bulk insert failed: ${err instanceof Error ? err.message : 'Unknown error'}. ${toInsert.length} row(s) were not imported.`)
   775→          skipped += toInsert.length
   776→        }
   777→      }
   778→    }
   779→
   780→    // Merge errors and skippedReasons so the UI can show every reason a row
   781→    // was not imported (validation error OR duplicate OR anything else).
   782→    const allReasons = [...errors, ...skippedReasons]
   783→
   784→    // Auto-sync Stock Overview from Production — when production rows are
   785→    // imported, the Stock module must reflect the latest daily totals
   786→    // automatically. We collect every date touched by THIS import and
   787→    // re-aggregate the matching Stock snapshot for each.
   788→    if (module === 'production' && imported > 0) {
   789→      const touchedDates = data
   790→        .map((row) => String(row.date || '').split('T')[0])
   791→        .filter(Boolean)
   792→      if (touchedDates.length > 0) {
   793→        try {
   794→          await syncStockForDates(touchedDates)
   795→        } catch (err) {
   796→          console.error('[import] Stock sync failed:', err)
   797→          // Don't fail the import — production rows were already saved.
   798→          // Just append a soft warning to the error list.
   799→          allReasons.push('Warning: production rows imported, but Stock Overview auto-sync failed. Check server logs.')
   800→        }
   801→      }
   802→    }
   803→
   804→    return NextResponse.json({
   805→      success: true,
   806→      imported,
   807→      skipped,
   808→      duplicatesSkipped,
   809→      total: data.length,
   810→      errors: allReasons.length > 0 ? allReasons.slice(0, 50) : undefined,
   811→    })
   812→  } catch (error) {
   813→    console.error('Error importing data:', error)
   814→    return NextResponse.json({ error: 'Failed to import data' }, { status: 500 })
   815→  }
   816→}

   818→// Map module name to its Mongoose model. Used by the bulk-insert path so we
   819→// can do a single `Model.insertMany(toInsert)` call instead of N creates.
   820→function getModelForModule(module: string) {
   821→  switch (module) {
   822→    case 'customers': return Customer
   823→    case 'production': return Production
   824→    case 'stock': return Stock
   825→    case 'dailySell': return DailySell
   826→    case 'customerPayment': return CustomerPayment
   827→    case 'labourPayment': return LabourPayment
   828→    case 'tractorPayment': return TractorPayment
   829→    case 'dustPurchase': return DustPurchase
   830→    case 'cementPurchase': return CementPurchase
   831→    case 'hardner': return Hardner
   832→    case 'electricity': return Electricity
   833→    case 'factoryStuff': return FactoryStuff
   834→    case 'payments': return Payment
   835→    case 'expenses': return Expense
   836→    // orders and dispatch need sequential numbering — handled separately.
   837→    default: return null
   838→  }
   839→}

   841→// Human-readable label for a duplicate row, used in the skip reason.
   842→function duplicateRowLabel(module: string, row: Record<string, unknown>): string {
   843→  switch (module) {
   844→    case 'customers':
   845→      return `customer "${row.name}" (mobile ${row.mobile})`
   846→    case 'production':
   847→      return `production entry on ${row.date}`
   848→    case 'stock':
   849→      return `stock entry for ${row.date}`
   850→    case 'dailySell':
   851→      return `daily sell for "${row.customerName}" on ${row.date} (₹${row.amount})`
   852→    case 'customerPayment':
   853→      return `payment by "${row.name}" on ${row.date} (₹${row.amount})`
   854→    case 'labourPayment':
   855→      return `payment to "${row.name}" on ${row.date} (₹${row.amount})`
   856→    case 'tractorPayment':
   857→      return `tractor payment to "${row.vendorName}" on ${row.date}`
   858→    case 'dustPurchase':
   859→      return `dust purchase from "${row.vendorName}" on ${row.date}`
   860→    case 'cementPurchase':
   861→      return `cement purchase from "${row.vendorName}" on ${row.date}`
   862→    case 'hardner':
   863→      return `hardner entry on ${row.date} (₹${row.amount})`
   864→    case 'electricity':
   865→      return `electricity entry on ${row.date} (₹${row.amount})`
   866→    case 'factoryStuff':
   867→      return `factory stuff "${row.itemName}" on ${row.date} (₹${row.amount})`
   868→    case 'orders':
   869→      return `order for customer ${row.customerId} on ${row.deliveryDate || row.date}`
   870→    case 'dispatch':
   871→      return `dispatch to customer ${row.customerId} on ${row.date}`
   872→    case 'payments':
   873→      return `payment for customer ${row.customerId} on ${row.date}`
   874→    case 'expenses':
   875→      return `expense "${row.category}" on ${row.date} (₹${row.amount})`
   876→    default:
   877→      return 'record'
   878→  }
   879→}
