     1→'use client'
     2→
     3→import * as React from 'react'
     4→import { api } from '@/lib/api'
     5→import { toast } from '@/hooks/use-toast'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Badge } from '@/components/ui/badge'
    10→import { Skeleton } from '@/components/ui/skeleton'
    11→import { Textarea } from '@/components/ui/textarea'
    12→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    13→import {
    14→  Select,
    15→  SelectContent,
    16→  SelectItem,
    17→  SelectTrigger,
    18→  SelectValue,
    19→} from '@/components/ui/select'
    20→import {
    21→  Table,
    22→  TableBody,
    23→  TableCell,
    24→  TableHead,
    25→  TableHeader,
    26→  TableRow,
    27→} from '@/components/ui/table'
    28→import {
    29→  Dialog,
    30→  DialogContent,
    31→  DialogDescription,
    32→  DialogFooter,
    33→  DialogHeader,
    34→  DialogTitle,
    35→} from '@/components/ui/dialog'
    36→import {
    37→  AlertDialog,
    38→  AlertDialogAction,
    39→  AlertDialogCancel,
    40→  AlertDialogContent,
    41→  AlertDialogDescription,
    42→  AlertDialogFooter,
    43→  AlertDialogHeader,
    44→  AlertDialogTitle,
    45→} from '@/components/ui/alert-dialog'
    46→import {
    47→  Wallet,
    48→  Plus,
    49→  Pencil,
    50→  Trash2,
    51→  Loader2,
    52→  IndianRupee,
    53→  Banknote,
    54→  Smartphone,
    55→  Landmark,
    56→  Users,
    57→  Upload,
    58→} from 'lucide-react'
    59→import ExcelImport from '@/components/erp/excel-import'
    60→
    61→// ── Types ───────────────────────────────────────────────────────────────────
    62→
    63→interface Customer {
    64→  id: string
    65→  name: string
    66→}
    67→
    68→interface Payment {
    69→  id: string
    70→  customerId: string
    71→  paymentType: string
    72→  amount: number
    73→  date: string
    74→  remarks: string
    75→  createdAt: string
    76→  customer: { id: string; name: string }
    77→}
    78→
    79→interface Order {
    80→  id: string
    81→  customerId: string
    82→  amount: number
    83→  status: string
    84→  createdAt: string
    85→  customer: { id: string; name: string }
    86→}
    87→
    88→interface PaymentFormData {
    89→  customerId: string
    90→  paymentType: string
    91→  amount: number | string
    92→  date: string
    93→  remarks: string
    94→}
    95→
    96→interface OutstandingEntry {
    97→  customerId: string
    98→  customerName: string
    99→  totalOrderAmount: number
   100→  totalPayments: number
   101→  outstanding: number
   102→}
   103→
   104→// ── Constants ───────────────────────────────────────────────────────────────
   105→
   106→const PAYMENT_TYPES = ['Cash', 'UPI', 'Bank Transfer'] as const
   107→
   108→const PAYMENT_TYPE_COLORS: Record<string, string> = {
   109→  Cash: 'bg-emerald-100 text-emerald-700 border-emerald-200',
   110→  UPI: 'bg-sky-100 text-sky-700 border-sky-200',
   111→  'Bank Transfer': 'bg-amber-100 text-amber-700 border-amber-200',
   112→}
   113→
   114→const PAYMENT_TYPE_ICONS: Record<string, React.ReactNode> = {
   115→  Cash: <Banknote className="size-3" />,
   116→  UPI: <Smartphone className="size-3" />,
   117→  'Bank Transfer': <Landmark className="size-3" />,
   118→}
   119→
   120→// ── Helpers ─────────────────────────────────────────────────────────────────
   121→
   122→const formatCurrency = (amount: number): string =>
   123→  new Intl.NumberFormat('en-IN', {
   124→    style: 'currency',
   125→    currency: 'INR',
   126→    minimumFractionDigits: 0,
   127→    maximumFractionDigits: 2,
   128→  }).format(amount)
   129→
   130→const formatDate = (dateStr: string): string =>
   131→  new Date(dateStr).toLocaleDateString('en-IN', {
   132→    day: '2-digit',
   133→    month: 'short',
   134→    year: 'numeric',
   135→  })
   136→
   137→const emptyForm: PaymentFormData = {
   138→  customerId: '',
   139→  paymentType: '',
   140→  amount: '',
   141→  date: '',
   142→  remarks: '',
   143→}
   144→
   145→// ── Component ───────────────────────────────────────────────────────────────
   146→
   147→export function PaymentModule() {
   148→  // State
   149→  const [payments, setPayments] = React.useState<Payment[]>([])
   150→  const [customers, setCustomers] = React.useState<Customer[]>([])
   151→  const [orders, setOrders] = React.useState<Order[]>([])
   152→  const [loading, setLoading] = React.useState(true)
   153→
   154→  // Dialog states
   155→  const [dialogOpen, setDialogOpen] = React.useState(false)
   156→  const [editingPayment, setEditingPayment] = React.useState<Payment | null>(null)
   157→  const [formData, setFormData] = React.useState<PaymentFormData>(emptyForm)
   158→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   159→
   160→  // Excel import
   161→  const [importOpen, setImportOpen] = React.useState(false)
   162→
   163→  // Delete confirmation
   164→  const [deleteTarget, setDeleteTarget] = React.useState<Payment | null>(null)
   165→  const [deleting, setDeleting] = React.useState(false)
   166→
   167→  // ── Computed: Summary values ─────────────────────────────────────────────
   168→  const totalReceived = React.useMemo(
   169→    () => payments.reduce((sum, p) => sum + Number(p.amount), 0),
   170→    [payments]
   171→  )
   172→
   173→  const cashPayments = React.useMemo(
   174→    () =>
   175→      payments
   176→        .filter((p) => p.paymentType === 'Cash')
   177→        .reduce((sum, p) => sum + Number(p.amount), 0),
   178→    [payments]
   179→  )
   180→
   181→  const upiBankPayments = React.useMemo(
   182→    () =>
   183→      payments
   184→        .filter((p) => p.paymentType === 'UPI' || p.paymentType === 'Bank Transfer')
   185→        .reduce((sum, p) => sum + Number(p.amount), 0),
   186→    [payments]
   187→  )
   188→
   189→  // ── Computed: Customer outstanding ───────────────────────────────────────
   190→  const outstandingEntries = React.useMemo<OutstandingEntry[]>(() => {
   191→    const map = new Map<string, OutstandingEntry>()
   192→
   193→    // Sum order amounts per customer (exclude Cancelled)
   194→    for (const order of orders) {
   195→      if (order.status === 'Cancelled') continue
   196→      const existing = map.get(order.customerId)
   197→      if (existing) {
   198→        existing.totalOrderAmount += Number(order.amount)
   199→      } else {
   200→        map.set(order.customerId, {
   201→          customerId: order.customerId,
   202→          customerName: order.customer?.name || 'Unknown',
   203→          totalOrderAmount: Number(order.amount),
   204→          totalPayments: 0,
   205→          outstanding: 0,
   206→        })
   207→      }
   208→    }
   209→
   210→    // Sum payments per customer
   211→    for (const payment of payments) {
   212→      const existing = map.get(payment.customerId)
   213→      if (existing) {
   214→        existing.totalPayments += Number(payment.amount)
   215→      } else {
   216→        map.set(payment.customerId, {
   217→          customerId: payment.customerId,
   218→          customerName: payment.customer?.name || 'Unknown',
   219→          totalOrderAmount: 0,
   220→          totalPayments: Number(payment.amount),
   221→          outstanding: 0,
   222→        })
   223→      }
   224→    }
   225→
   226→    // Calculate outstanding
   227→    for (const entry of map.values()) {
   228→      entry.outstanding = entry.totalOrderAmount - entry.totalPayments
   229→    }
   230→
   231→    // Return only customers with outstanding > 0, sorted by outstanding descending
   232→    return Array.from(map.values())
   233→      .filter((e) => e.outstanding > 0.01)
   234→      .sort((a, b) => b.outstanding - a.outstanding)
   235→  }, [orders, payments])
   236→
   237→  // ── Fetch data ───────────────────────────────────────────────────────────
   238→  const fetchPayments = React.useCallback(async () => {
   239→    setLoading(true)
   240→    try {
   241→      const res = await api.getPayments()
   242→      const paymentList = (res.payments as Payment[]).sort(
   243→        (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   244→      )
   245→      setPayments(paymentList)
   246→    } catch (err) {
   247→      toast({
   248→        title: 'Error',
   249→        description: err instanceof Error ? err.message : 'Failed to fetch payments',
   250→        variant: 'destructive',
   251→      })
   252→    } finally {
   253→      setLoading(false)
   254→    }
   255→  }, [])
   256→
   257→  const fetchCustomers = React.useCallback(async () => {
   258→    try {
   259→      const res = await api.getCustomers()
   260→      setCustomers(res.customers as Customer[])
   261→    } catch {
   262→      // Silently fail — customer dropdown will just be empty
   263→    }
   264→  }, [])
   265→
   266→  const fetchOrders = React.useCallback(async () => {
   267→    try {
   268→      const res = await api.getOrders()
   269→      setOrders(res.orders as Order[])
   270→    } catch {
   271→      // Silently fail — outstanding section will just be empty
   272→    }
   273→  }, [])
   274→
   275→  React.useEffect(() => {
   276→    fetchPayments()
   277→    fetchCustomers()
   278→    fetchOrders()
   279→  }, [fetchPayments, fetchCustomers, fetchOrders])
   280→
   281→  // ── Form handlers ────────────────────────────────────────────────────────
   282→  const openCreateDialog = () => {
   283→    setEditingPayment(null)
   284→    setFormData(emptyForm)
   285→    setDialogOpen(true)
   286→  }
   287→
   288→  const openEditDialog = (payment: Payment) => {
   289→    setEditingPayment(payment)
   290→    setFormData({
   291→      customerId: payment.customerId,
   292→      paymentType: payment.paymentType,
   293→      amount: payment.amount,
   294→      date: payment.date,
   295→      remarks: payment.remarks,
   296→    })
   297→    setDialogOpen(true)
   298→  }
   299→
   300→  const handleFormChange = (field: keyof PaymentFormData, value: string) => {
   301→    setFormData((prev) => ({ ...prev, [field]: value }))
   302→  }
   303→
   304→  const handleSubmit = async () => {
   305→    // Validation
   306→    if (!formData.customerId) {
   307→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   308→      return
   309→    }
   310→    if (!formData.paymentType) {
   311→      toast({ title: 'Validation Error', description: 'Please select a payment type', variant: 'destructive' })
   312→      return
   313→    }
   314→    if (!formData.amount || Number(formData.amount) <= 0) {
   315→      toast({ title: 'Validation Error', description: 'Amount must be greater than 0', variant: 'destructive' })
   316→      return
   317→    }
   318→    if (!formData.date) {
   319→      toast({ title: 'Validation Error', description: 'Date is required', variant: 'destructive' })
   320→      return
   321→    }
   322→
   323→    setFormSubmitting(true)
   324→    try {
   325→      const payload = {
   326→        customerId: formData.customerId,
   327→        paymentType: formData.paymentType,
   328→        amount: Number(formData.amount),
   329→        date: formData.date,
   330→        remarks: formData.remarks,
   331→      }
   332→
   333→      if (editingPayment) {
   334→        await api.updatePayment(editingPayment.id, payload)
   335→        toast({ title: 'Success', description: 'Payment updated successfully' })
   336→      } else {
   337→        await api.createPayment(payload)
   338→        toast({ title: 'Success', description: 'Payment recorded successfully' })
   339→      }
   340→
   341→      setDialogOpen(false)
   342→      setEditingPayment(null)
   343→      setFormData(emptyForm)
   344→      fetchPayments()
   345→      fetchOrders() // Refresh for outstanding calculations
   346→    } catch (err) {
   347→      toast({
   348→        title: 'Error',
   349→        description: err instanceof Error ? err.message : `Failed to ${editingPayment ? 'update' : 'record'} payment`,
   350→        variant: 'destructive',
   351→      })
   352→    } finally {
   353→      setFormSubmitting(false)
   354→    }
   355→  }
   356→
   357→  // ── Delete handler ──────────────────────────────────────────────────────
   358→  const handleDelete = async () => {
   359→    if (!deleteTarget) return
   360→    setDeleting(true)
   361→    try {
   362→      await api.deletePayment(deleteTarget.id)
   363→      toast({ title: 'Success', description: 'Payment deleted successfully' })
   364→      setDeleteTarget(null)
   365→      fetchPayments()
   366→      fetchOrders()
   367→    } catch (err) {
   368→      toast({
   369→        title: 'Error',
   370→        description: err instanceof Error ? err.message : 'Failed to delete payment',
   371→        variant: 'destructive',
   372→      })
   373→    } finally {
   374→      setDeleting(false)
   375→    }
   376→  }
   377→
   378→  // ── Render: Loading skeletons ───────────────────────────────────────────
   379→  const renderSkeletons = () =>
   380→    Array.from({ length: 5 }).map((_, i) => (
   381→      <TableRow key={i}>
   382→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   383→        <TableCell><Skeleton className="h-5 w-16" /></TableCell>
   384→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   385→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   386→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   387→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   388→      </TableRow>
   389→    ))
   390→
   391→  // ── Render: Payment type badge ──────────────────────────────────────────
   392→  const renderPaymentTypeBadge = (type: string) => (
   393→    <Badge className={`${PAYMENT_TYPE_COLORS[type] || 'bg-gray-100 text-gray-700 border-gray-200'} gap-1`}>
   394→      {PAYMENT_TYPE_ICONS[type]}
   395→      {type}
   396→    </Badge>
   397→  )
   398→
   399→  // ── Render: Payment dialog ──────────────────────────────────────────────
   400→  const renderPaymentDialog = () => (
   401→    <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   402→      <DialogContent className="sm:max-w-lg">
   403→        <DialogHeader>
   404→          <DialogTitle>
   405→            {editingPayment ? 'Edit Payment' : 'Receive Payment'}
   406→          </DialogTitle>
   407→          <DialogDescription>
   408→            {editingPayment
   409→              ? 'Update the payment details below.'
   410→              : 'Fill in the details to record a new payment.'}
   411→          </DialogDescription>
   412→        </DialogHeader>
   413→
   414→        <div className="grid gap-4 py-2">
   415→          {/* Customer */}
   416→          <div className="grid gap-2">
   417→            <Label htmlFor="payment-customer">
   418→              Customer <span className="text-destructive">*</span>
   419→            </Label>
   420→            <Select
   421→              value={formData.customerId}
   422→              onValueChange={(val) => handleFormChange('customerId', val)}
   423→            >
   424→              <SelectTrigger className="w-full">
   425→                <SelectValue placeholder="Select customer" />
   426→              </SelectTrigger>
   427→              <SelectContent>
   428→                {customers.map((customer) => (
   429→                  <SelectItem key={customer.id} value={customer.id}>
   430→                    {customer.name}
   431→                  </SelectItem>
   432→                ))}
   433→              </SelectContent>
   434→            </Select>
   435→          </div>
   436→
   437→          {/* Payment Type */}
   438→          <div className="grid gap-2">
   439→            <Label htmlFor="payment-type">
   440→              Payment Type <span className="text-destructive">*</span>
   441→            </Label>
   442→            <Select
   443→              value={formData.paymentType}
   444→              onValueChange={(val) => handleFormChange('paymentType', val)}
   445→            >
   446→              <SelectTrigger className="w-full">
   447→                <SelectValue placeholder="Select payment type" />
   448→              </SelectTrigger>
   449→              <SelectContent>
   450→                {PAYMENT_TYPES.map((type) => (
   451→                  <SelectItem key={type} value={type}>
   452→                    {type}
   453→                  </SelectItem>
   454→                ))}
   455→              </SelectContent>
   456→            </Select>
   457→          </div>
   458→
   459→          {/* Amount */}
   460→          <div className="grid gap-2">
   461→            <Label htmlFor="payment-amount">
   462→              Amount (₹) <span className="text-destructive">*</span>
   463→            </Label>
   464→            <div className="relative">
   465→              <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   466→              <Input
   467→                id="payment-amount"
   468→                type="number"
   469→                placeholder="0"
   470→                min="1"
   471→                className="pl-9"
   472→                value={formData.amount}
   473→                onChange={(e) => handleFormChange('amount', e.target.value)}
   474→              />
   475→            </div>
   476→          </div>
   477→
   478→          {/* Date */}
   479→          <div className="grid gap-2">
   480→            <Label htmlFor="payment-date">
   481→              Date <span className="text-destructive">*</span>
   482→            </Label>
   483→            <Input
   484→              id="payment-date"
   485→              type="date"
   486→              value={formData.date}
   487→              onChange={(e) => handleFormChange('date', e.target.value)}
   488→            />
   489→          </div>
   490→
   491→          {/* Remarks */}
   492→          <div className="grid gap-2">
   493→            <Label htmlFor="payment-remarks">Remarks</Label>
   494→            <Textarea
   495→              id="payment-remarks"
   496→              placeholder="Optional remarks about this payment..."
   497→              value={formData.remarks}
   498→              onChange={(e) => handleFormChange('remarks', e.target.value)}
   499→              className="min-h-[80px] resize-none"
   500→            />
   501→          </div>
   502→        </div>
   503→
   504→        <DialogFooter>
   505→          <Button
   506→            variant="outline"
   507→            onClick={() => setDialogOpen(false)}
   508→            disabled={formSubmitting}
   509→          >
   510→            Cancel
   511→          </Button>
   512→          <Button
   513→            onClick={handleSubmit}
   514→            disabled={formSubmitting}
   515→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   516→          >
   517→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   518→            {editingPayment ? 'Update Payment' : 'Record Payment'}
   519→          </Button>
   520→        </DialogFooter>
   521→      </DialogContent>
   522→    </Dialog>
   523→  )
   524→
   525→  // ── Render: Delete confirmation ─────────────────────────────────────────
   526→  const renderDeleteDialog = () => (
   527→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   528→      <AlertDialogContent>
   529→        <AlertDialogHeader>
   530→          <AlertDialogTitle>Delete Payment</AlertDialogTitle>
   531→          <AlertDialogDescription>
   532→            Are you sure you want to delete this payment of{' '}
   533→            <strong>{deleteTarget ? formatCurrency(Number(deleteTarget.amount)) : ''}</strong>{' '}
   534→            from <strong>{deleteTarget?.customer?.name || 'this customer'}</strong>? This action
   535→            cannot be undone.
   536→          </AlertDialogDescription>
   537→        </AlertDialogHeader>
   538→        <AlertDialogFooter>
   539→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   540→          <AlertDialogAction
   541→            onClick={handleDelete}
   542→            disabled={deleting}
   543→            className="bg-destructive text-white hover:bg-destructive/90"
   544→          >
   545→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   546→            Delete
   547→          </AlertDialogAction>
   548→        </AlertDialogFooter>
   549→      </AlertDialogContent>
   550→    </AlertDialog>
   551→  )
   552→
   553→  // ── Render: Main ────────────────────────────────────────────────────────
   554→  return (
   555→    <div className="space-y-6">
   556→      {/* Header */}
   557→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   558→        <div className="flex items-center gap-3">
   559→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   560→            <Wallet className="size-5" />
   561→          </div>
   562→          <div>
   563→            <h2 className="text-2xl font-bold tracking-tight">Payment Management</h2>
   564→            <p className="text-sm text-muted-foreground">
   565→              Track and manage customer payments
   566→            </p>
   567→          </div>
   568→        </div>
   569→        <div className="flex gap-2 w-full sm:w-auto">
   570→          <Button
   571→            variant="outline"
   572→            onClick={() => setImportOpen(true)}
   573→            className="w-full sm:w-auto"
   574→          >
   575→            <Upload className="size-4 mr-2" />
   576→            Import Excel
   577→          </Button>
   578→          <Button
   579→            onClick={openCreateDialog}
   580→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   581→          >
   582→            <Plus className="size-4" />
   583→            Receive Payment
   584→          </Button>
   585→        </div>
   586→      </div>
   587→
   588→      {/* Summary Cards */}
   589→      <div className="grid gap-4 sm:grid-cols-3">
   590→        <Card>
   591→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   592→            <CardTitle className="text-sm font-medium text-muted-foreground">
   593→              Total Received
   594→            </CardTitle>
   595→            <IndianRupee className="size-4 text-emerald-600" />
   596→          </CardHeader>
   597→          <CardContent>
   598→            {loading ? (
   599→              <Skeleton className="h-8 w-32" />
   600→            ) : (
   601→              <p className="text-2xl font-bold">{formatCurrency(totalReceived)}</p>
   602→            )}
   603→          </CardContent>
   604→        </Card>
   605→
   606→        <Card>
   607→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   608→            <CardTitle className="text-sm font-medium text-muted-foreground">
   609→              Cash Payments
   610→            </CardTitle>
   611→            <Banknote className="size-4 text-emerald-600" />
   612→          </CardHeader>
   613→          <CardContent>
   614→            {loading ? (
   615→              <Skeleton className="h-8 w-32" />
   616→            ) : (
   617→              <p className="text-2xl font-bold">{formatCurrency(cashPayments)}</p>
   618→            )}
   619→          </CardContent>
   620→        </Card>
   621→
   622→        <Card>
   623→          <CardHeader className="flex flex-row items-center justify-between pb-2">
   624→            <CardTitle className="text-sm font-medium text-muted-foreground">
   625→              UPI + Bank
   626→            </CardTitle>
   627→            <Landmark className="size-4 text-sky-600" />
   628→          </CardHeader>
   629→          <CardContent>
   630→            {loading ? (
   631→              <Skeleton className="h-8 w-32" />
   632→            ) : (
   633→              <p className="text-2xl font-bold">{formatCurrency(upiBankPayments)}</p>

   638→

   649→        <CardContent>

   665→                ) : payments.length === 0 ? (

   667→                    <TableCell colSpan={6} className="h-32 text-center text-muted-foreground">

   699→                            variant="ghost"
   700→                            size="icon"
   701→                            onClick={() => setDeleteTarget(payment)}
   702→                            title="Delete Payment"
   703→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   704→                          >
   705→                            <Trash2 className="size-4" />
   706→                          </Button>
   707→                        </div>
   708→                      </TableCell>
   709→                    </TableRow>
   710→                  ))
   711→                )}
   712→              </TableBody>
   713→            </Table>
   714→          </div>
   715→        </CardContent>
   716→      </Card>
   717→
   718→      {/* Customer Outstanding Section */}
   719→      <Card>
   720→        <CardHeader>
   721→          <CardTitle className="flex items-center gap-2">
   722→            <Users className="size-5" />
   723→            <span>Customer Outstanding Balances</span>
   724→          </CardTitle>
   725→        </CardHeader>
   726→        <CardContent>
   727→          {loading ? (
   728→            <div className="space-y-3">
   729→              {Array.from({ length: 3 }).map((_, i) => (
   730→                <Skeleton key={i} className="h-10 w-full" />
   731→              ))}
   732→            </div>
   733→          ) : outstandingEntries.length === 0 ? (
   734→            <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   735→              <Users className="size-10 mb-2 opacity-40" />
   736→              <p className="text-sm">No outstanding balances. All customers are paid up!</p>
   737→            </div>
   738→          ) : (
   739→            <div className="overflow-x-auto">
   740→              <Table>
   741→                <TableHeader>
   742→                  <TableRow>
   743→                    <TableHead>Customer</TableHead>
   744→                    <TableHead className="text-right">Total Orders (₹)</TableHead>
   745→                    <TableHead className="text-right">Total Paid (₹)</TableHead>

   748→                </TableHeader>
   749→                <TableBody>
   750→                  {outstandingEntries.map((entry) => (

   752→                      <TableCell className="font-medium whitespace-nowrap">
   753→                        {entry.customerName}
   754→                      </TableCell>
   755→                      <TableCell className="text-right whitespace-nowrap">
   756→                        {formatCurrency(entry.totalOrderAmount)}
   757→                      </TableCell>
   758→                      <TableCell className="text-right whitespace-nowrap">
   759→                        {formatCurrency(entry.totalPayments)}
   760→                      </TableCell>
   761→                      <TableCell className="text-right font-medium whitespace-nowrap">
   762→                        <span className="text-rose-600">{formatCurrency(entry.outstanding)}</span>
   763→                      </TableCell>
   764→                    </TableRow>
   765→                  ))}
   766→                </TableBody>
   767→              </Table>
   768→            </div>
   769→          )}
   770→        </CardContent>
   771→      </Card>
   772→
   773→      {/* Dialogs */}
   774→      {renderPaymentDialog()}
   775→      {renderDeleteDialog()}
   776→
   777→      <ExcelImport module="payments" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchPayments} />
   778→    </div>
   779→  )
   780→}
