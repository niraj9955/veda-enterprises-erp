   145→   145→// ── Component ───────────────────────────────────────────────────────────────
   146→   146→
   147→   147→export function OrderModule() {
   148→   148→  // State
   149→   149→  const [orders, setOrders] = React.useState<Order[]>([])
   150→   150→  const [search, setSearch] = React.useState('')
   151→   151→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   152→   152→  const [customers, setCustomers] = React.useState<Customer[]>([])
   153→   153→  const [loading, setLoading] = React.useState(true)
   154→   154→
   155→   155→  // Create dialog states
   156→   156→  const [createOpen, setCreateOpen] = React.useState(false)
   157→   157→  const [formData, setFormData] = React.useState<OrderFormData>(emptyForm)
   158→   158→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   159→   159→
   160→   160→  // Edit status dialog states
   161→   161→  const [editStatusOpen, setEditStatusOpen] = React.useState(false)
   162→   162→  const [editTarget, setEditTarget] = React.useState<Order | null>(null)
   163→   163→  const [editStatus, setEditStatus] = React.useState('')
   164→   164→  const [editSubmitting, setEditSubmitting] = React.useState(false)
   165→   165→
   166→   166→  // Excel import
   167→   167→  const [importOpen, setImportOpen] = React.useState(false)
   168→   168→
   169→   169→  // Delete confirmation
   170→   170→  const [deleteTarget, setDeleteTarget] = React.useState<Order | null>(null)
   171→   171→  const [deleting, setDeleting] = React.useState(false)
   172→   172→
   173→   173→  // ── Computed amount ─────────────────────────────────────────────────────
   174→   174→  // If items[] are present, total amount = sum of item amounts.
   175→   175→  // Otherwise fall back to top-level qty * rate.
   176→   176→  const computedAmount = React.useMemo(() => {
   177→   177→    if (formData.items.length > 0) {
   178→   178→      return formData.items.reduce((s, it) => s + (Number(it.amount) || 0), 0)
   179→   179→    }
   180→   180→    const qty = Number(formData.quantity) || 0
   181→   181→    const rate = Number(formData.rate) || 0
   182→   182→    return qty * rate
   183→   183→  }, [formData.quantity, formData.rate, formData.items])
   184→   184→
   185→   185→  // ── Item management ──────────────────────────────────────────────────────
   186→   186→  const updateItem = (idx: number, field: keyof OrderItem, value: string | number) => {
   187→   187→    setFormData((prev) => {
   188→   188→      const items = [...prev.items]
   189→   189→      const item = { ...items[idx] }
   190→   190→      if (field === 'description' || field === 'hsn' || field === 'unit') {
   191→   191→        (item as any)[field] = value as string
   192→   192→      } else {
   193→   193→        (item as any)[field] = Number(value) || 0
   194→   194→      }
   195→   195→      item.amount = item.quantity * item.rate
   196→   196→      items[idx] = item
   197→   197→      return { ...prev, items }
   198→   198→    })
   199→   199→  }
   200→   200→  const addItem = () => setFormData((prev) => ({ ...prev, items: [...prev.items, newItem()] }))
   201→   201→  const removeItem = (idx: number) =>
   202→   202→    setFormData((prev) => ({ ...prev, items: prev.items.filter((_, i) => i !== idx) }))
   203→   203→
   204→   204→  // ── Fetch orders ────────────────────────────────────────────────────────
   205→   205→  const fetchOrders = React.useCallback(async () => {
   206→   206→    setLoading(true)
   207→   207→    try {
   208→   208→      const res = await api.getOrders()
   209→   209→      const orderList = (res.orders as Order[]).sort(
   210→   210→        (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   211→   211→      )
   212→   212→      setOrders(orderList)
   213→   213→    } catch (err) {
   214→   214→      toast({
   215→   215→        title: 'Error',
   216→   216→        description: err instanceof Error ? err.message : 'Failed to fetch orders',
   217→   217→        variant: 'destructive',
   218→   218→      })
   219→   219→    } finally {
   220→   220→      setLoading(false)
   221→   221→    }
   222→   222→  }, [])
   223→   223→
   224→   224→  // ── Fetch customers (for dropdown) ─────────────────────────────────────
   225→   225→  const fetchCustomers = React.useCallback(async () => {
   226→   226→    try {
   227→   227→      const res = await api.getCustomers()
   228→   228→      setCustomers(res.customers as Customer[])
   229→   229→    } catch {
   230→   230→      // Silently fail — customer dropdown will just be empty
   231→   231→    }
   232→   232→  }, [])
   233→   233→
   234→   234→  // Debounced search
   235→   235→  React.useEffect(() => {
   236→   236→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   237→   237→    return () => clearTimeout(t)
   238→   238→  }, [search])
   239→   239→
   240→   240→  // Client-side filter
   241→   241→  const filteredOrders = React.useMemo(() => {
   242→   242→    if (!debouncedSearch.trim()) return orders
   243→   243→    const q = debouncedSearch.toLowerCase()
   244→   244→    return orders.filter((item: any) =>
   245→   245→      ['orderNumber', 'brickType', 'status', 'deliveryDate'].some((f) =>
   246→   246→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   247→   247→      )
   248→   248→    )
   249→   249→  }, [orders, debouncedSearch])
   250→   250→
   251→   251→  React.useEffect(() => {
   252→   252→    fetchOrders()
   253→   253→    fetchCustomers()
   254→   254→  }, [fetchOrders, fetchCustomers])
   255→   255→
   256→   256→  // ── Create dialog handlers ──────────────────────────────────────────────
   257→   257→  const openCreateDialog = () => {
   258→   258→    setFormData(emptyForm)
   259→   259→    setCreateOpen(true)
   260→   260→  }
   261→   261→
   262→   262→  const handleFormChange = (field: keyof OrderFormData, value: string) => {
   263→   263→    setFormData((prev) => ({ ...prev, [field]: value }))
   264→   264→  }
   265→   265→
   266→   266→  const handleCreateSubmit = async () => {
   267→   267→    if (!formData.customerId) {
   268→   268→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   269→   269→      return
   270→   270→    }
   271→   271→    if (!formData.brickType) {
   272→   272→      toast({ title: 'Validation Error', description: 'Please select a brick type', variant: 'destructive' })
   273→   273→      return
   274→   274→    }
   275→   275→    if (!formData.quantity || Number(formData.quantity) <= 0) {
   276→   276→      toast({ title: 'Validation Error', description: 'Quantity must be greater than 0', variant: 'destructive' })
   277→   277→      return
   278→   278→    }
   279→   279→    if (!formData.rate || Number(formData.rate) <= 0) {
   280→   280→      toast({ title: 'Validation Error', description: 'Rate must be greater than 0', variant: 'destructive' })
   281→   281→      return
   282→   282→    }
   283→   283→    if (!formData.deliveryDate) {
   284→   284→      toast({ title: 'Validation Error', description: 'Delivery date is required', variant: 'destructive' })
   285→   285→      return
   286→   286→    }
   287→   287→
   288→   288→    // Either items[] OR brickType+qty+rate must be present
   289→   289→    const hasItems = formData.items.length > 0 && formData.items.some((i) => i.description.trim())
   290→   290→    if (!hasItems && (!formData.brickType || !formData.quantity || !formData.rate)) {
   291→   291→      toast({
   292→   292→        title: 'Validation Error',
   293→   293→        description: 'Add at least one item, OR fill brick type + quantity + rate',
   294→   294→        variant: 'destructive',
   295→   295→      })
   296→   296→      return
   297→   297→    }
   298→   298→
   299→   299→    setFormSubmitting(true)
   300→   300→    try {
   301→   301→      const payload = {
   302→   302→        customerId: formData.customerId,
   303→   303→        brickType: formData.brickType,
   304→   304→        quantity: Number(formData.quantity) || 0,
   305→   305→        rate: Number(formData.rate) || 0,
   306→   306→        amount: computedAmount,
   307→   307→        deliveryDate: formData.deliveryDate,
   308→   308→        status: formData.status,
   309→   309→        items: formData.items.filter((i) => i.description.trim()),
   310→   310→      }
   311→   311→
   312→   312→      await api.createOrder(payload)
   313→   313→      toast({ title: 'Success', description: 'Order created successfully' })
   314→   314→      setCreateOpen(false)
   315→   315→      setFormData(emptyForm)
   316→   316→      fetchOrders()
   317→   317→    } catch (err) {
   318→   318→      toast({
   319→   319→        title: 'Error',
   320→   320→        description: err instanceof Error ? err.message : 'Failed to create order',
   321→   321→        variant: 'destructive',
   322→   322→      })
   323→   323→    } finally {
   324→   324→      setFormSubmitting(false)
   325→   325→    }
   326→   326→  }
   327→   327→
   328→   328→  // ── Edit status handlers ────────────────────────────────────────────────
   329→   329→  const openEditStatusDialog = (order: Order) => {
   330→   330→    setEditTarget(order)
   331→   331→    setEditStatus(order.status)
   332→   332→    setEditStatusOpen(true)
   333→   333→  }
   334→   334→
   335→   335→  const handleEditStatusSubmit = async () => {
   336→   336→    if (!editTarget) return
   337→   337→    setEditSubmitting(true)
   338→   338→    try {
   339→   339→      await api.updateOrder(editTarget.id, { status: editStatus })
   340→   340→      toast({ title: 'Success', description: 'Order status updated successfully' })
   341→   341→      setEditStatusOpen(false)
   342→   342→      setEditTarget(null)
   343→   343→      setEditStatus('')
   344→   344→      fetchOrders()
   345→   345→    } catch (err) {
   346→   346→      toast({
   347→   347→        title: 'Error',
   348→   348→        description: err instanceof Error ? err.message : 'Failed to update order status',
   349→   349→        variant: 'destructive',
   350→   350→      })
   351→   351→    } finally {
   352→   352→      setEditSubmitting(false)
   353→   353→    }
   354→   354→  }
   355→   355→
   356→   356→  // ── Delete handler ──────────────────────────────────────────────────────
   357→   357→  const handleDelete = async () => {
   358→   358→    if (!deleteTarget) return
   359→   359→    setDeleting(true)
   360→   360→    try {
   361→   361→      await api.deleteOrder(deleteTarget.id)
   362→   362→      toast({ title: 'Success', description: 'Order deleted successfully' })
   363→   363→      setDeleteTarget(null)
   364→   364→      fetchOrders()
   365→   365→    } catch (err) {
   366→   366→      toast({
   367→   367→        title: 'Error',
   368→   368→        description: err instanceof Error ? err.message : 'Failed to delete order',
   369→   369→        variant: 'destructive',
   370→   370→      })
   371→   371→    } finally {
   372→   372→      setDeleting(false)
   373→   373→    }
   374→   374→  }
   375→   375→
   376→   376→  // ── Render: Loading skeletons ───────────────────────────────────────────
   377→   377→  const renderSkeletons = () =>
   378→   378→    Array.from({ length: 5 }).map((_, i) => (
   379→   379→      <TableRow key={i}>
   380→   380→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   381→   381→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   382→   382→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   383→   383→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   384→   384→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   385→   385→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   386→   386→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   387→   387→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   388→   388→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   389→   389→      </TableRow>
   390→   390→    ))
   391→   391→
   392→   392→  // ── Render: Status badge ────────────────────────────────────────────────
   393→   393→  const renderStatusBadge = (status: string) => (
   394→   394→    <Badge className={STATUS_COLORS[status] || 'bg-gray-100 text-gray-700 border-gray-200'}>
   395→   395→      {status}
   396→   396→    </Badge>
   397→   397→  )
   398→   398→
   399→   399→  // ── Render: Create Order dialog ─────────────────────────────────────────
   400→   400→  const renderCreateDialog = () => (
   401→   401→    <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   402→   402→      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto">
   403→   403→        <DialogHeader>
   404→   404→          <DialogTitle>Create Order</DialogTitle>
   405→   405→          <DialogDescription>
   406→   406→            Fill in the details to create a new order. You can add multiple items below the brick type.
   407→   407→          </DialogDescription>
   408→   408→        </DialogHeader>
   409→   409→
   410→   410→        <div className="grid gap-4 py-2">
   411→   411→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   412→   412→          <CustomerSearchInput
   413→   413→            value={formData.customerId}
   414→   414→            onSelect={(c) => handleFormChange('customerId', c.id)}
   415→   415→            onClear={() => handleFormChange('customerId', '')}
   416→   416→            required
   417→   417→            label="Customer"
   418→   418→            placeholder="Type customer name or mobile to search..."
   419→   419→          />
   420→   420→
   421→   421→          {/* Brick Type — now optional since items[] can replace it */}
   422→   422→          <div className="grid gap-2">
   423→   423→            <Label htmlFor="order-brick">
   424→   424→              Brick Type <span className="text-muted-foreground text-xs font-normal">(optional when items are added below)</span>
   425→   425→            </Label>
   426→   426→            <Select
   427→   427→              value={formData.brickType}
   428→   428→              onValueChange={(val) => handleFormChange('brickType', val)}
   429→   429→            >
   430→   430→              <SelectTrigger className="w-full">
   431→   431→                <SelectValue placeholder="Select brick type (or skip if using items)" />
   432→   432→              </SelectTrigger>
   433→   433→              <SelectContent>
   434→   434→                {BRICK_TYPES.map((type) => (
   435→   435→                  <SelectItem key={type} value={type}>
   436→   436→                    {type}
   437→   437→                  </SelectItem>
   438→   438→                ))}
   439→   439→              </SelectContent>
   440→   440→            </Select>
   441→   441→          </div>
   442→   442→
   443→   443→          {/* Items section — multi-line item entry */}
   444→   444→          <div className="grid gap-2">
   445→   445→            <div className="flex items-center justify-between">
   446→   446→              <Label>Items <span className="text-muted-foreground text-xs font-normal">(add multiple products / brick types)</span></Label>
   447→   447→              <Button type="button" size="sm" variant="outline" onClick={addItem}>
   448→   448→                <Plus className="h-3 w-3 mr-1" /> Add Item
   449→   449→              </Button>
   450→   450→            </div>
   451→   451→            <div className="border rounded-md p-2 space-y-2 bg-muted/20">
   452→   452→              {formData.items.length === 0 ? (
   453→   453→                <p className="text-xs text-muted-foreground text-center py-2">
   454→   454→                  No items added. Use "Add Item" to add multiple products, OR skip this section and use brick type + qty + rate above.
   455→   455→                </p>
   456→   456→              ) : (
   457→   457→                <>
   458→   458→                  <div className="hidden md:grid grid-cols-12 gap-2 text-xs font-semibold text-muted-foreground px-1">
   459→   459→                    <div className="col-span-5">Description</div>
   460→   460→                    <div className="col-span-2">Qty</div>
   461→   461→                    <div className="col-span-2">Unit</div>
   462→   462→                    <div className="col-span-2">Rate (₹)</div>
   463→   463→                    <div className="col-span-1">Amount</div>
   464→   464→                  </div>
   465→   465→                  {formData.items.map((item, idx) => (
   466→   466→                    <div key={idx} className="grid grid-cols-12 gap-2 items-center">
   467→   467→                      <Input
   468→   468→                        className="col-span-12 md:col-span-5"
   469→   469→                        placeholder="Item description"
   470→   470→                        value={item.description}
   471→   471→                        onChange={(e) => updateItem(idx, 'description', e.target.value)}
   472→   472→                      />
   473→   473→                      <Input
   474→   474→                        type="number"
   475→   475→                        className="col-span-4 md:col-span-2"
   476→   476→                        placeholder="Qty"
   477→   477→                        value={item.quantity}
   478→   478→                        onChange={(e) => updateItem(idx, 'quantity', e.target.value)}
   479→   479→                      />
   480→   480→                      <Input
   481→   481→                        className="col-span-4 md:col-span-2"
