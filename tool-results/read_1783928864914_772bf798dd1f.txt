   160→  new Date(dateStr).toLocaleDateString('en-IN', {
   161→    day: '2-digit',

   166→const emptyForm: OrderFormData = {
   167→  customerId: '',
   168→  brickType: '',
   169→  quantity: '',
   170→  rate: '',

   173→  items: [],
   174→}
   175→
   176→// Default new line item — used when user clicks "Add Item"

   181→// ── Component ───────────────────────────────────────────────────────────────
   182→
   183→export function OrderModule() {
   184→  // State
   185→  const [orders, setOrders] = React.useState<Order[]>([])
   186→  const [search, setSearch] = React.useState('')
   187→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   188→  const [customers, setCustomers] = React.useState<Customer[]>([])
   189→  const [loading, setLoading] = React.useState(true)
   190→
   191→  // Create dialog states
   192→  const [createOpen, setCreateOpen] = React.useState(false)
   193→  const [formData, setFormData] = React.useState<OrderFormData>(emptyForm)
   194→  const [formSubmitting, setFormSubmitting] = React.useState(false)
   195→
   196→  // Edit status dialog states
   197→  const [editStatusOpen, setEditStatusOpen] = React.useState(false)
   198→  const [editTarget, setEditTarget] = React.useState<Order | null>(null)
   199→  const [editStatus, setEditStatus] = React.useState('')
   200→  const [editSubmitting, setEditSubmitting] = React.useState(false)
   201→
   202→  // Excel import
   203→  const [importOpen, setImportOpen] = React.useState(false)
   204→
   205→  // Delete confirmation
   206→  const [deleteTarget, setDeleteTarget] = React.useState<Order | null>(null)
   207→  const [deleting, setDeleting] = React.useState(false)
   208→
   209→  // ── Computed amount ─────────────────────────────────────────────────────
   210→  // If items[] are present, total amount = sum of item amounts.
   211→  // Otherwise fall back to top-level qty * rate.
   212→  const computedAmount = React.useMemo(() => {
   213→    if (formData.items.length > 0) {
   214→      return formData.items.reduce((s, it) => s + (Number(it.amount) || 0), 0)
   215→    }
   216→    const qty = Number(formData.quantity) || 0
   217→    const rate = Number(formData.rate) || 0
   218→    return qty * rate
   219→  }, [formData.quantity, formData.rate, formData.items])
   220→
   221→  // ── Item management ──────────────────────────────────────────────────────
   222→  const updateItem = (idx: number, field: keyof OrderItem, value: string | number) => {
   223→    setFormData((prev) => {
   224→      const items = [...prev.items]
   225→      const item = { ...items[idx] }
   226→      if (field === 'description' || field === 'hsn' || field === 'unit') {
   227→        (item as any)[field] = value as string
   228→      } else {
   229→        (item as any)[field] = Number(value) || 0
   230→      }
   231→      item.amount = item.quantity * item.rate
   232→      items[idx] = item
   233→      return { ...prev, items }
   234→    })
   235→  }
   236→  const addItem = () => setFormData((prev) => ({ ...prev, items: [...prev.items, newItem()] }))
   237→  const removeItem = (idx: number) =>
   238→    setFormData((prev) => ({ ...prev, items: prev.items.filter((_, i) => i !== idx) }))
   239→
   240→  // ── Fetch orders ────────────────────────────────────────────────────────
   241→  const fetchOrders = React.useCallback(async () => {
   242→    setLoading(true)
   243→    try {
   244→      const res = await api.getOrders()
   245→      const orderList = (res.orders as Order[]).sort(
   246→        (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   247→      )
   248→      setOrders(orderList)
   249→    } catch (err) {
   250→      toast({
   251→        title: 'Error',
   252→        description: err instanceof Error ? err.message : 'Failed to fetch orders',
   253→        variant: 'destructive',
   254→      })
   255→    } finally {
   256→      setLoading(false)
   257→    }
   258→  }, [])
   259→
   260→  // ── Fetch customers (for dropdown) ─────────────────────────────────────
   261→  const fetchCustomers = React.useCallback(async () => {
   262→    try {
   263→      const res = await api.getCustomers()
   264→      setCustomers(res.customers as Customer[])
   265→    } catch {
   266→      // Silently fail — customer dropdown will just be empty
   267→    }
   268→  }, [])
   269→
   270→  // Debounced search
   271→  React.useEffect(() => {
   272→    const t = setTimeout(() => setDebouncedSearch(search), 250)
   273→    return () => clearTimeout(t)
   274→  }, [search])
   275→
   276→  // Client-side filter
   277→  const filteredOrders = React.useMemo(() => {
   278→    if (!debouncedSearch.trim()) return orders
   279→    const q = debouncedSearch.toLowerCase()
   280→    return orders.filter((item: any) =>
   281→      ['orderNumber', 'brickType', 'status', 'deliveryDate'].some((f) =>
   282→        String((item as any)[f] ?? '').toLowerCase().includes(q)
   283→      )
   284→    )
   285→  }, [orders, debouncedSearch])
   286→
   287→  React.useEffect(() => {
   288→    fetchOrders()
   289→    fetchCustomers()
   290→  }, [fetchOrders, fetchCustomers])
   291→
   292→  // ── Create dialog handlers ──────────────────────────────────────────────
   293→  const openCreateDialog = () => {
   294→    setFormData(emptyForm)
   295→    setCreateOpen(true)
   296→  }
   297→
   298→  const handleFormChange = (field: keyof OrderFormData, value: string) => {
   299→    setFormData((prev) => ({ ...prev, [field]: value }))
   300→  }
   301→
   302→  const handleCreateSubmit = async () => {
   303→    // Unified empty-form check — show ONE popup instead of cascading errors
   304→    if (isFormEmpty([formData.customerId, formData.brickType, formData.quantity, formData.rate, formData.deliveryDate, formData.status])) {
   305→      toast(showPleaseFillDataToast())
   306→      return
   307→    }
   308→    if (!formData.customerId) {
   309→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   310→      return
   311→    }
   312→    if (!formData.brickType) {
   313→      toast({ title: 'Validation Error', description: 'Please select a brick type', variant: 'destructive' })
   314→      return
   315→    }
   316→    if (!formData.quantity || Number(formData.quantity) <= 0) {
   317→      toast({ title: 'Validation Error', description: 'Quantity must be greater than 0', variant: 'destructive' })
   318→      return
   319→    }
   320→    if (!formData.rate || Number(formData.rate) <= 0) {
   321→      toast({ title: 'Validation Error', description: 'Rate must be greater than 0', variant: 'destructive' })
   322→      return
   323→    }
   324→    if (!formData.deliveryDate) {
   325→      toast({ title: 'Validation Error', description: 'Delivery date is required', variant: 'destructive' })
   326→      return
   327→    }
   328→
   329→    // Either items[] OR brickType+qty+rate must be present
   330→    const hasItems = formData.items.length > 0 && formData.items.some((i) => i.description.trim())
   331→    if (!hasItems && (!formData.brickType || !formData.quantity || !formData.rate)) {
   332→      toast({
   333→        title: 'Validation Error',
   334→        description: 'Add at least one item, OR fill brick type + quantity + rate',
   335→        variant: 'destructive',
   336→      })
   337→      return
   338→    }
   339→
   340→    setFormSubmitting(true)
   341→    try {
   342→      const payload = {
   343→        customerId: formData.customerId,
   344→        brickType: formData.brickType,
   345→        quantity: Number(formData.quantity) || 0,
   346→        rate: Number(formData.rate) || 0,
   347→        amount: computedAmount,
   348→        deliveryDate: formData.deliveryDate,
   349→        status: formData.status,
   350→        items: formData.items.filter((i) => i.description.trim()),
   351→      }
   352→
   353→      await api.createOrder(payload)
   354→      toast({ title: 'Success', description: 'Order created successfully' })
   355→      setCreateOpen(false)
   356→      setFormData(emptyForm)
   357→      fetchOrders()
   358→    } catch (err) {
   359→      toast({
   360→        title: 'Error',
   361→        description: err instanceof Error ? err.message : 'Failed to create order',
   362→        variant: 'destructive',
   363→      })
   364→    } finally {
   365→      setFormSubmitting(false)
   366→    }
   367→  }
   368→
   369→  // ── Edit status handlers ────────────────────────────────────────────────
   370→  const openEditStatusDialog = (order: Order) => {
   371→    setEditTarget(order)
   372→    setEditStatus(order.status)
   373→    setEditStatusOpen(true)
   374→  }
   375→
   376→  const handleEditStatusSubmit = async () => {
   377→    if (!editTarget) return
   378→    setEditSubmitting(true)
   379→    try {
   380→      await api.updateOrder(editTarget.id, { status: editStatus })
   381→      toast({ title: 'Success', description: 'Order status updated successfully' })
   382→      setEditStatusOpen(false)
   383→      setEditTarget(null)
   384→      setEditStatus('')
   385→      fetchOrders()
   386→    } catch (err) {
   387→      toast({
   388→        title: 'Error',
   389→        description: err instanceof Error ? err.message : 'Failed to update order status',
   390→        variant: 'destructive',
   391→      })
   392→    } finally {
   393→      setEditSubmitting(false)
   394→    }
   395→  }
   396→
   397→  // ── Delete handler ──────────────────────────────────────────────────────
   398→  const handleDelete = async () => {
   399→    if (!deleteTarget) return
   400→    setDeleting(true)
   401→    try {
   402→      await api.deleteOrder(deleteTarget.id)
   403→      toast({ title: 'Success', description: 'Order deleted successfully' })
   404→      setDeleteTarget(null)
   405→      fetchOrders()
   406→    } catch (err) {
   407→      toast({
   408→        title: 'Error',
   409→        description: err instanceof Error ? err.message : 'Failed to delete order',
   410→        variant: 'destructive',
   411→      })
   412→    } finally {
   413→      setDeleting(false)
   414→    }
   415→  }
   416→
   417→  // ── Render: Loading skeletons ───────────────────────────────────────────
   418→  const renderSkeletons = () =>
   419→    Array.from({ length: 5 }).map((_, i) => (
   420→      <TableRow key={i}>
   421→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   422→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   423→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   424→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   425→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   426→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   427→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   428→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   429→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   430→      </TableRow>
   431→    ))
   432→
   433→  // ── Render: Status badge ────────────────────────────────────────────────
   434→  const renderStatusBadge = (status: string) => (
   435→    <Badge className={STATUS_COLORS[status] || 'bg-gray-100 text-gray-700 border-gray-200'}>
   436→      {status}
   437→    </Badge>
   438→  )
   439→
   440→  // ── Render: Create Order dialog ─────────────────────────────────────────
   441→  const renderCreateDialog = () => (
   442→    <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   443→      <DialogContent className="max-w-[95vw] sm:max-w-3xl max-h-[90vh] overflow-y-auto">
   444→        <DialogHeader>
   445→          <DialogTitle>Create Order</DialogTitle>
   446→          <DialogDescription>
   447→            Fill in the details to create a new order. You can add multiple items below the brick type.
   448→          </DialogDescription>
   449→        </DialogHeader>
   450→
   451→        <div className="grid gap-4 py-2">
   452→          {/* Customer — SEARCHABLE (replaces old dropdown) */}
   453→          <CustomerSearchInput
   454→            value={formData.customerId}
   455→            onSelect={(c) => handleFormChange('customerId', c.id)}
   456→            onClear={() => handleFormChange('customerId', '')}
   457→            required
   458→            label="Customer"
   459→            placeholder="Type customer name or mobile to search..."
   460→          />
   461→
   462→          {/* Brick Type — now optional since items[] can replace it */}
   463→          <div className="grid gap-2">
   464→            <Label htmlFor="order-brick">
   465→              Brick Type <span className="text-muted-foreground text-xs font-normal">(optional when items are added below)</span>
   466→            </Label>
   467→            <Select
   468→              value={formData.brickType}
   469→              onValueChange={(val) => handleFormChange('brickType', val)}
   470→            >
   471→              <SelectTrigger className="w-full">
   472→                <SelectValue placeholder="Select brick type (or skip if using items)" />
   473→              </SelectTrigger>
   474→              <SelectContent>
   475→                {BRICK_TYPES.map((type) => (
   476→                  <SelectItem key={type} value={type}>
   477→                    {type}
   478→                  </SelectItem>
   479→                ))}
   480→              </SelectContent>
   481→            </Select>
   482→          </div>
   483→
   484→          {/* Items section — multi-line item entry with paper block types */}
   485→          <div className="grid gap-2">
   486→            <div className="flex items-center justify-between">
   487→              <Label>Items <span className="text-muted-foreground text-xs font-normal">(select brick type for each line)</span></Label>
   488→              <Button type="button" size="sm" variant="outline" onClick={addItem}>
   489→                <Plus className="h-3 w-3 mr-1" /> Add Item
   490→              </Button>
   491→            </div>
   492→            <div className="border rounded-md p-2 space-y-2 bg-muted/20">
   493→              {formData.items.length === 0 ? (
   494→                <p className="text-xs text-muted-foreground text-center py-2">
   495→                  No items added. Use "Add Item" to add multiple paper block products, OR skip this section and use brick type + qty + rate above.
   496→                </p>
   497→              ) : (
   498→                <>
   499→                  <div className="hidden md:grid grid-cols-12 gap-2 text-xs font-semibold text-muted-foreground px-1">
   500→                    <div className="col-span-5">Brick Type</div>
   501→                    <div className="col-span-2">Qty</div>
   502→                    <div className="col-span-2">Unit</div>
   503→                    <div className="col-span-2">Rate (₹)</div>
   504→                    <div className="col-span-1">Amount</div>
   505→                  </div>
   506→                  {formData.items.map((item, idx) => (
   507→                    <div key={idx} className="grid grid-cols-12 gap-2 items-center">
   508→                      <Select
   509→                        value={item.description}
   510→                        onValueChange={(val) => updateItem(idx, 'description', val)}
   511→                      >
   512→                        <SelectTrigger className="col-span-12 md:col-span-5 h-9">
   513→                          <SelectValue placeholder="Select brick type" />
   514→                        </SelectTrigger>
   515→                        <SelectContent>
   516→                          {BRICK_TYPES.map((type) => (
   517→                            <SelectItem key={type} value={type}>
   518→                              {type}
   519→                            </SelectItem>
   520→                          ))}
   521→                        </SelectContent>
   522→                      </Select>
   523→                      <Input
   524→                        type="number"
   525→                        className="col-span-4 md:col-span-2"
   526→                        placeholder="Qty"
   527→                        value={item.quantity}
   528→                        onChange={(e) => updateItem(idx, 'quantity', e.target.value)}
   529→                      />
   530→                      <Input
   531→                        className="col-span-4 md:col-span-2"
   532→                        placeholder="Unit"
   533→                        value={item.unit}
   534→                        onChange={(e) => updateItem(idx, 'unit', e.target.value)}
   535→                      />
   536→                      <Input
   537→                        type="number"
   538→                        className="col-span-4 md:col-span-2"
   539→                        placeholder="Rate"
   540→                        value={item.rate}
   541→                        onChange={(e) => updateItem(idx, 'rate', e.target.value)}
   542→                      />
   543→                      <div className="col-span-11 md:col-span-1 flex items-center gap-1">
   544→                        <span className="text-xs font-medium flex-1">₹{(item.amount || 0).toLocaleString('en-IN')}</span>
   545→                        <Button
   546→                          type="button"
   547→                          size="icon"
   548→                          variant="ghost"
   549→                          className="h-7 w-7 text-destructive shrink-0"
   550→                          onClick={() => removeItem(idx)}
   551→                        >
   552→                          <X className="h-3 w-3" />
   553→                        </Button>
   554→                      </div>
   555→                    </div>
   556→                  ))}
   557→                </>
   558→              )}
   559→              {formData.items.length > 0 && (
   560→                <div className="flex justify-end pt-1 border-t text-sm font-medium">
   561→                  Total: ₹{computedAmount.toLocaleString('en-IN')}
   562→                </div>
   563→              )}
   564→            </div>
   565→          </div>
   566→
   567→          {/* Quantity & Rate — kept for backward compat / quick single-item orders */}
   568→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   569→            <div className="grid gap-2">
   570→              <Label htmlFor="order-qty">
   571→                Quantity <span className="text-muted-foreground text-xs font-normal">(if no items)</span>
   572→              </Label>
   573→              <Input
   574→                id="order-qty"
   575→                type="number"
   576→                placeholder="0"
   577→                min="1"
   578→                value={formData.quantity}
   579→                onChange={(e) => handleFormChange('quantity', e.target.value)}
   580→              />
   581→            </div>
   582→            <div className="grid gap-2">
   583→              <Label htmlFor="order-rate">
   584→                Rate (₹) <span className="text-muted-foreground text-xs font-normal">(if no items)</span>
   585→              </Label>
   586→              <div className="relative">
   587→                <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   588→                <Input
   589→                  id="order-rate"
   590→                  type="number"
   591→                  placeholder="0"
   592→                  min="1"
   593→                  className="pl-9"
   594→                  value={formData.rate}
   595→                  onChange={(e) => handleFormChange('rate', e.target.value)}
   596→                />
   597→              </div>
   598→            </div>
   599→          </div>
   600→
   601→          {/* Amount (auto-calculated) */}
   602→          <div className="grid gap-2">
   603→            <Label>Amount (₹)</Label>
   604→            <div className="flex h-9 items-center rounded-md border bg-muted/50 px-3 text-sm font-medium">
   605→              {formatCurrency(computedAmount)}
   606→            </div>
   607→          </div>
   608→
   609→          {/* Delivery Date */}
   610→          <div className="grid gap-2">
   611→            <Label htmlFor="order-date">
   612→              Delivery Date <span className="text-destructive">*</span>
   613→            </Label>
   614→            <Input
   615→              id="order-date"
   616→              type="date"
   617→              value={formData.deliveryDate}
   618→              onChange={(e) => handleFormChange('deliveryDate', e.target.value)}
   619→            />
   620→          </div>
   621→
   622→          {/* Status */}
   623→          <div className="grid gap-2">
   624→            <Label htmlFor="order-status">Status</Label>
   625→            <Select
   626→              value={formData.status}
   627→              onValueChange={(val) => handleFormChange('status', val)}
   628→            >
   629→              <SelectTrigger className="w-full">
   630→                <SelectValue placeholder="Select status" />
   631→              </SelectTrigger>
   632→              <SelectContent>
   633→                {ORDER_STATUSES.map((status) => (
   634→                  <SelectItem key={status} value={status}>
   635→                    {status}
   636→                  </SelectItem>
   637→                ))}
   638→              </SelectContent>
   639→            </Select>
   640→          </div>
   641→        </div>
   642→
   643→        <DialogFooter>
   644→          <Button
   645→            variant="outline"
   646→            onClick={() => setCreateOpen(false)}
   647→            disabled={formSubmitting}
   648→          >
   649→            Cancel
   650→          </Button>
   651→          <Button
   652→            onClick={handleCreateSubmit}
   653→            disabled={formSubmitting}
   654→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   655→          >
   656→            {formSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   657→            Create Order
   658→          </Button>
   659→        </DialogFooter>
   660→      </DialogContent>
   661→    </Dialog>
   662→  )
   663→
   664→  // ── Render: Edit Status dialog ──────────────────────────────────────────
   665→  const renderEditStatusDialog = () => (
   666→    <Dialog open={editStatusOpen} onOpenChange={setEditStatusOpen}>
   667→      <DialogContent className="max-w-[95vw] sm:max-w-md">
   668→        <DialogHeader>
   669→          <DialogTitle>Update Order Status</DialogTitle>
   670→          <DialogDescription>
   671→            Change the status for order <strong>{editTarget?.orderNumber}</strong>
   672→          </DialogDescription>
   673→        </DialogHeader>
   674→
   675→        <div className="grid gap-4 py-2">
   676→          <div className="grid gap-2">
   677→            <Label>Current Status</Label>
   678→            <div>{editTarget ? renderStatusBadge(editTarget.status) : null}</div>
   679→          </div>
   680→          <div className="grid gap-2">
   681→            <Label htmlFor="edit-status">New Status</Label>
   682→            <Select value={editStatus} onValueChange={setEditStatus}>
   683→              <SelectTrigger className="w-full">
   684→                <SelectValue placeholder="Select status" />
   685→              </SelectTrigger>
   686→              <SelectContent>
   687→                {ORDER_STATUSES.map((status) => (
   688→                  <SelectItem key={status} value={status}>
   689→                    {status}
   690→                  </SelectItem>
   691→                ))}
   692→              </SelectContent>
   693→            </Select>
   694→          </div>
   695→        </div>
   696→
   697→        <DialogFooter>
   698→          <Button
   699→            variant="outline"
   700→            onClick={() => setEditStatusOpen(false)}
   701→            disabled={editSubmitting}
   702→          >
   703→            Cancel
   704→          </Button>
   705→          <Button
   706→            onClick={handleEditStatusSubmit}
   707→            disabled={editSubmitting}
   708→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   709→          >
   710→            {editSubmitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   711→            Update Status
   712→          </Button>
   713→        </DialogFooter>
   714→      </DialogContent>
   715→    </Dialog>
   716→  )
   717→
   718→  // ── Render: Delete confirmation ─────────────────────────────────────────
   719→  const renderDeleteDialog = () => (
   720→    <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   721→      <AlertDialogContent>
   722→        <AlertDialogHeader>
   723→          <AlertDialogTitle>Delete Order</AlertDialogTitle>
   724→          <AlertDialogDescription>
   725→            Are you sure you want to delete order <strong>{deleteTarget?.orderNumber}</strong>? This
   726→            action cannot be undone.
   727→          </AlertDialogDescription>
   728→        </AlertDialogHeader>
   729→        <AlertDialogFooter>
   730→          <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   731→          <AlertDialogAction
   732→            onClick={handleDelete}
   733→            disabled={deleting}
   734→            className="bg-destructive text-white hover:bg-destructive/90"
   735→          >
   736→            {deleting && <Loader2 className="mr-2 size-4 animate-spin" />}
   737→            Delete
   738→          </AlertDialogAction>
   739→        </AlertDialogFooter>
   740→      </AlertDialogContent>
   741→    </AlertDialog>
   742→  )
   743→
   744→  // ── Render: Main ────────────────────────────────────────────────────────
   745→  return (
   746→    <div className="space-y-6">
   747→      {/* Header */}
   748→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   749→        <div className="flex items-center gap-3">
   750→          <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 text-emerald-700">
   751→            <ClipboardList className="size-5" />
   752→          </div>
   753→          <div>
   754→            <h2 className="text-2xl font-bold tracking-tight">Order Management</h2>
   755→            <p className="text-sm text-muted-foreground">
   756→              Track and manage customer orders
   757→            </p>
   758→          </div>
   759→        </div>
   760→        <div className="flex flex-col gap-2 sm:flex-row sm:w-auto">
   761→          <Button
   762→            variant="outline"
   763→            onClick={() => setImportOpen(true)}
   764→            className="w-full sm:w-auto"
   765→          >
   766→            <Upload className="size-4 mr-2" />
   767→            Import Excel
   768→          </Button>
   769→          <Button
   770→            onClick={openCreateDialog}
   771→            className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   772→          >
   773→            <Plus className="size-4" />
   774→            Create Order
   775→          </Button>
   776→        </div>
   777→      </div>
   778→
   779→      {/* Search */}
   780→      <Card>
   781→        <CardContent className="pt-6">
   782→          <div className="relative">
   783→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   784→            <Input
   785→              placeholder="Search across all fields (date, name, remarks, etc.)..."
   786→              value={search}
   787→              onChange={(e) => setSearch(e.target.value)}
   788→              className="pl-9"
   789→            />
   790→          </div>
   791→        </CardContent>
   792→      </Card>
   793→
   794→      {/* Table */}
   795→      <Card>
   796→        <CardHeader>
   797→          <CardTitle className="flex items-center justify-between gap-2 flex-wrap">
   798→            <span>Orders</span>
   799→            <Badge variant="secondary" className="bg-emerald-100 text-emerald-700 border-emerald-200">
   800→              {filteredOrders.length} of {orders.length} record{orders.length !== 1 ? 's' : ''}
   801→            </Badge>
   802→          </CardTitle>
   803→        </CardHeader>
   804→        <CardContent>
   805→          <div className="max-h-[60vh] overflow-auto rounded-md border">
   806→            <Table>
   807→              <TableHeader className="sticky top-0 bg-background z-10">
   808→                <TableRow>
   809→                  <TableHead>Order No.</TableHead>
   810→                  <TableHead>Customer</TableHead>
   811→                  <TableHead className="hidden sm:table-cell">Brick Type</TableHead>
   812→                  <TableHead className="text-right">Quantity</TableHead>
   813→                  <TableHead className="hidden sm:table-cell text-right">Rate (₹)</TableHead>
   814→                  <TableHead className="text-right">Amount (₹)</TableHead>
   815→                  <TableHead className="hidden md:table-cell">Delivery Date</TableHead>
   816→                  <TableHead>Status</TableHead>
   817→                  <TableHead className="text-right">Actions</TableHead>
   818→                </TableRow>
   819→              </TableHeader>
   820→              <TableBody>
   821→                {loading ? (
   822→                  renderSkeletons()
   823→                ) : filteredOrders.length === 0 ? (
   824→                  <TableRow>
   825→                    <TableCell colSpan={9} className="h-32 text-center text-muted-foreground">
   826→                      No orders yet. Click &quot;Create Order&quot; to get started.
   827→                    </TableCell>
   828→                  </TableRow>
   829→                ) : (
   830→                  filteredOrders.map((order) => (
   831→                    <TableRow key={order.id}>
   832→                      <TableCell className="font-medium whitespace-nowrap">
   833→                        {order.orderNumber}
   834→                      </TableCell>
   835→                      <TableCell>{order.customer?.name || '—'}</TableCell>
   836→                      <TableCell className="hidden sm:table-cell">
   837→                        <Badge variant="outline">{order.brickType}</Badge>
   838→                        {order.items && order.items.length > 0 && (
   839→                          <Badge variant="secondary" className="ml-1 text-[10px]">
   840→                            {order.items.length} item{order.items.length > 1 ? 's' : ''}
   841→                          </Badge>
   842→                        )}
   843→                      </TableCell>
   844→                      <TableCell className="text-right">
   845→                        {new Intl.NumberFormat('en-IN').format(order.quantity)}
   846→                      </TableCell>
   847→                      <TableCell className="hidden sm:table-cell text-right whitespace-nowrap">
   848→                        {formatCurrency(order.rate)}
   849→                      </TableCell>
   850→                      <TableCell className="text-right font-medium whitespace-nowrap">
   851→                        {formatCurrency(order.amount)}
   852→                      </TableCell>
   853→                      <TableCell className="hidden md:table-cell whitespace-nowrap">
   854→                        {formatDate(order.deliveryDate)}
   855→                      </TableCell>
   856→                      <TableCell>{renderStatusBadge(order.status)}</TableCell>
   857→                      <TableCell className="text-right">
   858→                        <div className="flex items-center justify-end gap-1">
   859→                          <Button
   860→                            variant="ghost"
   861→                            size="icon"
   862→                            onClick={() => openEditStatusDialog(order)}
   863→                            title="Edit Status"
   864→                            className="text-sky-600 hover:text-sky-700 hover:bg-sky-50"
   865→                          >
   866→                            <Pencil className="size-4" />
   867→                          </Button>
   868→                          <Button
   869→                            variant="ghost"
   870→                            size="icon"
   871→                            onClick={() => setDeleteTarget(order)}
   872→                            title="Delete Order"
   873→                            className="text-destructive hover:text-destructive hover:bg-destructive/10"
   874→                          >
   875→                            <Trash2 className="size-4" />
   876→                          </Button>
   877→                        </div>
   878→                      </TableCell>
   879→                    </TableRow>
   880→                  ))
   881→                )}
   882→              </TableBody>
   883→            </Table>
   884→          </div>
   885→        </CardContent>
   886→      </Card>
   887→
   888→      {/* Dialogs */}
   889→      {renderCreateDialog()}
   890→      {renderEditStatusDialog()}
   891→      {renderDeleteDialog()}
   892→
   893→      <ExcelImport module="orders" open={importOpen} onClose={() => setImportOpen(false)} onSuccess={fetchOrders} />
   894→    </div>
   895→  )
   896→}
