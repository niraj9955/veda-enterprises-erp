   247→   247→// Normalize a string for comparison: lowercase, remove special chars, spaces
   248→   248→function normalize(s: string): string {
   249→   249→  return s.toLowerCase().replace(/[^a-z0-9\u0900-\u097F]/g, '')
   250→   250→}

   252→   252→// Smart auto-detect mapping between Excel columns and module fields
   253→   253→function autoMapColumns(excelColumns: string[], moduleFields: { key: string; label: string; required: boolean; aliases: string[] }[]): Record<string, string> {
   254→   254→  const mapping: Record<string, string> = {} // excelColumn -> fieldKey
   255→   255→  const normalizedExcel = excelColumns.map((col) => ({ original: col, normalized: normalize(col) }))
   256→   256→
   257→   257→  for (const field of moduleFields) {
   258→   258→    let bestMatch = ''
   259→   259→    let bestScore = 0
   260→   260→
   261→   261→    for (const excelCol of normalizedExcel) {
   262→   262→      // Check exact match with key
   263→   263→      if (excelCol.normalized === normalize(field.key)) {
   264→   264→        bestMatch = excelCol.original
   265→   265→        bestScore = 100
   266→   266→        break
   267→   267→      }
   268→   268→
   269→   269→      // Check aliases
   270→   270→      for (const alias of field.aliases) {
   271→   271→        const normAlias = normalize(alias)
   272→   272→        if (excelCol.normalized === normAlias) {
   273→   273→          if (100 > bestScore) {
   274→   274→            bestMatch = excelCol.original
   275→   275→            bestScore = 100
   276→   276→          }
   277→   277→          break
   278→   278→        }
   279→   279→      }
   280→   280→
   281→   281→      if (bestScore >= 100) break
   282→   282→
   283→   283→      // Fuzzy match: check if alias is contained in column name or vice versa
   284→   284→      for (const alias of field.aliases) {
   285→   285→        const normAlias = normalize(alias)
   286→   286→        if (normAlias.length > 2 && (excelCol.normalized.includes(normAlias) || normAlias.includes(excelCol.normalized))) {
   287→   287→          const score = Math.min(normAlias.length, excelCol.normalized.length) / Math.max(normAlias.length, excelCol.normalized.length) * 80
   288→   288→          if (score > bestScore) {
   289→   289→            bestMatch = excelCol.original
   290→   290→            bestScore = score
   291→   291→          }
   292→   292→        }
   293→   293→      }
   294→   294→    }
   295→   295→
   296→   296→    if (bestMatch && bestScore >= 50) {
   297→   297→      mapping[bestMatch] = field.key
   298→   298→    }
   299→   299→  }
   300→   300→
   301→   301→  return mapping
   302→   302→}

   304→   304→// Auto-detect the module from Excel column headers
   305→   305→function autoDetectModule(columns: string[]): string | null {
   306→   306→  const normalized = columns.map(normalize)
   307→   307→  const scores: Record<string, number> = {}
   308→   308→
   309→   309→  for (const [moduleKey, template] of Object.entries(moduleTemplates)) {
   310→   310→    scores[moduleKey] = 0
   311→   311→    for (const field of template.fields) {
   312→   312→      for (const alias of field.aliases) {
   313→   313→        if (normalized.some((col) => col === normalize(alias) || (normalize(alias).length > 3 && col.includes(normalize(alias))))) {
   314→   314→          scores[moduleKey] += field.required ? 3 : 1
   315→   315→        }
   316→   316→      }
   317→   317→    }
   318→   318→  }
   319→   319→
   320→   320→  const bestModule = Object.entries(scores).sort((a, b) => b[1] - a[1])[0]
   321→   321→  return bestModule && bestModule[1] >= 3 ? bestModule[0] : null
   322→   322→}

   324→   324→// Transform data using mapping, also handle customer name -> ID resolution
   325→   325→function transformRow(
   326→   326→  row: Record<string, unknown>,
   327→   327→  mapping: Record<string, string>,
   328→   328→  customers?: { id: string; name: string }[]
   329→   329→): Record<string, unknown> {
   330→   330→  const result: Record<string, unknown> = {}
   331→   331→
   332→   332→  for (const [excelCol, fieldKey] of Object.entries(mapping)) {
   333→   333→    let value = row[excelCol]
   334→   334→
   335→   335→    // Handle customer name -> ID resolution
   336→   336→    if ((fieldKey === 'customerName') && customers && customers.length > 0) {
   337→   337→      const nameStr = String(value || '').trim().toLowerCase()
   338→   338→      const matched = customers.find((c) => c.name.toLowerCase() === nameStr)
   339→   339→      if (matched) {
   340→   340→        result['customerId'] = matched.id
   341→   341→      } else {
   342→   342→        result['customerId'] = String(value || '') // fallback: keep as-is for error
   343→   343→      }
   344→   344→      continue
   345→   345→    }
   346→   346→
   347→   347→    // Handle numeric fields
   348→   348→    if ([
   349→   349→      'quantityProduced', 'quantity', 'creditLimit', 'amount', 'rate',
   350→   350→      'zigZagGrey80', 'zigZagRed80', 'zigZagYellow80',
   351→   351→      'zigZagGrey60', 'zigZagRed60', 'zigZagYellow60',
   352→   352→      'curveStone', 'chequreTile', 'transportationCharge',
   353→   353→      'cement', 'dumbleGrey80', 'dumbleRed80', 'dumbleYellow80',
   354→   354→      'quantityTon', 'totalAmount', 'paidAmount', 'remainingAmount',
   355→   355→      'gst',
   356→   356→    ].includes(fieldKey)) {
   357→   357→      const numVal = Number(String(value || '').replace(/[^0-9.-]/g, '') || 0)
   358→   358→      result[fieldKey] = isNaN(numVal) ? 0 : numVal
   359→   359→      continue
   360→   360→    }
   361→   361→
   362→   362→    // Handle date fields - try to parse various formats
   363→   363→    if (['date', 'deliveryDate'].includes(fieldKey)) {
   364→   364→      // Excel may return dates as numeric serial numbers (e.g. 46178).
   365→   365→      // Detect that case and convert to YYYY-MM-DD before string parsing.
   366→   366→      if (typeof value === 'number' && Number.isFinite(value)) {
   367→   367→        // Excel serial date: days since 1899-12-30
   368→   368→        const ms = Math.round((value - 25569) * 86400 * 1000)
   369→   369→        const d = new Date(ms)
   370→   370→        if (!isNaN(d.getTime())) {
   371→   371→          result[fieldKey] = d.toISOString().split('T')[0]
   372→   372→          continue
   373→   373→        }
   374→   374→      }
   375→   375→      // Excel may also return a Date object if cellDates:true was used.
   376→   376→      if (value instanceof Date && !isNaN(value.getTime())) {
   377→   377→        result[fieldKey] = value.toISOString().split('T')[0]
   378→   378→        continue
   379→   379→      }
   380→   380→      result[fieldKey] = parseDate(String(value || ''))
   381→   381→      continue
   382→   382→    }
   383→   383→
   384→   384→    // Handle shift field - normalize
   385→   385→    if (fieldKey === 'shift') {
   386→   386→      const shiftStr = String(value || 'Morning').trim()
   387→   387→      const lower = shiftStr.toLowerCase()
   388→   388→      if (lower.includes('morn') || lower.includes('सुबह') || lower === '1' || lower === 'morning') {
   389→   389→        result[fieldKey] = 'Morning'
   390→   390→      } else if (lower.includes('eve') || lower.includes('शाम') || lower === '2' || lower === 'evening') {
   391→   391→        result[fieldKey] = 'Evening'
   392→   392→      } else if (lower.includes('night') || lower.includes('रात') || lower === '3' || lower === 'night') {
   393→   393→        result[fieldKey] = 'Night'
   394→   394→      } else {
   395→   395→        result[fieldKey] = shiftStr
   396→   396→      }
   397→   397→      continue
   398→   398→    }
   399→   399→
   400→   400→    // Handle payment type
   401→   401→    if (fieldKey === 'paymentType') {
   402→   402→      const ptStr = String(value || 'Cash').trim()
   403→   403→      const lower = ptStr.toLowerCase()
   404→   404→      if (lower.includes('cash') || lower.includes('नकद')) result[fieldKey] = 'Cash'
   405→   405→      else if (lower.includes('upi') || lower.includes('online')) result[fieldKey] = 'UPI'
   406→   406→      else if (lower.includes('bank') || lower.includes('transfer') || lower.includes('neft') || lower.includes('rtgs')) result[fieldKey] = 'Bank Transfer'
   407→   407→      else result[fieldKey] = ptStr
   408→   408→      continue
   409→   409→    }
   410→   410→
   411→   411→    // Handle expense category
   412→   412→    if (fieldKey === 'category') {
   413→   413→      const catStr = String(value || '').trim()
   414→   414→      const lower = catStr.toLowerCase()
   415→   415→      if (lower.includes('labour') || lower.includes('labor') || lower.includes('मजदूर') || lower.includes('wage')) result[fieldKey] = 'Labour'
   416→   416→      else if (lower.includes('coal') || lower.includes('कोयला')) result[fieldKey] = 'Coal'
   417→   417→      else if (lower.includes('diesel') || lower.includes('डीज़ल') || lower.includes('fuel')) result[fieldKey] = 'Diesel'
   418→   418→      else if (lower.includes('maint') || lower.includes('रखरखाव') || lower.includes('repair')) result[fieldKey] = 'Maintenance'
   419→   419→      else if (lower.includes('elect') || lower.includes('बिजली') || lower.includes('power')) result[fieldKey] = 'Electricity'
   420→   420→      else result[fieldKey] = catStr
   421→   421→      continue
   422→   422→    }
   423→   423→
   424→   424→    // Handle status
   425→   425→    if (fieldKey === 'status') {
   426→   426→      const stStr = String(value || 'Pending').trim()
   427→   427→      const lower = stStr.toLowerCase()
   428→   428→      if (lower.includes('pend') || lower.includes('लंबित')) result[fieldKey] = 'Pending'
   429→   429→      else if (lower.includes('process') || lower.includes('चल')) result[fieldKey] = 'Processing'
   430→   430→      else if (lower.includes('deliver') || lower.includes('भेजा') || lower.includes('complete')) result[fieldKey] = 'Delivered'
   431→   431→      else if (lower.includes('cancel') || lower.includes('रद्द')) result[fieldKey] = 'Cancelled'
   432→   432→      else result[fieldKey] = stStr
   433→   433→      continue
   434→   434→    }
   435→   435→
   436→   436→    result[fieldKey] = String(value || '')
   437→   437→  }
   438→   438→
   439→   439→  // For orders/dispatch: calculate amount if not mapped
   440→   440→  if (mapping) {
   441→   441→    const targetModule = Object.values(mapping)
   442→   442→    if (targetModule.includes('rate') && targetModule.includes('quantity') && !targetModule.includes('amount')) {
   443→   443→      const qty = Number(result.quantity || 0)
   444→   444→      const rate = Number(result.rate || 0)
   445→   445→      result.amount = qty * rate
   446→   446→    }
   447→   447→  }
   448→   448→
   449→   449→  return result
   450→   450→}

   452→   452→// Parse various date formats to YYYY-MM-DD
   453→   453→function parseDate(value: string): string {
   454→   454→  if (!value || value.trim() === '') return new Date().toISOString().split('T')[0]
   455→   455→
   456→   456→  const trimmed = value.trim()
   457→   457→
   458→   458→  // Already in YYYY-MM-DD format
   459→   459→  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed
   460→   460→
   461→   461→  // DD-MM-YYYY or DD/MM/YYYY
   462→   462→  const dmyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   463→   463→  if (dmyMatch) {
   464→   464→    const [, d, m, y] = dmyMatch
   465→   465→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   466→   466→  }
   467→   467→
   468→   468→  // DD-MM-YY or DD/MM/YY
   469→   469→  const dmyShortMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2})$/)
   470→   470→  if (dmyShortMatch) {
   471→   471→    const [, d, m, y] = dmyShortMatch
   472→   472→    return `20${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   473→   473→  }
   474→   474→
   475→   475→  // MM/DD/YYYY (US format) - only if first number > 12
   476→   476→  const mdyMatch = trimmed.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{4})$/)
   477→   477→  if (mdyMatch) {
   478→   478→    const [, m, d, y] = mdyMatch
   479→   479→    if (Number(m) > 12) {
   480→   480→      return `${y}-${d.padStart(2, '0')}-${m.padStart(2, '0')}`
   481→   481→    }
   482→   482→    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`
   483→   483→  }
   484→   484→
   485→   485→  // Try native Date parsing as fallback
   486→   486→  try {
   487→   487→    const date = new Date(trimmed)
   488→   488→    if (!isNaN(date.getTime())) {
   489→   489→      return date.toISOString().split('T')[0]
   490→   490→    }
   491→   491→  } catch {}
   492→   492→
   493→   493→  return trimmed
   494→   494→}
