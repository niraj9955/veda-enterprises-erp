   104→  orderId: string

   113→
   114→const BRICK_TYPES = ['Red Brick', 'Fly Ash Brick', 'Cement Brick', 'Hollow Block']
   115→
   116→const emptyForm: DispatchFormData = {
   117→  customerId: '',
   118→  orderId: '',
   119→  brickType: '',
   120→  quantity: '',
   121→  truckNumber: '',
   122→  driverName: '',
   123→  date: '',

   126→// ── Helpers ─────────────────────────────────────────────────────────────────
   127→
   128→const formatDate = (dateStr: string): string => {
   129→  if (!dateStr) return '—'
   130→  const date = new Date(dateStr + 'T00:00:00')
   131→  return date.toLocaleDateString('en-IN', {
   132→    day: '2-digit',
   133→    month: 'short',
   134→    year: 'numeric',
   135→  })
   136→}

   138→// ── Component ───────────────────────────────────────────────────────────────
   139→
   140→export function DispatchModule() {
   141→  // ── State ─────────────────────────────────────────────────────────────
   142→  const [dispatches, setDispatches] = React.useState<Dispatch[]>([])
   143→  const [loading, setLoading] = React.useState(true)
   144→
   145→  // Create dialog
   146→  const [createOpen, setCreateOpen] = React.useState(false)
   147→  const [formData, setFormData] = React.useState<DispatchFormData>(emptyForm)
   148→  const [submitting, setSubmitting] = React.useState(false)
   149→
   150→  // Dropdown data
   151→  const [customers, setCustomers] = React.useState<Customer[]>([])
   152→  const [orders, setOrders] = React.useState<Order[]>([])
   153→  const [dropdownsLoading, setDropdownsLoading] = React.useState(false)
   154→
   155→  // Challan dialog
   156→  const [challanDispatch, setChallanDispatch] = React.useState<Dispatch | null>(null)
   157→
   158→  // Delete confirmation
   159→  const [deleteTarget, setDeleteTarget] = React.useState<Dispatch | null>(null)
   160→  const [deleting, setDeleting] = React.useState(false)
   161→
   162→  // ── Fetch dispatches ──────────────────────────────────────────────────
   163→  const fetchDispatches = React.useCallback(async () => {
   164→    setLoading(true)
   165→    try {
   166→      const res = await api.getDispatches()
   167→      setDispatches(res.dispatches as Dispatch[])
   168→    } catch (err) {
   169→      toast({
   170→        title: 'Error',
   171→        description: err instanceof Error ? err.message : 'Failed to fetch dispatches',
   172→        variant: 'destructive',
   173→      })
   174→    } finally {
   175→      setLoading(false)
   176→    }
   177→  }, [])
   178→
   179→  React.useEffect(() => {
   180→    fetchDispatches()
   181→  }, [fetchDispatches])
   182→
   183→  // ── Fetch dropdown data when create dialog opens ──────────────────────
   184→  React.useEffect(() => {
   185→    if (!createOpen) return
   186→
   187→    const fetchDropdownData = async () => {
   188→      setDropdownsLoading(true)
   189→      try {
   190→        const [custRes, ordRes] = await Promise.all([
   191→          api.getCustomers(),
   192→          api.getOrders(),
   193→        ])
   194→        setCustomers(custRes.customers as Customer[])
   195→        setOrders(ordRes.orders as Order[])
   196→      } catch (err) {
   197→        toast({
   198→          title: 'Error',
   199→          description: 'Failed to load dropdown data',
   200→          variant: 'destructive',
   201→        })
   202→      } finally {
   203→        setDropdownsLoading(false)
   204→      }
   205→    }
   206→
   207→    fetchDropdownData()
   208→  }, [createOpen])
   209→
   210→  // ── Filter orders by selected customer ────────────────────────────────
   211→  const pendingOrdersForCustomer = React.useMemo(() => {
   212→    if (!formData.customerId) return []
   213→    return orders.filter(
   214→      (o) =>
   215→        o.customerId === formData.customerId &&
   216→        (o.status === 'Pending' || o.status === 'Processing')
   217→    )
   218→  }, [orders, formData.customerId])
   219→
   220→  // ── Auto-fill brick type when order is selected ───────────────────────
   221→  const handleOrderChange = (orderId: string) => {
   222→    setFormData((prev) => {
   223→      if (!orderId) return { ...prev, orderId: '' }
   224→      const selectedOrder = orders.find((o) => o.id === orderId)
   225→      return {
   226→        ...prev,
   227→        orderId,
   228→        brickType: selectedOrder?.brickType || prev.brickType,
   229→        quantity: selectedOrder ? String(selectedOrder.quantity) : prev.quantity,
   230→      }
   231→    })
   232→  }
   233→
   234→  // ── Create dispatch ───────────────────────────────────────────────────
   235→  const handleCreate = async () => {
   236→    if (!formData.customerId) {
   237→      toast({ title: 'Validation Error', description: 'Please select a customer', variant: 'destructive' })
   238→      return
   239→    }
   240→    if (!formData.brickType) {
   241→      toast({ title: 'Validation Error', description: 'Please select a brick type', variant: 'destructive' })
   242→      return
   243→    }
   244→    if (!formData.quantity || Number(formData.quantity) <= 0) {
   245→      toast({ title: 'Validation Error', description: 'Please enter a valid quantity', variant: 'destructive' })
   246→      return
   247→    }
   248→    if (!formData.truckNumber.trim()) {
   249→      toast({ title: 'Validation Error', description: 'Please enter the truck number', variant: 'destructive' })
   250→      return
   251→    }
   252→    if (!formData.driverName.trim()) {
   253→      toast({ title: 'Validation Error', description: 'Please enter the driver name', variant: 'destructive' })
   254→      return
   255→    }
   256→    if (!formData.date) {
   257→      toast({ title: 'Validation Error', description: 'Please select a date', variant: 'destructive' })
   258→      return
   259→    }
   260→
   261→    setSubmitting(true)
   262→    try {
   263→      const payload = {
   264→        customerId: formData.customerId,
   265→        orderId: formData.orderId || null,
   266→        brickType: formData.brickType,
   267→        quantity: Number(formData.quantity),
   268→        truckNumber: formData.truckNumber.trim(),
   269→        driverName: formData.driverName.trim(),
   270→        date: formData.date,
   271→      }
   272→
   273→      await api.createDispatch(payload)
   274→      toast({ title: 'Success', description: 'Dispatch created successfully' })
   275→      setCreateOpen(false)
   276→      setFormData(emptyForm)
   277→      fetchDispatches()
   278→    } catch (err) {
   279→      toast({
   280→        title: 'Error',
   281→        description: err instanceof Error ? err.message : 'Failed to create dispatch',
   282→        variant: 'destructive',
   283→      })
   284→    } finally {
   285→      setSubmitting(false)
   286→    }
   287→  }
   288→
   289→  // ── Delete dispatch ───────────────────────────────────────────────────
   290→  const handleDelete = async () => {
   291→    if (!deleteTarget) return
   292→    setDeleting(true)
   293→    try {
   294→      await api.deleteDispatch(deleteTarget.id)
   295→      toast({ title: 'Success', description: 'Dispatch deleted successfully' })
   296→      setDeleteTarget(null)
   297→      fetchDispatches()
   298→    } catch (err) {
   299→      toast({
   300→        title: 'Error',
   301→        description: err instanceof Error ? err.message : 'Failed to delete dispatch',
   302→        variant: 'destructive',
   303→      })
   304→    } finally {
   305→      setDeleting(false)
   306→    }
   307→  }
   308→
   309→  // ── Open create dialog ────────────────────────────────────────────────
   310→  const openCreateDialog = () => {
   311→    setFormData(emptyForm)
   312→    setCreateOpen(true)
   313→  }
   314→
   315→  // ── Print challan ─────────────────────────────────────────────────────
   316→  const handlePrint = () => {
   317→    window.print()
   318→  }
   319→
   320→  // ── Render: Loading skeletons ─────────────────────────────────────────
   321→  const renderSkeletons = () =>
   322→    Array.from({ length: 5 }).map((_, i) => (
   323→      <TableRow key={i}>
   324→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   325→        <TableCell><Skeleton className="h-4 w-28" /></TableCell>
   326→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   327→        <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   328→        <TableCell><Skeleton className="h-5 w-20" /></TableCell>
   329→        <TableCell><Skeleton className="h-4 w-24" /></TableCell>
   330→        <TableCell><Skeleton className="h-4 w-20" /></TableCell>
   331→        <TableCell><Skeleton className="h-8 w-20" /></TableCell>
   332→      </TableRow>
   333→    ))
   334→
   335→  // ── Render: Create dispatch dialog ────────────────────────────────────
   336→  const renderCreateDialog = () => (
   337→    <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   338→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   339→        <DialogHeader>
   340→          <DialogTitle className="flex items-center gap-2">
   341→            <Truck className="size-5 text-emerald-600" />
   342→            Create Dispatch
   343→          </DialogTitle>
   344→          <DialogDescription>
   345→            Fill in the details to create a new dispatch entry.
   346→          </DialogDescription>
   347→        </DialogHeader>
   348→
   349→        <div className="grid gap-4 py-2">
   350→          {/* Customer */}
   351→          <div className="grid gap-2">
   352→            <Label>
   353→              Customer <span className="text-destructive">*</span>
   354→            </Label>
   355→            <Select
   356→              value={formData.customerId}
   357→              onValueChange={(val) =>
   358→                setFormData((prev) => ({ ...prev, customerId: val, orderId: '', brickType: '', quantity: '' }))
   359→              }
   360→              disabled={dropdownsLoading}
   361→            >
   362→              <SelectTrigger className="w-full">
   363→                <SelectValue placeholder={dropdownsLoading ? 'Loading...' : 'Select customer'} />
   364→              </SelectTrigger>
   365→              <SelectContent>
   366→                {customers.map((c) => (
   367→                  <SelectItem key={c.id} value={c.id}>
   368→                    {c.name} — {c.mobile}
   369→                  </SelectItem>
   370→                ))}
   371→              </SelectContent>
   372→            </Select>
   373→          </div>
   374→
   375→          {/* Order (optional) */}
   376→          <div className="grid gap-2">
   377→            <Label>
   378→              Order <span className="text-muted-foreground text-xs font-normal">(optional)</span>
   379→            </Label>
   380→            <Select
   381→              value={formData.orderId}
   382→              onValueChange={handleOrderChange}
   383→              disabled={!formData.customerId || pendingOrdersForCustomer.length === 0}
   384→            >
   385→              <SelectTrigger className="w-full">
   386→                <SelectValue
   387→                  placeholder={
   388→                    !formData.customerId
   389→                      ? 'Select a customer first'
   390→                      : pendingOrdersForCustomer.length === 0
   391→                        ? 'No pending orders'
   392→                        : 'Select order'
   393→                  }
   394→                />
   395→              </SelectTrigger>
   396→              <SelectContent>
   397→                {pendingOrdersForCustomer.map((o) => (
   398→                  <SelectItem key={o.id} value={o.id}>
   399→                    {o.orderNumber} — {o.brickType} ({o.quantity.toLocaleString()})
   400→                  </SelectItem>
   401→                ))}
   402→              </SelectContent>
   403→            </Select>
   404→          </div>
   405→
   406→          {/* Brick Type */}
   407→          <div className="grid gap-2">
   408→            <Label>
   409→              Brick Type <span className="text-destructive">*</span>
   410→            </Label>
   411→            <Select
   412→              value={formData.brickType}
   413→              onValueChange={(val) => setFormData((prev) => ({ ...prev, brickType: val }))}
   414→            >
   415→              <SelectTrigger className="w-full">
   416→                <SelectValue placeholder="Select brick type" />
   417→              </SelectTrigger>
   418→              <SelectContent>
   419→                {BRICK_TYPES.map((type) => (
   420→                  <SelectItem key={type} value={type}>
   421→                    {type}
   422→                  </SelectItem>
   423→                ))}
   424→              </SelectContent>
   425→            </Select>
   426→          </div>
   427→
   428→          {/* Quantity */}
   429→          <div className="grid gap-2">
   430→            <Label htmlFor="dispatch-quantity">
   431→              Quantity <span className="text-destructive">*</span>
   432→            </Label>
   433→            <Input
   434→              id="dispatch-quantity"
   435→              type="number"
   436→              min="1"
   437→              placeholder="Enter quantity"
   438→              value={formData.quantity}
   439→              onChange={(e) => setFormData((prev) => ({ ...prev, quantity: e.target.value }))}
   440→            />
   441→          </div>
   442→
   443→          {/* Truck Number */}
   444→          <div className="grid gap-2">
   445→            <Label htmlFor="dispatch-truck">
   446→              Truck Number <span className="text-destructive">*</span>
   447→            </Label>
   448→            <Input
   449→              id="dispatch-truck"
   450→              placeholder="e.g. RJ-14-AB-1234"
   451→              value={formData.truckNumber}
   452→              onChange={(e) => setFormData((prev) => ({ ...prev, truckNumber: e.target.value }))}
   453→            />
   454→          </div>
   455→
   456→          {/* Driver Name */}
   457→          <div className="grid gap-2">
   458→            <Label htmlFor="dispatch-driver">
   459→              Driver Name <span className="text-destructive">*</span>
   460→            </Label>
   461→            <Input
   462→              id="dispatch-driver"
   463→              placeholder="Enter driver name"
   464→              value={formData.driverName}
   465→              onChange={(e) => setFormData((prev) => ({ ...prev, driverName: e.target.value }))}
   466→            />
   467→          </div>
   468→
   469→          {/* Date */}
   470→          <div className="grid gap-2">
   471→            <Label htmlFor="dispatch-date">
   472→              Date <span className="text-destructive">*</span>
   473→            </Label>
   474→            <Input
   475→              id="dispatch-date"
   476→              type="date"
   477→              value={formData.date}
   478→              onChange={(e) => setFormData((prev) => ({ ...prev, date: e.target.value }))}
   479→            />
   480→          </div>
   481→        </div>
   482→
   483→        <DialogFooter>
   484→          <Button
   485→            variant="outline"
   486→            onClick={() => setCreateOpen(false)}
   487→            disabled={submitting}
   488→          >
   489→            Cancel
   490→          </Button>
   491→          <Button
   492→            onClick={handleCreate}
   493→            disabled={submitting}
   494→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   495→          >
   496→            {submitting && <Loader2 className="mr-2 size-4 animate-spin" />}
   497→            Create Dispatch
   498→          </Button>
   499→        </DialogFooter>
   500→      </DialogContent>
   501→    </Dialog>
   502→  )
   503→
   504→  // ── Render: Challan / Print View dialog ───────────────────────────────
   505→  const renderChallanDialog = () => {
   506→    if (!challanDispatch) return null
   507→    const d = challanDispatch
   508→
   509→    return (
   510→      <Dialog open={!!challanDispatch} onOpenChange={(open) => !open && setChallanDispatch(null)}>
   511→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto print:max-w-none print:max-h-none print:p-0 print:shadow-none print:border-none">
   512→          {/* Screen-only close button header */}
   513→          <DialogHeader className="print:hidden">
   514→            <DialogTitle className="flex items-center gap-2">
   515→              <Eye className="size-5 text-emerald-600" />
   516→              Dispatch Challan
   517→            </DialogTitle>
   518→            <DialogDescription>
   519→              Preview and print the dispatch slip
   520→            </DialogDescription>
   521→          </DialogHeader>
   522→
   523→          {/* ── Challan Content (Print-friendly) ─────────────────────── */}
   524→          <div id="challan-content" className="print:p-6">
   525→            {/* Company Header */}
   526→            <div className="text-center mb-6 print:mb-8">
   527→              <div className="flex items-center justify-center gap-3 mb-2">
   528→                <Building2 className="size-8 text-emerald-600 print:text-black" />
   529→                <h1 className="text-2xl font-bold tracking-wide text-emerald-800 print:text-black">
   530→                  Veda Enterprises
   531→                </h1>
   532→              </div>
   533→              <p className="text-sm text-muted-foreground print:text-gray-600">
   534→                Brick Manufacturing &amp; Supply
   535→              </p>
   536→              <Separator className="mt-4 print:mt-6" />
   537→            </div>
   538→
   539→            {/* Title */}
   540→            <div className="text-center mb-6">
   541→              <h2 className="text-lg font-semibold underline underline-offset-4 print:text-black">
   542→                DISPATCH SLIP
   543→              </h2>
   544→            </div>
   545→
   546→            {/* Dispatch & Customer Info Grid */}
   547→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-6 text-sm print:text-black">
   548→              {/* Left column: Dispatch info */}
   549→              <div className="space-y-2">
   550→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   551→                  <span className="text-muted-foreground print:text-gray-500">Dispatch No:</span>
   552→                  <span className="font-semibold">{d.dispatchNumber}</span>
   553→                </div>
   554→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   555→                  <span className="text-muted-foreground print:text-gray-500">Date:</span>
   556→                  <span className="font-semibold">{formatDate(d.date)}</span>
   557→                </div>
   558→                {d.order && (
   559→                  <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   560→                    <span className="text-muted-foreground print:text-gray-500">Order Ref:</span>
   561→                    <span className="font-semibold">{d.order.orderNumber}</span>
   562→                  </div>
   563→                )}
   564→              </div>
   565→
   566→              {/* Right column: Customer info */}
   567→              <div className="space-y-2">
   568→                <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   569→                  <span className="text-muted-foreground print:text-gray-500">Customer:</span>
   570→                  <span className="font-semibold">{d.customer?.name || '—'}</span>
   571→                </div>
   572→                {d.customer?.mobile && (
   573→                  <div className="flex items-center justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   574→                    <span className="text-muted-foreground print:text-gray-500 flex items-center gap-1">
   575→                      <Phone className="size-3" /> Phone:
   576→                    </span>
   577→                    <span>{d.customer.mobile}</span>
   578→                  </div>
   579→                )}
   580→                {d.customer?.address && (
   581→                  <div className="flex items-start justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   582→                    <span className="text-muted-foreground print:text-gray-500 flex items-center gap-1 shrink-0">
   583→                      <MapPin className="size-3" /> Address:
   584→                    </span>
   585→                    <span className="text-right max-w-[60%]">{d.customer.address}</span>
   586→                  </div>
   587→                )}
   588→              </div>
   589→            </div>
   590→
   591→            <Separator className="mb-6 print:border-gray-400" />
   592→
   593→            {/* Transport Info */}
   594→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-6 text-sm print:text-black">
   595→              <div className="flex justify-between border-b border-dashed pb-1.5 print:border-gray-400">
   596→                <span className="text-muted-foreground print:text-gray-500">Truck Number:</span>
   597→                <span className="font-semibold">
   598→                  <Badge

   774→                      </TableCell>
